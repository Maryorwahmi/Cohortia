# main.py
import functions_framework
import json
import os
from google.cloud import compute_v1

@functions_framework.cloud_event
def vm_diagnostic_runbook(cloud_event):
  message_data = cloud_event.data.message.data
  if not message_data:
    print("No message data found.")
    return

  payload = json.loads(message_data)
  incident = payload.get('incident', {})
  vm_name = "my-test-vm"  # in practice parse from alert
  project_id = os.environ.get("GCP_PROJECT_ID")
  zone = "us-central1-a"

  try:
    compute_client = compute_v1.InstancesClient()
    request = compute_v1.GetInstanceRequest(project=project_id, zone=zone, instance=vm_name)
    vm_instance = compute_client.get(request=request)
    vm_status = vm_instance.status
    vm_ip = vm_instance.network_interfaces[0].network_ip if vm_instance.network_interfaces else "N/A"

    diagnostic_message = f"VM '{vm_name}' status: {vm_status}, ip: {vm_ip}"
  except Exception as e:
    diagnostic_message = f"Error fetching VM status: {e}"

  slack_webhook_url = os.environ.get('SLACK_WEBHOOK_URL')
  if slack_webhook_url:
    # post to Slack
    pass
  else:
    print(diagnostic_message)
Explanation
The function reads the alert payload, uses the Compute API to fetch instance state, and prints or posts a diagnostic message. Key points: parse defensively, handle exceptions, and use environment variables for configuration.
Deployment notes
Grant the function a Compute Viewer role and test it in staging. Keep it focused on diagnostics, not remediation.
