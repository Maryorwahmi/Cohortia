Module 4 › Chapter 4.1 › user_orders Walkthrough
Create UDT and table
-- Create a User-Defined Type for Address CREATE TYPE IF NOT EXISTS user_keyspace.address ( street text, city text, state text, zip_code text ); -- Create the user_orders table using frozen UDT, list, set, and map CREATE TABLE IF NOT EXISTS user_keyspace.user_orders ( user_id uuid, order_id uuid, order_status text STATIC, order_date timestamp, delivery_address frozen<address>, items list<frozen<tuple<text, int, decimal>>>, tags set<text>, shipping_details map<text, text>, PRIMARY KEY ((user_id, order_id)) );
Insert, update, and select
-- Insert data into user_orders INSERT INTO user_keyspace.user_orders ( user_id, order_id, order_status, order_date, delivery_address, items, tags, shipping_details ) VALUES ( a0eebc99-9c0b-4ef8-bb6d-6bb9bd380a11, b0eebc99-9c0b-4ef8-bb6d-6bb9bd380a12, 'Processing', '2023-10-26 10:00:00+0000', {street: '123 Main St', city: 'Anytown', state: 'CA', zip_code: '90210'}, [('Laptop', 1, 1200.00), ('Mouse', 1, 25.00)], {'electronics', 'urgent'}, {'carrier': 'UPS', 'tracking_number': 'ABC456'} ); -- Update static column order_status for the partition UPDATE user_keyspace.user_orders SET order_status = 'Shipped' WHERE user_id = a0eebc99-9c0b-4ef8-bb6d-6bb9bd380a11 AND order_id = b0eebc99-9c0b-4ef8-bb6d-6bb9bd380a12; -- Select elements and fields SELECT delivery_address.city, items[0], shipping_details['carrier'] FROM user_keyspace.user_orders WHERE user_id = a0eebc99-9c0b-4ef8-bb6d-6bb9bd380a11 AND order_id = b0eebc99-9c0b-4ef8-bb6d-6bb9bd380a12;
Scenario note
This combined design stores order-level metadata as static, structured addresses as a frozen UDT, and line items as a list, balancing read convenience with update cost considerations.
