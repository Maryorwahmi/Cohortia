Module 4 › Chapter 4.3 › News App Schema
Keyspace & Tables
CREATE KEYSPACE IF NOT EXISTS news_app
WITH replication = {'class': 'SimpleStrategy', 'replication_factor': 1};

USE news_app;
CREATE TABLE IF NOT EXISTS articles_by_author (
  author_id uuid,
  article_id uuid,
  publish_date timestamp,
  article_title text,
  article_content text,
  PRIMARY KEY ((author_id), publish_date, article_id)
) WITH CLUSTERING ORDER BY (publish_date DESC);
CREATE TABLE IF NOT EXISTS comments_by_article (
  article_id uuid,
  comment_id timeuuid,
  comment_text text,
  user_id uuid,
  user_name text,
  comment_time timestamp,
  PRIMARY KEY ((article_id), comment_time, comment_id)
) WITH CLUSTERING ORDER BY (comment_time DESC);
CREATE TABLE IF NOT EXISTS user_activity_feed (
  user_id uuid,
  activity_time timestamp,
  activity_type text,
  related_article_id uuid,
  related_article_title text,
  PRIMARY KEY ((user_id), activity_time)
) WITH CLUSTERING ORDER BY (activity_time DESC);
Design notes
  • Partition keys scope queries: author_id, article_id, user_id.
  • Clustering order returns newest-first for time-based lists.
  • user_name and related_article_title are denormalized for display.
