Starter script (replace content)
using UnityEngine;

public class PlayerController2D : MonoBehaviour
{
    public float moveSpeed = 5f;
    public float jumpForce = 10f;
    private Rigidbody2D rb;
    private bool isGrounded;

    public Transform groundCheck;
    public LayerMask groundLayer;

    void Start()
    {
        rb = GetComponent();
        if (groundCheck == null)
        {
            Debug.LogError("GroundCheck Transform not assigned!");
        }
    }

    void Update()
    {
        if (Input.GetButtonDown("Jump") && isGrounded)
        {
            rb.velocity = new Vector2(rb.velocity.x, jumpForce);
        }
    }

    void FixedUpdate()
    {
        isGrounded = Physics2D.OverlapCircle(groundCheck.position, 0.2f, groundLayer);
        float moveInput = Input.GetAxis("Horizontal");
        rb.velocity = new Vector2(moveInput * moveSpeed, rb.velocity.y);
    }
}
Scene checklist
  • Create Ground layer and assign to ground objects.
  • Add GroundCheck child and drag it into the script slot.
  • Set moveSpeed and jumpForce in Inspector for feel testing.
  • Freeze Z rotation on Rigidbody2D to prevent tumble.
Try this
Run the scene and press left/right and jump to confirm consistent behavior. Tweak values until controls feel responsive.
