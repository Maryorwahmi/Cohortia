Module 5 › Chapter 5.1 › Object Pooling — Pattern and Practice
Core implementation
A singleton pool that pre-instantiates objects and serves them on request prevents allocations during gameplay.
C# example
using System.Collections.Generic;
using UnityEngine;

public class ObjectPool : MonoBehaviour
{
    public static ObjectPool Instance { get; private set; }

    [SerializeField] private GameObject prefab;
    [SerializeField] private int initialPoolSize = 10;

    private Queue<GameObject> pool = new Queue<GameObject>();

    void Awake()
    {
        if (Instance == null) Instance = this; else Destroy(gameObject);

        for (int i = 0; i < initialPoolSize; i++)
        {
            GameObject obj = Instantiate(prefab);
            obj.SetActive(false);
            pool.Enqueue(obj);
        }
    }

    public GameObject GetPooledObject()
    {
        if (pool.Count > 0)
        {
            GameObject obj = pool.Dequeue();
            obj.SetActive(true);
            return obj;
        }
        else
        {
            GameObject obj = Instantiate(prefab);
            Debug.LogWarning("Object pool expanded at runtime. Consider increasing initialPoolSize.");
            return obj;
        }
    }

    public void ReturnPooledObject(GameObject obj)
    {
        obj.SetActive(false);
        pool.Enqueue(obj);
    }
}
Key Takeaway

Pool frequently spawned, short-lived objects to avoid runtime allocation and visible GC hitches.

Usage example
Call ObjectPool.Instance.GetPooledObject() instead of Instantiate, and ObjectPool.Instance.ReturnPooledObject(myObj) instead of Destroy. Pre-warm the pool to expected peak concurrency to avoid runtime expansion.
Common mistake
Not caching references or repeatedly calling GetComponent in Update loops — cache components and reuse references to reduce CPU cost.
