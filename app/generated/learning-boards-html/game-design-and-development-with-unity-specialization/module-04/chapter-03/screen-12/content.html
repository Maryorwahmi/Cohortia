Interaction flow
  1. Create a ray from the camera center using ViewportPointToRay.
  2. Use Physics.Raycast with a LayerMask and max distance.
  3. On hit, log or show prompt; on click, call Interact if the object implements IInteractable.
Debug tip: use Debug.DrawRay to visualize the ray while in the editor.
Example code
using UnityEngine;

public class PlayerInteraction : MonoBehaviour
{
  public float interactionDistance = 3f;
  public LayerMask interactableLayer; // Assign a layer for interactable objects

  void Update()
  {
    // Cast a ray from the camera's center forward
    Ray ray = Camera.main.ViewportPointToRay(new Vector3(0.5f, 0.5f, 0));
    RaycastHit hit;

    // Draw a debug ray in the Scene view (only visible in editor)
    Debug.DrawRay(ray.origin, ray.direction * interactionDistance, Color.red);

    if (Physics.Raycast(ray, out hit, interactionDistance, interactableLayer))
    {
      Debug.Log("Looking at: " + hit.collider.gameObject.name);

      if (Input.GetMouseButtonDown(0)) // Left mouse click
      {
        IInteractable interactable = hit.collider.GetComponent<IInteractable>();
        if (interactable != null)
        {
          interactable.Interact();
        }
        else
        {
          Debug.Log("Hit object is not interactable.");
        }
      }
    }
  }
}

public interface IInteractable
{
  void Interact();
}

public class Door : MonoBehaviour, IInteractable
{
  public void Interact()
  {
    Debug.Log("Door " + gameObject.name + " opened!");
  }
}
