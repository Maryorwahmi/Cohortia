// ThirdPersonCamera.cs
using UnityEngine;

public class ThirdPersonCamera : MonoBehaviour
{
    public Transform target; // Assign CameraTarget here
    public float distance = 5.0f;
    public float height = 2.0f;
    public float rotationSpeed = 3.0f;
    public float smoothSpeed = 0.125f;

    private float currentX = 0.0f;
    private float currentY = 0.0f;

    void LateUpdate()
    {
        if (target == null) return;

        currentX += Input.GetAxis("Mouse X") * rotationSpeed;
        currentY -= Input.GetAxis("Mouse Y") * rotationSpeed;
        currentY = Mathf.Clamp(currentY, -30f, 60f);

        Quaternion rotation = Quaternion.Euler(currentY, currentX, 0);
        Vector3 desiredPosition = target.position - rotation * Vector3.forward * distance + Vector3.up * height;
        Vector3 smoothedPosition = Vector3.Lerp(transform.position, desiredPosition, smoothSpeed);
        transform.position = smoothedPosition;
        transform.LookAt(target.position);
    }
}
This script demonstrates the common follow-and-look pattern. Clamping and smoothing parameters are the primary levers for feel.
Explanation cards
LateUpdate
Ensures the camera updates after player movement so the view matches the final character position for the frame.
Smoothing
Use Lerp or SmoothDamp for smooth motion; adjust smoothSpeed to control follow tightness.
Collision protection
Raycast from target to desired position and move camera to hit point to avoid clipping through walls.
Challenge
Adjust rotationSpeed and smoothSpeed so camera feels natural for both slow walking and quick turns.
