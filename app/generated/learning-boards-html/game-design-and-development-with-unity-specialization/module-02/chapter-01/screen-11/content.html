Full example
using UnityEngine; // Essential for Unity-specific classes like MonoBehaviour, Vector3, etc.

public class SimpleMovement : MonoBehaviour
{
    // Public variable, editable in the Unity Inspector
    public float moveSpeed = 5.0f;

    // Private variable, not directly editable in the Inspector
    private int score = 0;

    // Awake is called when the script instance is being loaded
    void Awake()
    {
        Debug.Log("SimpleMovement script has Woken Up!");
    }

    // Start is called before the first frame update
    void Start()
    {
        Debug.Log("SimpleMovement script has Started!");
        transform.position = new Vector3(0, 1, 0);
    }

    // Update is called once per frame
    void Update()
    {
        transform.Translate(Vector3.forward * moveSpeed * Time.deltaTime);

        if (Input.GetKeyDown(KeyCode.S))
        {
            score += 10;
            Debug.Log("Score: " + score);
        }
    }

    // FixedUpdate is called at a fixed interval, ideal for physics operations
    void FixedUpdate()
    {
        // Physics code would go here
    }
}
What to watch
Observe Console logs for Awake and Start to confirm order.
Runtime tweak
Adjust moveSpeed in the Inspector during Play to see immediate effect.
Misconception
Logging every frame is fine for short tests but can slow the Editor if left enabled long-term.
