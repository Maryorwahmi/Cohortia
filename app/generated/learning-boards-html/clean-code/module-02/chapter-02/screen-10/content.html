Refactored Code (good names)
# config_processor.py
DEFAULT_TIMEOUT_SECONDS = 600
DEFAULT_PORT = 8080
is_active = False

def process_input(raw_input_string):
    # raw_input_string is a string input
    if len(raw_input_string) > 0:
        global is_active
        is_active = True
        return raw_input_string.strip().upper()
    return ""

user_input = "  hello world "
processed_result = process_input(user_input)

if is_active:
    print(f"Processed: {processed_result} on port {DEFAULT_PORT}")
else:
    print("No input processed.")
Refactor notes
  • Constants renamed to ALL_CAPS and include units where helpful.
  • Booleans become predicates: is_active.
  • Function and parameter names describe action and content.
  • Intermediate names are explicit and searchable.
