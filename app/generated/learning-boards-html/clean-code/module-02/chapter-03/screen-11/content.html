Refactored Code
class UserManager:
    def __init__(self):
        self.users = {}  # {user_id: {"name": ..., "active": True, "last_login": ...}}

    def add_user(self, user_id, full_name, active_by_default=True):
        if user_id not in self.users:
            self.users[user_id] = {"name": full_name, "active": active_by_default, "last_login": None}
            return True
        return False

    def log_user_added(self, user_id):
        user = self.users.get(user_id)
        if user:
            print(f"User {user['name']} added.")

    def get_user_info(self, user_id):
        return self.users.get(user_id)

    def update_last_login(self, user_id):
        user = self.users.get(user_id)
        if user:
            user["last_login"] = "now"

    def is_user_active(self, user_id):
        user = self.users.get(user_id)
        return bool(user and user["active"])
Updated Usage
mgr = UserManager()
if mgr.add_user("u1", "Alice"):
    mgr.log_user_added("u1")
info = mgr.get_user_info("u1")
if info:
    mgr.update_last_login("u1")
print(mgr.is_user_active("u1"))
Key Takeaway

Refactored methods give explicit, testable behaviors and clearer call sites. Separate logging and state updates from pure queries.
