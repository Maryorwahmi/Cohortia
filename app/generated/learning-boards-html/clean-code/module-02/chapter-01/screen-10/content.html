Starter Code
# original_script.py
def p(d):
    # d is a list of tuples like (name_str, age_int, status_str)
    res = []
    for x in d:
        if x[1] > 18 and x[2] == "active":
            res.append(x[0].upper())
    return res

u_data = [
    ("alice", 25, "active"),
    ("bob", 17, "inactive"),
    ("charlie", 30, "active"),
    ("diana", 22, "pending")
]

o = p(u_data)
print(o)
