Before (No AOP)
class ProductService:
    def add_product(self, product_data):
        if not self.is_authorized("admin"):
            raise PermissionError("Not authorized")
        print(f"LOG: Entering add_product with {product_data}")
        # ... core business logic to add product ...
        print(f"LOG: Exiting add_product")
        return True

    def update_product(self, product_id, new_data):
        if not self.is_authorized("editor"):
            raise PermissionError("Not authorized")
        print(f"LOG: Entering update_product for {product_id}")
        # ... core business logic to update product ...
        print(f"LOG: Exiting update_product")
        return True
After (With Decorators)
class ProductService:
    @log_method_calls
    @authorize(roles=["admin"])
    def add_product(self, product_data):
        # core business logic to add product
        return {"id": 1, "name": product_data['name']}

    @log_method_calls
    @authorize(roles=["admin","editor"])
    def update_product(self, product_id, new_data):
        # core business logic to update product
        return {"id": product_id, "name": new_data['name']}
Benefit
Less duplication, clearer methods.
Risk
Implicit behavior can confuse new contributors if undocumented.
Mitigation
Document aspects and write tests for advised paths.
