Product status refactor
Original Product
class Product:
    def __init__(self, product_id, name, price, status: str):
        self.product_id = product_id
        self.name = name
        self.price = price
        self.status = status # "IN_STOCK", "OUT_OF_STOCK", "DISCONTINUED"

    def is_available_for_purchase(self):
        if self.status == "IN_STOCK":
            return True
        elif self.status == "OUT_OF_STOCK":
            return False
        elif self.status == "DISCONTINUED":
            return False
        else:
            print(f"Warning: Unknown status '{self.status}' for product {self.product_id}.")
            return False
Refactored pattern
from abc import ABC, abstractmethod

class ProductStatus(ABC):
    @abstractmethod
    def is_available_for_purchase(self):
        pass

    @abstractmethod
    def get_display_message(self):
        pass

class InStockStatus(ProductStatus):
    def is_available_for_purchase(self):
        return True
    def get_display_message(self):
        return "Available for immediate purchase."

class OutOfStockStatus(ProductStatus):
    def is_available_for_purchase(self):
        return False
    def get_display_message(self):
        return "Currently out of stock. Check back soon."

class DiscontinuedStatus(ProductStatus):
    def is_available_for_purchase(self):
        return False
    def get_display_message(self):
        return "This product is no longer available."
