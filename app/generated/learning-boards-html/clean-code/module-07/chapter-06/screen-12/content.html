from dataclasses import dataclass

@dataclass(frozen=True)
class DateRange:
    start_date: 'datetime.date'
    end_date: 'datetime.date'

    def overlaps_with(self, other: 'DateRange') -> bool:
        # True if ranges intersect
        return not (self.end_date <= other.start_date or self.start_date >= other.end_date)

class BookingManager:
    def check_availability(self, room_id, date_range: DateRange):
        print(f"Checking availability for room {room_id} from {date_range.start_date} to {date_range.end_date}.")
        for booking in self.bookings:
            if booking['room_id'] == room_id:
                if date_range.overlaps_with(booking['date_range']):
                    print("  Room is booked during the requested range.")
                    return False
        return True
Why this helps
The overlap logic now lives in one place, making tests simpler and preventing off-by-one mistakes across callers.
Key Takeaway
Encapsulate specialized logic with the data it operates on to reduce duplication and errors.
