# user_processor.py
def p_u_d(inp): # Process User Data
  res = {}
  parts = inp.split(',')
  if len(parts) == 2:
    nm = parts[0].strip()
    ag = int(parts[1].strip())
    if ag > 18:
      res['name'] = nm
      res['age'] = ag
      res['status'] = 'adult'
    else:
      res['name'] = nm
      res['age'] = ag
      res['status'] = 'minor'
  return res

# Example Usage:
u_input = "Alice,30"
processed_info = p_u_d(u_input)
print(processed_info)
