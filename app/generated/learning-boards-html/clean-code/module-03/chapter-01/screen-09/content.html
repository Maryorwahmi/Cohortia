Persistence helper
def _save_order_to_database(order_details):
    try:
        print(f"Saving order {order_details} to database...")
        # db.save(order_details)
        print("Order saved successfully.")
        return True
    except Exception as e:
        print(f"Error saving order: {e}")
        return False
