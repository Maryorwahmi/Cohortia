class InvalidFileDataError(Exception):
  def __init__(self, filepath, line_num, line_text):
    super().__init__(f"Invalid data in {filepath} at line {line_num}: '{line_text}'")
    self.filepath = filepath
    self.line_num = line_num
    self.line_text = line_text

def process_file_data(filepath):
  total_sum = 0
  try:
    with open(filepath, 'r') as f:
      for line_num, line in enumerate(f, 1):
        text = line.strip()
        try:
          value = int(text)
          total_sum += value
        except ValueError:
          # Treat non-integer lines as exceptional with context
          raise InvalidFileDataError(filepath, line_num, text)
    return total_sum
  except FileNotFoundError:
    raise
