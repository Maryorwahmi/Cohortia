import dlt
from pyspark.sql.functions import col, trim, lower

@dlt.table(comment="Cleaned customer data with trimmed strings and expectations.")
@dlt.expect("customer_id_not_null", "customer_id IS NOT NULL")
@dlt.expect_or_drop("age_positive", "age > 0")
@dlt.expect_or_warn("valid_email_format", "email LIKE '%@%.%'")
def cleaned_customer_data():
  return (
    dlt.read_stream("raw_customer_stream")
      .select(
        col("customer_id"),
        trim(col("first_name")).alias("first_name"),
        trim(col("last_name")).alias("last_name"),
        lower(trim(col("email"))).alias("email"),
        col("country"),
        col("age").cast("integer").alias("age")
      )
  )

@dlt.table(comment="Aggregated summary by country.")
def aggregated_customer_summary():
  return (
    dlt.read("cleaned_customer_data")
      .groupBy("country")
      .agg(count("customer_id").alias("total_customers"))
  )
Why this pattern
The warn pattern gives you time to collect data about format issues. Dropping invalid ages keeps averages sensible, while failing on missing IDs prevents incorrect totals from being published.
Check
After update, open the table's Data Quality panel to see counts for each expectation and links to event logs with offending rows.
