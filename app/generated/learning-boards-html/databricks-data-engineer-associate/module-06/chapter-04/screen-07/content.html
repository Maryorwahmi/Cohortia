Starter code (Python + Spark SQL)
# Create a catalog
spark.sql("CREATE CATALOG IF NOT EXISTS marketing_cohortia_uc")

# Create a schema inside the catalog
spark.sql("CREATE SCHEMA IF NOT EXISTS marketing_cohortia_uc.customer_data_cohortia_uc")

# Create a sample table with PII
spark.sql("""
  CREATE TABLE IF NOT EXISTS marketing_cohortia_uc.customer_data_cohortia_uc.customer_profiles_cohortia_uc (
    customer_id INT,
    name STRING,
    email STRING,
    segment STRING
  ) USING DELTA
""")

# Insert sample rows
spark.sql("INSERT INTO marketing_cohortia_uc.customer_data_cohortia_uc.customer_profiles_cohortia_uc VALUES (1, 'Alice Smith', 'alice.s@example.com', 'Premium')")
Grant and verify
-- Grant full SELECT to marketing_analysts
GRANT SELECT ON TABLE marketing_cohortia_uc.customer_data_cohortia_uc.customer_profiles_cohortia_uc TO `marketing_analysts_cohortia_uc`;

-- Grant column-level SELECT to junior analysts
GRANT SELECT (customer_id, name, segment) ON TABLE marketing_cohortia_uc.customer_data_cohortia_uc.customer_profiles_cohortia_uc TO `junior_analysts_cohortia_uc`;

-- Verify grants
SHOW GRANTS ON TABLE marketing_cohortia_uc.customer_data_cohortia_uc.customer_profiles_cohortia_uc;
Note: testing actual access often requires switching user context. Use groups to avoid per-user administration.
