Starter code
from pyspark.sql.types import StructType, StructField, IntegerType, DoubleType, DateType
from pyspark.sql.functions import sum, desc
from datetime import date

orders_data = [
  (101, 1, 150.75, date(2023,1,15)),
  (102, 2, 200.00, date(2023,1,16)),
  (103, 1, 50.25, date(2023,1,17)),
  (104, 3, 300.50, date(2023,1,18)),
  (105, 2, 75.00, date(2023,1,19))
]

orders_schema = StructType([
  StructField("order_id", IntegerType(), False),
  StructField("customer_id", IntegerType(), False),
  StructField("amount", DoubleType(), False),
  StructField("order_date", DateType(), False)
])

orders_df = spark.createDataFrame(orders_data, schema=orders_schema)
orders_df.createOrReplaceTempView("customer_orders")

# DataFrame API equivalent
from pyspark.sql.functions import sum as _sum

total_spent_df = orders_df.groupBy("customer_id").agg(_sum("amount").alias("total_spent")).orderBy(desc("total_spent"))
total_spent_df.show()
Key Takeaway

Replicate the same aggregation in both SQL and the DataFrame API to see identical results and inspect their plans with explain().
