from cryptography.hazmat.primitives import hashes
from cryptography.hazmat.primitives.asymmetric import rsa, padding
import json, base64, time

class VerifiableCredentialSystem:
    def __init__(self):
        # 1. Generate Issuer's (University) RSA key pair
        self.issuer_private_key = rsa.generate_private_key(
            public_exponent=65537,
            key_size=2048,
            backend=default_backend()
        )
        self.issuer_public_key = self.issuer_private_key.public_key()

    def issue_credential(self, holder_name, degree, graduation_date):
        # 2. Create Credential Data
        credential_data = {
            "holder": holder_name,
            "degree": degree,
            "graduation_date": graduation_date,
            "issuer": "Cohortia University",
            "timestamp": int(time.time())
        }
        credential_json = json.dumps(credential_data, sort_keys=True).encode('utf-8')

        # 3. Sign Credential with Issuer's private key
        signature = self.issuer_private_key.sign(
            credential_json,
            padding.PSS(
                mgf=padding.MGF1(hashes.SHA256()),
                salt_length=padding.PSS.MAX_LENGTH
            ),
            hashes.SHA256()
        )
        return {
            "credential": credential_data,
            "signature": base64.b64encode(signature).decode('utf-8')
        }

    def verify_credential(self, issued_vc, issuer_public_key_pem):
        credential_data = issued_vc["credential"]
        signature = base64.b64decode(issued_vc["signature"])
        # Reconstruct public key from PEM (simplified for brevity)
        verifier_public_key = serialization.load_pem_public_key(
            issuer_public_key_pem.encode('utf-8'),
            backend=default_backend()
        )
        credential_json = json.dumps(credential_data, sort_keys=True).encode('utf-8')

        # 4. Verify Credential with Issuer's public key
        try:
            verifier_public_key.verify(
                signature,
                credential_json,
                padding.PSS(
                    mgf=padding.MGF1(hashes.SHA256()),
                    salt_length=padding.PSS.MAX_LENGTH
                ),
                hashes.SHA256()
            )
            print(f"Credential for {credential_data['holder']} VERIFIED successfully!")
            return True
        except Exception as e:
            print(f"Credential for {credential_data['holder']} VERIFICATION FAILED: {e}")
            return False
Issuer's Role

Generates a key pair and uses its private key to cryptographically sign the credential data.

Verifier's Role

Uses the Issuer's public key to check the signature against the credential data, ensuring authenticity.

Key Takeaway

Cryptographic signatures are fundamental to VCs, providing a robust mechanism for trusted issuance and tamper-evident verification of digital attributes.
