import logging

logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')

def greet(name):
    logging.info(f"Greeting {name}")
    return f"Hello, {name.upper()}!"

if __name__ == '__main__':
    user_name = "DevOps Learner"
    print(greet(user_name))
