Definition docker-compose.yml structure
Example docker-compose.yml (root of project)
version: '3.8'

services:
  web:
    build: .
    ports:
      - "8000:5000"
    volumes:
      - ./app:/app
    environment:
      FLASK_APP: app.py
      FLASK_ENV: development
      DATABASE_URL: postgresql://user:password@db:5432/mydb
    depends_on:
      - db
    networks:
      - app-network

  db:
    image: postgres:13-alpine
    environment:
      POSTGRES_DB: mydb
      POSTGRES_USER: user
      POSTGRES_PASSWORD: password
    volumes:
      - pgdata:/var/lib/postgresql/data
    networks:
      - app-network

networks:
  app-network:
    driver: bridge

volumes:
  pgdata:
Explainer cards
build vs image
Use build to construct an image from a Dockerfile in your repo. Use image to pull an existing image from a registry.
ports
Map a host port to a container port so you can access services from your machine.
volumes
Named volumes persist data. Bind mounts let you edit code live during development.
Concrete example
Run this locally

Place this file at the project root, then run docker compose up to build and start web and db together.
