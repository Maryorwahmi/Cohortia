Python Lambda Function (Simplified)
import boto3
import os

ec2 = boto3.client('ec2')

def lambda_handler(event, context):
    print(f"Received alarm event: {event}")
    alarm_name = event['Records'][0]['Sns']['Subject']
    
    if "NAT Gateway Port Allocation Error" in alarm_name:
        print("Initiating NAT Gateway remediation...")
        vpc_id = os.environ.get('VPC_ID')
        public_subnet_id = os.environ.get('PUBLIC_SUBNET_ID')
        
        if not vpc_id or not public_subnet_id:
            print("VPC_ID or PUBLIC_SUBNET_ID not set. Cannot remediate.")
            return

        try:
            # 1. Allocate a new Elastic IP
            eip_response = ec2.allocate_address(Domain='vpc')
            eip_alloc_id = eip_response['AllocationId']
            print(f"Allocated new EIP: {eip_alloc_id}")

            # 2. Create a new NAT Gateway
            nat_gw_response = ec2.create_nat_gateway(
                SubnetId=public_subnet_id,
                AllocationId=eip_alloc_id,
                TagSpecifications=[{'ResourceType': 'natgateway', 'Tags': [{'Key': 'Name', 'Value': 'AutomatedRemediatedNATGateway'}]}]
            )
            new_nat_gw_id = nat_gw_response['NatGateway']['NatGatewayId']
            print(f"Created new NAT Gateway: {new_nat_gw_id}. Waiting...")

            # Wait for NAT Gateway to be available (crucial step)
            waiter = ec2.get_waiter('nat_gateway_available')
            waiter.wait(NatGatewayIds=[new_nat_gw_id])
            print(f"New NAT Gateway {new_nat_gw_id} is available.")

            # 3. Update route tables (conceptual)
            private_rt_id = os.environ.get('PRIVATE_ROUTE_TABLE_ID')
            if private_rt_id:
                print(f"Manually update route table {private_rt_id} to use {new_nat_gw_id}.")

        except Exception as e:
            print(f"Error during NAT Gateway remediation: {e}")
            raise

    return {
        'statusCode': 200,
        'body': 'Remediation process completed.'
    }
Code Explanation

Key steps in the remediation logic:

  • Event Parsing: Identifies the specific alarm that triggered the Lambda function.
  • EIP Allocation: Requests a new Elastic IP address for the new NAT Gateway.
  • NAT Gateway Creation: Provisions a new NAT Gateway in a specified public subnet.
  • Route Table Update (Conceptual): Points private route tables to the new, healthy NAT Gateway.
Key Takeaway

Lambda functions can automate complex remediation workflows, such as NAT Gateway failover, by programmatically interacting with AWS services.
