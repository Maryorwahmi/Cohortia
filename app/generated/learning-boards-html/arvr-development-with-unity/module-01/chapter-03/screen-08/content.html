PlaceOnPlane.cs
Instantiate a prefab at a detected plane position when the user taps the screen.
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.XR.ARFoundation;
using UnityEngine.XR.ARSubsystems;

public class PlaceOnPlane : MonoBehaviour
{
    [SerializeField]
    GameObject m_PlacedPrefab; // Assign your Cube Prefab here in the Inspector

    private ARRaycastManager m_RaycastManager;
    private static List<ARRaycastHit> s_Hits = new List<ARRaycastHit>();

    void Awake()
    {
        m_RaycastManager = GetComponent<ARRaycastManager>();
    }

    void Update()
    {
        if (Input.touchCount > 0)
        {
            Touch touch = Input.GetTouch(0);
            if (touch.phase == TouchPhase.Began)
            {
                if (m_RaycastManager.Raycast(touch.position, s_Hits, TrackableType.PlaneWithinPolygon))
                {
                    var hitPose = s_Hits[0].pose;
                    Instantiate(m_PlacedPrefab, hitPose.position, hitPose.rotation);
                }
            }
        }
    }
}
Explanation
  • ARRaycastManager handles AR raycasts into detected trackables like planes.
  • TrackableType.PlaneWithinPolygon ensures hits are inside detected plane boundaries.
  • Instantiate uses the hit pose position and rotation so objects orient with the surface.
Concrete scenario
Build and run on an ARCore-compatible Android device; tap a detected plane and check that the cube prefab appears and stays anchored to the surface as you move the phone.
Common mistake
Forgetting to assign the prefab in the Inspector. The script will run but nothing appears. Also ensure ARRaycastManager component exists on the same GameObject.
Key Takeaway

Use ARRaycastManager and trackable hits to reliably place prefabs; attach components and assign references before testing on device.
