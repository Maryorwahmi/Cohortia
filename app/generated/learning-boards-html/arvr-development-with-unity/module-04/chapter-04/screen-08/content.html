Module 5 › Chapter 4.4 › AR Anchor Manager
Placement lifecycle
  1. Raycast to find hit pose on a plane.
  2. Call arAnchorManager.AddAnchor(hitPose).
  3. Instantiate or parent object to the returned ARAnchor transform.
  4. On move, remove old anchor and create a new one.
Edge case
AddAnchor can fail if tracking is not ready. Always null-check and provide a fallback placement strategy.
Example integration
Below is the ARPlacementManager example modified to use ARAnchorManager. It demonstrates creating an anchor and parenting the placed object, including anchor removal when relocating the object.
using UnityEngine;
using UnityEngine.XR.ARFoundation;
using UnityEngine.XR.ARSubsystems;
using System.Collections.Generic;

public class ARPlacementManager : MonoBehaviour
{
    [SerializeField] private GameObject objectToPlace;
    private ARRaycastManager arRaycastManager;
    private ARAnchorManager arAnchorManager;
    private static List<ARRaycastHit> hits = new List<ARRaycastHit>();

    private GameObject placedObject;
    private ARAnchor placedObjectAnchor;

    void Awake()
    {
        arRaycastManager = GetComponent<ARRaycastManager>();
        arAnchorManager = GetComponent<ARAnchorManager>();
    }

    void Update()
    {
        if (Input.touchCount > 0)
        {
            Touch touch = Input.GetTouch(0);
            if (touch.phase == TouchPhase.Began)
            {
                if (arRaycastManager.Raycast(touch.position, hits, TrackableType.PlaneWithinPolygon))
                {
                    Pose hitPose = hits[0].pose;

                    if (placedObject == null)
                    {
                        placedObjectAnchor = arAnchorManager.AddAnchor(hitPose);
                        if (placedObjectAnchor != null)
                        {
                            placedObject = Instantiate(objectToPlace, placedObjectAnchor.transform);
                            placedObject.transform.localPosition = Vector3.zero;
                            placedObject.transform.localRotation = Quaternion.identity;
                        }
                    }
                    else
                    {
                        if (placedObjectAnchor != null)
                        {
                            arAnchorManager.RemoveAnchor(placedObjectAnchor);
                        }
                        placedObjectAnchor = arAnchorManager.AddAnchor(hitPose);
                        if (placedObjectAnchor != null)
                        {
                            placedObject.transform.SetParent(placedObjectAnchor.transform);
                            placedObject.transform.localPosition = Vector3.zero;
                            placedObject.transform.localRotation = Quaternion.identity;
                        }
                    }
                }
            }
        }
    }
}
Key Takeaway

Anchors should be created at confirmed hit poses and removed when relocating content to maintain a clean anchor lifecycle.
