Code Walkthrough

This example subscribes to frameReceived and sets ambient and directional lighting using safe HasValue checks and scaling.

using UnityEngine;
using UnityEngine.XR.ARFoundation;
using UnityEngine.XR.ARSubsystems;

public class ARLightEstimator : MonoBehaviour
{
    private ARCameraManager arCameraManager;

    void Awake()
    {
        arCameraManager = GetComponent<ARCameraManager>();
    }

    void OnEnable()
    {
        arCameraManager.frameReceived += OnCameraFrameReceived;
    }

    void OnDisable()
    {
        arCameraManager.frameReceived -= OnCameraFrameReceived;
    }

    void OnCameraFrameReceived(ARCameraFrameEventArgs eventArgs)
    {
        if (eventArgs.lightEstimation.averageColorTemperature.HasValue)
        {
            RenderSettings.ambientLight = Color.white; // Map color temp to a Color in practice
        }

        if (eventArgs.lightEstimation.averageIntensityInLumens.HasValue)
        {
            Light sceneLight = FindObjectOfType<Light>();
            if (sceneLight != null && sceneLight.type == LightType.Directional)
            {
                sceneLight.intensity = eventArgs.lightEstimation.averageIntensityInLumens.Value / 1000f; 
            }
        }
    }
}
Explanation & Tips
  • Always check HasValue before using light estimation fields.
  • Scale lumens to a Unity-friendly range; typical divisor values vary per scene.
  • Cache scene references and avoid heavy calls inside every frame callback.
  • Smooth updates with Lerp to avoid abrupt lighting changes.
Pro tip
Expose your Directional Light via inspector instead of using FindObjectOfType for better performance and reliability.
Key Takeaway

Use frame events to drive lighting but map and smooth values so the scene changes are believable and stable.
