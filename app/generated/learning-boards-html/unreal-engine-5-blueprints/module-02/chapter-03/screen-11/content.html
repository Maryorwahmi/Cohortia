// Simplified blueprint-style pseudocode for clarity
BP_PlayerCharacter:
  Variable CurrentHealth : Float = 100.0

Function ApplyDamage(DamageAmount: Float, DamageType: String) -> NewHealthValue: Float
  Entry:
    current = Get CurrentHealth
    result = current - DamageAmount
    Set CurrentHealth = result
    AppendStr = "Took " + ToString(DamageAmount) + " damage (" + DamageType + "). New Health: " + ToString(result)
    Print String AppendStr
  Return NewHealthValue = result

Event Graph:
  On Keyboard Spacebar Pressed
    Call ApplyDamage(DamageAmount = 10.0, DamageType = "Blunt Force")
