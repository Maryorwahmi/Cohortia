Starter code
def calculate_bmi(weight_kg, height_m, unit='metric'):
    """
    Calculates BMI given weight in kg and height in meters.
    Unit parameter is for demonstration of default values.
    """
    bmi = weight_kg / (height_m ** 2)
    return bmi

# Example calls shown on the right
Example calls
# Positional
bmi1 = calculate_bmi(70, 1.75)
print(f"BMI (positional): {bmi1}")

# Keyword
bmi2 = calculate_bmi(height_m=1.80, weight_kg=85, unit='imperial')
print(f"BMI (keyword): {bmi2}")

# Default unit
bmi3 = calculate_bmi(65, 1.60)
print(f"BMI (default unit): {bmi3}")

# Override unit
bmi4 = calculate_bmi(65, 1.60, unit='imperial')
print(f"BMI (override unit): {bmi4}")
