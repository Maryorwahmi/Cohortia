Starter code
# 1. Initialize empty inventory
inventory = {}

# 2. Add products
inventory["Laptop"] = {"quantity": 10, "price": 1200.00}
inventory["Mouse"] = {"quantity": 50, "price": 25.50}
inventory["Keyboard"] = {"quantity": 30, "price": 75.00}

# 3. Print inventory
print("Current Inventory:")
print(inventory)

# 4. Update Laptop quantity
inventory["Laptop"]["quantity"] = 8

# 5. Add Monitor
inventory["Monitor"] = {"quantity": 15, "price": 300.00}

# 6. Safe get
price = inventory.get("Headphones", {"price": 0.00})["price"]
print(price)

# 7. Remove Mouse
removed = inventory.pop("Mouse", None)

# 8. Iterate and print
print("
Updated Inventory (Product and Quantity):")
for product, info in inventory.items():
    print(product, info["quantity"])
