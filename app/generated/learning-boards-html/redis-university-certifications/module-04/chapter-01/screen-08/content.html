Node.js starter
const Redis = require('ioredis');

const redis = new Redis({
  host: 'localhost',
  port: 6379,
  db: 0
});

redis.on('connect', () => {
  console.log('Connected to Redis!');
});

redis.on('error', (err) => {
  console.error('Redis connection error:', err);
});

async function runRedisCommands() {
  try {
    await redis.ping();
    await redis.set('nodekey', 'Hello from Node.js!');
    const value = await redis.get('nodekey');
    console.log(`Value for 'nodekey': ${value}`);
  } finally {
    redis.quit();
  }
}

runRedisCommands();
Note: the example listens for events and calls quit in finally to close the client and its pool.
Observability and shutdown
  • Use connect and error events for logging and monitoring.
  • Call quit or disconnect during process termination to free sockets.
  • ioredis manages pooling by default; tune only when necessary.
Takeaway
Observe connection events and always perform a graceful quit on shutdown to avoid leaked sockets.
