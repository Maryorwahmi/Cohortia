import redis
import time
import random
import threading

r = redis.Redis(host='localhost', port=6379, db=0, decode_responses=True)

POST_ID = "post:123"
LIKES_COUNT_KEY = f"{POST_ID}:likes_count"
LIKED_BY_KEY = f"{POST_ID}:liked_by"

# Initialize Redis data
r.set(LIKES_COUNT_KEY, 0)
r.delete(LIKED_BY_KEY)

def like_post(post_id, user_id):
    with r.pipeline() as pipe:
        while True:
            try:
                pipe.watch(LIKES_COUNT_KEY, LIKED_BY_KEY)

                if pipe.sismember(LIKED_BY_KEY, user_id):
                    pipe.unwatch()
                    return False

                pipe.multi()
                pipe.incr(LIKES_COUNT_KEY)
                pipe.sadd(LIKED_BY_KEY, user_id)

                results = pipe.execute()
                if results is None:
                    time.sleep(random.uniform(0.01, 0.05))
                    continue
                else:
                    return True
            except redis.exceptions.WatchError:
                time.sleep(random.uniform(0.01, 0.05))
                continue
            except Exception:
                return False
Run multiple threads to simulate concurrent likes. Observe that WATCH aborts are retried and the final count matches the set size of likers.
Test idea
Start several threads that call like_post with different user ids and assert final count equals set cardinality.
Watchout
Checking membership before MULTI expands the watch window; keep logic short and retry on abort.
Key Takeaway
This activity concretely shows optimistic locking: abort on conflict, then retry with fresh state.
