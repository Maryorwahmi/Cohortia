Convert a C rectangle program to C++
Follow a practical, runnable example that replaces C I/O with streams and uses a reference parameter in a helper function.
Starter (C) template
#include <stdio.h>

int calculateAreaC(int length, int width) {
  return length * width;
}

int main() {
  int length, width;
  printf("Enter length: ");
  scanf("%d", &length);
  printf("Enter width: ");
  scanf("%d", &width);
  int area = calculateAreaC(length, width);
  printf("The area of the rectangle is: %d
", area);
  return 0;
}
Converted C++ example
#include <iostream>

int calculateArea(int& length, int& width) {
  return length * width;
}

int main() {
  int length, width;
  std::cout << "Enter length: ";
  std::cin >> length;
  std::cout << "Enter width: ";
  std::cin >> width;
  int area = calculateArea(length, width);
  std::cout << "The area of the rectangle is: " << area << std::endl;
  return 0;
}
Note: passing by reference here is optional; the example practices references. For simple calculations, passing by value is also acceptable.
