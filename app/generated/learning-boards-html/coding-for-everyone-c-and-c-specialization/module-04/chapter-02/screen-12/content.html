#include <iostream>
#include <string>

class Car {
private:
  std::string make;
  std::string model;
  int year;
  int speed;

public:
  Car() : make("Unknown"), model("Unknown"), year(2023), speed(0) {
    std::cout << "Default Car created." << std::endl;
  }

  Car(std::string carMake, std::string carModel, int carYear)
    : make(carMake), model(carModel), year(carYear), speed(0) {
    std::cout << "Car " << make << " " << model << " created." << std::endl;
  }

  ~Car() {
    std::cout << "Car " << make << " " << model << " destroyed." << std::endl;
  }

  void accelerate(int increment) {
    if (speed + increment <= 200) {
      speed += increment;
      std::cout << make << " accelerated. Current speed: " << speed << std::endl;
    } else {
      std::cout << "Cannot accelerate beyond max speed." << std::endl;
    }
  }

  int getSpeed() const { return speed; }
};

int main() {
  Car myCar; // default constructor
  Car* fancyCar = new Car("Ferrari", "488 GTB", 2022);
  fancyCar->accelerate(150);
  delete fancyCar; // destructor called
  return 0;
}
