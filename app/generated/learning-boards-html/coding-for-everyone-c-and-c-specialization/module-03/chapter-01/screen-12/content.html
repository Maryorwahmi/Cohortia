Starter Code
#include <stdio.h>
#include <stdlib.h>
#include <string.h> // For strlen, strcpy

#define INITIAL_CAPACITY 2
#define MAX_STRING_LENGTH 100

int main() {
    char **stringArray = NULL; // Pointer to an array of char pointers (strings)
    int capacity = INITIAL_CAPACITY;
    int count = 0;
    char buffer[MAX_STRING_LENGTH]; // Temporary buffer for reading input

    // Allocate initial memory for string pointers
    stringArray = (char **)malloc(capacity * sizeof(char *));
    if (stringArray == NULL) {
        fprintf(stderr, "Initial memory allocation failed!
");
        return 1;
    }

    printf("Enter names (type 'done' to finish):
");

    while (1) {
        printf("> ");
        if (fgets(buffer, MAX_STRING_LENGTH, stdin) == NULL) {
            break; // Error reading input
        }
        buffer[strcspn(buffer, "
")] = 0; // Remove newline character

        if (strcmp(buffer, "done") == 0) {
            break;
        }

        // Check if we need to reallocate
        if (count == capacity) {
            capacity *= 2; // Double the capacity
            char **tempArray = (char **)realloc(stringArray, capacity * sizeof(char *));
            if (tempArray == NULL) {
                fprintf(stderr, "Reallocation failed!
");
                // Free already allocated strings before exiting
                for (int i = 0; i < count; i++) {
                    free(stringArray[i]);
                }
                free(stringArray);
                return 1;
            }
            stringArray = tempArray;
            printf("Capacity increased to %d
", capacity);
        }

        // Allocate memory for the current string and copy it
        stringArray[count] = (char *)malloc((strlen(buffer) + 1) * sizeof(char));
        if (stringArray[count] == NULL) {
            fprintf(stderr, "String memory allocation failed for '%s'!
", buffer);
            // Handle error: free previous allocations and exit
            for (int i = 0; i < count; i++) {
                free(stringArray[i]);
            }
            free(stringArray);
            return 1;
        }
        strcpy(stringArray[count], buffer);
        count++;
    }

    printf("
--- Collected Names ---
");
    for (int i = 0; i < count; i++) {
        printf("%d: %s
", i + 1, stringArray[i]);
    }

    // TODO: Free all dynamically allocated memory (strings and array of pointers)
    // HINT: You need two loops or a loop and a final free.

    return 0;
}
What to implement
  • After printing collected names, free each string with free(stringArray[i]).
  • Then free the pointer array with free(stringArray) and set stringArray to NULL.
  • On any allocation failure, ensure previously allocated strings are freed before exiting.
