// app_config.h
#ifndef APP_CONFIG_H
#define APP_CONFIG_H

#define MAX_VALUE 100

#ifdef DEBUG_MODE
  #include <stdio.h>
  #define DEBUG_PRINT(msg) printf("DEBUG: %s
", msg)
#else
  #define DEBUG_PRINT(msg) do {} while (0)
#endif

#endif // APP_CONFIG_H

// utility.c
#include "app_config.h"
#include <stdio.h>

int calculate_sum(int a, int b) {
  DEBUG_PRINT("Entering calculate_sum function.");
  int sum = a + b;
  if (sum > MAX_VALUE) {
    DEBUG_PRINT("Sum exceeded MAX_VALUE.");
    return MAX_VALUE;
  }
  DEBUG_PRINT("Exiting calculate_sum function.");
  return sum;
}
How to test
Compile normally to run without debug prints. Compile with -DDEBUG_MODE to enable debug output and observe behavior changes.
Learning check
Did include guards prevent duplicate declarations? Did DEBUG_PRINT appear only when defined?
Next step
Optionally add a makefile with targets for debug and release builds to automate testing.
