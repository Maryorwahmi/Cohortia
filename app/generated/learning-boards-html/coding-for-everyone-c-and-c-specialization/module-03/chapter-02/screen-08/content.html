Write integers to a binary file
#include <stdio.h>
#include <stdlib.h>

int main() {
  int numbers[] = {10,20,30,40,50};
  int count = sizeof(numbers) / sizeof(numbers[0]);
  FILE *binFile = fopen("numbers.bin", "wb");
  if (binFile == NULL) { perror("Error opening binary file for writing"); return 1; }
  size_t written = fwrite(numbers, sizeof(int), count, binFile);
  if (written != count) { perror("Partial write"); }
  fclose(binFile);
  return 0;
}
