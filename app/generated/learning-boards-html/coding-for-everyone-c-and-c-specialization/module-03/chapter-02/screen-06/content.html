Reading lines safely
#include <stdio.h>

int main() {
  FILE *inFile = fopen("data.txt", "r");
  char buffer[256];
  if (inFile == NULL) {
    perror("Error opening file for reading");
    return 1;
  }
  if (fgets(buffer, sizeof(buffer), inFile) != NULL) {
    printf("Read line: %s", buffer);
  }
  fclose(inFile);
  return 0;
}
Notes
  • fgets preserves the newline if it fits; trim it for comparisons.
  • Check fscanf return value to confirm parsed items.
