class Fraction {
private:
  int numerator;
  int denominator;
  void simplify() {
    if (denominator == 0) { numerator = 0; denominator = 1; return; }
    if (numerator == 0) { denominator = 1; return; }
    int d = std::gcd(std::abs(numerator), std::abs(denominator));
    numerator /= d; denominator /= d;
    if (denominator < 0) { numerator = -numerator; denominator = -denominator; }
  }
public:
  Fraction(int num=0, int den=1): numerator(num), denominator(den) {
    if (denominator == 0) {
      std::cerr << "Error: Denominator cannot be zero. Setting to 1." << std::endl;
      denominator = 1;
    }
    simplify();
  }

};

Example expectation
1/2 + 1/3 yields 5/6. 1/2 * 1/3 yields 1/6. Fractions should be simplified.
Common pitfall
Forgetting to simplify or to normalize sign on the denominator leads to inconsistent comparisons.
Key Takeaway

Implement the overloads, test edge cases, and ensure results are simplified and consistent.
