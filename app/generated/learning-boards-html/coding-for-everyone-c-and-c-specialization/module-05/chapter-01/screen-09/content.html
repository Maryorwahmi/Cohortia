Starter & Solution
Below is a completed implementation matching the activity steps: Shape with protected color; Circle and Rectangle call the base constructor and implement calculateArea.
#include <iostream>
#include <string>
#include <cmath> // For M_PI

class Shape {
protected:
    std::string color;
public:
    Shape(const std::string& c) : color(c) {
        std::cout << "Shape constructor called for color: " << color << std::endl;
    }
    void displayColor() const {
        std::cout << "This shape is " << color << "." << std::endl;
    }
    ~Shape() {
        std::cout << "Shape destructor called for color: " << color << std::endl;
    }
};

class Circle : public Shape {
private:
    double radius;
public:
    Circle(const std::string& c, double r) : Shape(c), radius(r) {
        std::cout << "Circle constructor called for a " << color << " circle." << std::endl;
    }
    double calculateArea() const { return M_PI * radius * radius; }
    ~Circle() { std::cout << "Circle destructor called for a " << color << " circle." << std::endl; }
};

class Rectangle : public Shape {
private:
    double width;
    double height;
public:
    Rectangle(const std::string& c, double w, double h) : Shape(c), width(w), height(h) {
        std::cout << "Rectangle constructor called for a " << color << " rectangle." << std::endl;
    }
    double calculateArea() const { return width * height; }
    ~Rectangle() { std::cout << "Rectangle destructor called for a " << color << " rectangle." << std::endl; }
};

int main() {
    Circle myCircle("Red", 5.0);
    Rectangle myRectangle("Blue", 4.0, 6.0);
    myCircle.displayColor();
    std::cout << "Circle area: " << myCircle.calculateArea() << std::endl;
    myRectangle.displayColor();
    std::cout << "Rectangle area: " << myRectangle.calculateArea() << std::endl;
    return 0;
}
Explanation
  • Shape's constructor sets the protected color and prints a message to confirm initialization.
  • Circle and Rectangle call Shape(c) in their initializer lists so color is set before derived setup.
  • calculateArea implements formulas from the activity: pi·r² for Circle and width·height for Rectangle.
Key Takeaway

Follow the activity steps to practice initializer lists and protected members; run the main to observe constructor and destructor ordering.
