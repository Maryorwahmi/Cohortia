Starter code (my_profile dot c)
#include <stdio.h>

int main() {
    char my_initial = 'C';
    int my_age = 25;
    float my_height_meters = 1.80;
    char my_favorite_letter = 'Z';

    printf("Hello Cohortia! My initial is %c.
", my_initial);
    printf("I am %d years old.
", my_age);
    printf("My height is %.2f meters.
", my_height_meters);

}
