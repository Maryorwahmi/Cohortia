Starter code (run this file)
import hashlib
import json
import time

def calculate_hash(block_data):
  """Calculates the SHA256 hash of a block's data."""
  block_string = json.dumps(block_data, sort_keys=True).encode()
  return hashlib.sha256(block_string).hexdigest()

# --- Block 1 ---
block1_data = {
  "index": 1,
  "timestamp": time.time(),
  "transactions": [
    {"sender": "Alice", "recipient": "Bob", "amount": 10},
    {"sender": "Charlie", "recipient": "David", "amount": 5}
  ],
  "previous_hash": "0"
}
block1_hash = calculate_hash(block1_data)

print("--- Block 1 ---")
print(f"Data: {json.dumps(block1_data, indent=2)}")
print(f"Hash: {block1_hash}
")

# --- Block 2 ---
block2_data = {
  "index": 2,
  "timestamp": time.time(),
  "transactions": [
    {"sender": "Eve", "recipient": "Frank", "amount": 20}
  ],
  "previous_hash": block1_hash
}
block2_hash = calculate_hash(block2_data)

print("--- Block 2 ---")
print(f"Data: {json.dumps(block2_data, indent=2)}")
print(f"Hash: {block2_hash}
")

# --- Verification ---
print("--- Verification ---")
print(f"Is Block 2's previous_hash equal to Block 1's hash? {block2_data['previous_hash'] == block1_hash}")

What to observe
  • Both block hashes are printed; notice their values differ.
  • After tampering, block one’s hash changes and block two’s stored previous_hash no longer matches.
  • This mismatch is how nodes detect tampering across copies.
Key Takeaway

Run the script and tamper with block1 to observe immutability in action: hashes change and links break, making tampering obvious.
