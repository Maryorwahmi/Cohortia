#[program]
pub mod my_counter_app {
  use super::*;

  pub fn initialize(ctx: Context<Initialize>) -> Result<()> {
    let counter_account = &mut ctx.accounts.counter_account;
    counter_account.count = 0;
    msg!("Counter account initialized with count: {}", counter_account.count);
    Ok(())
  }

  pub fn increment(ctx: Context<Increment>) -> Result<()> {
    let counter_account = &mut ctx.accounts.counter_account;
    counter_account.count += 1;
    msg!("Counter incremented to: {}", counter_account.count);
    Ok(())
  }
}

#[derive(Accounts)]
pub struct Initialize<'info> {
  #[account(init, payer = user, space = 8 + 8)]
  pub counter_account: Account<'info, Counter>,
  #[account(mut)]
  pub user: Signer<'info>,
  pub system_program: Program<'info, System>,
}

#[derive(Accounts)]
pub struct Increment<'info> {
  #[account(mut)]
  pub counter_account: Account<'info, Counter>,
}

#[account]
pub struct Counter {
  pub count: u64,
}
Explanation & checks
  • init: creates counter_account and allocates space.
  • payer = user: user pays creation cost and must sign.
  • space = 8 + 8: 8 bytes for Anchor discriminator, 8 bytes for u64 count.
  • mut: marks accounts whose data will change during the instruction.
Common error
Undersizing space or forgetting payer as a signer causes build or runtime failures when Anchor constructs or writes the account.
Key Takeaway

Anchor attributes are declarative: they perform the heavy lifting for account creation so you can focus on program logic.
