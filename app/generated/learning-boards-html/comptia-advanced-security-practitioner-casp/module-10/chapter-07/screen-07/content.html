# --- Secure manage_post implementation
@ app.route('/api/v1/posts/<post_id>', methods=['GET', 'PUT', 'DELETE'])
@ jwt_required
def manage_post(post_id):
    post = posts_db.get(post_id)
    if not post:
        return jsonify({"message": "Post not found"}), 404

    current_user_id = request.user_id

    if request.method == 'GET':
        return jsonify(post)

    elif request.method == 'PUT':
        data = request.get_json()
        if not data:
            return jsonify({"message": "Request must be JSON"}), 400

        # Authorization: only the author can update
        if current_user_id != post.get('author_id'):
            return jsonify({"message": "Forbidden: not the author"}), 403

        # Input validation: ensure title and content meet basic rules
        title = data.get('title')
        content = data.get('content')
        if not title or not content:
            return jsonify({"message": "Missing title or content"}), 400
        if len(title) < 3 or len(title) > 200:
            return jsonify({"message": "Title length invalid"}), 400
        if len(content) < 10:
            return jsonify({"message": "Content too short"}), 400

        post.update({'title': title, 'content': content})
        return jsonify({"message": "Post updated", "post": post})

    elif request.method == 'DELETE':
        # Authorization: only the author can delete
        if current_user_id != post.get('author_id'):
            return jsonify({"message": "Forbidden: not the author"}), 403
        del posts_db[post_id]
        return jsonify({"message": "Post deleted"})
Explanation
Authorization check
Compare current user identity to the post author before allowing update or delete operations; return a 403 if they differ.
Input validation
Verify required fields are present and meet length constraints to prevent malformed data and injection attempts.
Test scenarios
Try an update with a non-owner token; expect forbidden. Try owner update with missing fields; expect a bad request.
Key Takeaway

Enforce authorization first, validate inputs second, then mutate state. The order matters for security.
