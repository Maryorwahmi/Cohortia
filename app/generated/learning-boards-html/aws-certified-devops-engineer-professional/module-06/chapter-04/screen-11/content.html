Lambda: NotifyIncidentLambda (excerpt)
# notify_incident_lambda.py
import json
import os
import boto3

sns_client = boto3.client('sns')
SNS_TOPIC_ARN = os.environ.get('SNS_TOPIC_ARN')

def lambda_handler(event, context):
  print("Received event for notification:")
  print(json.dumps(event, indent=2))
  message = f"Incident Detected! Details: {json.dumps(event)}"
  if SNS_TOPIC_ARN:
    sns_client.publish(TopicArn=SNS_TOPIC_ARN, Message=message, Subject="CRITICAL INCIDENT ALERT")
  return {'statusCode':200,'body':json.dumps('Notification processed!')}
Deploy this function with an execution role that allows sns:Publish and set SNS_TOPIC_ARN as an environment variable. Create AttemptRemediationLambda as a logging placeholder that returns remediationStatus: SUCCESS for testing.
