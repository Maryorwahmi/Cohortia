<!-- SecureWebApp construct (escaped) --> import * as cdk from 'aws-cdk-lib'; import { Construct } from 'constructs'; import * as ec2 from 'aws-cdk-lib/aws-ec2'; import * as ecs from 'aws-cdk-lib/aws-ecs'; import * as ecs_patterns from 'aws-cdk-lib/aws-ecs-patterns'; import * as rds from 'aws-cdk-lib/aws-rds'; export class SecureWebApp extends Construct { constructor(scope: Construct, id: string, props: SecureWebAppProps) { super(scope, id); const cluster = new ecs.Cluster(this, 'Cluster', { vpc: props.vpc }); new ecs_patterns.ApplicationLoadBalancedFargateService(this, 'FargateService', { cluster, cpu: 256, memoryLimitMiB: 512, desiredCount: 1, taskImageOptions: { image: ecs.ContainerImage.fromRegistry('amazon/amazon-ecs-sample'), containerPort: 80, }, publicLoadBalancer: true, }); new rds.DatabaseInstance(this, 'Database', { engine: rds.DatabaseInstanceEngine.mysql({ version: rds.MysqlEngineVersion.VER_8_0_28 }), instanceType: ec2.InstanceType.of(ec2.InstanceClass.T3, ec2.InstanceSize.MICRO), vpc: props.vpc, databaseName: props.databaseName, credentials: rds.Credentials.fromUsername(props.databaseUser), allocatedStorage: 20, }); } }
Notes & Best Practices
  • Expose minimal properties that vary across teams and keep defaults for everything else.
  • Document outputs such as ARNs and endpoints so consumers know what to reference.
  • Run cdk synth to inspect the resources the construct creates — check IAM and cost-sensitive items.
Key Takeaway

Constructs let you distribute infrastructure patterns like a library, improving consistency and reducing accidental misconfiguration.
