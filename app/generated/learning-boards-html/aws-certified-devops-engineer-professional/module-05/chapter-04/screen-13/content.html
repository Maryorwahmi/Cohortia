Terminal commands (sequence)
# 1) Create DLQ
aws sqs create-queue --queue-name SnsAlertDlq

# 2) Get DLQ ARN
DLQ_ARN=$(aws sqs get-queue-attributes --queue-name SnsAlertDlq --attribute-names QueueArn --query 'Attributes.QueueArn' --output text)

# 3) Create SNS topic
aws sns create-topic --name DevOpsCriticalAlerts

# 4) Subscribe HTTPS (mock PagerDuty)
aws sns subscribe --topic-arn  --protocol https --notification-endpoint  \
  --attributes "FilterPolicy={\"severity\":[\"CRITICAL\",\"HIGH\"]},RedrivePolicy={\"deadLetterTargetArn\":\"$DLQ_ARN\"}"
