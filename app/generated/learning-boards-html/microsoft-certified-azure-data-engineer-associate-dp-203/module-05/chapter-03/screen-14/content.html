Starter code (excerpt)
from pyspark.sql.functions import col, avg, to_date
from pyspark.sql.window import Window

# assume df_sensor_readings exists and ReadingDate is a date column
window_spec = Window.partitionBy('DeviceID').orderBy('ReadingDate').rowsBetween(-2, 0)
df_with_avg = df_sensor_readings.withColumn('ThreeDayMovingAvgTemp', avg('Temperature').over(window_spec))
df_with_avg.show()

# inspect partitions
print('Original partitions:', df_sensor_readings.rdd.getNumPartitions())
# repartition and coalesce
df_repart = df_sensor_readings.repartition(5)
print('After repartition:', df_repart.rdd.getNumPartitions())
df_final = df_repart.coalesce(2)
print('After coalesce:', df_final.rdd.getNumPartitions())
