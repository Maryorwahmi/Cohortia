T-SQL examples

Run these in your Synapse SQL Pool to create tables with distribution and index options.
-- Create a Hash Distributed Fact Table
CREATE TABLE FactSales_Synapse (
  SalesKey BIGINT NOT NULL,
  DateKey INT NOT NULL,
  ProductKey INT NOT NULL,
  CustomerKey INT NOT NULL,
  OrderQuantity INT NOT NULL,
  SalesAmount DECIMAL(18, 2) NOT NULL
)
WITH (
  DISTRIBUTION = HASH(ProductKey), -- Distribute by ProductKey
  CLUSTERED COLUMNSTORE INDEX
);

-- Create a Replicated Dimension Table
CREATE TABLE DimProduct_Synapse (
  ProductKey INT NOT NULL,
  ProductID NVARCHAR(50) NOT NULL,
  ProductName NVARCHAR(255) NOT NULL,
  ProductCategory NVARCHAR(100)
)
WITH (
  DISTRIBUTION = REPLICATE,
  CLUSTERED COLUMNSTORE INDEX
);

-- Create a Round Robin Staging Table
CREATE TABLE Staging_RawSales (
  RawData NVARCHAR(MAX)
)
WITH (
  DISTRIBUTION = ROUND_ROBIN,
  HEAP
);

Notes & examples

  • Hash on ProductKey when DimProduct is joined frequently on that column.
  • Replicate small dimension tables to avoid shuffle during joins.
  • Use round robin when ingesting raw data for fastest writes, then transform into distributed fact tables.
