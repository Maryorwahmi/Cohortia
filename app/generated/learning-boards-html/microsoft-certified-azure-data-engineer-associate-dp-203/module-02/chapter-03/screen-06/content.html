Module 2 › Chapter 2.3 › BULK INSERT walkthrough
Workflow steps
  1. Create staging table representing raw file fields
  2. Configure external data source and credentials for blob access
  3. Run BULK INSERT with format and batch options
  4. Validate and transform rows in staging
  5. MERGE into production within a controlled transaction
Key tip
Use FIRSTROW to skip headers and enable batch sizes to limit transaction log growth during imports.
Example — BULK INSERT + MERGE
-- Assume 'mydataengdb' is the target database
USE mydataengdb;
GO

-- Create a temporary staging table for bulk import
CREATE TABLE StagingOrders (
    OrderID INT,
    CustomerID INT,
    OrderDate DATE,
    TotalAmount DECIMAL(10, 2)
);
GO

BULK INSERT StagingOrders
FROM 'https://yourstorageaccount.blob.core.windows.net/data/orders.csv'
WITH (
    DATA_SOURCE = 'MyAzureBlobStorage',
    FORMAT = 'CSV',
    FIELDTERMINATOR = ',',
    ROWTERMINATOR = '
',
    FIRSTROW = 2
);
GO

-- After bulk insert, transform and merge
MERGE INTO FactSales AS Target
USING StagingOrders AS Source
ON Target.OrderID = Source.OrderID
WHEN MATCHED THEN
    UPDATE SET
        Target.OrderDate = Source.OrderDate,
        Target.TotalAmount = Source.TotalAmount
WHEN NOT MATCHED THEN
    INSERT (DateKey, ProductKey, CustomerKey, Quantity, UnitPrice)
    VALUES (
        (SELECT DateKey FROM DimDate WHERE FullDate = Source.OrderDate),
        (SELECT ProductKey FROM DimProduct WHERE ProductID = Source.ProductID),
        Source.CustomerID,
        Source.Quantity,
        Source.UnitPrice
    );
GO
