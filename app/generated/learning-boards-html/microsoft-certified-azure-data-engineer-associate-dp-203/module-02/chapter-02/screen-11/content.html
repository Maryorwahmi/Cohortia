Starter script
-- Use the correct database
USE mydataengdb;
GO

-- Drop tables if they exist
IF OBJECT_ID('FactSales', 'U') IS NOT NULL DROP TABLE FactSales;
IF OBJECT_ID('DimProduct', 'U') IS NOT NULL DROP TABLE DimProduct;
IF OBJECT_ID('DimDate', 'U') IS NOT NULL DROP TABLE DimDate;
GO

-- Create tables (abbreviated for the activity)
CREATE TABLE DimDate (
  DateKey INT PRIMARY KEY,
  FullDate DATE NOT NULL UNIQUE,
  Year INT NOT NULL,
  Month INT NOT NULL,
  Day INT NOT NULL
);

CREATE TABLE DimProduct (
  ProductKey INT IDENTITY(1,1) PRIMARY KEY,
  ProductName NVARCHAR(255) NOT NULL,
  Category NVARCHAR(100)
);

CREATE TABLE FactSales (
  SalesKey BIGINT IDENTITY(1,1) PRIMARY KEY,
  DateKey INT NOT NULL,
  ProductKey INT NOT NULL,
  Quantity INT NOT NULL,
  UnitPrice DECIMAL(10, 2) NOT NULL,
  TotalAmount AS (Quantity * UnitPrice),
  SaleTimestamp DATETIME DEFAULT GETDATE(),
  CONSTRAINT FK_FactSales_DimDate FOREIGN KEY (DateKey) REFERENCES DimDate(DateKey),
  CONSTRAINT FK_FactSales_DimProduct FOREIGN KEY (ProductKey) REFERENCES DimProduct(ProductKey)
);

CREATE NONCLUSTERED INDEX IX_FactSales_DateProduct ON FactSales (DateKey, ProductKey);
Verification steps
  1. Run the script in a test DB to create objects.
  2. Insert sample rows for multiple dates and products.
  3. Run SELECT joins and view execution plans to confirm index usage.
  4. Increase sample size to better surface realistic plans.
Remember: avoid running DROP statements in production. Use a safe dev database for iteration.
