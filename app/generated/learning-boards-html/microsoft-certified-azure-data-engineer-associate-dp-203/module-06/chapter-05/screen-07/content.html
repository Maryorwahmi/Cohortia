Example PySpark source read
from pyspark.sql import SparkSession
from pyspark.sql.functions import col, from_json
from pyspark.sql.types import StructType, StructField, StringType, DoubleType

# schema for incoming JSON
json_schema = StructType([
  StructField("deviceId", StringType(), True),
  StructField("temperature", DoubleType(), True),
  StructField("humidity", DoubleType(), True),
  StructField("timestamp", StringType(), True)
])

# read from Event Hubs
df = spark.readStream.format("eventhubs").options(**eh_conf).load()
parsed_df = df.withColumn("body_str", col("body").cast(StringType())).withColumn("json_data", from_json(col("body_str"), json_schema)).select(
  col("json_data.deviceId").alias("DeviceId"),
  col("json_data.temperature").alias("Temperature"),
  col("json_data.humidity").alias("Humidity"),
  col("enqueuedTime").alias("EventHubEnqueuedTime")
)
Notes & pitfalls
  • Always define an explicit JSON schema to avoid runtime parsing errors.
  • Cast the binary body to string before parsing JSON.
  • Prefer enqueuedTime or an event timestamp from the payload for windowing to maintain event-time semantics.
Check: If messages change schema, update the StructType instead of relying on inference.
