Starter script
#!/usr/bin/env python3

import paramiko
import requests
import json
import sys
import logging

# --- Configuration ---
REMOTE_HOST = 'your_remote_server_ip_or_hostname'
REMOTE_USER = 'your_ssh_user'
PRIVATE_KEY_PATH = '/home/your_user/.ssh/id_rsa'
DISK_THRESHOLD = 80
NOTIFICATION_API_URL = 'https://jsonplaceholder.typicode.com/posts'

# Configure logging
logging.basicConfig(
    filename='/var/log/remote_monitor.log',
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s'
)

def send_notification(message):
    logging.info(f"Attempting to send notification: {message}")
    payload = {
        "title": "Disk Usage Alert",
        "body": message,
        "userId": 1
    }
    try:
        response = requests.post(NOTIFICATION_API_URL, json=payload)
        response.raise_for_status()
        logging.info(f"Notification sent successfully. Response: {response.json()}")
        print(f"Notification sent: {message}")
    except requests.exceptions.RequestException as e:
        logging.error(f"Failed to send notification: {e}")
        print(f"Error sending notification: {e}", file=sys.stderr)
Operational notes
  • Replace placeholders for host, user, and key before running.
  • Run the script with controlled permissions so logs are not world-readable.
  • Test failure modes: no SSH, authentication failure, and API down.
