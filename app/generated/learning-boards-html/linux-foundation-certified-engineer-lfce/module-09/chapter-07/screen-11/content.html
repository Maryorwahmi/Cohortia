#!/usr/bin/env python3

import argparse
import subprocess
import logging
import sys

logging.basicConfig(
    filename='/var/log/disk_monitor.log',
    level=logging.INFO,
    format='%(asctime)s - %(levelname)s - %(message)s'
)

def check_disk_usage(path, threshold):
    logging.info(f"Checking disk usage for path: {path} with threshold: {threshold}%")
    try:
        result = subprocess.run(['df', '-h', path], capture_output=True, text=True, check=True)
        output_lines = result.stdout.strip().split('
')
        if len(output_lines) < 2:
            logging.error("Unexpected output from df -h")
            print(f"Error: Unexpected output from df -h for path: {path}", file=sys.stderr)
            return
        disk_line = output_lines[1]
        parts = disk_line.split()
        usage_percent = int(parts[4].strip('%'))
        if usage_percent > threshold:
            msg = f"WARNING: Disk usage for {path} is {usage_percent}% which exceeds {threshold}%!"
            logging.warning(msg)
            print(msg, file=sys.stderr)
        else:
            msg = f"INFO: Disk usage for {path} is {usage_percent}%, within threshold."
            logging.info(msg)
            print(msg)
    except FileNotFoundError:
        logging.error("Command 'df' not found.")
        print("Error: Command 'df' not found.", file=sys.stderr)
    except subprocess.CalledProcessError as e:
        logging.error(f"df -h failed: {e.stderr.strip()}")
        print(f"Error: df -h failed. {e.stderr.strip()}", file=sys.stderr)
    except Exception as e:
        logging.critical(f"Unexpected error: {e}", exc_info=True)
        print(f"Critical Error: {e}", file=sys.stderr)
Adjust the logging path or run with elevated permissions for /var/log. For testing, create the log file with permissive mode or run as a user with write access to that path.
Testing checklist
  • Run with a low threshold to trigger warning paths.
  • Simulate missing df by adjusting PATH to observe FileNotFoundError.
  • Inspect /var/log/disk_monitor.log for matching entries.
Key Takeaway

A clear, well-logged script reduces time to diagnose disk problems in production environments.
