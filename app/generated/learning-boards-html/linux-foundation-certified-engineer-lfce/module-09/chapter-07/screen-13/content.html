# Example parsing strategy
lines = result.stdout.strip().split('
')
# header may be first line; find the line matching the mount path
for line in lines[1:]:
    parts = line.split()
    if parts and parts[-1] == path:
        try:
            usage_percent = int(parts[4].strip('%'))
        except (IndexError, ValueError):
            logging.error(f"Could not parse usage from line: {line}")
            raise
        break
else:
    logging.error("No matching mount line found")
    raise ValueError("Mount path not found in df output")
