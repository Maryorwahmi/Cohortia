Cleanup script (/usr/local/bin/daily_cleanup.sh)
#!/bin/bash
LOG_FILE="/var/log/cleanup_script.log"
TARGET_DIR="/var/log/old_app_logs"

echo "$(date): Starting daily cleanup in $TARGET_DIR" >> "$LOG_FILE"

if [ ! -d "$TARGET_DIR" ]; then
    echo "$(date): Error: Target directory $TARGET_DIR does not exist." >> "$LOG_FILE"
    exit 1
fi

find "$TARGET_DIR" -type f -mtime +30 -exec rm {} \; >> "$LOG_FILE" 2>&1

if [ $? -eq 0 ]; then
    echo "$(date): Cleanup completed successfully." >> "$LOG_FILE"
else
    echo "$(date): Cleanup encountered errors." >> "$LOG_FILE"
fi
Service and Timer units
# /etc/systemd/system/daily-cleanup.service
[Unit]
Description=Daily log file cleanup service
After=network.target

[Service]
Type=oneshot
ExecStart=/usr/local/bin/daily_cleanup.sh
User=root
Group=root
StandardOutput=journal
StandardError=journal

# /etc/systemd/system/daily-cleanup.timer
[Unit]
Description=Runs daily log file cleanup

[Timer]
OnCalendar=*-*-* 03:30:00
Persistent=true
Unit=daily-cleanup.service

[Install]
WantedBy=timers.target
Verification steps
  1. chmod +x the script and create the target directory with a test file.
  2. sudo systemctl daemon-reload
  3. sudo systemctl enable --now daily-cleanup.timer
  4. Check timers: systemctl list-timers --all | grep daily-cleanup
  5. Inspect logs: journalctl -u daily-cleanup.service and cat /var/log/cleanup_script.log
