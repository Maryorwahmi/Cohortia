Starter Template (annotated)
#!/bin/bash

# --- Variables ---
THRESHOLD_GB=10
LOGFILE="system_health.log"

# --- User Input ---
read -p "Enter directory to check: " CHECK_DIR

# --- Directory Check ---
if [ ! -d "$CHECK_DIR" ]; then
  echo "Directory does not exist: $CHECK_DIR" | tee -a "$LOGFILE"
  exit 1
fi

# --- Health Report ---
echo "--- System Health Report ---" | tee -a "$LOGFILE"
echo "Date: $(date)" | tee -a "$LOGFILE"
echo "Uptime: $(uptime -p)" | tee -a "$LOGFILE"

echo -e "
--- Disk Usage for $CHECK_DIR ---" | tee -a "$LOGFILE"
df -h "$CHECK_DIR" | awk 'NR==2 {print "Total: "$2", Used: "$3", Free: "$4"}' | tee -a "$LOGFILE"

echo -e "
--- Memory Usage ---" | tee -a "$LOGFILE"
free -h | tee -a "$LOGFILE"

echo -e "
--- Top 5 CPU Processes ---" | tee -a "$LOGFILE"
ps aux --sort=-%cpu | head -n 6 | tee -a "$LOGFILE"
