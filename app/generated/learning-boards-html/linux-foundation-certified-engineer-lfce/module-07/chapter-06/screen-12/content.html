Sudoers lines to add (via visudo)
# 2. Demo full admin for alice (use with caution)
alice ALL=(ALL) NOPASSWD: ALL

# 3. bob restricted to nginx management
Cmnd_Alias NGINX_MGMT = /usr/sbin/systemctl restart nginx, /usr/sbin/systemctl status nginx, /usr/bin/tail -f /var/log/nginx/access.log
bob ALL=(root) NGINX_MGMT

# 4. app_monitor service account with NOPASSWD for app service
Cmnd_Alias APP_MGMT = /usr/bin/systemctl restart appservice, /usr/bin/systemctl status appservice
app_monitor ALL=(root) NOPASSWD: APP_MGMT

# 5. Enable sudo logging
Defaults    logfile=/var/log/sudo.log
Defaults    log_input
Defaults    log_output
Testing commands
# Test alice (should not ask)
sudo -u alice sudo -i

# Test bob (should ask for password)
sudo -u bob sudo systemctl status nginx

# Test app_monitor (no password)
sudo -u app_monitor sudo systemctl restart appservice

# Check logs
sudo tail /var/log/sudo.log
Note: The alice NOPASSWD: ALL line is for demonstration only. Replace with a narrow alias for real environments.
