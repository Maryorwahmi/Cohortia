Starter commands
# Step 1: Generate key (local)
ssh-keygen -t ed25519 -C "prod_server_key"

# Step 2: Deploy public key
ssh-copy-id -i ~/.ssh/id_ed25519.pub testuser@remote_server_ip

# Step 3: Backup and edit sshd_config (remote)
sudo cp /etc/ssh/sshd_config /etc/ssh/sshd_config.bak
# Edit: Port 2222, PermitRootLogin no, PasswordAuthentication no, PubkeyAuthentication yes, AllowUsers testuser adminuser
sudo sshd -t
Restricted user example
# Create restricted account and script (remote)
sudo useradd -m git_sync
sudo install -m 755 -o root -g root -d /usr/local/bin
echo '#!/bin/bash' | sudo tee /usr/local/bin/git-shell
echo 'echo "This account is for Git access only. Interactive shell is disabled."' | sudo tee -a /usr/local/bin/git-shell
echo 'exit 1' | sudo tee -a /usr/local/bin/git-shell
sudo vi /etc/ssh/sshd_config
# Add at end:
# Match User git_sync
#   ForceCommand /usr/local/bin/git-shell
#   PermitTunnel no
#   AllowAgentForwarding no
#   X11Forwarding no
#   PermitOpen none
sudo sshd -t && sudo systemctl restart sshd
