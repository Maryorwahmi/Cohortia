XFS steps
# format and mount
sudo mkfs.xfs /dev/vg_xfs/lv_data
sudo mkdir -p /mnt/xfs_data
sudo mount /dev/vg_xfs/lv_data /mnt/xfs_data
# add test data
sudo dd if=/dev/urandom of=/mnt/xfs_data/large_file bs=1M count=500

# extend LV and grow filesystem
sudo lvextend -L +1G /dev/vg_xfs/lv_data
sudo xfs_growfs /mnt/xfs_data
# verify
df -h /mnt/xfs_data
Btrfs steps
# format and mount
sudo mkfs.btrfs /dev/vg_btrfs/lv_data
sudo mkdir -p /mnt/btrfs_root
sudo mount /dev/vg_btrfs/lv_data /mnt/btrfs_root

# create subvolume and mount
sudo btrfs subvolume create /mnt/btrfs_root/home_users
sudo mkdir -p /mnt/home_users_mount
sudo mount -o subvol=home_users /dev/vg_btrfs/lv_data /mnt/home_users_mount

echo "User data" | sudo tee /mnt/home_users_mount/user_file.txt

# snapshot and verify
sudo btrfs subvolume snapshot -r /mnt/btrfs_root/home_users /mnt/btrfs_root/home_users_snap_readonly
sudo btrfs subvolume list /mnt/btrfs_root

# extend and resize
sudo lvextend -L +1G /dev/vg_btrfs/lv_data
sudo btrfs filesystem resize max /mnt/btrfs_root
# verify
df -h /mnt/btrfs_root
