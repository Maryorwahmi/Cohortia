Build and Scan a Secure Non-Root Nginx Image

Starter Dockerfile (lab)
# Dockerfile
FROM alpine:latest
RUN apk add --no-cache nginx && \
    adduser -D -g 'nginx' nginx && \
    mkdir -p /var/cache/nginx /var/run/nginx && \
    chown -R nginx:nginx /var/cache/nginx /var/run/nginx /var/log/nginx

COPY default.conf /etc/nginx/conf.d/default.conf
RUN echo "<h1>Hello Secure Nginx!</h1>" > /usr/share/nginx/html/index.html

USER nginx
EXPOSE 80
CMD ["nginx","-g","daemon off;"]
Run and verify
docker build -t secure-nginx:latest .
docker run -d --name secure-web -p 8080:80 --read-only --cap-drop ALL --cap-add NET_BIND_SERVICE secure-nginx:latest
docker exec -it secure-web whoami # expect nginx
Attempt a write inside the container and confirm permission denied to demonstrate read-only root.
Scan
trivy image secure-nginx:latest
