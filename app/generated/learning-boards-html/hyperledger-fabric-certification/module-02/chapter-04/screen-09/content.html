CreateAsset code (excerpt)
CreateAsset validates argument count, builds the asset struct, marshals to JSON, and calls PutState. Errors are returned as shim.Error to signal failed simulation.
// CreateAsset creates a new asset
func (t *SimpleChaincode) CreateAsset(stub shim.ChaincodeStubInterface, args []string) peer.Response {
  if len(args) != 5 {
    return shim.Error("Incorrect number of arguments. Expecting 5 for CreateAsset.")
  }

  asset := Asset{
    ID:    args[0],
    Color: args[1],
    Size:  json.Number(args[2]).Int64(),
    Owner: args[3],
    Value: json.Number(args[4]).Int64(),
  }

  assetAsBytes, err := json.Marshal(asset)
  if err != nil {
    return shim.Error(fmt.Sprintf("Failed to marshal asset: %s", err))
  }

  err = stub.PutState(asset.ID, assetAsBytes)
  if err != nil {
    return shim.Error(fmt.Sprintf("Failed to create asset: %s", err))
  }

  return shim.Success(assetAsBytes)
}
ReadAsset code (excerpt)
// ReadAsset reads an asset from the ledger
func (t *SimpleChaincode) ReadAsset(stub shim.ChaincodeStubInterface, args []string) peer.Response {
  if len(args) != 1 {
    return shim.Error("Incorrect number of arguments. Expecting 1 for ReadAsset.")
  }

  assetID := args[0]
  assetAsBytes, err := stub.GetState(assetID)
  if err != nil {
    return shim.Error(fmt.Sprintf("Failed to read asset: %s", err))
  }
  if assetAsBytes == nil {
    return shim.Error(fmt.Sprintf("Asset with ID %s does not exist", assetID))
  }

  return shim.Success(assetAsBytes)
}
Key Takeaway

Always validate args and conversions, marshal reliably, and return explicit errors so endorsers produce the same read/write set.
