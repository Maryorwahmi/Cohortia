Unit test example
func TestCreateAsset_Authorized(t *testing.T) {
  scc := new(SmartContract)
  mockStub := shim.NewMockStub("mycc", scc)
  mockStub.SetCreator(getTestCreator("Org1MSP"))

  response := mockStub.MockInvoke("1", [][]byte{
    []byte("CreateAsset"), []byte("asset1"), []byte("blue"), []byte("10"), []byte("Alice"),
  })
  assert.Equal(t, int32(peer.Response_OK), response.Status)
}

func TestCreateAsset_Unauthorized(t *testing.T) {
  scc := new(SmartContract)
  mockStub := shim.NewMockStub("mycc", scc)
  mockStub.SetCreator(getTestCreator("Org2MSP"))

  response := mockStub.MockInvoke("1", [][]byte{
    []byte("CreateAsset"), []byte("asset2"), []byte("red"), []byte("20"), []byte("Bob"),
  })
  assert.Equal(t, int32(peer.Response_ERROR), response.Status)
  assert.Contains(t, response.Message, "client not authorized")
}
Test structure
  • Arrange: instantiate contract and mock stub.
  • Act: set creator and invoke the function.
  • Assert: check status and messages for expected outcomes.
Developer tip
Keep helper functions for creating mock creators so tests are readable and focused on behavior.
