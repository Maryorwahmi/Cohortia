Create sample sales_transactions

CREATE TABLE sales_transactions (
  transaction_id SERIAL PRIMARY KEY,
  product_id INT,
  customer_id INT,
  transaction_date DATE DEFAULT CURRENT_DATE,
  amount NUMERIC(10,2),
  region TEXT
);

-- Insert 500,000 rows (example script)
INSERT INTO sales_transactions (product_id, customer_id, transaction_date, amount, region)
SELECT
  (random() * 1000)::int + 1,
  (random() * 50000)::int + 1,
  (CURRENT_DATE - (random() * 365 * 5)::int)::DATE,
  (random() * 1000)::numeric(10,2) + 1,
  CASE (random() * 4)::int
    WHEN 0 THEN 'North'
    WHEN 1 THEN 'South'
    WHEN 2 THEN 'East'
    WHEN 3 THEN 'West'
    ELSE 'Central' END
FROM generate_series(1, 500000);

ANALYZE sales_transactions;
