Terminal Demo
Follow the three-terminal setup and reproduce CPU and I/O workloads safely on a lab instance.
# Terminal 1: htop
# Terminal 2: iostat -x 1
# Terminal 3: psql
-- Create CPU function and run it
CREATE OR REPLACE FUNCTION cpu_intensive_task() RETURNS VOID AS $$
DECLARE i INT := 0; result NUMERIC := 0;
BEGIN FOR i IN 1..10000000 LOOP result := result + sqrt(i) / (i + 1); END LOOP; RAISE NOTICE 'done'; END; $$ LANGUAGE plpgsql;
SELECT cpu_intensive_task();

-- Create io_test table and insert many rows to stress disk
CREATE TABLE io_test (id SERIAL PRIMARY KEY, data TEXT DEFAULT repeat('A',1000));
INSERT INTO io_test (data) SELECT repeat(md5(random()::text),10) FROM generate_series(1,100000);
Observation checklist
  • During CPU test, htop shows high %user on postgres processes; pg_stat_activity shows the function running.
  • During I/O test, iostat shows %util and await spikes; vmstat shows if swapping occurs.
  • Use pg_stat_statements to find expensive queries after the test completes.
Cleanup

DROP FUNCTION cpu_intensive_task(); DROP TABLE io_test; Always run labs on a non-production instance.
