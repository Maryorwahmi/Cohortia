-- Step 2: Create the customer_data table
CREATE TABLE customer_data (
    customer_id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    csr_id VARCHAR(50) NOT NULL
);

-- Insert sample data
INSERT INTO customer_data (name, csr_id) VALUES
('John Doe', 'csr_alice'),
('Jane Smith', 'csr_bob');

-- Step 4: Create CSR user roles
CREATE ROLE csr_alice WITH LOGIN PASSWORD 'AliceSecurePass!';
CREATE ROLE csr_bob WITH LOGIN PASSWORD 'BobSecurePass!';

GRANT SELECT, INSERT, UPDATE, DELETE ON customer_data TO csr_alice;
GRANT SELECT, INSERT, UPDATE, DELETE ON customer_data TO csr_bob;

-- Enable RLS
ALTER TABLE customer_data ENABLE ROW LEVEL SECURITY;

-- Create RLS policy
CREATE POLICY csr_access_policy ON customer_data
FOR ALL
TO csr_alice, csr_bob
USING (csr_id = current_user)
WITH CHECK (csr_id = current_user);

-- Optionally force RLS
ALTER TABLE customer_data FORCE ROW LEVEL SECURITY;
Expected tests & outcomes
- Connect as csr_alice and run SELECT; only csr_alice rows should appear.
- INSERT with csr_id = 'csr_alice' should succeed; INSERT with csr_id = 'csr_bob' should fail due to WITH CHECK.
- Attempt to UPDATE csr_id to another user; the update should fail if it violates WITH CHECK.
- Connect as a role with BYPASSRLS (if granted) and observe full visibility, then remove BYPASSRLS and re-test.
Key Takeaway

Hands-on tests that include failing writes are the most reliable way to validate RLS behavior.
