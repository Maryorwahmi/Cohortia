Task 1: UPSERT
-- Insert new product SKU003
INSERT INTO products (sku, name, price, stock_quantity)
VALUES ('SKU003', 'Gaming Headset', 75.00, 200);

-- Upsert SKU001 to update price and stock on conflict
INSERT INTO products (sku, name, price, stock_quantity)
VALUES ('SKU001', 'Wireless Mouse', 28.00, 160)
ON CONFLICT (sku) DO UPDATE
SET price = EXCLUDED.price,
    stock_quantity = EXCLUDED.stock_quantity;
Task 2: UPDATE and RETURNING
UPDATE products
SET stock_quantity = 80
WHERE sku = 'SKU002'
RETURNING product_id, name, stock_quantity AS new_stock;
Task 3: Data-modifying CTE for audit
WITH old AS (
  SELECT product_id, price
  FROM products
  WHERE sku = 'SKU003'
),
upd AS (
  UPDATE products
  SET price = 79.00
  WHERE sku = 'SKU003'
  RETURNING product_id, price AS new_price
)
INSERT INTO product_audit_log (product_id, action_type, old_price, new_price)
SELECT o.product_id, 'PRICE_CHANGE', o.price, u.new_price
FROM old o
JOIN upd u USING (product_id);
