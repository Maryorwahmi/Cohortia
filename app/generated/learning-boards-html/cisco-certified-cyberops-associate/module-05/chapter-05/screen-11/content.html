# Import the regular expression module
import re
import os

def analyze_apache_log(log_file_path):
    """
    Analyzes an Apache access log file for suspicious patterns.
    """
    suspicious_patterns = [
        r"union select",      # Common SQL Injection payload
        r"etc/passwd",        # Path Traversal attempt
        r"eval\(",            # PHP Code Execution attempt
        r"wp-admin",          # Common WordPress admin panel access
        r"(\.\./){2,}",       # Multiple directory traversals
        r"select.+from",      # Generic SQL query
        r"xss",               # Cross-Site Scripting attempt
    ]
    
    # Compile regex patterns for efficiency
    compiled_patterns = [(re.compile(pattern, re.IGNORECASE), pattern) for pattern in suspicious_patterns]
    
    found_incidents = []
    if not os.path.exists(log_file_path):
        print(f"Error: Log file not found at {log_file_path}")
        return []

    print(f"Analyzing log file: {log_file_path}")
    try:
        with open(log_file_path, 'r', errors='ignore') as f:
            for line_num, line in enumerate(f, 1):
                for compiled_pattern, original_pattern in compiled_patterns:
                    if compiled_pattern.search(line):
                        found_incidents.append({
                            "line_num": line_num,
                            "log_entry": line.strip(),
                            "matched_pattern": original_pattern
                        })
                        break 
        
        if found_incidents:
            print("
--- Suspicious Activities Found ---")
            for incident in found_incidents:
                print(f"Line {incident['line_num']}: {incident['log_entry']}")
                print(f"  Matched Pattern: '{incident['matched_pattern']}'
")
        else:
            print("No suspicious activities found.")
            
    except Exception as e:
        print(f"An error occurred during log analysis: {e}")
        
    return found_incidents

The `re` Module

Python's built-in module for regular expression operations. It provides functions for pattern matching, searching, and manipulation of strings, crucial for parsing complex text like log files.

Suspicious Patterns List

A collection of regular expressions, each designed to detect a specific type of malicious activity or anomaly within log entries. Examples include SQL injection, path traversal, and code execution attempts.

Compiling for Efficiency

Regular expressions are compiled into pattern objects using re.compile(). This pre-processing step significantly improves performance, especially when the same patterns are used repeatedly against many lines of data, as in log analysis.

Key Takeaway

Defining and compiling regular expressions is the foundation for efficiently identifying known malicious patterns within large log datasets, a critical step in automated incident detection.
