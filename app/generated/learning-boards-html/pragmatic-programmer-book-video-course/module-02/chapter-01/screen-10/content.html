app.py (after refactor)
from flask import Flask, request, jsonify
from validators import is_valid_email, is_valid_password

app = Flask(__name__)

@app.route('/register', methods=['POST'])
def register_user():
  data = request.get_json()
  email = data.get('email')
  password = data.get('password')

  if not is_valid_email(email):
    return jsonify({"error": "Invalid email format"}), 400

  if not is_valid_password(password):
    return jsonify({"error": "Password must be at least 8 characters and contain a digit"}), 400

  print(f"Registering user: {email}")
  return jsonify({"message": "User registered successfully"}), 201

@app.route('/update_profile', methods=['POST'])
def update_profile():
  data = request.get_json()
  email = data.get('email')
  password = data.get('password')

  if email and not is_valid_email(email):
    return jsonify({"error": "Invalid email format"}), 400

  if password and not is_valid_password(password):
    return jsonify({"error": "Password must be at least 8 characters and contain a digit"}), 400

  print(f"Updating profile for: {email if email else 'current user'}")
  return jsonify({"message": "Profile updated successfully"}), 200

if __name__ == '__main__':
  app.run(debug=True)
Testing checklist
  • Run unit tests for validators.
  • Run integration tests for register and update routes.
  • Verify error messages and status codes are unchanged.
Common pitfall
Forgetting to change all callers or leaving slightly different implementations in other places will defeat the refactor. Use search tools to find duplicates.
