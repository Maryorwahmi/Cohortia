Interfaces
Define contracts
Use language abstractions — in Python, abstract base classes — to describe required behavior without implementation details.
interfaces.py (visual)
import abc

class IDataLoader(abc.ABC):
  @abc.abstractmethod
  def load_data(self, source: str) -> list:
    pass

class IReporter(abc.ABC):
  @abc.abstractmethod
  def generate_report(self, data: list):
    pass
main.py (refactored)
from interfaces import IDataLoader, IReporter

class CSVDataLoader(IDataLoader):
  def load_data(self, filepath: str) -> list:
    print(f"Loading data from CSV: {filepath}")
    return [{"id": 1, "name": "Alice"}, {"id": 2, "name": "Bob"}]

class ConsoleReporter(IReporter):
  def generate_report(self, data: list):
    print("--- Console Report ---")
    for item in data:
      print(f"ID: {item['id']}, Name: {item['name']}")
    print("----------------------")

class ReportGenerator:
  def __init__(self, data_loader: IDataLoader, reporter: IReporter):
    self.data_loader = data_loader
    self.reporter = reporter

  def generate(self, source_path: str):
    data = self.data_loader.load_data(source_path)
    self.reporter.generate_report(data)
Composition
Create concrete instances at program start and pass them into ReportGenerator. The orchestration never constructs concrete implementations itself.
Verification
Write a unit test that constructs ReportGenerator with a fake loader returning deterministic data to validate orchestration behavior.
