Process Flow Unit Tests (Mock)
Guidelines
  • Mock external repository methods only.
  • Test validation and error flows deterministically.
  • Assert only essential interactions to avoid brittle tests.
Example (pytest + unittest.mock)
from unittest.mock import Mock
import pytest

# Assume ProductService and Product classes are imported

def test_create_product_valid_calls_repo_add():
  mock_repo = Mock()
  service = ProductService(repository=mock_repo)
  product = service.create_product('p1', 'Widget', 9.99)
  mock_repo.add.assert_called_once()
  assert product.id == 'p1'

def test_create_product_invalid_price_raises():
  mock_repo = Mock()
  service = ProductService(repository=mock_repo)
  with pytest.raises(ValueError):
    service.create_product('p2', 'Gadget', 0)
