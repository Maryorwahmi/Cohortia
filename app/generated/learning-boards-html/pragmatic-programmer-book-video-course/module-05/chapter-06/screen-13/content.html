Starter files
# my_app.py
def add(a, b):
    return a + b

def subtract(a, b):
    return a - b

# test_my_app.py
import pytest
from my_app import add, subtract

def test_add():
    assert add(1, 2) == 3
    assert add(0, 0) == 0
    assert add(-1, 1) == 0

def test_subtract():
    assert subtract(5, 2) == 3
    assert subtract(10, 10) == 0
    assert subtract(0, 5) == -5

# requirements.txt
# pytest==7.4.0
# pytest-cov==4.1.0
Example Makefile
.PHONY: all test clean

all: test

test:
	pip install -r requirements.txt
	pytest --cov=my_app

clean:
	find . -type d -name "__pycache__" -exec rm -rf {} +
	rm -f .pytest_cache .coverage
Run make test to execute dependency install and tests. Run make clean to remove artifacts. In CI, call make test to reproduce the same steps automatically.
