Code Walkthrough (example)
// index.js
function calculateDiscountedPrice(price, discountPercentage) {
  if (discountPercentage > 100) {
    console.log("Discount cannot be more than 100%");
    return price;
  }
  const discountAmount = price * (discountPercentage / 100);
  return price - discountAmount;
}

let productPrice = 200;
let discount = 10; // This should be 15 for a special promotion

console.log("Original price:", productPrice);
console.log("Discounted price:", calculateDiscountedPrice(productPrice, discount));

// A variable name that needs refactoring
let disc = 5;
console.log("Another discounted price:", calculateDiscountedPrice(productPrice, disc));
Try using the editor's Rename Symbol on 'disc' to change it to 'additionalDiscount' and observe all references update automatically.
Refactor Tips
  • Always preview changes before applying.
  • Run tests or execute a small script after refactor.
  • Prefer small, frequent refactors over large rewrites.
Key Takeaway

Automated refactors reduce error-prone manual edits, but always validate them with previews and tests.
