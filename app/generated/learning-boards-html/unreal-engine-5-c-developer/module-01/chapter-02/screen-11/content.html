Code example
#include <iostream>

int main()
{
  // Declare and initialize
  float num1 = 0.0f;
  float num2 = 0.0f;

  std::cout << "Enter the first number: ";
  std::cin >> num1;

  std::cout << "Enter the second number: ";
  std::cin >> num2;

  float sum = num1 + num2;
  float difference = num1 - num2;
  float product = num1 * num2;
  float quotient = 0.0f;
  if (num2 != 0.0f) {
      quotient = num1 / num2;
  } else {
      std::cout << "Warning: Division by zero is not allowed." << std::endl;
  }

  // Remainder: cast to int for modulo
  int intNum1 = static_cast(num1);
  int intNum2 = static_cast(num2);
  if (intNum2 != 0) {
      int remainder = intNum1 % intNum2;
      std::cout << "Remainder: " << remainder << std::endl;
  }

  std::cout << "Sum: " << sum << std::endl;
  std::cout << "Quotient: " << quotient << std::endl;

  return 0;
}
Explanation
  • Variables are declared with initial values to ensure safe starting state.
  • std::cin reads input; be aware it stops at whitespace for strings.
  • Division guarded with a zero check prevents runtime warnings or incorrect results.
  • Modulo requires integer operands; cast floats when appropriate and understand truncation behavior.
