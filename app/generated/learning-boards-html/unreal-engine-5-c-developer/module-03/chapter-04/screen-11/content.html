TryInteract example
// MyPlayerCharacter.cpp - simplified
void AMyPlayerCharacter::TryInteract()
{
  FVector Start = GetActorLocation();
  FVector End = Start + GetActorForwardVector() * 200.0f;

  FHitResult Hit;
  FCollisionQueryParams Params;
  Params.AddIgnoredActor(this);

  if (GetWorld()->LineTraceSingleByChannel(Hit, Start, End, ECC_Visibility, Params))
  {
    AActor* HitActor = Hit.GetActor();
    if (HitActor && HitActor->Implements<UInteractable>())
    {
      IInteractable::Execute_OnInteract(HitActor, this);
    }
    else
    {
      UE_LOG(LogTemp, Warning, TEXT("Hit non-interactable"));
    }
  }
  else
  {
    UE_LOG(LogTemp, Warning, TEXT("Nothing to interact with"));
  }
}
Test checklist
  • Place ADoor and APickupItem in level and set bIsLocked where relevant.
  • Ensure player pawn is AMyPlayerCharacter and bind TryInteract to the input.
  • Observe on-screen debug messages or logs when interacting.
Common mistake

Forgetting Params.AddIgnoredActor(this) can cause self hits; include it to avoid false positives.
