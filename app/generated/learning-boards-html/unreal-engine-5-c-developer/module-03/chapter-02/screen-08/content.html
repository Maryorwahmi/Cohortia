AWeapon (header)
// Weapon.h (excerpt)
UCLASS()
class MYPROJECT_API AWeapon : public AActor
{
  GENERATED_BODY()
public:
  UFUNCTION(BlueprintCallable, Category = "Weapon")
  virtual void Fire();
};
ARifle override
// Rifle.h (excerpt)
UCLASS()
class MYPROJECT_API ARifle : public AWeapon
{
  GENERATED_BODY()
public:
  UFUNCTION(BlueprintCallable, Category = "Weapon")
  virtual void Fire() override;
};
WeaponManager iteration
// WeaponManager.cpp (excerpt)
for (AWeapon* Weapon : WeaponsInInventory)
{
  if (Weapon)
  {
    Weapon->Fire(); // Calls Rifle or Pistol Fire() at runtime
  }
}
Use this pattern to keep managers generic and make adding new weapon types require only a new derived class with Fire() implemented.
