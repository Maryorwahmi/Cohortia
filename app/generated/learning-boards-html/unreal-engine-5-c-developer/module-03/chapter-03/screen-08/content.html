// MyDoor.cpp
#include "MyDoor.h"
#include "Engine/Engine.h"
#include "Kismet/KismetMathLibrary.h"

AMyDoor::AMyDoor()
{
    PrimaryActorTick.bCanEverTick = true;
    bIsOpen = false;
    DoorOpenDuration = 1.5f;
    CurrentOpenAlpha = 0.0f;
}

void AMyDoor::BeginPlay()
{
    Super::BeginPlay();
    if (GEngine) GEngine->AddOnScreenDebugMessage(-1, 5.f, FColor::Yellow, TEXT("Door initialized."));
}

void AMyDoor::Tick(float DeltaTime)
{
    Super::Tick(DeltaTime);
    UpdateDoorPosition(DeltaTime);
}

void AMyDoor::UpdateDoorPosition(float DeltaTime)
{
    float TargetAlpha = bIsOpen ? 1.0f : 0.0f;
    CurrentOpenAlpha = FMath::FInterpTo(CurrentOpenAlpha, TargetAlpha, DeltaTime, 1.0f / DoorOpenDuration);
}

void AMyDoor::OpenDoor()
{
    if (!bIsOpen) { bIsOpen = true; }
}

void AMyDoor::CloseDoor()
{
    if (bIsOpen) { bIsOpen = false; }
}

bool AMyDoor::IsDoorOpen() const
{
    return bIsOpen;
}
What to notice
  • Constructor sets defaults so the actor behaves predictably in editor and during play.
  • Tick calls a single UpdateDoorPosition to keep animation logic centralized.
  • OpenDoor/CloseDoor flip state only through controlled methods.
Best practice: centralize state mutation so any rule, like clamping or triggering effects, always runs in one place.
Key Takeaway

Keep mutation and update logic together and expose only intended actions to external systems.
