Top-down: world-relative movement
Modified Move
void AMyPlayerPawn::Move(const FInputActionValue& Value)
{
  FVector2D MovementVector = Value.Get<FVector2D>();

  if (Controller && (MovementVector.X != 0.0f || MovementVector.Y != 0.0f))
  {
    const FVector WorldForward = FVector::ForwardVector; // (1,0,0)
    const FVector WorldRight   = FVector::RightVector;   // (0,1,0)

    AddMovementInput(WorldForward, MovementVector.Y * MovementSpeed * GetWorld()->GetDeltaSeconds());
    AddMovementInput(WorldRight,   MovementVector.X * MovementSpeed * GetWorld()->GetDeltaSeconds());
  }
}
Why use it
  • Consistent controls in a fixed-camera top-down view.
  • Movement unaffected by camera rotation.
  • Useful when the camera is canonical, like maps or strategy views.
