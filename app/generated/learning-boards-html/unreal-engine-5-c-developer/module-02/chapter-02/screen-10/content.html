MyFirstActor.cpp (excerpt)
#include "MyFirstActor.h"
#include "Engine/Engine.h"

AMyFirstActor::AMyFirstActor()
{
  PrimaryActorTick.bCanEverTick = true;
  MyFloatValue = 100.0f;
}

void AMyFirstActor::BeginPlay()
{
  Super::BeginPlay();
  if (GEngine)
  {
    GEngine->AddOnScreenDebugMessage(-1, 5.f, FColor::Green, TEXT("MyFirstActor has begun play!"));
  }
  UE_LOG(LogTemp, Warning, TEXT("MyFirstActor has begun play from C++ log!"));
}

void AMyFirstActor::Tick(float DeltaTime)
{
  Super::Tick(DeltaTime);
  // per-frame logic
}

void AMyFirstActor::PrintMyMessage()
{
  if (GEngine)
  {
    GEngine->AddOnScreenDebugMessage(-1, 5.f, FColor::Yellow, FString::Printf(TEXT("MyFloatValue is: %f"), MyFloatValue));
  }
  UE_LOG(LogTemp, Log, TEXT("PrintMyMessage called. MyFloatValue: %f"), MyFloatValue);
}
Explanation and Tips
  • Constructor sets defaults and tick behavior before BeginPlay runs.
  • BeginPlay is ideal for runtime initialization and one-time debug messages.
  • Use Tick sparingly; prefer events for infrequent updates.
  • Use GEngine->AddOnScreenDebugMessage plus UE_LOG to validate both on-screen and log output.
