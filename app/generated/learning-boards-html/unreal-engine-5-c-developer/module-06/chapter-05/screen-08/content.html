// MyUtilityPlugin.Build.cs
using UnrealBuildTool;

public class MyUtilityPlugin : ModuleRules
{
  public MyUtilityPlugin(ReadOnlyTargetRules Target) : base(Target)
  {
    PCHUsage = ModuleRules.PCHUsageMode.UseExplicitOrSharedPCHs;

    PublicDependencyModuleNames.AddRange(
      new string[]
      {
        "Core",
        "CoreUObject",
        "Engine"
      }
    );

    PrivateDependencyModuleNames.AddRange(new string[] { });
  }
}
Explanation
  • PublicDependencyModuleNames: modules your public API needs.
  • PrivateDependencyModuleNames: internal build-time dependencies.
  • Keep dependencies minimal to reduce coupling.
Key Takeaway
Accurately declare Core, CoreUObject, Engine as needed; missing declarations cause compile errors, extra ones increase coupling.
