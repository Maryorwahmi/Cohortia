// MyHealthComponent.cpp (demo implementation)
#include "MyHealthComponent.h"

UMyHealthComponent::UMyHealthComponent()
{
    // Initialize HealthData with a default MaxHealth
    HealthData = MakeShareable(new FHealthData(100.0f));
}

void UMyHealthComponent::BeginPlay()
{
    Super::BeginPlay();
}

void UMyHealthComponent::ApplyDamage(float DamageAmount)
{
    if (!HealthData.IsValid()) return;
    float Old = HealthData->CurrentHealth;
    HealthData->CurrentHealth = FMath::Max(0.0f, HealthData->CurrentHealth - DamageAmount);
    float Delta = HealthData->CurrentHealth - Old;
    OnHealthChanged.Broadcast(HealthData->CurrentHealth, Delta);
}

void UMyHealthComponent::ApplyHealing(float HealAmount)
{
    if (!HealthData.IsValid()) return;
    float Old = HealthData->CurrentHealth;
    HealthData->CurrentHealth = FMath::Min(HealthData->MaxHealth, HealthData->CurrentHealth + HealAmount);
    float Delta = HealthData->CurrentHealth - Old;
    OnHealthChanged.Broadcast(HealthData->CurrentHealth, Delta);
}

float UMyHealthComponent::GetCurrentHealth() const
{
    return HealthData.IsValid() ? HealthData->CurrentHealth : 0.0f;
}
Note: This code demonstrates the intended behavior. In your exercise, ensure you validate pointers before dereferencing and clamp values to avoid invalid state.
