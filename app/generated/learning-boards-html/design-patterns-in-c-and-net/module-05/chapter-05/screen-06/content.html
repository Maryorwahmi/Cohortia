using System;using System.Threading;using System.Threading.Tasks;public class ThreadSafeCounter{    private int _count = 0;    private readonly object _lockObject = new object(); // A private, readonly object for locking    public void Increment()    {        lock (_lockObject) // Acquire the lock        {            // This is the critical section            // Only one thread can be inside this block at a time            _count++;            Console.WriteLine($"Thread {Thread.CurrentThread.ManagedThreadId}: Count = {_count}");        } // Release the lock automatically when exiting the block    }    public int GetCount()    {        // Reading shared state also needs protection if it can be modified concurrently        lock (_lockObject)        {            return _count;        }    }}public class LockDemo{    public static async Task Main(string[] args)    {        ThreadSafeCounter counter = new ThreadSafeCounter();        List<Task> tasks = new List<Task>();        Console.WriteLine("Starting concurrent increments...");        for (int i = 0; i < 10; i++)        {            tasks.Add(Task.Run(() =>            {                for (int j = 0; j < 1000; j++)                {                    counter.Increment();                }            }));        }        await Task.WhenAll(tasks); // Wait for all tasks to complete        Console.WriteLine($"
Final Count: {counter.GetCount()} (Expected: 10000)");    }}
Code Breakdown
  • _lockObject: A dedicated, private object used as the lock target.
  • lock (_lockObject): Ensures exclusive access to the code block.
  • _count++: The critical section where shared state is modified.
  • Automatic Release: The lock is released when the block is exited, even on exceptions.
Why it matters

Without the lock, multiple threads could read, increment, and write _count concurrently, leading to an incorrect final value due to lost updates.

Key Takeaway

The lock keyword provides a simple and effective way to protect shared mutable state within a single process, ensuring data consistency in concurrent operations.
