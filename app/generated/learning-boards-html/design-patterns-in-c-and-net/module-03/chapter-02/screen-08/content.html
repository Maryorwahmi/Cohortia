CoffeeDecorator & Concrete Decorators
public abstract class CoffeeDecorator : ICoffee
{
  protected ICoffee _decoratedCoffee;

  public CoffeeDecorator(ICoffee coffee)
  {
    _decoratedCoffee = coffee;
  }

  public virtual string GetDescription()
  {
    return _decoratedCoffee.GetDescription();
  }

  public virtual double GetCost()
  {
    return _decoratedCoffee.GetCost();
  }
}

public class MilkDecorator : CoffeeDecorator
{
  public MilkDecorator(ICoffee coffee) : base(coffee) { }

  public override string GetDescription()
  {
    return _decoratedCoffee.GetDescription() + ", with Milk";
  }

  public override double GetCost()
  {
    return _decoratedCoffee.GetCost() + 1.5;
  }
}
Testing the chain
Compose: var coffee = new SugarDecorator(new MilkDecorator(new SimpleCoffee())); Verify the description includes both toppings and the cost is the sum of base and toppings.
Common pitfall
Avoid decorators that perform unrelated responsibilities or that change return semantics expected by the interface; keep each decorator focused and predictable.
