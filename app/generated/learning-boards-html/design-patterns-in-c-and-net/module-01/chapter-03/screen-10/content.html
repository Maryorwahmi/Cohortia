Refactor outline
public interface ILogger { void LogError(string message); }
public class SqlDatabaseLogger : ILogger { /* ... */ }
public class OrderProcessor
{
  private readonly ILogger _logger;
  public OrderProcessor(ILogger logger) { _logger = logger; }
}

// At composition root:
ILogger logger = new SqlDatabaseLogger();
var processor = new OrderProcessor(logger);
This refactor enables unit tests to inject a mock ILogger and production code to choose a real implementation at startup. It isolates responsibility for object creation to the composition root.
