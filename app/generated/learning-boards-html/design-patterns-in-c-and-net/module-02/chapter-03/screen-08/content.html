Client code
The client composes with the abstract factory and uses only abstract product members.
// Client: Uses the Abstract Factory and Abstract Products.
public class Application
{
    private IGUIFactory _factory;
    private IButton _button;
    private ICheckbox _checkbox;

    public Application(IGUIFactory factory)
    {
        _factory = factory;
        _button = _factory.CreateButton();
        _checkbox = _factory.CreateCheckbox();
    }

    public void Paint()
    {
        Console.WriteLine(_button.Render());
        Console.WriteLine(_checkbox.Render());
        Console.WriteLine($"Is checkbox checked? {_checkbox.IsChecked()}");
    }
}

public class Program
{
    public static void Main(string[] args)
    {
        Console.WriteLine("--- Windows Application ---");
        Application windowsApp = new Application(new WindowsGUIFactory());
        windowsApp.Paint();
        Console.WriteLine();

        Console.WriteLine("--- Mac Application ---");
        Application macApp = new Application(new MacGUIFactory());
        macApp.Paint();
        Console.WriteLine();
    }
}
Runtime behavior
  • Swap the concrete factory to change the entire family returned to the client.
  • Client code remains unchanged; only factory instantiation differs.
  • Dependency injection makes switching and testing straightforward.
Example output
Windows and Mac runs show different Render strings but identical client calls.
