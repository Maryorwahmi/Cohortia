Starter code overview
The starter defines IDbConnection and IDbCommand, concrete SQL Server classes, and a client that takes an IDbFactory. Your tasks are marked TODO in the starter code: implement Oracle products and OracleFactory, then use OracleFactory in Main.

// Client
public class DatabaseClient
{
    private IDbFactory _factory;
    private IDbConnection _connection;
    private IDbCommand _command;

    public DatabaseClient(IDbFactory factory, string connectionString)
    {
        _factory = factory;
        _connection = _factory.CreateConnection(connectionString);
        _command = _factory.CreateCommand();
        _command.SetConnection(_connection);
    }

    public void RunDatabaseOperations(string query)
    {
        _connection.Open();
        _command.ExecuteQuery(query);
        _connection.Close();
    }
}
Lab guidance
