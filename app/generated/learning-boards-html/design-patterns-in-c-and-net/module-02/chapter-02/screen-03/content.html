Hands-on activity: Logger refactor
Refactor ConsoleLogger into Product/Creator so Application accepts a LoggerCreator. This prepares the system for file or database loggers in future without modifying Application.
Starter code (concept)
// ConsoleLogger
public class ConsoleLogger
{
  public void Log(string message)
  {
    Console.WriteLine($"[Console Log] {message}");
  }
}

// Application has direct instantiation
public class Application
{
  private ConsoleLogger _logger;

  public Application()
  {
    _logger = new ConsoleLogger();
  }

  public void Run()
  {
    _logger.Log("Application started.");
    _logger.Log("Application finished.");
  }
}
