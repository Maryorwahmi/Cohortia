Factory Pattern Reimagined with DI

Using the DI container to dynamically resolve service implementations.

Externalized Creation

DI container manages object creation, replacing explicit factory classes.

Factory Method via Lambda

Register `AddTransient` with a lambda to define dynamic creation logic.

Key Takeaway

DI allows you to implement the Factory pattern declaratively by registering factory methods, making object creation configurable and testable without manual factory classes.

Program.cs
using Microsoft.Extensions.DependencyInjection;
using System;

public interface IMessageSender
{
    void Send(string message);
}

public class EmailSender : IMessageSender
{
    public void Send(string message) => Console.WriteLine($"Emailing: {message}");
}

public class SmsSender : IMessageSender
{
    public void Send(string message) => Console.WriteLine($"SMSing: {message}");
}

public class NotificationService
{
    private readonly IMessageSender _sender;

    public NotificationService(IMessageSender sender)
    {
        _sender = sender;
    }

    public void Notify(string message)
    {
        _sender.Send(message);
    }
}

public class Program
{
    public static void Main(string[] args)
    {
        var serviceCollection = new ServiceCollection();

        // Register IMessageSender with a factory method (Factory pattern via DI)
        serviceCollection.AddTransient<IMessageSender>(provider =>
        {
            bool useEmail = true; // Logic from config or context
            if (useEmail)
            {
                return new EmailSender();
            }
            else
            {
                return new SmsSender();
            }
        });

        serviceCollection.AddTransient<NotificationService>();

        var serviceProvider = serviceCollection.BuildServiceProvider();

        var notificationService = serviceProvider.GetService<NotificationService>();
        notificationService.Notify("Hello from Cohortia!");
    }
}
