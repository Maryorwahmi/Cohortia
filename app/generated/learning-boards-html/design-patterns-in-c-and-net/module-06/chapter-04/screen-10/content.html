Module 6 › Chapter 6.4 › LoggingNotifierDecorator Test
Test class template
using Xunit;
using Moq;

public class LoggingNotifierDecoratorTests
{
  [Fact]
  public void Send_ShouldLogBeforeAndAfterAndCallWrappedNotifier()
  {
    // Arrange
    var mockWrappedNotifier = new Mock<INotifier>();
    var mockLogger = new Mock<ILogger>();
    var decorator = new LoggingNotifierDecorator(mockWrappedNotifier.Object, mockLogger.Object);
    string testMessage = "Test Message";

    // Act
    decorator.Send(testMessage);

    // Assert
    mockLogger.Verify(l => l.LogInfo(It.Is<string>(s => s.Contains("Before sending: "))), Times.Once());
    mockWrappedNotifier.Verify(n => n.Send(testMessage), Times.Once());
    mockLogger.Verify(l => l.LogInfo(It.Is<string>(s => s.Contains("After sending: "))), Times.Once());
  }
}
Explanation
The test arranges mocks, constructs the decorator, acts by calling Send, then asserts three interactions: logger before, wrapped notifier send, and logger after. Using It.Is<string> lets you check message content without matching the exact interpolated string.
Pro tip
Prefer verifying presence and counts over strict ordering unless the sequence is critical to correctness.
Key Takeaway

This test pattern is reusable for other decorators. Mock external collaborators and assert added behavior plus delegation.
