Refactored Code Example
A compact Strategy-based replacement for the old switch-based calculator.
public interface IDiscountStrategy
{
  double ApplyDiscount(Order order, double basePrice);
}

public class SeasonalDiscountStrategy : IDiscountStrategy
{
  public double ApplyDiscount(Order order, double basePrice)
  {
    return basePrice * 0.90; // 10% off
  }
}

public class LoyaltyDiscountStrategy : IDiscountStrategy
{
  public double ApplyDiscount(Order order, double basePrice)
  {
    if (order.IsLoyaltyCustomer)
      return basePrice * 0.85; // 15% off

    return basePrice;
  }
}

public class NewDiscountCalculator
{
  private IDiscountStrategy _strategy;
  public NewDiscountCalculator(IDiscountStrategy strategy)
  {
    _strategy = strategy ?? throw new ArgumentNullException(nameof(strategy));
  }

  public double CalculateFinalPrice(Order order)
  {
    double basePrice = order.GetTotalPrice();
    return _strategy.ApplyDiscount(order, basePrice);
  }
}
Example
Swap strategies in Main to see different final prices without touching NewDiscountCalculator.
