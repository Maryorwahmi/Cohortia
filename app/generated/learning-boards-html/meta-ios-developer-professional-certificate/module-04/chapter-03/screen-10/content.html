// ObservableObject: UserProfileManager
class UserProfileManager: ObservableObject {
  @Published var userName: String = "Guest User"
  @Published var userEmail: String = "guest@example.com"
  @Published var isLoggedIn: Bool = false

  init() {
    // Simulate login after delay
    DispatchQueue.main.asyncAfter(deadline: .now() + 2) {
      self.userName = "John Doe"
      self.userEmail = "john.doe@cohortia.com"
      self.isLoggedIn = true
    }
  }
}

// Consumer: UserGreetingView
struct UserGreetingView: View {
  @EnvironmentObject var userProfile: UserProfileManager
  @Environment(\.colorScheme) var colorScheme
  var body: some View {
    Text("Hello, \(userProfile.userName)!")
  }
}
Explanation & verification
  1. Create the manager with @StateObject at root to own it.
  2. Attach it with .environmentObject(manager) so descendants can access it.
  3. In a nested view declare @EnvironmentObject to receive it and react to @Published updates.
Try this
Run the app and watch the simulated login update the greeting after two seconds. Toggle the login state at the root and confirm the greeting updates immediately.
Key Takeaway

This example shows a complete flow from ownership to provision to consumption and how system environment values adapt UI styling.
