Starter code
Use @State for the bill amount string and tip percentage; computed properties do the math.
import SwiftUI

struct TipCalculatorView: View {
    @State private var billAmountString: String = ""
    @State private var tipPercentage: Double = 15.0 // Default 15%

    var billAmount: Double {
        return Double(billAmountString) ?? 0.0
    }

    var tipAmount: Double {
        return billAmount * (tipPercentage / 100.0)
    }

    var totalAmount: Double {
        return billAmount + tipAmount
    }

    var body: some View {
        NavigationView {
            Form {
                Section("Enter Bill Amount") {
                    TextField("Amount", text: $billAmountString)
                        .keyboardType(.decimalPad)
                }

                Section("Select Tip Percentage") {
                    Slider(value: $tipPercentage, in: 0...25, step: 1) {
                        Text("Tip: \(Int(tipPercentage))%")
                    } minimumValueLabel: { Text("0%") } maximumValueLabel: { Text("25%") }
                    Text("Tip: \(Int(tipPercentage))%")
                }

                Section("Summary") {
                    HStack { Text("Bill Amount:"); Spacer(); Text(String(format: "$%.2f", billAmount)) }
                    HStack { Text("Tip Amount (\(Int(tipPercentage))%):"); Spacer(); Text(String(format: "$%.2f", tipAmount)) }
                    HStack { Text("Total Amount:"); Spacer(); Text(String(format: "$%.2f", totalAmount)).fontWeight(.bold) }
                }
            }
            .navigationTitle("Tip Calculator")
        }
    }
}
Try this
Refactor the slider into a child component that accepts a binding to tipPercentage to practice @Binding.
Teaching bullets
  • Keep input storage as String when using TextField and convert to Double safely for calculations.
  • Use Slider bound to a Double @State for immediate updates.
  • Format currency on display; handle empty or invalid input as zero.
  • Refactor controls into children using @Binding to increase reuse.
Key Takeaway

The tip calculator demonstrates how @State and @Binding produce a responsive, declarative form with clear ownership.
