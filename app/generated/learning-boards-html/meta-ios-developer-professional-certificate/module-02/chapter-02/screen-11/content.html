Why use previews
  • See immediate visual feedback while coding.
  • Test different sizes, modes, and example states.
  • Faster iteration than running the simulator.
Preview snippet
struct ProfileCardView_Previews: PreviewProvider {
  static var previews: some View {
    ProfileCardView()
      .previewLayout(.sizeThatFits)
      .padding()
  }
}
