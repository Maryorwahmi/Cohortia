Starter code
struct Task: Identifiable {
  let id = UUID()
  var name: String
  var isCompleted: Bool = false
}

struct DynamicToDoListView: View {
  @State private var tasks: [Task] = [
    Task(name: "Learn SwiftUI Layout", isCompleted: true),
    Task(name: "Build a Product Card", isCompleted: false),
    Task(name: "Master Navigation", isCompleted: false)
  ]
  @State private var newTaskName: String = ""
  @State private var showCompletedOnly: Bool = false

  var filteredTasks: [Task] {
    if showCompletedOnly { return tasks.filter { $0.isCompleted } }
    else { return tasks }
  }

  var body: some View {
    // UI omitted here — see challenge instructions in chapter
  }
}
Binding pattern

Inside ForEach, use Binding(get:set:) to update tasks using its index. This avoids trying to bind directly to the loop variable.

Challenge

Implement the ForEach for filteredTasks and a Toggle bound via a Binding initializer that finds the task index and updates tasks[index].isCompleted.

Key Takeaway

This exercise ties together ForEach, Identifiable, conditional filtering, and ScrollView to form a complete dynamic feature.
