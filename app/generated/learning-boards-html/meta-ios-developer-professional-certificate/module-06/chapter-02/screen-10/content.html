Starter ViewModel
import Foundation

class CalculatorViewModel: ObservableObject {
  @Published var result: Double = 0.0

  func add(_ a: Double, _ b: Double) { result = a + b }
  func subtract(_ a: Double, _ b: Double) { result = a - b }
  func multiply(_ a: Double, _ b: Double) { result = a * b }
  func divide(_ a: Double, _ b: Double) { if b == 0 { result = .nan } else { result = a / b } }
}
Task: create a Unit Testing Bundle, add CalculatorViewModelTests, implement setUpWithError, and write tests for each operation including division by zero.
Example test methods
import XCTest
@testable import YourAppName

final class CalculatorViewModelTests: XCTestCase {
  var vm: CalculatorViewModel!

  override func setUpWithError() throws { vm = CalculatorViewModel() }

  func testAddition() throws {
    vm.add(2, 3)
    XCTAssertEqual(vm.result, 5.0)
  }

  func testDivisionByZero() throws {
    vm.divide(5, 0)
    XCTAssertTrue(vm.result.isNaN)
  }
}
Key Takeaway

Write small, focused tests; run them frequently and keep each test independent to catch regressions early.
