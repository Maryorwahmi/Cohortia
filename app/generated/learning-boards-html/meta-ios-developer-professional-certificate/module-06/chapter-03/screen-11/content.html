Suggested replacement
import SwiftUI
import Kingfisher

struct ContentView: View {
  let imageUrl = URL(string: "https://picsum.photos/id/237/200/300")!

  var body: some View {
    VStack {
      Text("Image from URL")
        .font(.title)
        .padding()

      KFImage(imageUrl)
        .resizable()
        .placeholder { Image(systemName: "arrow.triangle.2.circlepath") }
        .aspectRatio(contentMode: .fit)
        .frame(width: 200, height: 300)
        .padding()
    }
  }
}
