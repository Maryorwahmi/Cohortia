func createNewPost(post: Post, completion: @escaping (Result<Post, Error>) -> Void) {
  var request = URLRequest(url: URL(string: createPostAPIURLString)!)
  request.httpMethod = "POST"
  request.setValue("application/json", forHTTPHeaderField: "Content-Type")

  do {
    let jsonData = try JSONEncoder().encode(post)
    request.httpBody = jsonData
  } catch {
    completion(.failure(error))
    return
  }

  URLSession.shared.dataTask(with: request) { data, response, error in
    if let error = error { completion(.failure(error)); return }
    guard let httpResponse = response as? HTTPURLResponse,
          (200...299).contains(httpResponse.statusCode) else {
      let code = (response as? HTTPURLResponse)?.statusCode ?? -1
      completion(.failure(NSError(domain:"HTTPError", code: code, userInfo:nil)))
      return
    }
    guard let data = data else { completion(.failure(URLError(.zeroBytesInResource))); return }
    do {
      let decoder = JSONDecoder()
      let createdPost = try decoder.decode(Post.self, from: data)
      completion(.success(createdPost))
    } catch { completion(.failure(error)) }
  }.resume()
}
Concrete example: sending a Post with userId 1. If server returns 201 and a JSON post with an ID, decoding succeeds and you can display the created post details in the UI.
