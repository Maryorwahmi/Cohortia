Fetch and decode
URLSession.shared.dataTask(with: url) { data, response, error in
  if let error = error { /* network error */; return }
  guard let http = response as? HTTPURLResponse,
        (200...299).contains(http.statusCode) else { /* status error */; return }
  guard let data = data else { /* no data */; return }
  do {
    let post = try JSONDecoder().decode(Post.self, from: data)
    // use post
  } catch { /* decoding error */ }
}.resume()
Example: fetching /posts/1 returns JSON for a single post; decode Post.self. If the API returns an array instead, decode([Post].self, from: data) to match the shape.
