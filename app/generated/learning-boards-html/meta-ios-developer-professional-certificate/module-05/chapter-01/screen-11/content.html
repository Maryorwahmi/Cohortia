Starter: ContentView
struct ContentView: View {
  @AppStorage("isDarkModeEnabled") var isDarkModeEnabled: Bool = false
  @AppStorage("welcomeMessage") var welcomeMessage: String = "Hello, Cohortia Learner!"

  var body: some View {
    NavigationView {
      VStack {
        Text(welcomeMessage)
          .font(.largeTitle)
          .padding()

        Spacer()

        // Navigation in starter code omitted here for brevity
      }
      .background(isDarkModeEnabled ? Color.black : Color.white)
      .foregroundColor(isDarkModeEnabled ? .white : .black)
    }
  }
}
Testing checklist
  • Change the welcome message and relaunch the app to confirm persistence.
  • Toggle dark mode and confirm the background and text colors update on next launch.
  • Call removeObject(forKey:) for cleanup on logout and verify values reset.
Pro tip

Use the simulator to terminate and relaunch your app to validate that saved values persist beyond the running session.
