let sender = ensure_signed(origin)?;
let (owner, _item, current_bid, end_block) = <Auctions<T>>::get(auction_id)
    .ok_or(Error::AuctionDoesNotExist)?;

ensure!(sender != *owner, Error::CannotBidOnOwnAuction);
ensure!(bid_amount > *current_bid, Error::BidTooLow);
ensure!(frame_system::Module::::block_number() < *end_block, Error::AuctionAlreadyEnded);

<Auctions<T>>::insert(auction_id, (owner.clone(), _item.clone(), bid_amount.clone(), *end_block));
<CurrentBidders<T>>::insert(auction_id, sender.clone());

Self::deposit_event(RawEvent::BidPlaced(auction_id, sender.clone(), bid_amount));
