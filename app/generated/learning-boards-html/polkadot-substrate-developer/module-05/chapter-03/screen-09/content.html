signAndSend pattern
const unsub = await transfer.signAndSend(alice, { nonce }, ({ status, events, dispatchError }) => {
  if (status.isInBlock) {
    console.log('Included in block', status.asInBlock.toHex());
  }
  if (status.isFinalized) {
    if (dispatchError) {
      if (dispatchError.isModule) {
        const decoded = api.registry.findError(dispatchError.asModule);
        console.log('Dispatch Error:', decoded.section, decoded.name);
      } else {
        console.log('Dispatch Error:', dispatchError.toString());
      }
    }
    events.forEach(({ event: { section, method, data } }) => {
      console.log(`${section}.${method}:`, data.toString());
    });
    unsub();
    api.disconnect();
  }
});
Make sure to call unsub() after finalization to stop the callback stream and close the connection when your script completes to avoid leaking sockets.
