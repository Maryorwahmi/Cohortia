Query.js example

// query.js
const { ApiPromise, WsProvider } = require('@polkadot/api');

async function main () {
  const provider = new WsProvider('ws://127.0.0.1:9944');
  const api = await ApiPromise.create({ provider });

  console.log('--- Chain Information ---');
  const chain = await api.rpc.system.chain();
  console.log(`Connected to chain: ${chain}`);

  console.log('
--- Querying State ---');
  const blockNumber = await api.query.system.number();
  console.log(`Current block number: ${blockNumber.toNumber()}`);

  const existentialDeposit = api.consts.balances.existentialDeposit;
  console.log(`Existential Deposit: ${existentialDeposit.toHuman()} units`);

  const totalIssuance = await api.query.balances.totalIssuance();
  console.log(`Total Issuance: ${totalIssuance.toHuman()} units`);

  await api.disconnect();
}

main().catch(console.error);
Run this script with node query.js while your Substrate dev node is running. Expect numeric conversions for block number and human-readable strings for token amounts.
Tip
If a session pallet is not present, check api.query.system.events() as a fallback to inspect recent events.
