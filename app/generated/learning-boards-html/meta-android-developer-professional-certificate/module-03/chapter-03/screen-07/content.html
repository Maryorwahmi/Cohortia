Write to internal storage
// Conceptual Kotlin
fun writeToFile(context: Context, filename: String, content: String) {
  try {
    context.openFileOutput(filename, Context.MODE_PRIVATE).use {
      it.write(content.toByteArray())
      // file written
    }
  } catch (e: IOException) {
    // handle error
  }
}
Use use() to ensure streams are closed and catch IOExceptions to handle disk or permission errors.
Read from internal storage
// Conceptual Kotlin
fun readFromFile(context: Context, filename: String): String? {
  try {
    context.openFileInput(filename).use { inputStream ->
      val reader = BufferedReader(InputStreamReader(inputStream))
      val sb = StringBuilder()
      var line: String?
      while (reader.readLine().also { line = it } != null) {
        sb.append(line).append("
")
      }
      return sb.toString()
    }
  } catch (e: FileNotFoundException) {
    return null
  } catch (e: IOException) {
    return null
  }
}
Key Takeaway

Use buffered readers, close streams safely, and handle both missing files and IO errors while running on a background thread.
