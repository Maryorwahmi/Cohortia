UserAdapter.kt
class UserAdapter(private val userList: List<User>) :
  RecyclerView.Adapter<UserAdapter.UserViewHolder>() {

  class UserViewHolder(itemView: View) : RecyclerView.ViewHolder(itemView) {
    val userNameTextView: TextView = itemView.findViewById(R.id.userNameTextView)
    val userEmailTextView: TextView = itemView.findViewById(R.id.userEmailTextView)

    fun bind(user: User) {
      userNameTextView.text = user.name
      userEmailTextView.text = user.email
      itemView.setOnClickListener {
        Toast.makeText(itemView.context, "Clicked on ${user.name}", Toast.LENGTH_SHORT).show()
      }
    }
  }

  override fun onCreateViewHolder(parent: ViewGroup, viewType: Int): UserViewHolder {
    val itemView = LayoutInflater.from(parent.context)
      .inflate(R.layout.item_user, parent, false)
    return UserViewHolder(itemView)
  }

  override fun onBindViewHolder(holder: UserViewHolder, position: Int) {
    val currentUser = userList[position]
    holder.bind(currentUser)
  }

  override fun getItemCount(): Int {
    return userList.size
  }
}
Explanation
  • Adapter holds the data and creates ViewHolders when needed.
  • ViewHolder caches views and exposes a bind method for assignments.
  • onBindViewHolder delegates to bind so the adapter logic stays thin.
Common mistake

Avoid heavy work in bind such as synchronous network calls or expensive computations — do those off-thread.
