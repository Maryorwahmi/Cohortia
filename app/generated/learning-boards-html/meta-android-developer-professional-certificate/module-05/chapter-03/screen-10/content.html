// snippet from TaskViewModelTest
@get:Rule
val instantTaskExecutorRule = InstantTaskExecutorRule()
private val mockRepository: TaskRepository = mock()
private val testDispatcher = StandardTestDispatcher()

@Before
fun setup() {
  Dispatchers.setMain(testDispatcher)
  viewModel = TaskViewModel(mockRepository)
}

@Test
fun `getAllTasks emits tasks from repository`() = runTest {
  val tasks = listOf(
    Task(1, "Test Task 1", "Desc 1"),
    Task(2, "Test Task 2", "Desc 2")
  )
  whenever(mockRepository.getAllTasks()).thenReturn(flowOf(tasks))

  var emittedTasks: List? = null
  viewModel.allTasks.observeForever { emittedTasks = it }

  testDispatcher.scheduler.advanceUntilIdle()

  assertEquals(tasks, emittedTasks)
  verify(mockRepository).getAllTasks()
}
Explanation
  • Arrange: stub repository.getAllTasks to return a Flow with tasks.
  • Act: observe the LiveData to start collection.
  • Advance: run pending coroutines with advanceUntilIdle.
  • Assert: the observed value equals expected tasks and repository method was invoked.
Key Takeaway

This pattern makes ViewModel tests deterministic and verifies both data flow and interactions.
