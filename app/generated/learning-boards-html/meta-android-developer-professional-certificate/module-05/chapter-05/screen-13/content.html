1) Add Hilt dependencies
// project-level build.gradle
buildscript {
  dependencies {
    classpath 'com.google.dagger:hilt-android-gradle-plugin:2.51.1'
  }
}

// app/build.gradle plugins
plugins {
  id 'kotlin-kapt'
  id 'com.google.dagger.hilt.android'
}

// app dependencies
implementation 'com.google.dagger:hilt-android:2.51.1'
kapt 'com.google.dagger:hilt-android-compiler:2.51.1'
2) Application class
@HiltAndroidApp
class TaskApplication : Application() {}

// AndroidManifest.xml
3) Provide dependencies
@Module
@InstallIn(SingletonComponent::class)
object AppModule {
  @Singleton @Provides
  fun provideAppDatabase(@ApplicationContext context: Context): AppDatabase = AppDatabase.getDatabase(context)

  @Singleton @Provides
  fun provideTaskDao(db: AppDatabase): TaskDao = db.taskDao()

  @Singleton @Provides
  fun provideTaskRepository(dao: TaskDao): TaskRepository = TaskRepository(dao)
}
4) Inject ViewModel & Activity
@HiltViewModel
class TaskViewModel @Inject constructor(
  private val repository: TaskRepository
): ViewModel() { ... }

@AndroidEntryPoint
class TaskListActivity : AppCompatActivity() {
  private val taskViewModel: TaskViewModel by viewModels()
}
Key Takeaway

Follow the steps: add dependencies, annotate the Application, provide modules, and inject the ViewModel and Activity to simplify DI and testing.
