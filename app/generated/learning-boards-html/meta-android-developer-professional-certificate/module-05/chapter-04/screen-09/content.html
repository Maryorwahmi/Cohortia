@Test
fun toggleTaskCompletion_updatesUI() {
    val taskTitle = "Clean Room"
    val taskDescription = "Vacuum and dust"

    // Add a task
    onView(withId(R.id.editText_task_title))
        .perform(typeText(taskTitle), closeSoftKeyboard())
    onView(withId(R.id.editText_task_description))
        .perform(typeText(taskDescription), closeSoftKeyboard())
    onView(withId(R.id.button_add_task)).perform(click())

    // Toggle completion by clicking the item
    onView(withText(taskTitle)).perform(click())

    // Verification may require a custom matcher; here we check it is still displayed
    onView(withText(taskTitle)).check(matches(isDisplayed()))
}

@Test
fun deleteTask_removesFromList() {
    val taskTitle = "Task to Delete"

    // Add task and then perform long click to delete
    onView(withId(R.id.editText_task_title)).perform(typeText(taskTitle), closeSoftKeyboard())
    onView(withId(R.id.button_add_task)).perform(click())

    onView(withText(taskTitle)).perform(longClick())

    // Verify the task no longer exists
    onView(withText(taskTitle)).check(doesNotExist())
}
Testing tip

Prefer explicit delete controls or test-only hooks so tests can trigger removal reliably without complex gestures.

Explanation
  • Use longClick or a delete button to trigger removal in tests.
  • Check absence with doesNotExist() to confirm the view was removed from the hierarchy.
  • Custom visual checks may need custom matchers for color or strikethrough.
Common pitfall

Using not(isDisplayed()) can pass even when the view still exists but is hidden; use doesNotExist() for removal assertions.
