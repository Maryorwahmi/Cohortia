// src/androidTest/java/com/cohortia/taskapp/ui/tasklist/TaskListActivityTest.kt
@RunWith(AndroidJUnit4::class)
class TaskListActivityTest {

    @get:Rule
    val activityRule = ActivityScenarioRule(TaskListActivity::class.java)

    @Test
    fun addTask_displaysInList() {
        val taskTitle = "Buy Groceries"
        val taskDescription = "Milk, Eggs, Bread"

        onView(withId(R.id.editText_task_title))
            .perform(typeText(taskTitle), closeSoftKeyboard())

        onView(withId(R.id.editText_task_description))
            .perform(typeText(taskDescription), closeSoftKeyboard())

        onView(withId(R.id.button_add_task))
            .perform(click())

        onView(withText(taskTitle))
            .check(matches(isDisplayed()))

        onView(withText(taskDescription))
            .check(matches(isDisplayed()))
    }
}
Concrete example
Typing and clicking Add should result in the new task text being visible on screen.
Common mistake
Omitting closeSoftKeyboard causes the keyboard to cover the Add button and may make the click fail.
Explanation
  • ActivityScenarioRule launches the activity and handles lifecycle in tests.
  • onView(withId(...)) finds a view by resource id; withText finds by visible text.
  • perform runs actions such as typeText and click; closeSoftKeyboard hides the keyboard.
  • check with matches(assertion) verifies view state like isDisplayed().
Key Takeaway

Follow the find-act-assert pattern and handle the soft keyboard to avoid false negatives in UI tests.
