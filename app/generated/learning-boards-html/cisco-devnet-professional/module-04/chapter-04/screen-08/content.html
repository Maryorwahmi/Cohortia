Configuring VLANs with NETCONF: Connection & Locking

Setting up the `ncclient` connection and securing the running configuration.

`ncclient` Library

Python library for NETCONF client operations, simplifying RPC calls and session management.

Datastore Locking

The `m.lock(target='running')` operation ensures exclusive access to the device's configuration.

from ncclient import manager

HOST = "your_cisco_ios_xe_device_ip"
PORT = 830
USER = "your_username"
PASS = "your_password"

def create_vlan_netconf(vlan_id, vlan_name):
    # Establish connection
    with manager.connect(host=HOST, port=PORT, username=USER, password=PASS,
                         hostkey_verify=False, device_params={'name': 'iosxe'}) as m:
        print(f"--- Connected to device for VLAN {vlan_id} config ---")
        try:
            # Lock the running datastore
            m.lock(target='running')
            print("Running datastore locked.")
            # ... configuration payload and commit will follow ...
        except Exception as e:
            print(f"Error: {e}")
        finally:
            m.unlock(target='running')
            print("Running datastore unlocked.")
