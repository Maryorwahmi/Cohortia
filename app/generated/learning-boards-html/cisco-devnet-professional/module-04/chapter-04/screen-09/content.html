Configuring VLANs with NETCONF: Payload & Commit

Constructing the XML payload and committing changes to the device.

XML Configuration Payload

The configuration is defined in XML, adhering to the device's YANG model for strict validation.

`edit_config` and `commit`

`edit_config` applies changes to the datastore, and `commit` makes them active on the device.

from ncclient import manager

def create_vlan_netconf(vlan_id, vlan_name):
    # ... connection and lock ...
        try:
            m.lock(target='running')
            print("Running datastore locked.")

            # XML payload for VLAN creation
            config_payload = f"""
            <config>
              <native xmlns="http://cisco.com/ns/yang/Cisco-IOS-XE-native">
                <vlan>
                  <vlan-list>
                    <vlan-id>{vlan_id}</vlan-id>
                    <name>{vlan_name}</name>
                  </vlan-list>
                </vlan>
              </native>
            </config>
            """
            print(f"
--- Sending NETCONF config for VLAN {vlan_id} ---")
            m.edit_config(target='running', config=config_payload)
            print("Configuration sent. Committing changes...")
            m.commit()
            print(f"VLAN {vlan_id} ({vlan_name}) created successfully via NETCONF.")

        except Exception as e:
            print(f"Error creating VLAN {vlan_id} via NETCONF: {e}")
        finally:
            m.unlock(target='running')
            print("Running datastore unlocked.")
