// test/Counter.js
const { expect } = require("chai");
const { ethers } = require("hardhat");

describe("Counter", function () {
  let Counter;
  let counter;
  let owner;
  let addr1;

  beforeEach(async function () {
    Counter = await ethers.getContractFactory("Counter");
    [owner, addr1] = await ethers.getSigners();
    counter = await Counter.deploy();
    await counter.deployed();
  });

  it("Should have an initial count of 0", async function () {
    expect(await counter.getCount()).to.equal(0);
  });

  it("Should increment the count by 1", async function () {
    await counter.increment();
    expect(await counter.getCount()).to.equal(1);
  });

  it("Should revert if decrementing from zero", async function () {
    await expect(counter.decrement()).to.be.revertedWith("Count cannot be negative");
  });

  it("Should allow other accounts to increment", async function () {
    await counter.connect(addr1).increment();
    expect(await counter.getCount()).to.equal(1);
  });
});
Explanation points
  • Initial state test confirms constructor behavior.
  • State-changing functions require awaiting the transaction before asserting.
  • revertedWith checks specific error messages from require.
  • connect(signer) simulates calls from different accounts for access testing.
Key Takeaway

Write focused tests that validate one behavior and one assertion per it. Use revert checks for error paths.
