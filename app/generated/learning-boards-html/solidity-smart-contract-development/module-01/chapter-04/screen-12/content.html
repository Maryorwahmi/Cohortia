Counter.sol (complete)
// contracts/Counter.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

contract Counter {
    uint256 public count;
    address public owner;

    event CounterIncremented(address indexed by, uint256 newCount);

    constructor() {
        count = 0;
        owner = msg.sender;
    }

    function increment() public {
        require(msg.sender == owner, "Only owner can increment");
        count++;
        emit CounterIncremented(msg.sender, count);
    }

    function getCount() public view returns (uint256) {
        return count;
    }

    function reset() public {
        require(msg.sender == owner, "Only owner can reset");
        count = 0;
        emit CounterIncremented(msg.sender, count);
    }
}
Compile steps
1. Save Counter.sol in contracts/.
2. Run npx hardhat compile in the project root.
3. Inspect artifacts/ to confirm the compiled output.
Key Takeaway

Build the contract incrementally, run the compiler often, and test owner-only flows and event logs to validate behavior.
