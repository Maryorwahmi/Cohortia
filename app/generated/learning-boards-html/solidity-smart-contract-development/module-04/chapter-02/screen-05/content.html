Definition
onlyOwner modifier
A small, auditable check that prevents unauthorized callers from executing sensitive functions.
Test: add an unauthorized caller test to assert require failure for protected functions.
When to use
Single-owner contracts, administrative tasks, and emergency controls.
When to consider RBAC
If you need many admins with different responsibilities, RBAC scales better than a single owner address.
Secure example (excerpt)
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract SecureAccessControl {
    address public owner;
    uint256 public criticalValue;

    constructor() {
        owner = msg.sender;
        criticalValue = 100;
    }

    modifier onlyOwner() {
        require(msg.sender == owner, "Not owner");
        _;
    }

    // SECURE: Only the owner can change the critical value
    function setCriticalValue(uint256 _newValue) public onlyOwner {
        criticalValue = _newValue;
    }

    // SECURE: Only the owner can transfer ownership
    function transferOwnership(address _newOwner) public onlyOwner {
        require(_newOwner != address(0), "New owner cannot be zero address");
        owner = _newOwner;
    }
}
Takeaway
Apply checks first, validate inputs like zero address, and keep modifier usage consistent across all sensitive functions.
