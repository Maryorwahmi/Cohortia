Vulnerable withdrawal pattern

Solidity example (vulnerable)
<!-- code -->
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract VulnerableWithdrawal {
    mapping(address => uint) public balances;

    constructor() payable {
        balances[msg.sender] = msg.value;
    }

    function deposit() public payable {
        balances[msg.sender] += msg.value;
    }

    function withdraw(uint _amount) public {
        require(balances[msg.sender] >= _amount, "Insufficient balance");

        // VULNERABLE: External call before state update
        (bool success, ) = msg.sender.call{value: _amount}("");
        require(success, "Withdrawal failed");

        balances[msg.sender] -= _amount; // State updated AFTER external call
    }

    function getBalance() public view returns (uint) {
        return address(this).balance;
    }
}
Explanation
  • The external call occurs before the balance is decreased, which creates the reentry window.
  • A malicious recipient can run code to call withdraw again inside its fallback, while the balance still appears unchanged.
  • Tests should deploy an attacker contract with a fallback that calls withdraw to emulate the exploit.
Example scenario
If an attacker deposited 1 ether and then re-entered withdraw for 1 ether repeatedly, the contract can be drained.
Common mistake

Assuming a transfer or call is safe without reordering state updates is a frequent oversight.
