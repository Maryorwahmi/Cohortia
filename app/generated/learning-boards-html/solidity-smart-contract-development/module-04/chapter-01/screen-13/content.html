Lab: implement vulnerable and secure vaults

Vulnerable contract (starter)
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract EtherVaultVulnerable {
    mapping(address => uint256) public deposits;

    function deposit() public payable {
        deposits[msg.sender] += msg.value;
    }

    function withdraw(uint256 _amount) public {
        require(deposits[msg.sender] >= _amount, "Insufficient funds");

        // TODO: make an external call here BEFORE updating state to create the vulnerability

        // deposits[msg.sender] -= _amount; // update state AFTER external call to be vulnerable
    }

    function getContractBalance() public view returns (uint256) {
        return address(this).balance;
    }
}
Lab steps & test plan
  1. Complete withdraw to call msg.sender.call{{value:_amount}} before decrementing deposits.
  2. Write an attacker contract with a fallback that calls withdraw when receiving Ether.
  3. Write Hardhat tests that assert the vulnerable vault is drained and the secure vault resists the exploit.
Remember: keep the vulnerable pattern in the starter so you can demonstrate the exploit, and move the state update before the call in the secure contract.
