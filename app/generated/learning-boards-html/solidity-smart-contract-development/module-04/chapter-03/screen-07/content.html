Using OpenZeppelin's Pausable
import "@openzeppelin/contracts/access/Ownable.sol";
import "@openzeppelin/contracts/security/Pausable.sol";

contract MyPausableContract is Ownable, Pausable {
    function deposit() public payable whenNotPaused {
        // ... deposit logic ...
    }

    function withdraw() public whenNotPaused {
        // ... withdraw logic ...
    }

    function emergencyPause() public onlyOwner {
        _pause();
    }

    function resumeOperations() public onlyOwner {
        _unpause();
    }
}
When to pause

Pause during active exploits or when a critical bug is discovered. Use it to stop state-changing operations while you patch or migrate.

Check

Test that functions revert with whenNotPaused and that only authorized roles can toggle pause.
