Logic Contract (excerpt)
// Simplified logic contract
contract LogicV1 {
  uint public counter;

  function initialize(uint _initialValue) public {
    counter = _initialValue;
  }

  function increment() public {
    counter++;
  }

  function getVersion() public pure returns (string memory) {
    return "V1";
  }
}
Note: constructors are avoided in logic contracts so initialization happens in the proxy's storage context.
Proxy (excerpt)
contract SimpleProxy {
  address public implementation;
  address public owner;

  constructor(address _impl) {
    implementation = _impl;
    owner = msg.sender;
  }

  fallback() external payable {
    (bool success, bytes memory data) = implementation.delegatecall(msg.data);
    // return or revert with returned data
  }
}
The fallback forwards calldata to the implementation and returns the implementation's response to the original caller.
Key Takeaway

Always initialize and call through the proxy so state lives in the proxy and logic can be upgraded safely.
