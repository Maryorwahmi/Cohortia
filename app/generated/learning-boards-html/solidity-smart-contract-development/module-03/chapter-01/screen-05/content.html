Owned contract (example)
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract Owned {
  address public owner;
  constructor() { owner = msg.sender; }
  modifier onlyOwner() {
    require(msg.sender == owner, "Caller is not the owner");
    _;
  }
  function transferOwnership(address _newOwner) public onlyOwner {
    require(_newOwner != address(0), "New owner cannot be the zero address");
    owner = _newOwner;
  }
}
Explanation
  • The modifier enforces that only the owner can run guarded functions.
  • Because Derived inherits Owned, it gets owner and onlyOwner without extra code.
  • Overriding an onlyOwner function must retain or strengthen the access check.
Common Mistake

Removing onlyOwner in an override effectively opens access to sensitive operations. Always re-apply or call parent's guarded logic.
