Payable deposit with validation
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.4;

contract ExampleDeposit {
  mapping(address => uint) public balances;
  event Deposit(address indexed sender, uint amount, uint balance);
  error InvalidInput(uint input);

  function depositFunds() public payable {
    if (msg.value == 0) {
      revert InvalidInput(0);
    }
    balances[msg.sender] += msg.value;
    emit Deposit(msg.sender, msg.value, balances[msg.sender]);
  }
}
Run in Remix, send a small value, then inspect the transaction logs to confirm the Deposit event and the updated balances mapping. If msg.value is zero, the function reverts with a custom error payload that clients can parse.
