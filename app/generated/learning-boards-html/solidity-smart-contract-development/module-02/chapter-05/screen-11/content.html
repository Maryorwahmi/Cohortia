Starter code walkthrough
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.4;

contract SimpleTokenTransfer {
  mapping(address => uint) public balances;
  event Transfer(address indexed from, address indexed to, uint amount);
  error InsufficientBalance(address sender, uint requested, uint available);
  error ZeroAmountTransfer();

  constructor() { balances[msg.sender] = 1000; }

  function transfer(address _to, uint _amount) public {
    if (_amount == 0) revert ZeroAmountTransfer();
    if (balances[msg.sender] < _amount) revert InsufficientBalance(msg.sender, _amount, balances[msg.sender]);
    balances[msg.sender] -= _amount;
    balances[_to] += _amount;
    emit Transfer(msg.sender, _to, _amount);
  }
}
Testing scenarios
  • • Successful transfer: balances change and Transfer event appears.
  • • Zero amount: transaction reverts with ZeroAmountTransfer.
  • • Insufficient balance: revert returns InsufficientBalance with parameters.
Inspect the revert payload to see the custom error signature and parameters; this allows richer client-side handling than parsing strings.
