Counter.sol (excerpt)
// contracts/Counter.sol // SPDX-License-Identifier: MIT pragma solidity ^0.8.0; contract Counter { uint public count; constructor() { count = 0; } function increment() public { count++; } function decrement() public { require(count > 0, "Count cannot go below zero"); count--; } function getCount() public view returns (uint) { return count; } }
test/Counter.js (excerpt)
const { expect } = require("chai"); const { ethers } = require("hardhat"); describe("Counter", function () { let Counter; let counter; let owner; beforeEach(async function () { Counter = await ethers.getContractFactory("Counter"); [owner] = await ethers.getSigners(); counter = await Counter.deploy(); await counter.deployed(); }); it("Should set the initial count to 0", async function () { expect(await counter.getCount()).to.equal(0); }); it("Should increment the count by 1", async function () { await counter.increment(); expect(await counter.getCount()).to.equal(1); }); it("Should revert if count is already 0", async function () { await expect(counter.decrement()).to.be.revertedWith("Count cannot go below zero"); }); });
Common mistake

Missing await on transactions or not redeploying per test can cause flaky or interdependent tests.
