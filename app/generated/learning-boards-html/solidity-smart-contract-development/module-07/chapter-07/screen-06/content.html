Listener script (Node / Hardhat)
Attach contract.on and keep the process alive for live updates.
// scripts/listen_events.js
const { ethers } = require("hardhat");

async function main() {
  const CONTRACT_ADDRESS = "YOUR_NEWLY_DEPLOYED_COUNTER_ADDRESS"; // replace
  const CounterFactory = await ethers.getContractFactory("Counter");
  const counter = CounterFactory.attach(CONTRACT_ADDRESS);

  console.log(`Listening for CountChanged events on ${CONTRACT_ADDRESS}...`);

  counter.on("CountChanged", (user, newCount, action, event) => {
    console.log(`--- Event Detected ---`);
    console.log(`User: ${user}`);
    console.log(`New Count: ${newCount.toString()}`);
    console.log(`Action: ${action}`);
    console.log(`Transaction Hash: ${event.transactionHash}`);
    console.log(`Block Number: ${event.blockNumber}`);
    console.log(`----------------------`);
  });

  await new Promise(() => {});
}

main().catch((error) => {
  console.error(error);
  process.exitCode = 1;
});
  • Ensure the provider supports subscriptions (WebSocket) for near-real-time streaming.
  • In tests you can use HTTP polling providers with short intervals, but WebSockets are preferred for production listeners.
  • Attach error handling to reconnect on dropped connections in production services.

Common mistake

Running the listener with an HTTP provider expecting push events will not deliver real-time results; verify provider capabilities first.
