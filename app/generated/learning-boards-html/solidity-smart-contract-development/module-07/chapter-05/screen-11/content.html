Example script
This script reads the Counter, increments, waits, then decrements and confirms each change.
// scripts/interact_counter.js
const { ethers } = require("hardhat");

async function main() {
  const CONTRACT_ADDRESS = "YOUR_DEPLOYED_COUNTER_ADDRESS"; // replace
  const [signer] = await ethers.getSigners();
  const CounterFactory = await ethers.getContractFactory("Counter");
  const counter = CounterFactory.attach(CONTRACT_ADDRESS);

  console.log("--- Initial State ---");
  let currentCount = await counter.count();
  console.log(`Current count: ${'${currentCount.toString()}'}');

  console.log("--- Incrementing Counter ---");
  const incTx = await counter.connect(signer).increment();
  await incTx.wait();
  console.log("Increment transaction successful!");

  currentCount = await counter.count();
  console.log(`New count: ${'${currentCount.toString()}'}');
}

main().catch((error) => {
  console.error(error);
  process.exitCode = 1;
});
