Module 7 › Chapter 7.1 › Hardhat Configuration
Key points
  • Require the Hardhat toolbox and dotenv.
  • Read SEPOLIA_RPC_URL and PRIVATE_KEY from .env.
  • Specify chainId for sepolia to avoid cross-network errors.
Example snippet
// hardhat.config.js
require("@nomicfoundation/hardhat-toolbox");
require("dotenv").config(); // Load environment variables

const SEPOLIA_RPC_URL = process.env.SEPOLIA_RPC_URL || "https://eth-sepolia.g.alchemy.com/v2/YOUR_ALCHEMY_API_KEY";
const PRIVATE_KEY = process.env.PRIVATE_KEY || "YOUR_TEST_WALLET_PRIVATE_KEY";

module.exports = {
  solidity: "0.8.20",
  networks: {
    hardhat: {
      // local settings
    },
    sepolia: {
      url: SEPOLIA_RPC_URL,
      accounts: PRIVATE_KEY !== undefined ? [PRIVATE_KEY] : [],
      chainId: 11155111, // Sepolia's chain ID
    },
  },
};
Note: Replace placeholders in .env and never commit the file.
