Test outline
Arrange: fake repo and real validator (to be spied on). Act: place_order with valid data. Assert: order saved and validator called.
# test_order_service.py
import unittest
from unittest.mock import patch
from order_service import OrderService, OrderValidator
from fake_order_repository import FakeOrderRepository

class TestOrderService(unittest.TestCase):
  def test_place_valid_order_with_fake_repo_and_spy_validator(self):
    fake_repo = FakeOrderRepository()
    validator = OrderValidator()
    service = OrderService(validator, fake_repo)

    order_data = {"items": [{"id": "P1", "qty": 1}]}

    with patch.object(validator, 'is_valid_order', wraps=validator.is_valid_order) as spy:
      saved = service.place_order(order_data)
      spy.assert_called_once_with(order_data)

    assert saved is not None
    assert 'id' in saved
    assert saved['id'].startswith('ORD_')
Note: wraps lets the real validation run so the test verifies both the call and the real validation logic.
Check your understanding
  • Why prefer a spy here instead of replacing the validator with a stub? Because we want to verify validation and exercise its logic in the flow.
  • What would a mock add? Strict interaction-based assertions that replace behavior entirely.
