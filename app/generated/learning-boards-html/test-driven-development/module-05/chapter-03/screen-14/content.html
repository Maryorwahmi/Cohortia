class UserRepository:
    # ... existing methods ...

    def find_user_by_username(self, username):
        conn = self.db_conn_factory()
        try:
            user = conn.execute("SELECT * FROM users WHERE username = ?", (username,)).fetchone()
            return dict(user) if user else None
        finally:
            conn.close()
