Bad test (excerpt)
import pytest
from user_service import UserService

@pytest.fixture(scope="module")
def user_service_with_db():
    service = UserService(database_connection_string="real_db_prod")
    service.clear_all_users()
    yield service
    service.clear_all_users()

def test_register_and_get_user(user_service_with_db):
    user_service_with_db.register_user("alice", "password123")
    user = user_service_with_db.get_user("alice")
    assert user.username == "alice"
    assert user.email is None
Problems: hits a real DB, mixes two behaviors, and uses a long-lived fixture that may affect other tests.
Good test (excerpt)
from unittest.mock import Mock

@pytest.fixture
def mock_user_repository():
    return Mock()

@pytest.fixture
def user_service(mock_user_repository):
    return UserService(mock_user_repository)

def test_register_user_success(user_service, mock_user_repository):
    username = "bob"
    password = "secure_password"

    result = user_service.register_user(username, password)

    assert result is True
    mock_user_repository.add.assert_called_once_with({"username": username, "password": password})
Improvement: uses a mock, isolates behavior, and asserts interactions explicitly.
