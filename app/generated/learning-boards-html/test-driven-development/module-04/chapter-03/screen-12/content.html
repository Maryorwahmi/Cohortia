Starter test example
# test_notification_service.py
from notification_service import NotificationService
from unittest.mock import Mock

def test_send_email_notification():
    mock_email_sender = Mock()
    service = NotificationService(mock_email_sender)

    service.send_email_notification('user@example.com', 'Hello', 'Welcome')

    mock_email_sender.send.assert_called_once_with('user@example.com', 'Hello', 'Welcome')
Activity steps
