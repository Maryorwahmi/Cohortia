# test_order_processor.py
import pytest
from order_processor import OrderProcessor  # class not implemented yet
from unittest.mock import Mock

def test_process_new_order_successfully():
    mock_payment_gateway = Mock()
    mock_inventory_service = Mock()
    processor = OrderProcessor(mock_payment_gateway, mock_inventory_service)

    order_details = {"order_id": "ORD001", "items": [{"product_id": "P001", "quantity": 2}], "total_amount": 200.00}

    confirmation = processor.process_order(order_details)

    assert confirmation["status"] == "confirmed"
    assert confirmation["order_id"] == "ORD001"
    mock_payment_gateway.charge.assert_called_once_with(200.00)
    mock_inventory_service.deduct_stock.assert_called_once_with("P001", 2)
This test names the API and interactions. Notice how constructor injection and collaborator method names are determined by the assertions.
Teaching points
  • Tests define the constructor signature by showing which dependencies must exist.
  • Asserting calls on mocks clarifies collaborator APIs and interaction patterns.
  • Avoid asserting internal state; prefer observable results and interactions.
Misconception

Asserting private internals can lock you into brittle designs. Test the observable contract instead.
