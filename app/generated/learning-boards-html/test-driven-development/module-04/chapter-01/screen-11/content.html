# authenticator.py
class Authenticator:
    def __init__(self):
        self._users = {}

    def register_user(self, username: str, password: str) -> bool:
        if username in self._users:
            return False
        self._users[username] = password
        return True
This minimal code gets the test green and leaves room for future refactors.
