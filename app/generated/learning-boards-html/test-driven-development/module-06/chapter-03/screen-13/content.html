Example using unittest.mock
from legacy_calculator import PriceCalculator
from unittest.mock import Mock

def test_calculation_with_mocks():
    mock_tax = Mock()
    mock_tax.get_tax_rate.return_value = 0.05
    mock_discount = Mock()
    mock_discount.get_discount_percentage.return_value = 0.10

    calculator = PriceCalculator(tax_service=mock_tax, discount_service=mock_discount)
    assert calculator.calculate_final_price(100, 42, "US") == 94.50
This test isolates the arithmetic: 100 * (1 - 0.10) = 90; 90 * (1 + 0.05) = 94.5, rounded to 94.5. Keep both characterization and unit tests to cover behavior and logic.
