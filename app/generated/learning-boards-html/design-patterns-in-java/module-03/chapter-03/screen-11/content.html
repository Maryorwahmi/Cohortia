Keep code modular: define contracts with interfaces and use components for implementations so Spring can wire them automatically.
Starter code snippets
// MessageService.java
public interface MessageService {
  String getMessage();
}

// GreetingService.java
@Component
public class GreetingService implements MessageService {
  @Override
  public String getMessage() { return "Hello from GreetingService!"; }
}

// NotificationService.java
@Component
public class NotificationService {
  private final MessageService messageService;
  public NotificationService(MessageService messageService) { this.messageService = messageService; }
  public String sendNotification() { return "Sending notification: " + messageService.getMessage(); }
}
