Application client
// Client code
public class Application {
    private GUIFactory factory;
    private Button button;
    private Checkbox checkbox;

    public Application(GUIFactory factory) {
        this.factory = factory;
    }

    public void createUI() {
        button = factory.createButton();
        checkbox = factory.createCheckbox();
    }

    public void paintUI() {
        button.paint();
        checkbox.paint();
    }
}
Choosing a factory at startup
// Choose factory based on environment
String os = System.getProperty("os.name").toLowerCase();
GUIFactory factory;
if (os.contains("windows")) {
    factory = new WindowsFactory();
} else if (os.contains("mac")) {
    factory = new MacOSFactory();
} else {
    factory = new WindowsFactory(); // fallback
}
Application app = new Application(factory);
app.createUI();
app.paintUI();
Design reminder

Always swap the factory to change product families. Do not mix factories for different product types inside the same client scope.
