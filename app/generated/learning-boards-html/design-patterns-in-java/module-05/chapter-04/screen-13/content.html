// Employee class
class Employee {
  private String name;
  private String id;
  public Employee(String name, String id) { this.name = name; this.id = id; }
  public String getName() { return name; }
  public String getId() { return id; }
  @Override public String toString() { return "Employee [ID: " + id + ", Name: " + name + "]"; }
}
