Definition › No-allocation code patterns
Coroutine: Bad vs Good
// BAD: Allocates new WaitForSeconds every loop
IEnumerator MyCoroutineBad()
{
    while (true)
    {
        yield return new WaitForSeconds(1f);
        // ...
    }
}

// GOOD: Cache WaitForSeconds
private WaitForSeconds _oneSecondWait = new WaitForSeconds(1f);
IEnumerator MyCoroutineGood()
{
    while (true)
    {
        yield return _oneSecondWait;
        // ...
    }
}
String handling pattern
// BAD: Concatenate every frame
void Update()
{
    scoreText.text = "Score: " + playerScore + " - Lives: " + lives;
}

// BETTER: Update only when values change
private int _lastScore = -1;
void UpdateScoreUI()
{
    if (_lastScore != playerScore)
    {
        _lastScore = playerScore;
        scoreText.text = $"Score: {playerScore} - Lives: {lives}";
    }
}
Measure after these refactors. Removing repeated allocations in hot paths yields visible drops in GC allocation rate and fewer small-object allocations on the Profiler timeline.
