Code example
// Simple Object Pool Example
public class BulletPool : MonoBehaviour
{
    public GameObject bulletPrefab;
    public int poolSize = 20;
    private Queue<GameObject> _bulletPool = new Queue<GameObject>();

    void Awake()
    {
        for (int i = 0; i < poolSize; i++)
        {
            GameObject bullet = Instantiate(bulletPrefab);
            bullet.SetActive(false);
            _bulletPool.Enqueue(bullet);
        }
    }

    public GameObject GetBullet()
    {
        if (_bulletPool.Count > 0)
        {
            GameObject bullet = _bulletPool.Dequeue();
            bullet.SetActive(true);
            return bullet;
        }
        else
        {
            Debug.LogWarning("Bullet pool exhausted, instantiating new bullet.");
            return Instantiate(bulletPrefab);
        }
    }

    public void ReturnBullet(GameObject bullet)
    {
        bullet.SetActive(false);
        _bulletPool.Enqueue(bullet);
    }
}
Walkthrough & tips
  • Pre-allocate in Awake to avoid runtime spikes.
  • Reset object state on enable: position, velocity, timers.
  • Decide whether to expand the pool or log a warning on exhaustion.
  • Parent pooled objects under the spawner for Hierarchy cleanliness.
Key Takeaway

Replace Instantiate/Destroy with Get/Return calls and reset state to prevent unexpected behavior and reduce allocations.
