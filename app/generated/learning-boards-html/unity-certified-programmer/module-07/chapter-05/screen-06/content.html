Starter code (leaky)
// LeakyManager.cs
using UnityEngine;
using System.Collections.Generic;

public class LeakyManager : MonoBehaviour
{
  public GameObject leakyPrefab;
  private List _leakedObjects = new List();
  private List _leakedStrings = new List();

  void Update()
  {
    if (Input.GetKeyDown(KeyCode.Space))
    {
      GameObject newObj = Instantiate(leakyPrefab, transform.position, Quaternion.identity);
      _leakedObjects.Add(newObj);
      _leakedStrings.Add("Leaked String " + Time.frameCount);
    }

    if (Input.GetKeyDown(KeyCode.Return))
    {
      Resources.UnloadUnusedAssets();
    }
  }
}
Fixed pattern
// Fixed example
void OnUserTriggerUnload()
{
  foreach (GameObject obj in _leakedObjects)
  {
    Destroy(obj);
  }
  _leakedObjects.Clear();
  _leakedStrings.Clear();
  Resources.UnloadUnusedAssets();
}
Concrete result
After clearing references and destroying objects, snapshots show reduced managed counts and smaller retained sizes for related assets.
Misconception
Destroying objects without clearing the lists that reference them prevents GC from reclaiming memory; both steps are required.
