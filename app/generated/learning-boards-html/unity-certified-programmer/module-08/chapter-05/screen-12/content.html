Starter code
// PlatformQualityManager.cs (attach to GameObject)
using UnityEngine;
using UnityEngine.UI;

public class PlatformQualityManager : MonoBehaviour
{
    public Text infoText; // Assign in Inspector

    void Start()
    {
        if (infoText == null) return;

        QualitySettings.SetQualityLevel(1);
        Application.targetFrameRate = 60;

#if UNITY_ANDROID || UNITY_IOS
        QualitySettings.SetQualityLevel(0);
        Application.targetFrameRate = 30;
#endif

        UpdateInfoText();
    }

    void Update()
    {
        UpdateInfoText();
    }

    void UpdateInfoText()
    {
        infoText.text = $"Platform: {Application.platform}\nQuality: {QualitySettings.names[QualitySettings.GetQualityLevel()]}\nTarget FPS: {Application.targetFrameRate}\nActual FPS: {Mathf.RoundToInt(1.0f / Time.unscaledDeltaTime)}";
    }
}
Steps
  1. Create InfoText UI and assign it in the Inspector.
  2. Attach script, switch platform, and run in Editor to observe different code paths.
  3. Build to device and profile with Unity Profiler for real results.
Debug tip
Use scripting define symbols for development overlays and wrap them in #if so release builds remain clean.
Key Takeaway

This lab shows how compile-time directives and Player Settings produce reproducible platform differences you can observe and measure.
