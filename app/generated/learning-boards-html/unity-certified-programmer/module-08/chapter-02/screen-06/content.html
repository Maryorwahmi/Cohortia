Module 08 › Chapter 8.2 › PlayerWeapon Example
Example Code
public class PlayerWeapon : MonoBehaviour
{
    public WeaponData currentWeapon; // Reference to a ScriptableObject asset
    private float nextFireTime;

    void Update()
    {
        if (currentWeapon == null) return;
        if (Input.GetButton("Fire1") && Time.time >= nextFireTime)
        {
            FireWeapon();
            nextFireTime = Time.time + 1f / currentWeapon.fireRate;
        }
    }

    void FireWeapon()
    {
        // Use data from the asset to control behavior
        if (currentWeapon.projectilePrefab != null)
        {
            Instantiate(currentWeapon.projectilePrefab, transform.position + transform.forward, transform.rotation);
        }
    }
}
Annotated Notes
  • Reference the asset via a public field so designers can assign it.
  • Read fields at runtime to drive timing, damage, and prefab instantiation.
  • Check for null asset or missing prefab to avoid runtime errors.
Analogy: asset is the blueprint; MonoBehaviour is the contractor that follows it.
Misconception: ScriptableObjects are not a replacement for component state like health or position during play.
