Excerpt
[CustomEditor(typeof(ItemData))]
public class ItemDataEditor : Editor
{
    public override void OnInspectorGUI()
    {
        ItemData itemData = (ItemData)target;
        EditorGUILayout.LabelField("Item Configuration", EditorStyles.boldLabel);
        itemData.itemName = EditorGUILayout.TextField("Item Name", itemData.itemName);
        if (GUILayout.Button("Simulate Use Item"))
        {
            itemData.UseItem();
        }
        if (GUI.changed) EditorUtility.SetDirty(itemData);
    }
}
Practical Tips
  • Use EditorGUILayout for consistent layout across editor themes.
  • Mark objects dirty after changes to ensure persistence.
  • Provide clear labels and small helper buttons for frequent tasks.
Pro Tip
Keep inspectors focused: if a tool grows complex, extract it into an Editor Window instead of cramming too many controls into the Inspector.
