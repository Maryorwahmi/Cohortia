Code Walkthrough AIAgentPool
Inspect the pool mechanics and practical considerations for use in a scene.
// Simplified view of the example from the lesson content
public class AIAgentPool : MonoBehaviour
{
    public GameObject agentPrefab;
    public int poolSize = 10;
    private Queue<GameObject> agentPool = new Queue<GameObject>();

    void Awake()
    {
        for (int i = 0; i < poolSize; i++)
        {
            GameObject agent = Instantiate(agentPrefab);
            agent.SetActive(false);
            agentPool.Enqueue(agent);
        }
    }

    public GameObject GetAgent()
    {
        if (agentPool.Count > 0)
        {
            GameObject agent = agentPool.Dequeue();
            agent.SetActive(true);
            return agent;
        }
        else
        {
            Debug.LogWarning("Pool exhausted, instantiating new agent. Consider increasing pool size.");
            return Instantiate(agentPrefab);
        }
    }

    public void ReturnAgent(GameObject agent)
    {
        agent.SetActive(false);
        agentPool.Enqueue(agent);
    }
}
Walkthrough Notes
  • Awake pre-populates the queue to avoid runtime instantiation spikes.
  • GetAgent activates an object; remember to initialize agent-specific state there.
  • ReturnAgent deactivates and re-enqueues but must also clear references and timers.
  • Fallback Instantiate reduces failure but may reintroduce spikes; prefer growing pool at low risk times.
Example Caveat
When agents have NavMeshAgents, ensure NavMesh state or path data is reset so the reused agent doesn't resume a stale path or cause movement glitches.
Check: Run profiler before and after enabling pool. Spawn the same number of agents and compare spawn-time CPU and GC frequency.
