SphereLauncher.cs
using UnityEngine;

public class SphereLauncher : MonoBehaviour
{
    [SerializeField] private float launchForce = 500f;
    [SerializeField] private Vector3 launchDirection = Vector3.forward;
    private Rigidbody rb;

    void Start()
    {
        rb = GetComponent<Rigidbody>();
        if (rb == null)
        {
            Debug.LogError("SphereLauncher requires a Rigidbody component!", this);
            enabled = false;
            return;
        }

        launchDirection = launchDirection.normalized;
        rb.AddForce(launchDirection * launchForce, ForceMode.Impulse);
        Debug.Log($"Launched sphere with force: {launchForce} in direction: {launchDirection}");
    }
}
Notes & best practices
  • Use SerializeField to make variables editable in the Inspector while keeping them private.
  • Normalize direction to ensure predictable force magnitude.
  • Disable the script if the Rigidbody is missing to avoid null-reference issues.
Try this
Set launchDirection to Vector3.up to make the sphere jump instead of moving forward.
