Add listeners and initialize
public class UIManager : MonoBehaviour
{
public Slider volumeSlider;
public Toggle musicToggle;
public TMP_InputField playerNameInputField;
void Start()
{
if (volumeSlider != null) {
volumeSlider.onValueChanged.AddListener(SetVolume);
volumeSlider.value = AudioListener.volume;
}
if (musicToggle != null) {
musicToggle.onValueChanged.AddListener(ToggleMusic);
musicToggle.isOn = AudioListener.volume > 0;
}
if (playerNameInputField != null) {
playerNameInputField.onEndEdit.AddListener(SetPlayerName);
}
}
void SetVolume(float v) { AudioListener.volume = v; Debug.Log("Volume set to: " + v); }
void ToggleMusic(bool on) { AudioListener.volume = on ? 1f : 0f; Debug.Log("Music: " + on); }
void SetPlayerName(string n) { Debug.Log("Player: " + n); }
}
Explanation
  • Always null-check public fields before adding listeners to avoid runtime errors.
  • Initialize UI values from the current game state so the UI reflects reality when opened.
  • Use onEndEdit for final text capture to avoid per-character processing cost.
Pro tip
Use Debug.Log inside handlers during development to confirm events are firing before adding game logic.
Key Takeaway

Programmatic listeners give explicit control and make initialization straightforward, but remember to match expected parameter types.
