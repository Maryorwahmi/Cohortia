Implement PlayerHealth & DamageDealer

public class PlayerHealth : MonoBehaviour
{
    public int currentHealth = 100;
    public int maxHealth = 100;

    public void TakeDamage(int amount)
    {
        currentHealth -= amount;
        if (currentHealth < 0) currentHealth = 0;
        Debug.Log("Player took " + amount + " damage. Current Health: " + currentHealth);
        if (currentHealth == 0) Die();
    }

    private void Die() { Debug.Log("Player has been defeated!"); gameObject.SetActive(false); }
}

public class DamageDealer : MonoBehaviour
{
    public int damageAmount = 25;
    void OnTriggerEnter(Collider other)
    {
        PlayerHealth playerHealth = other.GetComponent();
        if (playerHealth != null) playerHealth.TakeDamage(damageAmount);
    }
}
Testing checklist
  • Add PlayerHealth to a player GameObject.
  • Create DamageZone with BoxCollider (Is Trigger) and Rigidbody (Is Kinematic).
  • Attach DamageDealer and set damageAmount, then run and observe Debug.Log messages.
Common mistake
Do not use new PlayerHealth(); it creates an unattached instance instead of the component on the GameObject.
