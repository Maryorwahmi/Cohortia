Starter code
A conceptual, in-memory registry that supports register and discover operations.
import random

class ServiceRegistry:
  def __init__(self):
    self.services = {}  # {'service_name': [{'ip': '...', 'port': '...'}, ...]}

  def register(self, service_name, ip, port):
    if service_name not in self.services:
      self.services[service_name] = []
    self.services[service_name].append({'ip': ip, 'port': port})
    print(f"Registered {service_name} instance: {ip}:{port}")

  def discover(self, service_name):
    instances = self.services.get(service_name, [])
    if not instances:
      print(f"No instances found for {service_name}")
      return None
    chosen_instance = random.choice(instances)
    print(f"Discovered {service_name} instance: {chosen_instance['ip']}:{chosen_instance['port']}")
    return chosen_instance

# Simulate usage:
registry = ServiceRegistry()
registry.register('ProductService', '192.168.1.100', '8081')
registry.register('ProductService', '192.168.1.101', '8081')
for _ in range(3):
  registry.discover('ProductService')
Run this locally to see registration prints and different discovery choices. Expand it by adding heartbeats and TTLs for more realism.
Limitations to note
  • No persistence—state lost on restart.
  • No health checks or TTL-based removal.
  • No authentication or access controls for registration calls.
