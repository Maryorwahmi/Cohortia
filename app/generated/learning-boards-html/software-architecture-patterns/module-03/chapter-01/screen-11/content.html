package com.cohortia.productservice.controller;

import com.cohortia.productservice.model.Product;
import com.cohortia.productservice.repository.ProductRepository;
import org.springframework.web.bind.annotation.*;

@RestController
@RequestMapping("/products")
public class ProductController {
  private final ProductRepository repo;
  public ProductController(ProductRepository repo) { this.repo = repo; }

  @PostMapping
  public Product create(@RequestBody Product p) { return repo.save(p); }

  @GetMapping("/{id}")
  public Product get(@PathVariable Long id) { return repo.findById(id).orElse(null); }

  @PutMapping("/{id}")
  public Product update(@PathVariable Long id, @RequestBody Product p) {
    p.setId(id);
    return repo.save(p);
  }
}

Terminal demo
# Create a product
curl -s -X POST -H "Content-Type: application/json" -d '{"name":"Lamp","description":"Desk lamp","price":19.99}' http://localhost:8080/products

# Retrieve the created product
curl -s http://localhost:8080/products/1
These commands create and read product data in ProductService's private H2 database. They demonstrate ownership and local persistence.
