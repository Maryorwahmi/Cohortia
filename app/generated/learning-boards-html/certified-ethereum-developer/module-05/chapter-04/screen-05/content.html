Code Example
// src/components/TokenBalanceDisplay.tsx
'use client';

import { useEffect, useState } from 'react';
import { ethers } from 'ethers';
import { ERC20_ABI } from '../utils/abis';

export default function TokenBalanceDisplay({ provider, account, tokenAddress }) {
  const [balance, setBalance] = useState('0');
  const [symbol, setSymbol] = useState('');
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState(null);

  useEffect(() => {
    async function fetchTokenData() {
      if (!provider || !account || !tokenAddress) {
        setBalance('0');
        setSymbol('');
        return;
      }
      setIsLoading(true);
      setError(null);
      try {
        const tokenContract = new ethers.Contract(tokenAddress, ERC20_ABI, provider);
        const rawBalance = await tokenContract.balanceOf(account);
        const decimals = await tokenContract.decimals();
        const formattedBalance = ethers.formatUnits(rawBalance, decimals);
        const tokenSymbol = await tokenContract.symbol();
        setBalance(formattedBalance);
        setSymbol(tokenSymbol);
      } catch (err) {
        setError(`Failed to fetch token data: ${err.message}`);
        setBalance('0');
        setSymbol('');
      } finally {
        setIsLoading(false);
      }
    }
    fetchTokenData();
  }, [provider, account, tokenAddress]);

  // Render uses conditional UI for loading/error/empty
}
Explanation
  • Use an effect to fetch balance and metadata when provider, account, or address change.
  • Convert raw units with ethers.formatUnits so values are human-readable.
  • Keep loading and error state to guide the user and avoid blank UI.
Concrete example

A user connects MetaMask, the component shows a spinner while fetching, then displays "Your CHT Balance: 123.45". If fetch fails, show "Failed to fetch token data" with a retry action in the page layer.

Key Takeaway

Always format on-chain values and make loading and errors explicit to avoid confusing raw numbers.
