ImageStorage contract
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract ImageStorage {
    string public currentImageCID;
    address public owner;

    event ImageCIDUpdated(address indexed updater, string newCID);

    constructor() {
        owner = msg.sender;
        currentImageCID = "QmUNLLsPekqG2qK1V2s6rYf2A1C1C1C1C1C1C1C1C"; // Default placeholder CID
    }

    function setImageCID(string memory _newCID) public {
        require(msg.sender == owner, "Only owner can set CID");
        currentImageCID = _newCID;
        emit ImageCIDUpdated(msg.sender, _newCID);
    }
}
The contract stores only a small string and emits an event when updated so the frontend can react to changes.
Practical tip
Use events to update the UI instead of polling. Keep storage minimal to reduce gas across transactions.
Access control
Restrict who can change CIDs to avoid malicious replacements; the example uses a simple owner pattern.
