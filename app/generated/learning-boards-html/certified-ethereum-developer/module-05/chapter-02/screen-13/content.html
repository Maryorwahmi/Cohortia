Example deploy.js
const hre = require("hardhat");

async function main() {
  const MyContract = await hre.ethers.getContractFactory("MyContract");
  const myContract = await MyContract.deploy();
  await myContract.waitForDeployment();

  console.log(`MyContract deployed to: ${myContract.target}`);
}

main().catch((error) => {
  console.error(error);
  process.exitCode = 1;
});
After running the script with --network localhost, copy the printed address into your .env.local as NEXT_PUBLIC_CONTRACT_ADDRESS and ensure the frontend imports the contract ABI from contracts/.
Verification
Call a read-only method like greeting in the deployed MyContract from the frontend and confirm the returned value matches the contract state.
Pro Tip

Automate address propagation to the frontend with a small script that writes .env.local after deployment during local CI or dev workflows.
