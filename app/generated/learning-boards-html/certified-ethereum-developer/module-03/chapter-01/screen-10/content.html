Build Ownable, Pausable, and MyToken
// Ownable
contract Ownable {
  address public owner;
  constructor() { owner = msg.sender; }
  modifier onlyOwner() { require(msg.sender == owner, "Only owner can call this function"); _; }
}

// Pausable inherits Ownable
contract Pausable is Ownable {
  bool public paused = false;
  modifier whenNotPaused() { require(!paused, "Contract is paused"); _; }
  function pause() public onlyOwner whenNotPaused { paused = true; }
  function unpause() public onlyOwner { paused = false; }
}

// MyToken inherits Pausable
contract MyToken is Pausable {
  mapping(address => uint256) public balances;
  function transfer(address recipient, uint256 amount) public whenNotPaused returns (bool) { /* ... */ }
}
Checklist
  • Implement onlyOwner and transferOwnership safely.
  • Ensure whenNotPaused prevents transfers while paused.
  • Test pause/unpause and validate transfer behavior in both states.
Scenario
If deployer pauses the contract before a token sale, all transfers must revert until unpause is called by the owner.
Key Takeaway

Compose simple, audited base contracts for cross-cutting concerns and test their interactions thoroughly.
