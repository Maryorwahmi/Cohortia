Module 7 › Chapter 6.7 › Hardhat test example
Representative test (excerpt)
// test/MyToken.test.js
const { expect } = require("chai");
const { ethers } = require("hardhat");

describe("MyToken", function () {
  let MyToken, myToken, owner, addr1, addr2;

  beforeEach(async function () {
    MyToken = await ethers.getContractFactory("MyToken");
    [owner, addr1, addr2] = await ethers.getSigners();
    myToken = await MyToken.deploy(1000);
    await myToken.deployed();
  });

  it("transfers tokens between accounts", async function () {
    await myToken.transfer(addr1.address, 50);
    expect(await myToken.balanceOf(addr1.address)).to.equal(50);
    await myToken.connect(addr1).transfer(addr2.address, 50);
    expect(await myToken.balanceOf(addr2.address)).to.equal(50);
  });
});
Explanation
  • beforeEach deploys a fresh contract instance to isolate tests.
  • Use .connect(signer) to simulate transactions from different accounts in the suite.
  • Assert both success cases and expected revert messages for error conditions.
Common mistake
Sharing state between tests causes order-dependent failures; isolate with fresh deploys.
Key Takeaway

Write tests that read like examples: arrange setup, act, then assert results and failures.
