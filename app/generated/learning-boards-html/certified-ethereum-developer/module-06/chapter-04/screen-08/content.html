pragma solidity ^0.8.0;

import "@openzeppelin/contracts/access/AccessControl.sol";

contract MyAccessControlledToken is AccessControl {
  bytes32 public constant MINTER_ROLE = keccak256("MINTER_ROLE");
  bytes32 public constant PAUSER_ROLE = keccak256("PAUSER_ROLE");

  constructor() {
    _setupRole(DEFAULT_ADMIN_ROLE, msg.sender);
    _setupRole(MINTER_ROLE, msg.sender);
  }

  function mint(address to, uint256 amount) public onlyRole(MINTER_ROLE) {
    // minting logic
  }
}
What to watch for
  • DEFAULT_ADMIN_ROLE holders can grant and revoke roles; protect it carefully.
  • Roles are represented by bytes32 constants and checked by onlyRole.
  • Grant only what is necessary to each role and test every protected function.
Concrete example
Grant MINTER_ROLE to a minting service and use onlyRole to prevent unauthorized minting calls.
Pro tip
Consider using a multi-sig to hold DEFAULT_ADMIN_ROLE to protect role management operations.
