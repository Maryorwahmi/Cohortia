pragma solidity ^0.8.0;

import "@openzeppelin/contracts/access/AccessControl.sol";

contract MyRBACControlledToken is AccessControl {
  bytes32 public constant MINTER_ROLE = keccak256("MINTER_ROLE");
  bytes32 public constant PAUSER_ROLE = keccak256("PAUSER_ROLE");

  uint256 public totalSupply;
  mapping(address => uint256) public balances;
  bool public pausedState = false;

  constructor() {
    _setupRole(DEFAULT_ADMIN_ROLE, msg.sender);
    _setupRole(MINTER_ROLE, msg.sender);
    _setupRole(PAUSER_ROLE, msg.sender);
  }

  function mint(address to, uint256 amount) public onlyRole(MINTER_ROLE) {
    require(!pausedState, "Contract is paused");
    totalSupply += amount;
    balances[to] += amount;
  }
}
Testing checklist
  • Verify non-minter cannot mint and that grantRole allows minting.
  • Pause the contract and verify minting is blocked until unpause.
  • Revoke roles and confirm revoked accounts lose abilities immediately.
Hands-on activity
Implement and deploy MyRBACControlledToken in Remix, then run the test checklist to validate role behavior.
Key takeaway
Combine role checks with state validation and test both negative and positive cases thoroughly.
