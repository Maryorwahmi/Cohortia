Module 6 › Chapter 6.6 › Hands-on: Simple Oracle
Starter Code
Deploy these two contracts locally. Then change price and observe the dependent contract's output.
<!-- Escaped Solidity starter code -->
<!-- SPDX-License-Identifier: MIT -->
<!-- pragma solidity ^0.8.0; -->

<!-- contract SimplePriceOracle { -->
<!--     address public owner; -->
<!--     uint256 public currentPrice; -->

<!--     constructor() { -->
<!--         owner = msg.sender; -->
<!--         currentPrice = 10000; -->
<!--     } -->

<!--     modifier onlyOwner() { -->
<!--         require(msg.sender == owner, "Not owner"); -->
<!--         _; -->
<!--     } -->

<!--     function setPrice(uint256 _newPrice) public onlyOwner { -->
<!--         currentPrice = _newPrice; -->
<!--     } -->

<!--     function getPrice() public view returns (uint256) { -->
<!--         return currentPrice; -->
<!--     } -->
<!-- } -->

<!-- contract PriceDependentContract { -->
<!--     SimplePriceOracle public oracle; -->
<!--     uint256 public constant BASE_VALUE = 1 ether; -->

<!--     constructor(address _oracleAddress) { -->
<!--         oracle = SimplePriceOracle(_oracleAddress); -->
<!--     } -->

<!--     function calculateValueBasedOnPrice() public view returns (uint256) { -->
<!--         uint256 price = oracle.getPrice(); -->
<!--         return (BASE_VALUE * price) / 100; -->
<!--     } -->
<!-- } -->
Activity steps
  1. Deploy SimplePriceOracle and set price to 10000.
  2. Deploy PriceDependentContract with the oracle address.
  3. Call calculateValueBasedOnPrice and record the output.
  4. Change the oracle price to 1000 or 100000 and observe the change immediately.
  5. Discuss mitigation options and implement one if you have time.
Expected learning
See how a single owner can control on-chain inputs and cause downstream contracts to behave unexpectedly.
Reflection

How would adding a 10-minute TWAP or requiring two independent updates change the exploitability?
