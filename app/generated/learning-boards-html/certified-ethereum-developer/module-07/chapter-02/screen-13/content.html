Contract example
// contracts/MyToken.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "@openzeppelin/contracts/token/ERC20/ERC20.sol";

contract MyToken is ERC20 {
    constructor(string memory name, string memory symbol, uint256 initialSupply) ERC20(name, symbol) {
        _mint(msg.sender, initialSupply);
    }
}
Deployment script
// scripts/deployMyToken.js
const { ethers } = require("hardhat");

async function main() {
  const [deployer] = await ethers.getSigners();
  console.log("Deploying MyToken with the account:", deployer.address);

  const initialSupply = ethers.utils.parseUnits("1000000", 18);
  const MyToken = await ethers.getContractFactory("MyToken");
  const myToken = await MyToken.deploy("Cohortia Token", "CHT", initialSupply);
  await myToken.deployed();
  console.log("MyToken deployed to:", myToken.address);
  console.log("Initial supply minted to deployer:", (await myToken.balanceOf(deployer.address)).toString());
}

main().catch((error) => { console.error(error); process.exit(1); });
Hands-on steps
  1. Install OpenZeppelin and create the MyToken.sol contract.
  2. Configure SEPOLIA_RPC_URL and PRIVATE_KEY in .env.
  3. Fund your deployer from a Sepolia faucet.
  4. Run npx hardhat run scripts/deployMyToken.js --network sepolia and copy the address.
  5. Verify the contract and confirm the minted balance matches expectations.
