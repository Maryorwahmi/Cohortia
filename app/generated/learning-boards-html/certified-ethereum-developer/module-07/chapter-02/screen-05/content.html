Example hardhat.config.js

// hardhat.config.js
require("@nomicfoundation/hardhat-toolbox");
require("dotenv").config();

const SEPOLIA_RPC_URL = process.env.SEPOLIA_RPC_URL || "https://eth-sepolia.g.alchemy.com/v2/YOUR_ALCHEMY_API_KEY";
const MAINNET_RPC_URL = process.env.MAINNET_RPC_URL || "https://eth-mainnet.g.alchemy.com/v2/YOUR_ALCHEMY_API_KEY";
const PRIVATE_KEY = process.env.PRIVATE_KEY;

module.exports = {
  solidity: "0.8.20",
  networks: {
    hardhat: {},
    sepolia: {
      url: SEPOLIA_RPC_URL,
      accounts: PRIVATE_KEY !== undefined ? [PRIVATE_KEY] : [],
      chainId: 11155111,
    },
    mainnet: {
      url: MAINNET_RPC_URL,
      accounts: PRIVATE_KEY !== undefined ? [PRIVATE_KEY] : [],
      chainId: 1,
    },
  },
  etherscan: { apiKey: process.env.ETHERSCAN_API_KEY },
};
Explanation & Checklist
  • Always load secrets with dotenv and process.env to avoid committing keys.
  • Provide accounts as an array when PRIVATE_KEY exists; otherwise keep blank for local testing.
  • Match chainId to the network to avoid signing errors at send time.
  • Include plugins like etherscan and gasReporter if you intend to verify or measure gas.
Concrete example
If you set SEPOLIA_RPC_URL to your Alchemy endpoint and place your deployer PRIVATE_KEY in .env, running a deploy with --network sepolia will sign with that key and send the transaction to Sepolia.
Key Takeaway

Validate your config by compiling and running a short deploy to confirm network connectivity and signing behavior.
