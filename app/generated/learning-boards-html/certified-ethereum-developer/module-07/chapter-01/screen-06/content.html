Example Script
// scripts/deploy.js
const { ethers } = require("hardhat");

async function main() {
  const [deployer] = await ethers.getSigners();

  console.log("Deploying contracts with the account:", deployer.address);
  console.log("Account balance:", (await deployer.getBalance()).toString());

  const MyToken = await ethers.getContractFactory("MyToken");
  const myToken = await MyToken.deploy("My Awesome Token", "MAT");

  await myToken.deployed();

  console.log("MyToken deployed to:", myToken.address);
}

main()
  .then(() => process.exit(0))
  .catch((error) => {
    console.error(error);
    process.exit(1);
  });
What to notice
  • getSigners returns accounts available to the configured provider; the first is used as deployer.
  • getContractFactory prepares compiled artifacts and a factory with deploy convenience.
  • deploy returns a contract instance; await deployed waits for mining confirmation.
Pro tip
Add logic to write the deployed address and ABI to a JSON file immediately after deployment. This makes front-end integration repeatable and avoids manual copy-paste errors.
