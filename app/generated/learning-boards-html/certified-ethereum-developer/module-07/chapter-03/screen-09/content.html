Contract
Greeter stores and returns a greeting string. Below is the contract source as used in the example.
// src/Greeter.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract Greeter {
  string private _greeting;

  constructor(string memory greeting_) { _greeting = greeting_; }

  function greet() public view returns (string memory) { return _greeting; }

  function setGreeting(string memory greeting_) public { _greeting = greeting_; }
}
Deployment Script
This script demonstrates the vm.startBroadcast pattern and logs the deployed address.
// script/DeployGreeter.s.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "forge-std/Script.sol";
import "../src/Greeter.sol";

contract DeployGreeter is Script {
  function run() public returns (Greeter greeter) {
    vm.startBroadcast();
    greeter = new Greeter("Hello, Foundry!");
    vm.stopBroadcast();
    console.log("Greeter deployed to:", address(greeter));
  }
}
Remember to set environment variables before broadcasting.
