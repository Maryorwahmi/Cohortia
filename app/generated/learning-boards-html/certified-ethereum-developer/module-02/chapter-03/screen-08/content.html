// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract AccessControl {
  address public owner;
  uint private _value;

  modifier onlyOwner() {
    require(msg.sender == owner, "Only the contract owner can call this function.");
    _;
  }

  constructor() {
    owner = msg.sender;
    _value = 0;
  }

  function setValue(uint newValue) public onlyOwner {
    _value = newValue;
  }

  function getValueExternal() external view returns (uint) {
    return _value;
  }

  function calculateSum(uint a, uint b) public pure returns (uint) {
    return a + b;
  }
}
Testing checklist
  • Call setValue from non-owner and expect revert with the modifier message.
  • Call getValueExternal from an external address — it should return the value and be gas-free off-chain.
  • Check that calculateSum returns deterministic results and does not access storage.
Concept callouts
owner is public; _value is private; onlyOwner manages access; view and pure show mutability intent.
Key takeaway
Reading example contracts shows how visibility, mutability, modifiers, and error checks combine to express both behavior and security intent.
