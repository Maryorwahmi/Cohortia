// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "@openzeppelin/contracts/token/ERC721/ERC721.sol";
import "@openzeppelin/contracts/access/Ownable.sol";
import "@openzeppelin/contracts/utils/Counters.sol";

contract MyNFT is ERC721, Ownable {
  using Counters for Counters.Counter;
  Counters.Counter private _tokenIdCounter;

  string private _baseTokenURI;

  constructor(string memory name_, string memory symbol_, string memory baseTokenURI_)
    ERC721(name_, symbol_)
    Ownable(msg.sender)
  {
    _baseTokenURI = baseTokenURI_;
  }

  function _baseURI() internal view override returns (string memory) {
    return _baseTokenURI;
  }

  function setBaseURI(string memory baseTokenURI_) public onlyOwner {
    _baseTokenURI = baseTokenURI_;
  }
}
Explanation cards
Counters.Counter
Provides safe incrementing tokenIds; avoids collisions when minting sequential tokens.
Ownable
Restricts sensitive functions like mint and setBaseURI to the contract owner to protect supply and metadata.
_baseURI()
Returns the base URI used by tokenURI to compose the metadata location for each tokenId.
