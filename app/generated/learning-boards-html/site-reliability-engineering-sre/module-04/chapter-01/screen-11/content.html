Python Script: Service Restart Example
import subprocess
import time

SERVICE_NAME = "nginx"

def check_service_status(service):
    cmd = ["systemctl", "is-active", service]
    result = subprocess.run(cmd, capture_output=True, text=True)
    return result.stdout.strip()

def restart_service(service):
    print(f"Attempting to restart {service}...")
    cmd = ["sudo", "systemctl", "restart", service]
    subprocess.run(cmd, check=True)
    print(f"{service} restarted successfully.")

if __name__ == "__main__":
    status = check_service_status(SERVICE_NAME)
    print(f"{SERVICE_NAME} status: {status}")
    
    if status != "active":
        print(f"{SERVICE_NAME} is not active. Restarting...")
        try:
            restart_service(SERVICE_NAME)
            time.sleep(5) # Give service time to start
            if check_service_status(SERVICE_NAME) == "active":
                print(f"{SERVICE_NAME} is now active.")
            else:
                print(f"Failed to activate {SERVICE_NAME} after restart.")
        except subprocess.CalledProcessError as e:
            print(f"Error restarting {SERVICE_NAME}: {e}")
    else:
        print(f"{SERVICE_NAME} is already active.")
Code Explanation

This Python script checks the status of a service (e.g., Nginx) using `systemctl`. If the service is not active, it attempts to restart it and then verifies its status again. This demonstrates basic automated health checking and remediation.
