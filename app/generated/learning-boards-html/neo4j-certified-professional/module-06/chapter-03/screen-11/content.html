Starter Schema (file: schema.graphql)
# Define the User type
type User {
  id: ID!
  username: String!
  email: String
  posts: [Post!]
  friends: [User!]
}

# Define the Post type
type Post {
  id: ID!
  content: String!
  timestamp: String!
  author: User
}

# Define the root Query type
type Query {
  user(username: String!): User
  post(id: ID!): Post
}
Playground Query
query GetUserAndFriends($username: String!) {
  user(username: $username) {
    username
    email
    friends {
      username
      email
    }
  }
}
