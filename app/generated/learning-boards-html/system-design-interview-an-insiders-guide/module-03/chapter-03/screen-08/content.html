LRU Cache — Starter Code
A compact implementation using OrderedDict for access ordering.
Code
from collections import OrderedDict

class LRUCache:
  def __init__(self, capacity: int):
    self.cache = OrderedDict()
    self.capacity = capacity

  def get(self, key: int) -> int:
    if key not in self.cache:
      return -1
    # Move the accessed item to the end to mark it as most recently used
    value = self.cache.pop(key)
    self.cache[key] = value
    return value

  def put(self, key: int, value: int) -> None:
    if key in self.cache:
      # If key exists, update its value and move to end
      self.cache.pop(key)
    elif len(self.cache) >= self.capacity:
      # If cache is full, evict the least recently used item (first item)
      self.cache.popitem(last=False)
    # Add the new item (or updated item) to the end
    self.cache[key] = value

# Example Usage:
cache = LRUCache(2)
cache.put(1, 1)
cache.put(2, 2)
print(f"Get(1): {cache.get(1)}")    # returns 1
cache.put(3, 3)                     # evicts key 2
print(f"Get(2): {cache.get(2)}")    # returns -1 (not found)
cache.put(4, 4)                     # evicts key 1
print(f"Get(1): {cache.get(1)}")    # returns -1 (not found)
print(f"Get(3): {cache.get(3)}")    # returns 3
print(f"Get(4): {cache.get(4)}")    # returns 4
Explanation & Example
The example usage shows the movement of items to the most-recently-used end and eviction of the least recently used entry when capacity is reached.
Common mistake
Not updating recency on get will degrade LRU behavior and evict active items.
Key Takeaway

LRU requires updating order on both get and put; for production add thread-safety or use a distributed cache for scale.
