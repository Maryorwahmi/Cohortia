Starter Code
from collections import deque

class LRUCache:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.cache = {}  # Stores key: value
        self.order = deque() # Stores keys in order of recency (MRU at right, LRU at left)

    def _mark_as_recently_used(self, key):
        # Remove key from its current position and add to the right (MRU)
        if key in self.order:
            self.order.remove(key)
        self.order.append(key)

    def get(self, key: int) -> int:
        if key not in self.cache:
            return -1
        # Mark as recently used
        self._mark_as_recently_used(key)
        return self.cache[key]

    def put(self, key: int, value: int) -> None:
        if key in self.cache:
            self.cache[key] = value
            self._mark_as_recently_used(key)
        else:
            if len(self.cache) >= self.capacity:
                # Evict LRU item
                lru_key = self.order.popleft()
                del self.cache[lru_key]
            self.cache[key] = value
            self.order.append(key)
Note: deque.remove is O(n) in CPython. For strict O(1) removals use a custom doubly-linked list or an ordered map that supports node moves.
Walkthrough
  • get: lookup in map, if present move key to MRU end and return value.
  • put(existing): update value and mark recent.
  • put(new): if full, remove LRU from left and from map, then insert new key at right.
Example: capacity 2, put(1), put(2), get(1) => order [2,1]; put(3) evicts 2.
