Starter code
from flask import Flask, request, jsonify
import time
import random
from prometheus_client import Counter, generate_latest, Histogram

app = Flask(__name__)

HTTP_REQUESTS_TOTAL = Counter('http_requests_total', 'Total HTTP requests', ['method','endpoint','status'])
HTTP_REQUEST_DURATION_SECONDS = Histogram('http_request_duration_seconds', 'HTTP request duration in seconds', ['method','endpoint'])

@app.route('/data', methods=['GET'])
def get_data():
  start_time = time.time()
  status_code = 200
  try:
    time.sleep(random.uniform(0.05, 0.2))
    if random.random() < 0.10:
      status_code = 500
      raise ValueError('Simulated error')
    return jsonify({'message':'ok'}), status_code
  except Exception as e:
    return jsonify({'message':str(e)}), status_code
  finally:
    HTTP_REQUESTS_TOTAL.labels(request.method, request.path, status_code).inc()
    HTTP_REQUEST_DURATION_SECONDS.labels(request.method, request.path).observe(time.time() - start_time)

@app.route('/metrics')
def metrics():
  return generate_latest(), 200
Notes
  • Keep label cardinality low.
  • Observe durations in seconds consistently.
  • Increment counters with status as a label for error-rate queries.
Try this

Run the app, hit /data repeatedly, then view /metrics to confirm counters and histogram are present.
