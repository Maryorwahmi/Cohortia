Starter Flask code
from flask import Flask, jsonify
import random

app = Flask(__name__)

# Simulated DB connection flag
db_connected = False

def check_db_connection():
  global db_connected
  if not db_connected:
    if random.random() > 0.3:
      db_connected = True
  if db_connected and random.random() < 0.05:
    db_connected = False
  return db_connected

@app.route('/healthz')
def liveness_probe():
  return jsonify({"status": "UP"}), 200

@app.route('/readyz')
def readiness_probe():
  if check_db_connection():
    return jsonify({"status": "READY"}), 200
  return jsonify({"status": "NOT_READY", "reason": "Database connection failed"}), 503

if __name__ == '__main__':
  check_db_connection()
  app.run(host='0.0.0.0', port=5000)
