Building an Asynchronous Email Sender

Extending Celery for Real-World Scenarios

Scenario & Task

You have a web service that needs to send a large number of personalized emails. Sending each email synchronously would be too slow and block the UI.

Task: Extend the Celery example to create an asynchronous email sending service.

# Modify proj/tasks.py
# ... (existing code) ...
import random

@app.task
def send_personalized_email(recipient_email, subject, body_template):
    print(f"Sending email to {recipient_email} - Subject: '{subject}'")
    time.sleep(random.uniform(1, 3)) # Simulate email sending time
    final_body = body_template.replace("{recipient}", recipient_email.split('@')[0])
    print(f"Email to {recipient_email} sent. Body snippet: '{final_body[:50]}...' ")
    return {"status": "sent", "recipient": recipient_email, "subject": subject}
# Modify client.py
from proj.tasks import add, process_image, send_personalized_email
import time

print("Sending tasks...")
# ... (existing add and process_image tasks) ...

print("
Sending email tasks...")
email_tasks = []
recipients = [
    "alice@example.com", "bob@example.com", "charlie@example.com",
    "david@example.com", "eve@example.com", "frank@example.com"
]
subject = "Welcome to Our Service!"
body_template = "Hello {recipient},

Thank you for joining..."

for recipient in recipients:
    async_result_email = send_personalized_email.delay(recipient, subject, body_template)
    email_tasks.append(async_result_email)
    print(f"Email task for {recipient} sent. Task ID: {async_result_email.id}")

print("
Waiting for all email results...")
for i, task in enumerate(email_tasks):
    try:
        result = task.get(timeout=10)
        print(f"Email task {i+1} result: {result}")
    except Exception as e:
        print(f"Email task {i+1} failed: {e}")

print("
All tasks processed and results retrieved.")
