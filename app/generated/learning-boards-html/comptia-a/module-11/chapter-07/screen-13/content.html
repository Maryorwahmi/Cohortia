Implementing RBAC for the backend ServiceAccount in the lab
Create minimal Role and verify with API token tests.
Eyebrow: Process Flow
RBAC manifests
# serviceaccount.yaml
apiVersion: v1
kind: ServiceAccount
metadata:
  name: backend-sa
  namespace: my-secure-app

# role.yaml
apiVersion: rbac.authorization.k8s.io/v1
kind: Role
metadata:
  name: pod-getter
  namespace: my-secure-app
rules:
- apiGroups: [""]
  resources: ["pods"]
  verbs: ["get"]

# rolebinding.yaml
apiVersion: rbac.authorization.k8s.io/v1
kind: RoleBinding
metadata:
  name: backend-sa-pod-getter
  namespace: my-secure-app
subjects:
- kind: ServiceAccount
  name: backend-sa
  namespace: my-secure-app
roleRef:
  kind: Role
  name: pod-getter
  apiGroup: rbac.authorization.k8s.io
Validation commands
# get pod name
kubectl get pod -n my-secure-app -l app=backend -o jsonpath='{.items[0].metadata.name}'

# inside backend pod
kubectl exec -n my-secure-app  -- curl -s -k -H "Authorization: Bearer $(cat /var/run/secrets/kubernetes.io/serviceaccount/token)" \
  https://kubernetes.default.svc/api/v1/namespaces/my-secure-app/pods/
# Expected: pod JSON

# disallowed call
kubectl exec -n my-secure-app  -- curl -s -k -H "Authorization: Bearer $(cat /var/run/secrets/kubernetes.io/serviceaccount/token)" \
  https://kubernetes.default.svc/apis/apps/v1/deployments
# Expected: 403 Forbidden
