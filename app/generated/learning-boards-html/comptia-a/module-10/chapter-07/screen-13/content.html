@app.route('/api/v1/posts/', methods=['GET', 'PUT', 'DELETE'])
@jwt_required
def manage_post(post_id):
    post = posts_db.get(post_id)
    if not post:
        return jsonify({"message": "Post not found"}), 404

    current_user_id = request.user_id

    if request.method == 'GET':
        return jsonify(post)

    elif request.method == 'PUT':
        data = request.get_json()
        if not data:
            return jsonify({"message": "Request must be JSON"}), 400

        # Authorization: ensure current user is the author
        if current_user_id != post.get('author_id'):
            return jsonify({"message": "Forbidden: not the post author"}), 403

        # Input validation (simple example)
        title = data.get('title')
        content = data.get('content')
        if not title or not content:
            return jsonify({"message": "Missing title or content"}), 400
        if len(title) < 3 or len(content) < 5:
            return jsonify({"message": "Title or content too short"}), 400

        post.update({"title": title, "content": content})
        return jsonify({"message": "Post updated", "post": post})

    elif request.method == 'DELETE':
        if current_user_id != post.get('author_id'):
            return jsonify({"message": "Forbidden: not the post author"}), 403
        del posts_db[post_id]
        return jsonify({"message": "Post deleted"})
Test Cases
Successful PUT: author user123 updates post 101 with valid title and content.
Unauthorized PUT: user456 attempts to update post 101 and receives 403.
Invalid input: author sends too-short title and gets 400.

Key Takeaway

Always check ownership for modifications and validate inputs server-side to stop BOLA and malformed updates.
