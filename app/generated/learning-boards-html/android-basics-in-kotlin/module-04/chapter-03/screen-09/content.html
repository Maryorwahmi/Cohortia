Data class example
// User.kt
import android.os.Parcelable
import kotlinx.parcelize.Parcelize

@Parcelize
data class User(val id: Int, val name: String, val email: String) : Parcelable
This class can now be attached to an Intent with putExtra("USER_DATA", user) and restored with getParcelableExtra.
Send and receive
// Sending
val user = User(1, "Alice", "alice@example.com")
val intent = Intent(this, SecondActivity::class.java).apply {
  putExtra("USER_DATA", user)
}
startActivity(intent)

// Receiving
val user: User? = intent.getParcelableExtra("USER_DATA")
if (user != null) {
  receivedMessageTextView.text = "Welcome, ${'${'}user.name${'}'}!"
} else {
  receivedMessageTextView.text = "No user data received."
}
Tip: Make sure the same key string is used when sending and receiving. Consider defining it as a constant to avoid typos.
