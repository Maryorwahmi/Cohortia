import pandas as pd
import plotly.graph_objects as go
from datetime import date, timedelta
import random

# 1. Generate Stock Data
start_date = date(2023, 1, 1)
num_days = 20
prices = []
current_price = 100.0
for i in range(num_days):
    change = random.uniform(-2.0, 2.0)
    current_price += change
    prices.append(max(50.0, round(current_price, 2)))

dates = [start_date + timedelta(days=i) for i in range(num_days)]
stock_data = pd.DataFrame({'Date': dates, 'Price': prices})
print(stock_data.head())

# 2. Create Plotly Chart
fig = go.Figure(data=[go.Scatter(x=stock_data['Date'], y=stock_data['Price'], mode='lines+markers')])
fig.update_layout(title='Simulated Stock Price Trend', xaxis_title='Date', yaxis_title='Price ($)', hovermode='x unified')

# 3. Export to HTML
output_file = 'stock_price_chart.html'
fig.write_html(output_file, include_plotlyjs='cdn')

print(f"Interactive stock chart saved to {output_file}")
