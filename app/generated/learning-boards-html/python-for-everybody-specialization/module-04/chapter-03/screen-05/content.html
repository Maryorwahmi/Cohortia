Code Walkthrough
import sqlite3

conn = None
try:
    conn = sqlite3.connect('my_first_database.db')
    cursor = conn.cursor()

    cursor.execute('''
        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            name TEXT NOT NULL,
            email TEXT UNIQUE NOT NULL
        )
    ''')

    user_name = "Alice Wonderland"
    user_email = "alice@example.com"
    cursor.execute("INSERT INTO users (name, email) VALUES (?, ?)", (user_name, user_email))
    print(f"Inserted user: {user_name}")

    conn.commit()

except sqlite3.IntegrityError as e:
    print(f"Data integrity error (e.g., duplicate email): {e}")
except sqlite3.Error as e:
    print(f"A database error occurred: {e}")
finally:
    if conn:
        conn.close()
Checklist
  • Use placeholders in execute()
  • Call conn.commit() after inserts
  • Handle IntegrityError for unique constraints
  • Always close the connection in finally
Key Takeaway

Explicit commits and finally blocks keep your data safe and your app predictable.
