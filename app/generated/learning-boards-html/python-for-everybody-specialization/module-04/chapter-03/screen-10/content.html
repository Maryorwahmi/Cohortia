import sqlite3
import datetime

conn = None
try:
    conn = sqlite3.connect('ecommerce_data.db')
    cursor = conn.cursor()

    # --- Insert Customers ---
    customers_to_insert = [
        ("Alice Johnson", "alice@example.com"),
        ("Bob Williams", "bob@example.com"),
        ("Charlie Davis", "charlie@example.com")
    ]
    cursor.executemany("INSERT INTO Customers (name, email) VALUES (?, ?)", customers_to_insert)
    print(f"Inserted {cursor.rowcount} customers.")
    conn.commit()

    # --- Insert Orders ---
    orders_to_insert = [
        (1, datetime.date.today().strftime('%Y-%m-%d'), 120.50),
        (1, (datetime.date.today() - datetime.timedelta(days=7)).strftime('%Y-%m-%d'), 45.99),
        (2, datetime.date.today().strftime('%Y-%m-%d'), 250.00),
        (3, (datetime.date.today() - datetime.timedelta(days=3)).strftime('%Y-%m-%d'), 75.25)
    ]
    cursor.executemany("INSERT INTO Orders (customer_id, order_date, total_amount) VALUES (?, ?, ?)", orders_to_insert)
    print(f"Inserted {cursor.rowcount} orders.")
    conn.commit()

except sqlite3.IntegrityError as e:
    print(f"Data integrity error: {e}")
except sqlite3.Error as e:
    print(f"A database error occurred: {e}")
finally:
    if conn:
        conn.close()
        print("Database connection closed.")
