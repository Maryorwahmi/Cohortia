Starter inefficient spawner
<gdscript>
extends Node2D

@export var enemy_scene: PackedScene
@export var enemy_count: int = 1000

func _ready():
  if enemy_scene:
    for i in range(enemy_count):
      var enemy = enemy_scene.instantiate()
      enemy.position = Vector2(randf_range(0, 1024), randf_range(0, 600))
      add_child(enemy)
</gdscript>
This creates many nodes and often leads to high draw and script overhead. Profile to confirm.
Optimized MultiMesh approach
<gdscript>
extends Node2D

@export var enemy_texture: Texture2D
@export var enemy_count: int = 1000
@onready var multi_mesh_instance: MultiMeshInstance2D = $MultiMeshInstance2D

func _ready():
  if enemy_texture:
    multi_mesh_instance.multimesh = MultiMesh.new()
    multi_mesh_instance.multimesh.mesh = QuadMesh.new()
    multi_mesh_instance.multimesh.texture = enemy_texture
    multi_mesh_instance.multimesh.transform_format = MultiMesh.TRANSFORM_2D
    multi_mesh_instance.multimesh.instance_count = enemy_count
    for i in range(enemy_count):
      var transform = Transform2D()
      transform.origin = Vector2(randf_range(0, 1024), randf_range(0, 600))
      multi_mesh_instance.multimesh.set_instance_transform_2d(i, transform)
</gdscript>
This shifts rendering to a single batched draw call. Keep logic separate if each enemy needs behavior.
Key Takeaway

MultiMesh reduces draw calls but does not replace per-instance AI — combine approaches as needed.
