TextEditor (visual)
public class TextEditor {
    private TextFormatter formatter;

    public TextEditor(TextFormatter formatter) { this.formatter = formatter; }

    public void setFormatter(TextFormatter formatter) { this.formatter = formatter; }

    public void publishText(String text) {
        String formattedText = formatter.format(text);
        System.out.println("Formatted Text: " + formattedText);
    }
}
