Template skeleton
abstract class ReportGenerator{
  public final void generateReport(){
    System.out.println("--- Starting Report Generation ---");
    Object raw = fetchData();
    Object processed = processData(raw);
    String out = formatOutput(processed);
    saveReport(out);
    System.out.println("--- Report Generation Finished ---");
  }
  protected abstract Object fetchData();
  protected abstract Object processData(Object rawData);
  protected abstract String formatOutput(Object processedData);
  protected void saveReport(String content){ System.out.println("Saving report content (first 50 chars): " + content.substring(0, Math.min(content.length(),50)) + "..."); }
}
