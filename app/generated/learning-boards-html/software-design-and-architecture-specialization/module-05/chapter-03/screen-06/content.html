User Service (concept)
# User Service: exposes GET /users/<id>
from flask import Flask, jsonify
user_app = Flask(__name__)
users_db = { '1': {'name':'Alice'}}

@user_app.route('/users/')
def get_user(user_id):
    user = users_db.get(user_id)
    if user:
        print('User Service: fetched user', user_id)
        return jsonify(user)
    return jsonify({'message':'User not found'}), 404

Order Service (concept)
# Order Service makes an HTTP call to User Service
from flask import Flask, request, jsonify
import requests
order_app = Flask(__name__)
orders_db = {}

@order_app.route('/orders', methods=['POST'])
def create_order():
    data = request.get_json()
    try:
        r = requests.get(f'http://127.0.0.1:5001/users/{data.get("user_id")}')
        if r.status_code != 200:
            return jsonify({'message':'Invalid user_id'}), 400
    except requests.exceptions.ConnectionError:
        return jsonify({'message':'User service unavailable'}), 503
    # save order...
    return jsonify({'order_id':'1','status':'pending'}), 201
Example behavior
Order creation calls User Service synchronously to validate users, and fails if the user service is unavailable.
Misconception
Assuming network calls are infallible. Always plan for retries, timeouts, and graceful degradation.
Pro tip
Add circuit breakers and bulkheads to protect other services from cascading failures.
