Key code excerpts
from abc import ABC, abstractmethod class PaymentMethod(ABC): @abstractmethod def process_payment(self, amount: float) -> bool: pass class CreditCardPayment(PaymentMethod): def __init__(self, card_number, expiration_date, cvv): self.__card_number = card_number self.__expiration_date = expiration_date self.__cvv = cvv def process_payment(self, amount: float) -> bool: print(f"Processing ${amount:.2f} via Credit Card {self.__card_number[-4:] }...") return amount > 0 class PayPalPayment(PaymentMethod): def __init__(self, email): self.__email = email def process_payment(self, amount: float) -> bool: print(f"Processing ${amount:.2f} via PayPal account {self.__email}...") return amount > 0 and "@" in self.__email
