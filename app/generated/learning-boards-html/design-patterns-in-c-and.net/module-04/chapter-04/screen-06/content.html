Key excerpt (iterator core)
public class DinerMenuIterator : IEnumerator<MenuItem>
{
  private MenuItem[] _list;
  private int _position = -1;

  public DinerMenuIterator(MenuItem[] list)
  {
    _list = list;
  }

  public MenuItem Current
  {
    get
    {
      try
      {
        return _list[_position];
      }
      catch (IndexOutOfRangeException)
      {
        throw new InvalidOperationException();
      }
    }
  }

  object IEnumerator.Current => Current;

  public bool MoveNext()
  {
    _position++;
    return (_position < _list.Length && _list[_position] != null);
  }

  public void Reset() { _position = -1; }
  public void Dispose() { }
}
Explanation
Position starts before first element. MoveNext advances and returns whether a valid element exists. Current reads the element or raises if invalid.
Concrete example
DinerMenu constructs a DinerMenuIterator with its internal array; foreach uses that iterator under the hood.
Misconception
Assuming MoveNext will skip removed items safely — concurrent modification may cause exceptions unless guarded.
