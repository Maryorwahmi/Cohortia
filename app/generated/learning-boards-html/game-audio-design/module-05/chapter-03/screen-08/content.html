WeaponAudio snippet
using UnityEngine;

public class WeaponAudio : MonoBehaviour
{
    public AudioSource weaponAudioSource; // Assign in Inspector
    public AudioClip shootSound;          // Assign in Inspector
    public AudioClip reloadSound;         // Assign in Inspector

    void Update()
    {
        if (Input.GetMouseButtonDown(0)) // Left click to shoot
        {
            if (shootSound != null)
            {
                weaponAudioSource.PlayOneShot(shootSound);
            }
        }

        if (Input.GetKeyDown(KeyCode.R)) // R key to reload
        {
            if (reloadSound != null)
            {
                weaponAudioSource.PlayOneShot(reloadSound);
            }
        }
    }
}
Practical notes
  • PlayOneShot avoids changing the AudioSource.clip, keeping persistent loops intact.
  • Use volumeScale to lower repetitive sounds, e.g., PlayOneShot(clip, 0.7f).
  • Randomize small pitch offsets for natural variation: audioSource.pitch = 1f + Random.Range(-0.05f,0.05f).
Key Takeaway

PlayOneShot is the efficient, simple way to layer short SFX on a single AudioSource.
