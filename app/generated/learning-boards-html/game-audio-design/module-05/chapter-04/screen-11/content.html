Steps
  1. Import audio clips: looping music, explosion, coin.
  2. Create MainAudioMixer asset and open Mixer window.
  3. Add Music and SFX groups under Master and nest children as needed.
  4. Assign BGM_Player AudioSource Output to MainAudioMixer → Music; assign SFX_Player to SFX.

  6. Create Gameplay and Paused snapshots and configure group volumes.
Script starter
using UnityEngine;
using UnityEngine.Audio;

public class SnapshotController : MonoBehaviour
{
  public AudioMixer mainMixer; // Assign MainAudioMixer here
  public AudioMixerSnapshot gameplaySnapshot; // Assign Gameplay snapshot
  public AudioMixerSnapshot pausedSnapshot;   // Assign Paused snapshot

  private bool isPaused = false;

  void Start()
  {
    if (gameplaySnapshot != null)
    {
      gameplaySnapshot.TransitionTo(0.1f);
    }
  }

  void Update()
  {
    if (Input.GetKeyDown(KeyCode.P))
    {
      isPaused = !isPaused;
      if (isPaused && pausedSnapshot != null) pausedSnapshot.TransitionTo(0.7f);
      else if (!isPaused && gameplaySnapshot != null) gameplaySnapshot.TransitionTo(0.7f);
    }

    if (Input.GetKeyDown(KeyCode.Space) && GetComponent() != null)
    {
      GetComponent().PlayOneShot(GetComponent().clip);
    }
  }
}
Test checklist
Play, press P to toggle paused snapshot, press Space to play SFX, and verify reverb and volume changes. Fix missing inspector links if behavior is absent.
