<code>using UnityEngine;
using UnityEngine.UI; // Required for Button component

public class ButtonSoundPlayer : MonoBehaviour
{
  public AudioSource buttonAudioSource; // Assign this in the Inspector

  void Start()
  {
    // Get the Button component on this GameObject
    Button btn = GetComponent<Button>();
    if (btn != null)
    {
      // Add a listener to the button's onClick event
      btn.onClick.AddListener(PlayButtonSound);
    }
    else
    {
      Debug.LogError("Button component not found on this GameObject.", this);
    }

    if (buttonAudioSource == null)
    {
      Debug.LogError("Button AudioSource not assigned!", this);
    }
  }

  void PlayButtonSound()
  {
    if (buttonAudioSource != null && buttonAudioSource.clip != null)
    {
      buttonAudioSource.Play();
    }
    else
    {
      Debug.LogWarning("Attempted to play button sound, but AudioSource or clip is missing.");
    }
  }
}
</code>
Explanation

Assign the AudioSource on the same Button GameObject. The script registers to the Button's onClick event and plays the clip when clicked. It logs helpful errors if components are missing.

Common debugging checks: ensure the AudioSource has a clip, Play On Awake is unchecked, and the GameObject is active in the scene.

Pro Tip

For many buttons, consider a shared audio player to centralize volume control. For independent customization, keep AudioSources local to each button.
