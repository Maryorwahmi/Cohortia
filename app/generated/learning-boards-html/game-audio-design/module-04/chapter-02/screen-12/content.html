Conceptual UI
Intensity slider (0–100)
Faders: base, tension, danger (driven by Intensity)
Waveform stack: visualize layer energies while moving the slider
Pseudo-code (conceptual)
function map(value, inMin, inMax, outMin, outMax) {
  let t = (value - inMin) / (inMax - inMin)
  t = Math.max(0, Math.min(1, t)) // clamp
  // apply easing here if desired
  return outMin + t * (outMax - outMin)
}

// Example usage
tension_volume = map(Intensity, 30, 60, 0, 1)
danger_volume = map(Intensity, 60, 100, 0, 1)
base_level = 1 - 0.2 * danger_volume // reduce base when danger peaks
