use testdb;
for (let i = 0; i < 1000; i++) {
  db.loadtest.insertOne({ data: "some_data_" + i, timestamp: new Date() });

After running the loop, re-run rs.status() and rs.printReplicationInfo() to watch optimeDate advance and observe any change in oplog window or lag.
