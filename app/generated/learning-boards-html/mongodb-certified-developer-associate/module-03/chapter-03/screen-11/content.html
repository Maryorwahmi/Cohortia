Starter commands
// 1. Create 2dsphere index
db.events.createIndex({ "location": "2dsphere" })

// 2. Create text index with weights
db.events.createIndex({ "name": "text", "description": "text" }, { weights: { name: 5, description: 1 } })
Sample queries
// 3. Events within 10km of NYC
db.events.find({
  "location": { "$near": { "$geometry": { "type": "Point", "coordinates": [ -74.0060, 40.7128 ] }, "$maxDistance": 10000 } }
})

// 4. Text search "music festival" sorted by relevance
db.events.find(
  { "$text": { "$search": "\"music festival\"" } },
  { "score": { "$meta": "textScore" } }
).sort({ "score": { "$meta": "textScore" } })

// 5. Tagged outdoor events within a box
db.events.find({
  "tags": "outdoor",
  "location": { "$geoWithin": { "$box": [ [ -74.02, 40.70 ], [ -73.98, 40.72 ] ] } }
})
Key Takeaway

Follow the createIndex patterns, verify with getIndexes, and combine text and geospatial filters to build location-aware search UX.
