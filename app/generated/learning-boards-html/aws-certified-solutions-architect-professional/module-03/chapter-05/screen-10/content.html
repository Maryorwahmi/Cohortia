Create destination and enable versioning
# Create destination bucket in us-west-2
aws s3api create-bucket \
    --bucket my-company-dr-backups \
    --create-bucket-configuration LocationConstraint=us-west-2 \
    --region us-west-2

# Enable versioning on destination
aws s3api put-bucket-versioning \
    --bucket my-company-dr-backups \
    --versioning-configuration Status=Enabled \
    --region us-west-2

# Enable versioning on source
aws s3api put-bucket-versioning \
    --bucket my-company-data \
    --versioning-configuration Status=Enabled \
    --region us-east-1
Replication configuration example
{
    "Rules": [
        {
            "ID": "ReplicateBackups",
            "Status": "Enabled",
            "Priority": 1,
            "Filter": { "Prefix": "backups/" },
            "Destination": {
                "Bucket": "arn:aws:s3:::my-company-dr-backups",
                "StorageClass": "STANDARD",
                "Account": "<YOUR_ACCOUNT_ID>"
            }
        }
    ],
    "Role": "arn:aws:iam::<YOUR_ACCOUNT_ID>:role/S3ReplicationRole"
}
After applying, upload test objects to s3://my-company-data/backups/ and verify s3://my-company-dr-backups/backups/ receives them.
