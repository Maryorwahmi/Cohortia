Lambda@Edge Handler (Viewer Request)
exports.handler = async (event) => {
  const request = event.Records[0].cf.request;
  const headers = request.headers;

  // Default variant
  let abTestVariant = 'A';

  // Check for an A/B test cookie
  if (headers.cookie) {
    for (let i = 0; i < headers.cookie.length; i++) {
      if (headers.cookie[i].value.includes('abtest=B')) {
        abTestVariant = 'B';
        break;
      }
    }
  }

  // If no cookie, randomly assign
  if (!headers.cookie || !headers.cookie.some(c => c.value.includes('abtest='))) {
    if (Math.random() < 0.5) {
      abTestVariant = 'B';
    }
    // To persist the variant, add Set-Cookie in a response trigger.
  }

  if (abTestVariant === 'B') {
    request.uri = '/variantB' + request.uri;
  }

  return request;
};
Concrete detail
Routing happens before origin fetch. Persisting variant requires setting cookie in a response event.
Key Takeaway

Separate routing (request event) and persistence (response event) when implementing A/B tests at the edge.
