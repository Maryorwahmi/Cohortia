This code demonstrates pool creation, cache-aside checks, and set-with-expire to store JSON.
import redis
import json
import time

# Configuration and pool
pool = redis.ConnectionPool(host='localhost', port=6379, db=0)
r = redis.Redis(connection_pool=pool)

# Simulated DB fetch
def get_user_data_from_db(user_id):
  time.sleep(2)
  return {"id": user_id, "name": f"User {user_id}", "email": f"user{user_id}@example.com"}

# Cache-aside function
def get_cached_user_data(user_id):
  cache_key = f"user:{user_id}"
  cached_data = r.get(cache_key)
  if cached_data:
    return json.loads(cached_data)
  user_data = get_user_data_from_db(user_id)
  r.setex(cache_key, 300, json.dumps(user_data))
  return user_data
Notes & checks
1. Ensure pool is created once and reused across requests.
2. Use JSON serialization consistently when storing structured data.
3. Always set expirations to avoid stale or unbounded cache growth.
