ioredis example

const Redis = require('ioredis');

const redis = new Redis({ host: 'localhost', port: 6379, db: 0 });

redis.on('connect', () => console.log('Connected to Redis!'));
redis.on('error', err => console.error('Redis connection error:', err));

async function run() {
  try {
    await redis.ping();
    await redis.set('nodekey', 'Hello from Node.js!');
    const value = await redis.get('nodekey');
    console.log(`Value for 'nodekey': ${value}`);
  } catch (err) {
    console.error('Error executing Redis commands:', err);
  } finally {
    await redis.quit();
  }
}

run();
This async pattern uses event listeners and try-catch to handle errors. Call quit on shutdown so connections are closed gracefully.
Notes
  • ioredis handles pooling behaviors for many topologies internally.
  • Listen for error events to log and react to connection failures.
  • Call quit() on shutdown to free sockets and allow the process to exit.
Pro tip
Treat ECONNREFUSED as an operational signal: verify service status and firewall rules before retrying programmatically.
