Module 04 › Chapter 4.3 › Hardened unit example
Unit file (example)
[Unit]
Description=My Web Application
After=network.target

[Service]
ExecStart=/usr/bin/mywebapp-server
User=mywebapp
Group=mywebapp
PrivateTmp=true
ProtectSystem=full
ProtectHome=true
NoNewPrivileges=true
CapabilityBoundingSet=~CAP_SYS_ADMIN ~CAP_DAC_OVERRIDE ~CAP_CHOWN ~CAP_FOWNER ~CAP_KILL ~CAP_SETGID ~CAP_SETUID ~CAP_NET_ADMIN ~CAP_NET_RAW
RestrictAddressFamilies=AF_INET AF_UNIX
MemoryLimit=256M
TasksMax=50
ReadWritePaths=/var/log/mywebapp /var/cache/mywebapp
ReadOnlyPaths=/etc/mywebapp/config.d

[Install]
WantedBy=multi-user.target
Annotations
  • User and Group — drop root where possible.
  • PrivateTmp & ProtectSystem — isolate temp and keep system files read-only.
  • CapabilityBoundingSet — explicitly remove unnecessary kernel powers.
  • MemoryLimit & TasksMax — stop resource exhaustion.
  • ReadWritePaths/ReadOnlyPaths — whitelist specific writable and readable paths.
