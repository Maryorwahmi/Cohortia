Network setup
# Example static IP with nmcli
sudo nmcli connection modify enp1s0 ipv4.method manual ipv4.addresses 192.168.1.10/24 ipv4.gateway 192.168.1.1 ipv4.dns "192.168.1.100"
sudo nmcli connection up enp1s0

# Verify DNS
ping -c 3 google.com
nslookup rhevh-01.example.com
Note
Replace example addresses and names with your environment values and confirm both forward and reverse DNS.
Packages & Firewall
# Enable repos and install cockpit plugin
sudo subscription-manager repos --enable=rhel-8-for-x86_64-baseos-rpms \
  --enable=rhel-8-for-x86_64-appstream-rpms --enable=rhv-4-for-rhel-8-x86_64-rpms
sudo dnf install cockpit-ovirt-dashboard -y
sudo systemctl enable --now cockpit.socket

# Firewall rules
sudo firewall-cmd --permanent --add-service=cockpit
sudo firewall-cmd --permanent --add-port=16514/tcp
sudo firewall-cmd --permanent --add-port=5432/tcp
sudo firewall-cmd --permanent --add-port=5900-6923/tcp
sudo firewall-cmd --reload
KVM verification
# Check CPU virtualization flags and libvirt
grep -E 'vmx|svm' /proc/cpuinfo || echo "No virtualization flags found"
sudo systemctl status libvirtd
Safety note

Verify mounts and SSH connectivity before running hosted-engine --deploy to reduce recovery work if the deploy fails.
