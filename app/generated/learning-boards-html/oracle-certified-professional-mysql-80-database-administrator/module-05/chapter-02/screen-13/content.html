GRANT REPLICATION SLAVE ON *.* TO 'repl_user'@'192.168.1.101';

# Run dump
mysqldump -u root -p --all-databases --single-transaction --flush-logs --master-data=2 > /tmp/full_backup.sql

# On replica: import and configure
mysql -u root -p < /tmp/full_backup.sql
CHANGE REPLICATION SOURCE TO ... ;
START REPLICA;
