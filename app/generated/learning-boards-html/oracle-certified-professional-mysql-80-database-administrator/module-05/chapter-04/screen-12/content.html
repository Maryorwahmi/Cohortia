Module 5 › Chapter 5.4 › Activity: Enable semi‑sync
Source checklist
SHOW PLUGINS; -- look for rpl_semi_sync_source
INSTALL PLUGIN rpl_semi_sync_source SONAME 'mysql_semi_sync_source.so';
SET GLOBAL rpl_semi_sync_source_enabled = ON;
SET GLOBAL rpl_semi_sync_source_timeout = 10000;
Replica checklist
SHOW PLUGINS; -- look for rpl_semi_sync_replica
INSTALL PLUGIN rpl_semi_sync_replica SONAME 'mysql_semi_sync_replica.so';
SET GLOBAL rpl_semi_sync_replica_enabled = ON;
Expected verification
SHOW STATUS LIKE 'Rpl_semi_sync%'; -- Expected key rows: Rpl_semi_sync_source_status | ON Rpl_semi_sync_source_clients | 1 Rpl_semi_sync_source_no_times_no_slaves | 0
