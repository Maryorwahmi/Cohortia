Module 6 › Chapter 6.5 › Unity Analytics
Enable Analytics
Open Window → General → Services, link your project to an organization, and toggle Analytics on to start recording events to your dashboard.
Design guidance
  • Start small: GameStart, LevelComplete, CollectiblePickup, PowerUpUse.
  • Include meaningful parameters such as completion time or item name to add context.
  • Respect privacy: collect only what you need and follow platform rules.
Pro tip

Use Debug.Log to confirm events are triggered in the Editor before checking the online dashboard.

Analytics Example (C#)
Place this on a manager GameObject and call the custom events from gameplay scripts.
// AnalyticsExample.cs
using UnityEngine;
using UnityEngine.Analytics; // Make sure to include this namespace
using System.Collections.Generic; // For Dictionary

public class AnalyticsExample : MonoBehaviour
{
    void Start()
    {
        // Send a simple event when the game starts
        AnalyticsEvent.GameStart(); 
        Debug.Log("Analytics: GameStart event sent.");
    }

    public void PlayerDied(int enemiesKilled, float timeAlive)
    {
        // Send a custom event with parameters when a player dies
        AnalyticsEvent.Custom("PlayerDeath", new Dictionary
        {
            { "enemies_killed", enemiesKilled },
            { "time_alive_seconds", timeAlive }
        });
        Debug.Log($"Analytics: PlayerDeath event sent (Kills: {enemiesKilled}, Time: {timeAlive}).");
    }

    public void LevelCompleted(string levelName, float completionTime)
    {
        // Another custom event
        AnalyticsEvent.LevelComplete(levelName, new Dictionary
        {
            { "completion_time_seconds", completionTime }
        });
        Debug.Log($"Analytics: LevelComplete event sent for {levelName}.");
    }
}
After running the Editor you should see Debug.Log output confirming GameStart and any custom events you trigger.
