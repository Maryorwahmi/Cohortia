Component overview
// src/components/LoginButton.tsx
import React from 'react';
import { Pressable, Text, ActivityIndicator } from 'react-native';

const LoginButton = ({ onPress, title, isLoading=false }) => {
  const handlePress = async () => { if (!isLoading) await onPress(); };
  return (
    <Pressable testID='login-button' disabled={isLoading} onPress={handlePress}>
      {isLoading ? <ActivityIndicator testID='loading-spinner' /> : <Text>{title}</Text>}
    </Pressable>
  );
};
Test highlights
it('calls onPress when pressed', async () => {
  const mockOnPress = jest.fn().mockResolvedValueOnce(undefined);
  render(<LoginButton title='Login' onPress={mockOnPress} />);
  fireEvent.press(screen.getByTestId('login-button'));
  await waitFor(() => expect(mockOnPress).toHaveBeenCalledTimes(1));
});
Edge case
Assert that isLoading disables the button and shows the spinner so users cannot double-submit.
Key Takeaway

Combine jest.fn, fireEvent, and waitFor to verify async handlers and conditional rendering.
