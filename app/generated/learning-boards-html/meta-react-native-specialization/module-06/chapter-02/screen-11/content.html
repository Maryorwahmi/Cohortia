Component behavior
UserList fetches users on mount, shows an ActivityIndicator while loading, displays the list on success, and an error message on failure.
// src/components/UserList.tsx
import React, { useEffect, useState } from 'react';
import { Text, View, FlatList, ActivityIndicator } from 'react-native';
import { fetchUsers } from '../services/api';

const UserList: React.FC = () => {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  useEffect(() => { /* fetchUsers and set state */ }, []);
  return loading ? <ActivityIndicator testID='loading-indicator' /> : error ? <Text testID='error-message'>{error}</Text> : <FlatList data={users} renderItem={({item})=><Text testID={`user-name-${item.id}`}>{item.name}</Text> /> />;
};
