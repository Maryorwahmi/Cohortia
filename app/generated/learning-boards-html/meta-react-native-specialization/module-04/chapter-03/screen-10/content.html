Before (Fetch)
async function fetchPostsWithFetch() {
  try {
    const res = await fetch('https://jsonplaceholder.typicode.com/posts');
    if (!res.ok) throw new Error('Network response was not ok');
    const data = await res.json();
    setPosts(data);
  } catch (err) {
    // handle errors
  }
}
After (Axios)
async function fetchPostsWithAxios() {
  try {
    const response = await axios.get('https://jsonplaceholder.typicode.com/posts');
    setPosts(response.data);
  } catch (err) {
    // inspect err.response / err.request / err.message
  }
}
Notice the removal of manual ok checks and json parsing; response.data is ready to use.
Pro tip
After refactor, remove redundant error branches and move shared headers into an interceptor or instance.
