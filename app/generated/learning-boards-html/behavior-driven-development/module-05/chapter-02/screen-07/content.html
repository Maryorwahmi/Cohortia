Hooks.java (example)
This example shows ordering, tagged hooks, and cleanup responsibilities in code.
Key Takeaway

Keep hooks focused on technical setup and teardown; avoid embedding business logic that belongs in Background or scenarios.

import io.cucumber.java.After;
import io.cucumber.java.Before;
import io.cucumber.java.Scenario;

public class Hooks {
  public static WebDriver driver;

  @Before(order = 0)
  public void setupBase() {
    System.out.println("Base setup for every scenario...");
    // e.g., system properties, shared test config
  }

  @Before("@database")
  public void setupDatabase() {
    System.out.println("Preparing database for @database scenario...");
    // create schema, seed data, or clear tables
  }

  @Before("@ui")
  public void setupBrowser() {
    System.out.println("Initializing browser for UI tests...");
    // driver = new ChromeDriver();
  }

  @After
  public void tearDown(Scenario scenario) {
    if (scenario.isFailed() && driver != null) {
      System.out.println("Scenario failed - taking screenshot...");
    }
    if (driver != null) {
      driver.quit();
      driver = null;
    }
  }
}
