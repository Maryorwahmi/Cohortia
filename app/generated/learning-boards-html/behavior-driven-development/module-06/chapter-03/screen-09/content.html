Good: Page Object
public class RegistrationPage {
    private WebDriver driver;
    public RegistrationPage(WebDriver driver) { this.driver = driver; }
    public void navigateTo() { driver.get("http://localhost:8080/register"); }
    public void enterEmail(String email) { driver.findElement(By.id("email")).sendKeys(email); }
    public void clickRegister() { driver.findElement(By.id("registerButton")).click(); }
}

@Given("I am on the registration page")
public void i_am_on_the_registration_page() {
    new RegistrationPage(driver).navigateTo();
}
