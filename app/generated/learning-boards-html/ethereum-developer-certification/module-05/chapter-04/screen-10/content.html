Complete example
async function setStoredNumber(newNumber) {
  if (!simpleStorageContract || !currentAccount) {
    transactionStatus.textContent = 'Error: Contract not loaded or account not connected.';
    return;
  }

  if (newNumber === '' || isNaN(newNumber) || parseInt(newNumber) < 0) {
    alert('Please enter a valid positive number.');
    return;
  }

  transactionStatus.textContent = 'Estimating gas...';
  try {
    const estimate = await simpleStorageContract.methods.setNumber(newNumber).estimateGas({ from: currentAccount });
    const gasLimit = Math.floor(estimate * 1.2);

    transactionStatus.textContent = 'Sending transaction... Please confirm in MetaMask.';
    const receipt = await simpleStorageContract.methods.setNumber(newNumber).send({ from: currentAccount, gas: gasLimit });

    transactionStatus.textContent = `Transaction successful! Hash: ${receipt.transactionHash.substring(0,10)}...`;
    await getContractData();
    newNumberInput.value = '';
  } catch (error) {
    if (error.code === 4001) {
      transactionStatus.textContent = 'Transaction rejected by user.';
    } else if (error.message && error.message.includes('insufficient funds')) {
      transactionStatus.textContent = 'Insufficient funds for transaction. Please top up your account.';
    } else {
      transactionStatus.textContent = `Transaction failed: ${error.message}`;
    }
    console.error('Error sending transaction:', error);
  }
}
Explanation and checks
  • Input validation prevents trivial reverts and bad UX.
  • Estimating gas before opening MetaMask avoids surprise failures.
  • Adding a buffer prevents marginal out-of-gas errors on complex state changes.
Key Takeaway

Follow validate → estimate → send → await receipt → refresh pattern and provide clear status messages for users.
