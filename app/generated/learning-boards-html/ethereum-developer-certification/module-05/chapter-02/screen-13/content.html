// app.js (outline)
let web3; let currentAccount = null;
async function connectWallet(){
  if (window.ethereum){
    try{
      const accounts = await window.ethereum.request({ method: 'eth_requestAccounts' });
      currentAccount = accounts[0];
      web3 = new Web3(window.ethereum);
      // update UI and listeners
    } catch(err){ /* handle 4001 etc */ }
  } else { console.error('MetaMask not installed'); }
}

Follow the connect pattern, initialize web3 with window.ethereum, and set up listeners for dynamic wallet changes while testing thoroughly in the browser.
