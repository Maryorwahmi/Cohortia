async function getContractData() {
  if (!simpleStorageContract) {
    console.error('Contract not loaded.');
    return;
  }
  try {
    const storedNum = await simpleStorageContract.methods.getNumber().call();
    document.getElementById('displayNumber').textContent = storedNum;

    const sumResult = await simpleStorageContract.methods.add(10, 20).call();
    document.getElementById('displaySum').textContent = sumResult;

  } catch (error) {
    console.error('Error reading from contract:', error);
    document.getElementById('displayNumber').textContent = 'Error';
    document.getElementById('displaySum').textContent = 'Error';
  }
}
Note
Calls return strings for uint256; use parseInt for small values or BN/BigInt for large ones.
Pitfall
Logging without await prints a Promise instead of the resolved value.
Example scenario
A DApp displays a stored game score and a computed bonus. Use getNumber().call() for the score and add(base,bonus).call() for the computed value, then show both on the page.
Key takeaway
Wrap calls in try/catch, await them, update UI, and choose proper numeric conversion strategy.
