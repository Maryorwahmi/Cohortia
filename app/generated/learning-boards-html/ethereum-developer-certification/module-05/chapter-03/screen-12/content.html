const simpleStorageABI = [ /* YOUR_ABI_JSON_HERE */ ];
const simpleStorageContractAddress = 'YOUR_CONTRACT_ADDRESS_HERE';
let simpleStorageContract;

async function loadContractInstance() {
  if (web3 && simpleStorageContractAddress && simpleStorageABI) {
    simpleStorageContract = new web3.eth.Contract(simpleStorageABI, simpleStorageContractAddress);
    await getContractData();
  }
}

async function getContractData() {
  if (!simpleStorageContract) return;
  try {
    const storedNum = await simpleStorageContract.methods.getNumber().call();
    document.getElementById('displayNumber').textContent = storedNum;
    const sumResult = await simpleStorageContract.methods.add(10, 20).call();
    document.getElementById('displaySum').textContent = sumResult;
  } catch (err) {
    document.getElementById('displayNumber').textContent = 'Error';
    document.getElementById('displaySum').textContent = 'Error';
  }
}
