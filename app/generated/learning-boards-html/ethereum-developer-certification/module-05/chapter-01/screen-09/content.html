Module 5 › Chapter 5.1 › Connect to Ganache
connect.js
const Web3 = require('web3');

// --- YOUR GANACHE RPC URL HERE ---
const ganacheRpcUrl = 'http://127.0.0.1:8545';

const web3 = new Web3(ganacheRpcUrl);

async function checkConnection() {
  try {
    const networkId = await web3.eth.net.getId();
    console.log('Successfully connected to network ID:', networkId);

    const latestBlock = await web3.eth.getBlockNumber();
    console.log('Latest block number:', latestBlock);

    const accounts = await web3.eth.getAccounts();
    console.log('First account address:', accounts[0]);

    const balanceWei = await web3.eth.getBalance(accounts[0]);
    const balanceEth = web3.utils.fromWei(balanceWei, 'ether');
    console.log(`Balance of first account: ${'${balanceEth}'} ETH`);

  } catch (error) {
    console.error('Failed to connect to Ganache or fetch data:', error.message);
    console.error('Please ensure Ganache is running and the RPC URL is correct.');
  }
}

checkConnection();
Expected terminal output
Successfully connected to network ID: 5777
Latest block number: 12
First account address: 0x1234...abcd
Balance of first account: 100 ETH
Troubleshooting
  • Confirm Ganache is running and RPC URL matches the displayed address.
  • Check firewall or port conflicts if connection is refused.
  • Ensure web3 package is installed locally in the project.
Key Takeaway

A short verification script gives immediate feedback that your development environment, provider, and Web3.js setup are correct.
