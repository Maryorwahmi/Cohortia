// scripts/deploy.js
const hre = require("hardhat");

async function main() {
  const MyContract = await hre.ethers.getContractFactory("MyContract");
  const [deployer] = await hre.ethers.getSigners();
  console.log("Deploying contract with the account:", deployer.address);

  const myContract = await MyContract.deploy("Hello, Cohortia!");
  await myContract.deployed();
  console.log("MyContract deployed to:", myContract.address);

  let currentGreeting = await myContract.greeting();
  console.log("Current greeting:", currentGreeting);

  const tx = await myContract.setGreeting("Greetings from Hardhat!");
  await tx.wait();

  currentGreeting = await myContract.greeting();
  console.log("New greeting:", currentGreeting);
}

main().then(() => process.exit(0)).catch((error) => { console.error(error); process.exit(1); });
Explanation & callouts
  • Require Hardhat to access hre and the injected ethers instance.
  • getContractFactory returns the deployment blueprint for MyContract.
  • getSigners gives accounts; the deployer signs the deployment transaction.
  • Await deployed() to ensure the address is confirmed before using it.
  • Call greeting() for a view; call setGreeting() to change state and await tx.wait() to confirm.
Common mistake
Reading state immediately after a mutating call without awaiting tx.wait() leads to stale values.

Run npx hardhat node in a separate terminal, then npx hardhat run scripts/deploy.js and watch the printed greetings change.
