const { expect } = require("chai");
const { ethers } = require("hardhat");

describe("MyContract", function () {
  let MyContract, myContract, owner, addr1;

  async function deployMyContractFixture() {
    [owner, addr1] = await ethers.getSigners();
    MyContract = await ethers.getContractFactory("MyContract");
    myContract = await MyContract.deploy("Initial Greeting");
    await myContract.deployed();
    return { myContract, owner, addr1 };
  }

  beforeEach(async function () {
    ({ myContract, owner, addr1 } = await loadFixture(deployMyContractFixture()));
  });

  it("Should set the right greeting upon deployment", async function () {
    expect(await myContract.greeting()).to.equal("Initial Greeting");
  });
});
Explanation
  • Imports bring in assertion helpers and the runtime ethers object.
  • The fixture deploys once; beforeEach resets to the snapshot.
  • Each it asserts a single behavior for quick diagnosis when failures occur.
