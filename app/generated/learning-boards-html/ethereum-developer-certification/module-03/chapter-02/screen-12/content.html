pragma solidity ^0.8.0;

contract FunctionAndControl {
    address public owner;
    uint256 public value = 100;
    uint256[] private numbers;

    modifier onlyOwner() {
        require(msg.sender == owner, "Caller is not the owner");
        _;
    }

    modifier greaterThanZero(uint _amount) {
        require(_amount > 0, "Amount must be greater than zero");
        _;
    }

    constructor() {
        owner = msg.sender;
        numbers.push(10);
        numbers.push(20);
        numbers.push(30);
    }

    function setValue(uint256 _newValue) public onlyOwner greaterThanZero(_newValue) {
        require(_newValue <= 1000, "Value cannot exceed 1000");
        value = _newValue;
    }
}
Highlights
  • Modifiers: onlyOwner centralizes access control.
  • Visibility: owner is public; numbers is private to prevent direct external reads.
  • Safety: require guards inputs and loop limits.
  • Mutability: setValue is non-view because it updates state; multiply is pure elsewhere.
Reflection
Which checks catch invalid inputs, and which checks guard internal invariants? Separate those concerns in your designs.
