Files to create
IPausable.sol (interface)
// IPausable.sol
interface IPausable {
  event Paused(address account);
  event Unpaused(address account);

  function paused() external view returns (bool);
  function pause() external;
  function unpause() external;
}
Pausable.sol (abstract)
// Pausable.sol
abstract contract Pausable is IPausable {
  bool private _paused;
  constructor(){ _paused = false; }
  function paused() public view virtual override returns (bool) { return _paused; }

  modifier whenNotPaused(){ require(!_paused, "Pausable: paused"); _; }
  modifier whenPaused(){ require(_paused, "Pausable: not paused"); _; }

  function _pause() internal virtual { require(!_paused, "Pausable: already paused"); _paused = true; emit Paused(msg.sender); }
  function _unpause() internal virtual { require(_paused, "Pausable: not paused"); _paused = false; emit Unpaused(msg.sender); }

  function pause() public virtual override {}
  function unpause() public virtual override {}
}
MyPausableToken.sol (implementation)
// MyPausableToken.sol
contract MyPausableToken is Pausable {
  address public owner;
  event ImportantActionDone(address indexed by, uint256 timestamp);

  modifier onlyOwner(){ require(msg.sender == owner, "Only owner can call this function"); _; }

  constructor(){ owner = msg.sender; }

  function pause() public override onlyOwner { _pause(); }
  function unpause() public override onlyOwner { _unpause(); }

  function doSomethingImportant() public whenNotPaused {
    emit ImportantActionDone(msg.sender, block.timestamp);
  }

  function getContractBalance() public view returns (uint256) {
    return address(this).balance;
  }
}
Verification steps
  1. Deploy MyPausableToken and call pause; observe Paused emitted.
  2. Call doSomethingImportant while paused and confirm it reverts.
  3. Unpause and call doSomethingImportant; confirm ImportantActionDone event.
Key Takeaway

This activity demonstrates how interfaces define the contract, abstract contracts provide partial implementations and events, and children implement ownership and expose safe, audited behavior.
