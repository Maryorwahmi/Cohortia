Module 5 › Chapter 4.4 › Configuring hardhat.config.js
Key fields for deployment
  • solidity: compiler version to ensure bytecode compatibility
  • networks: list of network objects with url, accounts, and chainId
  • etherscan: apiKey for contract verification
Security note

Never hardcode PRIVATE_KEY in the config. Use dotenv to load values from a local .env file kept out of version control.

Example snippet
// hardhat.config.js
require("@nomicfoundation/hardhat-toolbox");
require("dotenv").config();

const SEPOLIA_RPC_URL = process.env.SEPOLIA_RPC_URL;
const PRIVATE_KEY = process.env.PRIVATE_KEY;

module.exports = {
  solidity: "0.8.19",
  networks: {
    hardhat: {},
    sepolia: {
      url: SEPOLIA_RPC_URL || "",
      accounts: PRIVATE_KEY ? [PRIVATE_KEY] : [],
      chainId: 11155111,
    },
  },
  etherscan: {
    apiKey: process.env.ETHERSCAN_API_KEY,
  }
};
