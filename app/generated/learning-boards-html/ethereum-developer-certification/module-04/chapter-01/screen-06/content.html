test/Counter.js
Mocha structure, deployment, and assertions.
const { expect } = require("chai");
const { ethers } = require("hardhat");

describe("Counter", function () {
  let Counter;
  let counter;
  let owner;

  beforeEach(async function () {
    Counter = await ethers.getContractFactory("Counter");
    [owner] = await ethers.getSigners();
    counter = await Counter.deploy();
    await counter.deployed();
  });

  it("Should have an initial count of 0", async function () {
    expect(await counter.getCount()).to.equal(0);
  });

  it("Should increment the count by 1", async function () {
    await counter.increment();
    expect(await counter.getCount()).to.equal(1);
  });

  it("Should not allow decrementing below zero", async function () {
    await expect(counter.decrement()).to.be.revertedWith("Count cannot go below zero");
  });
});
Explanation & checklist
  • ContractFactory: ethers.getContractFactory obtains the deployable contract artifact.
  • Signers: ethers.getSigners returns accounts you can use to call contract methods as different users.
  • Deployment in beforeEach: ensures each it runs against a fresh instance.
  • Await everything: await deployments and transactions to avoid race conditions in tests.
Common mistake

Not awaiting contract.deployed() or transaction receipts can make assertions run before state updates, producing false negatives.

Key Takeaway

Write tests that deploy fresh contracts, await async operations, and assert both values and revert messages to capture behavior reliably.
