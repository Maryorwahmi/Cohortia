Test Skeleton
// test/TokenSale.js (skeleton)
const { expect } = require("chai");
const { ethers } = require("hardhat");

describe("TokenSale integration", function () {
  let Token, token, TokenSale, tokenSale, owner, buyer;
  const TOKEN_PRICE = ethers.utils.parseUnits("1", "wei"); // example price in Wei

  beforeEach(async function () {
    Token = await ethers.getContractFactory("ERC20Token");
    TokenSale = await ethers.getContractFactory("TokenSale");
    [owner, buyer] = await ethers.getSigners();

    token = await Token.deploy(ethers.utils.parseUnits("1000", 18));
    await token.deployed();

    tokenSale = await TokenSale.deploy(token.address, TOKEN_PRICE);
    await tokenSale.deployed();

    // Arrange: transfer tokens to tokenSale so it can sell
    await token.transfer(tokenSale.address, ethers.utils.parseUnits("500", 18));
  });

  it("Deployment sets correct initial state", async function () {
    expect(await tokenSale.token()).to.equal(token.address);
    expect(await tokenSale.tokenPrice()).to.equal(TOKEN_PRICE);
  });

  // More tests: successful purchase, insufficient funds, insufficient tokens
});
Example assertion

A successful purchase test should assert the TokensPurchased event, buyer token balance increased by expected token amount, and tokenSale Ether balance increased by msg.value. Keep token amount calculations explicit so the test documents intent.

Pro Tip

Convert values using ethers.utils to avoid unit mistakes. Be explicit about decimals and tokenPrice units.

Common Mistake

Not transferring tokens into the sale contract during setup. Tests should arrange contract balances, not assume them.
