Code (C) — linear_search
#include <stdio.h>
#include <stdbool.h> // For boolean type

// Function to perform linear search
int linear_search(int arr[], int size, int target) {
  for (int i = 0; i < size; i++) {
    if (arr[i] == target) {
      return i; // Target found at index i
    }
  }
  return -1; // Target not found
}

int main() {
  int numbers[] = {10, 5, 8, 12, 3, 7};
  int size = sizeof(numbers) / sizeof(numbers[0]);
  int target1 = 8;
  int target2 = 99;

  int result1 = linear_search(numbers, size, target1);
  if (result1 != -1) {
    printf("Target %d found at index %d.
", target1, result1);
  } else {
    printf("Target %d not found.
", target1);
  }

  int result2 = linear_search(numbers, size, target2);
  if (result2 != -1) {
    printf("Target %d found at index %d.
", target2, result2);
  } else {
    printf("Target %d not found.
", target2);
  }

  return 0;
}
Trace and costs
  • Each loop iteration does one comparison arr[i] == target; count these to estimate time.
  • An early return reduces iterations — best case constant time when match is first.
  • When not found, the loop examines all elements: worst-case linear time proportional to size.
Concrete walkthrough
With numbers {10,5,8,12,3,7} and target 8, the function checks 10, then 5, then 8 and returns index 2 — three comparisons total.
