Starter C code
#include <stdio.h>

long long power(int base, int exp) {
  // Base case: If exponent is 0, result is 1
  if (exp == 0) {
    return 1;
  }
  // Recursive step: base * power(base, exp - 1)
  else {
    return base * power(base, exp - 1);
  }
}

int main() {
  printf("2^5 = %lld
", power(2, 5));
  printf("3^0 = %lld
", power(3, 0));
  printf("10^3 = %lld
", power(10, 3));
  return 0;
}
