Starter Code
def calculate(num1, num2, op):
  """
  Performs a basic arithmetic operation on two numbers.
  Returns a number or an error message string.
  """
  if op == '+':
    return num1 + num2
  elif op == '-':
    return num1 - num2
  elif op == '*':
    return num1 * num2
  elif op == '/':
    if num2 == 0:
      return "Error: Division by zero!"
    return num1 / num2
  else:
    return "Error: Invalid operator."
