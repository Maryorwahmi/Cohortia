Transaction example
import sqlite3

conn = sqlite3.connect('my_application.db')
cursor = conn.cursor()

# Ensure the users table exists
cursor.execute('''
    CREATE TABLE IF NOT EXISTS users (
        id INTEGER PRIMARY KEY,
        username TEXT NOT NULL UNIQUE,
        email TEXT NOT NULL
    )
''')
conn.commit()

try:
    cursor.execute("INSERT INTO users (username, email) VALUES (?, ?)", ("bob", "bob@example.com"))
    print("Inserted user 'bob'.")

    cursor.execute("UPDATE users SET email = ? WHERE username = ?", ("new.alice@example.com", "alice"))
    if cursor.rowcount == 0:
        raise ValueError("User 'alice' not found for update, simulating a failure.")
    print("Updated user 'alice'.")

    conn.commit()
    print("Transaction committed: Both operations saved.")

except Exception as e:
    conn.rollback()
    print(f"Transaction rolled back: {e}")

finally:
    cursor.close()
    conn.close()
What this demonstrates
If the update for 'alice' fails, the earlier insert of 'bob' is also undone by rollback, preserving atomicity.
Verify after
Open a fresh connection and run a SELECT to list users and confirm whether the transaction was committed or rolled back.
