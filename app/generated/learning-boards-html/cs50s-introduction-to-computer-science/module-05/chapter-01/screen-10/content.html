Module 5 › Chapter 5.1 › Schema Example
Students
CREATE TABLE Students (
  StudentID INT PRIMARY KEY,
  FirstName VARCHAR(50) NOT NULL,
  LastName VARCHAR(50) NOT NULL,
  Email VARCHAR(100) UNIQUE,
  EnrollmentDate DATE
);
Courses
CREATE TABLE Courses (
  CourseID VARCHAR(10) PRIMARY KEY,
  CourseName VARCHAR(100) NOT NULL,
  Credits INT
);
Enrollments
CREATE TABLE Enrollments (
  EnrollmentID INT PRIMARY KEY AUTOINCREMENT,
  StudentID INT,
  CourseID VARCHAR(10),
  Grade CHAR(2),
  EnrollmentDate DATE,
  FOREIGN KEY (StudentID) REFERENCES Students(StudentID),
  FOREIGN KEY (CourseID) REFERENCES Courses(CourseID)
);
Design note
The Enrollments table models many-to-many relationships and ensures each enrollment refers to valid students and courses via foreign keys.
