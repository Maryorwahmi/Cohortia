Starter code for a simple copy program. Note the explicit argc check and error paths that close any already-opened streams.
#include <stdio.h>
int main(int argc, char *argv[])
{
  if (argc != 3) {
    printf("Usage: %s <input> <output>
", argv[0]);
    return 1;
  }
  FILE *in = fopen(argv[1], "r");
  if (in == NULL) { printf("Error: Could not open %s.
", argv[1]); return 1; }
  FILE *out = fopen(argv[2], "w");
  if (out == NULL) { printf("Error: Could not open %s.
", argv[2]); fclose(in); return 1; }

  int c;
  while ((c = fgetc(in)) != EOF) {
    fputc(c, out);
  }

  fclose(in);
  fclose(out);
  printf("File copied successfully.
");
  return 0;
}
Try compiling and running with different argument counts and file names to see the error handling. For binary-safe copies open both files with "rb" and "wb" and consider using fread/fwrite for performance.
