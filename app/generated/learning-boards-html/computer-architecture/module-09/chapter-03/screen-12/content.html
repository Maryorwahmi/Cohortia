Core loop
// Pseudocode for a 2-bit saturating counter update
index = (PC >> 2) & (BHT_size-1)
state = BHT[index]   // 0..3
predict = (state & 0b10) != 0
if predict != actual_outcome:
  mispredictions += 1
// update state
if actual_outcome == TAKEN:
  state = min(state + 1, 3)
else:
  state = max(state - 1, 0)
BHT[index] = state
