Full contract (abridged for clarity)
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract AdvancedStorage {
    uint256 private data; 
    address public owner; 
    event DataChanged(address indexed user, uint256 oldData, uint256 newData);

    constructor(uint256 initialData) {
        data = initialData;
        owner = msg.sender;
    }

    function setData(uint256 _newData) public {
        require(msg.sender == owner, "Only the owner can set the data.");
        uint256 oldData = data;
        data = _newData;
        emit DataChanged(msg.sender, oldData, _newData);
    }

    function getData() public view returns (uint256) { return data; }
}
Note: make sure names used in emit match the actual local and state variables to avoid compilation errors.
Explanation & examples
  • Constructor: runs once and sets initial data and owner using the deploying address.
  • msg.sender: the address that initiated the transaction; used here to assign ownership and log the event source.
  • Event indexing: indexed allows efficient searches for the user field in logs.
Concrete example
If deployer sets initialData to 42, getData returns 42. If the owner calls setData(123), DataChanged is emitted with oldData 42 and newData 123.
Common mistake
Emitting a misspelled variable or forgetting to set constructor args will fail compilation or deployment.
