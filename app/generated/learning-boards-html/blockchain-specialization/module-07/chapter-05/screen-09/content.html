Hardhat deploy & mint

A minimal contract and deployment script illustrate the exact operations you will run on a testnet.

// contracts/MyNFT.sol
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.19;

import "@openzeppelin/contracts/token/ERC721/ERC721.sol";
import "@openzeppelin/contracts/access/Ownable.sol";
import "@openzeppelin/contracts/utils/Counters.sol";

contract MyNFT is ERC721, Ownable {
    using Counters for Counters.Counter;
    Counters.Counter private _tokenIdCounter;

    constructor() ERC721("MyAwesomeNFT","MANFT") {}

    function safeMint(address to, string memory uri) public onlyOwner {
        uint256 newItemId = _tokenIdCounter.current();
        _tokenIdCounter.increment();
        _safeMint(to, newItemId);
        // _setTokenURI(newItemId, uri); // optional pattern via ERC721URIStorage
    }
}
Deployment script highlights
  • Configure provider and private key via environment variables.
  • Deploy the contract factory and wait for confirmation.
  • Mint with safeMint(owner, tokenURI) and confirm token ownership.
Key Takeaway

Keep secrets out of source control, pin metadata before minting, and verify tokenURI and ownerOf after minting.
