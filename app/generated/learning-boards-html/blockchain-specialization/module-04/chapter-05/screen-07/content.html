Code Walkthrough

AccessControl → ManagedToken

// Base contract for basic ownership
contract AccessControl {
  address public owner;
  event OwnershipTransferred(address indexed previousOwner, address indexed newOwner);
  constructor() { owner = msg.sender; }
  modifier onlyOwner() { require(msg.sender == owner, "Caller is not the owner"); _; }
  function transferOwnership(address _newOwner) public onlyOwner { require(_newOwner != address(0)); owner = _newOwner; emit OwnershipTransferred(msg.sender, _newOwner); }
}

// Derived contract
contract ManagedToken is AccessControl {
  string public name; string public symbol; uint256 public totalSupply; mapping(address => uint256) public balances;
  constructor(string memory _name, string memory _symbol) { name = _name; symbol = _symbol; }
  function mint(address _to, uint256 _amount) public onlyOwner { totalSupply += _amount; balances[_to] += _amount; }
}
Runtime scenario
Deploy ManagedToken on Remix using JavaScript VM with constructor args "MyToken","MTK". The deployer is owner and can mint. Switching accounts to a non-owner and calling mint should revert.
Misconception
Token transfer does not change contract ownership. transfer updates balances; transferOwnership updates who can call onlyOwner functions.
Key Takeaway

Use the base contract to centralize access control and keep derived contracts focused on domain logic like minting and transfers.
