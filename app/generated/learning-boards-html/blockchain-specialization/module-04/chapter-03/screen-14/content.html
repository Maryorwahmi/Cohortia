Build and test a voting contract

Starter code (paste into Remix)
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract VotingContract {
    address public owner;
    string[] public candidateNames;
    mapping(uint256 => uint256) public votes; // candidateId => voteCount
    mapping(address => bool) public hasVoted;

    error NotOwner();
    error AlreadyVoted();
    error InvalidCandidateId();

    constructor(string[] memory _candidateNames) {
        owner = msg.sender;
        require(_candidateNames.length > 0, "Must provide at least one candidate.");
        candidateNames = _candidateNames;
    }

    function addCandidate(string memory _name) public {
        if (msg.sender != owner) revert NotOwner();
        candidateNames.push(_name);
    }

    function vote(uint256 _candidateId) public {
        require(!hasVoted[msg.sender], "You have already voted.");
        require(_candidateId < candidateNames.length, "Invalid candidate ID.");

        votes[_candidateId]++;
        hasVoted[msg.sender] = true;
    }

    function getTotalVotes() public view returns (uint256) {
        uint256 total = 0;
        for (uint256 i = 0; i < candidateNames.length; i++) {
            total += votes[i];
        }
        return total;
    }
}
Note: Deploy with a small candidate array like ["Alice","Bob","Charlie"] to keep loops safe during testing.
