# Build for production
npm run build

# Create .env with Pinata keys
PINATA_API_KEY=YOUR_PINATA_API_KEY
PINATA_SECRET_API_KEY=YOUR_PINATA_SECRET_API_KEY

# Run deploy script
node deploy-ipfs.js

# Output: IPFS CID and gateway URLs
Ensure .env is excluded from source control and use a secrets manager in CI for automation.
