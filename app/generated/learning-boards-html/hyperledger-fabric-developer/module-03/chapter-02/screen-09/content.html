func (s *SimpleChaincode) createCar(stub shim.ChaincodeStubInterface, args []string) peer.Response {
// 1. Argument Validation
if len(args) != 6 {
return shim.Error("Incorrect number of arguments. Expecting 6: ID, Make, Model, Color, Owner, Value")
}
carID := args[0]
// ... other args ...
value, err := strconv.Atoi(args[5])
if err != nil { return shim.Error(fmt.Sprintf("Invalid value: %s", args[5])) }
// 2. Check for existing car
carAsBytes, err := stub.GetState(carID)
if err != nil { return shim.Error(fmt.Sprintf("Failed to get car: %s", err)) }
if carAsBytes != nil { return shim.Error(fmt.Sprintf("Car %s already exists", carID)) }
// 3. Create Car struct & Marshal to JSON
car := Car{ID: carID, Make: args[1], Model: args[2], Color: args[3], Owner: args[4], Value: value}
carJSONasBytes, err := json.Marshal(car)
if err != nil { return shim.Error(fmt.Sprintf("Failed to marshal: %s", err)) }
// 4. Store on ledger
err = stub.PutState(carID, carJSONasBytes)
if err != nil { return shim.Error(fmt.Sprintf("Failed to create car: %s", err)) }
return shim.Success(carJSONasBytes)
}
