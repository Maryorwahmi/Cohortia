func (s *SimpleChaincode) transferCar(stub shim.ChaincodeStubInterface, args []string) peer.Response {
// 1. Argument Validation
if len(args) != 2 {
return shim.Error("Incorrect number of arguments. Expecting 2: Car ID, New Owner")
}
carID := args[0]
newOwner := args[1]
// 2. Retrieve existing car
carAsBytes, err := stub.GetState(carID)
if err != nil { return shim.Error(fmt.Sprintf("Failed to get car: %s", err)) }
if carAsBytes == nil { return shim.Error(fmt.Sprintf("Car %s does not exist", carID)) }
// 3. Unmarshal, Modify, Marshal
car := Car{}
err = json.Unmarshal(carAsBytes, &car)
if err != nil { return shim.Error(fmt.Sprintf("Failed to unmarshal: %s", err)) }
car.Owner = newOwner // Update the owner
carJSONasBytes, err := json.Marshal(car)
if err != nil { return shim.Error(fmt.Sprintf("Failed to marshal updated car: %s", err)) }
// 4. Store updated car on ledger
err = stub.PutState(carID, carJSONasBytes)
if err != nil { return shim.Error(fmt.Sprintf("Failed to transfer car: %s", err)) }
return shim.Success(carJSONasBytes)
}

Implementing transferCar (Update)

Modifying an existing car's ownership

Key Steps for Update

  • Retrieve the existing asset using stub.GetState.
  • Unmarshal the JSON bytes into a Go struct.
  • Modify the desired fields of the struct.
  • Marshal the updated struct back into JSON bytes.
  • Use stub.PutState to overwrite the old record with the new, updated data.
Key Takeaway

The retrieve-modify-save pattern (GetState -> Unmarshal -> Modify -> Marshal -> PutState) is fundamental for all asset updates on the ledger.
