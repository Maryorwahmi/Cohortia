func (s *SimpleChaincode) deleteCar(stub shim.ChaincodeStubInterface, args []string) peer.Response {
// 1. Argument Validation
if len(args) != 1 {
return shim.Error("Incorrect number of arguments. Expecting 1: Car ID")
}
carID := args[0]
// 2. Check if car exists before deleting
carAsBytes, err := stub.GetState(carID)
if err != nil { return shim.Error(fmt.Sprintf("Failed to get car: %s", err)) }
if carAsBytes == nil { return shim.Error(fmt.Sprintf("Car %s does not exist, cannot delete", carID)) }
// 3. Delete from ledger
err = stub.DelState(carID)
if err != nil { return shim.Error(fmt.Sprintf("Failed to delete car: %s", err)) }
return shim.Success([]byte(fmt.Sprintf("Car %s deleted", carID)))
}

Removing car assets from the ledger

Key Steps

  • Validate the single argument (Car ID).
  • Crucially, check if the asset exists using stub.GetState before attempting deletion.
  • Use stub.DelState(carID) to remove the key-value pair from the ledger.
  • Return success or an error based on the operation's outcome.
Key Takeaway

Always verify an asset's existence before calling stub.DelState to ensure you're deleting the correct item and to provide meaningful error messages.
