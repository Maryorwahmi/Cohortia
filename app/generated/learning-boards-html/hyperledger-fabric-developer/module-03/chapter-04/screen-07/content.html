Testing Chaincode Queries: TestQueryCar

Validating data retrieval and handling non-existent entries.

Query Test Scenarios
1

Setup Data: First, use createCar via MockInvoke to populate the mock ledger with a car to query.

2

Successful Query: Invoke queryCar, assert shim.OK, unmarshal the payload, and verify the returned car's properties.

3

Non-Existent Query: Invoke queryCar for an ID that doesn't exist. Assert shim.ERROR and check for an appropriate error message.

Key Takeaway

Thorough query testing ensures your chaincode reliably retrieves data and provides clear feedback for missing entries.

GoLang TestQueryCar Example
// TestQueryCar tests the queryCar function
func TestQueryCar(t *testing.T) {
    scc := new(SimpleChaincode)
    stub := test.NewMockStub("simplechaincode", scc)
    stub.MockInit("1", nil)

    // First, create a car to query
    createArgs := util.ToChaincodeArgs("createCar", "CAR2", "BMW", "X5", "black", "Bob", "30000")
    stub.MockInvoke("2", createArgs)

    // Test case 1: Successful query
    queryArgs := util.ToChaincodeArgs("queryCar", "CAR2")
    response := stub.MockInvoke("3", queryArgs)
    assert.Equal(t, int32(shim.OK), response.GetStatus(), "queryCar failed for CAR2")
    assert.NotEmpty(t, response.GetPayload(), "queryCar returned empty payload")

    var car Car
    err := json.Unmarshal(response.GetPayload(), &car)
    assert.NoError(t, err, "Failed to unmarshal queried car")
    assert.Equal(t, "Bob", car.Owner, "Queried car owner mismatch")

    // Test case 2: Querying a non-existent car
    queryArgs = util.ToChaincodeArgs("queryCar", "NONEXISTENT")
    response = stub.MockInvoke("4", queryArgs)
    assert.Equal(t, int32(shim.ERROR), response.GetStatus(), "queryCar should fail for non-existent car")
    assert.Contains(t, string(response.GetMessage()), "does not exist", "Error message should indicate non-existent car")
}
