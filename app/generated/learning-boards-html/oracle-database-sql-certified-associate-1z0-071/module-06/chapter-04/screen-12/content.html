Module 6 › Chapter 6.4 › Activity: projects table
Starter SQL (commented)
-- CREATE TABLE projects
CREATE TABLE projects (
    project_id      NUMBER(6) NOT NULL,
    project_name    VARCHAR2(100) NOT NULL,
    start_date      DATE DEFAULT SYSDATE,
    end_date        DATE,
    status          VARCHAR2(20) DEFAULT 'In Progress',
    budget          NUMBER(12,2)
);

-- ALTER to add manager_id
ALTER TABLE projects ADD (manager_id NUMBER(6));

-- MODIFY project_name to 150
ALTER TABLE projects MODIFY (project_name VARCHAR2(150));

-- RENAME status
ALTER TABLE projects RENAME COLUMN status TO project_status;

-- INSERT sample
INSERT INTO projects (project_id, project_name, end_date, budget, manager_id)
VALUES (1, 'Website Redesign', TO_DATE('2024-12-31','YYYY-MM-DD'), 50000.00, 101);

-- VERIFY
-- DESCRIBE projects; SELECT * FROM projects;

-- CLEANUP
DROP TABLE projects;
Verification tips
  • Run DESCRIBE to confirm column types and defaults.
  • SELECT a few rows to confirm defaults populated as expected.
  • Use TO_DATE when inserting dates to avoid format issues.
Key Takeaway
Follow a staged approach: create, add nullable columns with defaults, backfill, then enforce constraints.
