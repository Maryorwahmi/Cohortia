Server code (excerpt)
from flask import Flask, request, jsonify
import sqlite3

app = Flask(__name__)

@app.route('/api/readings', methods=['POST'])
def add_reading():
    data = request.get_json()
    timestamp = data.get('timestamp')
    sensor_id = data.get('sensor_id')
    value = data.get('value')

    conn = sqlite3.connect('iot_data.db')
    c = conn.cursor()
    c.execute("INSERT INTO sensor_readings VALUES (?, ?, ?)",
              (timestamp, sensor_id, value))
    conn.commit()
    conn.close()
    return jsonify({"status": "success"}), 201
This handler parses JSON, inserts into SQLite, and returns 201 on success. Validate fields before inserting in production.

@app.route('/api/readings/<sensor_id>', methods=['GET'])
def get_readings(sensor_id):
    conn = sqlite3.connect('iot_data.db')
    c = conn.cursor()
    c.execute("SELECT * FROM sensor_readings WHERE sensor_id = ? ORDER BY timestamp DESC LIMIT 10",
              (sensor_id,))
    rows = c.fetchall()
    conn.close()
    return jsonify(rows)
