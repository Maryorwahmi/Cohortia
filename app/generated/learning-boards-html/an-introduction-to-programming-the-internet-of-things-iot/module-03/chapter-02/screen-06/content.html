Process HTTP GET example
Code (Ethernet client)
#include <SPI.h>
#include <Ethernet.h>

byte mac[] = { 0xDE, 0xAD, 0xBE, 0xEF, 0xFE, 0xED };
IPAddress ip(192, 168, 1, 100);
IPAddress server(192, 168, 1, 50);  // Local server IP

EthernetClient client;

void setup() {
  Serial.begin(9600);
  Ethernet.begin(mac, ip);
  delay(1000);

  if (client.connect(server, 80)) {
    Serial.println("Connected to server");
    client.println("GET /data?temp=25.5&hum=60 HTTP/1.1");
    client.println("Host: 192.168.1.50");
    client.println("Connection: close");
    client.println();
  } else {
    Serial.println("Connection failed");
  }
}

void loop() {
  if (client.available()) {
    char c = client.read();
    Serial.print(c);
  }
  if (!client.connected()) {
    client.stop();
    delay(60000);  // Send data every minute
  }
}
Explanation
  • Set a MAC and IP, then initialize Ethernet using Ethernet.begin.
  • Open a client connection to the server IP and port 80 to send HTTP lines.
  • Include sensor values as URL query parameters for a simple telemetry send.
  • Read and print the server response, then handle disconnects and delays between sends.
Example check
Run this sketch and watch Serial. On success you will see 'Connected to server' and the server response printed; inspect the server log to find the data parameters.
Security note

HTTP GET sends data in the URL and is not encrypted; avoid sensitive information and consider secure alternatives for production.
