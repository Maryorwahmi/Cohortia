#include <stdio.h>

int main() {
  int score;
  printf("--- Grade Calculator ---
");
  printf("Enter the student's numerical score (0-100): ");
  if (scanf("%d", &score) != 1) {
    printf("Invalid input. Please enter a number.
");
    return 1;
  }
  while (getchar() != '
' && !feof(stdin));

  if (score < 0 || score > 100) {
    printf("Error: Score must be between 0 and 100.
");
  } else if (score >= 90) {
    printf("Grade: A
");
  } else if (score >= 80) {
    printf("Grade: B
");
  } else if (score >= 70) {
    printf("Grade: C
");
  } else if (score >= 60) {
    printf("Grade: D
");
  } else {
    printf("Grade: F
");
  }

  const char* status = (score >= 60 && score <= 100) ? "Pass" : "Fail";
  printf("Overall Status: %s
", status);
  return 0;
}
