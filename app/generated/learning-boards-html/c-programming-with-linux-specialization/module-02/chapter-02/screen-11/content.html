Starter code (preview)
#include <stdio.h>

int main() {
  int a = 10, b = 3;
  double x = 10.0, y = 3.0;
  int counter = 0;

  printf("a / b = %d
", a / b); // integer division
  printf("(double)a / b = %.2f
", (double)a / b); // floating-point

  int val1 = ++counter; // prefix
  int val2 = counter++; // postfix
  printf("counter = %d, val1 = %d, val2 = %d
", counter, val1, val2);

  return 0;
}
