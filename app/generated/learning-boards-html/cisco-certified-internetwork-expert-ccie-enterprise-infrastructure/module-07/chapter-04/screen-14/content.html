Activity Goals
  • Connect to multiple devices using netmiko.
  • Collect a set of show commands and save outputs to files.
  • Log successes and failures for auditability.
Try this
Add show cdp neighbors detail to the commands_to_gather list and confirm the corresponding file is created for each reachable device.
Starter template
from netmiko import ConnectHandler, NetmikoAuthenticationException, NetmikoTimeoutException
import logging
import os

logging.basicConfig(filename="device_info.log", level=logging.INFO,
                    format="%(asctime)s - %(levelname)s - %(message)s")

devices = [
  {"device_type": "cisco_ios", "host": "192.168.1.101", "username": "admin", "password": "cisco", "secret": "enable_password"},
  {"device_type": "cisco_ios", "host": "192.168.1.102", "username": "admin", "password": "cisco", "secret": "enable_password"}
]

commands_to_gather = ["show version", "show ip interface brief", "show cdp neighbors detail"]

output_dir = "device_outputs"
if not os.path.exists(output_dir):
  os.makedirs(output_dir)

for device in devices:
  device_name = device["host"]
  logging.info(f"Attempting to connect to {device_name}")
  try:
    with ConnectHandler(**device) as net_connect:
      logging.info(f"Connected to {device_name}")
      for cmd in commands_to_gather:
        output = net_connect.send_command(cmd)
        filename = f"{output_dir}/{device_name}_{cmd.replace(' ', '_')}.txt"
        with open(filename, "w") as f:
          f.write(output)
        logging.info(f"Saved output of '{cmd}' for {device_name} to {filename}")
  except NetmikoAuthenticationException:
    logging.error(f"Authentication failed for {device_name}")
  except NetmikoTimeoutException:
    logging.error(f"Connection timed out for {device_name}")
  except Exception as e:
    logging.critical(f"Unexpected error for {device_name}: {e}", exc_info=True)

logging.info("Script finished.")
Key Takeaway

This activity produces raw evidence and logs. Use it as a scaffold to add parsing, credential sources, and retries later.
