// scripts/interact.js
require("dotenv").config();
const { ethers } = require("hardhat");

async function main() {
  const contractAddress = process.env.COUNTER_CONTRACT_ADDRESS;
  const privateKey = process.env.PRIVATE_KEY;

  const provider = new ethers.JsonRpcProvider(process.env.SEPOLIA_RPC_URL);
  const signer = new ethers.Wallet(privateKey, provider);

  const CounterArtifact = require("../artifacts/contracts/Counter.sol/Counter.json");
  const counterABI = CounterArtifact.abi;

  const counterContract = new ethers.Contract(contractAddress, counterABI, signer);

  let currentCount = await counterContract.count();
  console.log("Current count:", currentCount.toString());

  const tx = await counterContract.increment();
  await tx.wait();
  console.log("Increment tx:", tx.hash);
}

main().catch(console.error);
Running the script
Set COUNTER_CONTRACT_ADDRESS and PRIVATE_KEY in .env, then run node scripts/interact.js. Watch the terminal output for the current count, transaction hash, and updated count after confirmation.
Key Takeaway

This interact script demonstrates the complete read-write-confirm cycle and the use of the transaction hash for verification.
