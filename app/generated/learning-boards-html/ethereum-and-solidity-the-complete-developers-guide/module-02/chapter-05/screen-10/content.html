pragma solidity ^0.8.0;

contract MyToken {
    string public name = "My Cohortia Token";
    string public symbol = "MCT";
    uint8 public decimals = 18;
    uint public totalSupply;

    mapping(address => uint) public balances;

    address public owner;

    // TODO: Declare a Transfer event (from, to, amount)
    // TODO: Declare an Approval event (owner, spender, amount)

    constructor(uint _initialSupply) {
        owner = msg.sender;
        totalSupply = _initialSupply * (10 ** uint(decimals));
        balances[owner] = totalSupply;
    }

    function transfer(address _to, uint _value) public returns (bool success) {
        require(balances[msg.sender] >= _value, "Insufficient balance.");
        require(_to != address(0), "Cannot transfer to zero address.");

        balances[msg.sender] -= _value;
        balances[_to] += _value;

        // TODO: Emit the Transfer event
        emit Transfer(msg.sender, _to, _value);
        return true;
    }

    function approve(address _spender, uint _value) public returns (bool success) {
        // TODO: Implement approval logic and emit Approval event
        emit Approval(msg.sender, _spender, _value);
        return true;
    }
}
Lab steps
  1. Create MyToken.sol and paste starter code.
  2. Declare Transfer and Approval events with indexed addresses.
  3. Add emit statements after state updates in transfer and approve.
  4. Deploy to the JavaScript VM and call transfer and approve.
  5. Open the transaction logs in Remix and inspect topics and data.
Tip
Deploy with a small initial supply to make testing transfers simple, and check the logs tab under the transaction details.
