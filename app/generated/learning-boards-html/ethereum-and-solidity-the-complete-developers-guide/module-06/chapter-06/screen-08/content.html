Test skeleton (excerpt)
const { expect } = require("chai");
const { ethers } = require("hardhat");

describe("Tx.origin Vulnerability", function () {
  let VulnerableWalletTxOrigin, vulnerableWalletTxOrigin;
  let MaliciousRelay, maliciousRelay;
  let owner, attacker;

  beforeEach(async function () {
    [owner, attacker] = await ethers.getSigners();

    VulnerableWalletTxOrigin = await ethers.getContractFactory("VulnerableWalletTxOrigin", owner);
    vulnerableWalletTxOrigin = await VulnerableWalletTxOrigin.deploy({ value: ethers.utils.parseEther("1") });

    MaliciousRelay = await ethers.getContractFactory("MaliciousRelay", attacker);
    maliciousRelay = await MaliciousRelay.deploy(vulnerableWalletTxOrigin.address);
  });

  it("Should demonstrate tx.origin phishing attack", async function () {
    const transferAmount = ethers.utils.parseEther("0.5");
    await maliciousRelay.connect(owner).phishingAttack(owner.address, transferAmount);
    expect(await vulnerableWalletTxOrigin.getBalance(attacker.address)).to.equal(transferAmount);
  });
});
This example shows how to reproduce the exploit. Extend it by asserting owner balances and testing the patched contract expecting reverts.
Testing checklist
  • Simulate owner calling the relay to reproduce the exploit.
  • Assert precise balance changes in the contract’s internal storage.
  • Deploy patched contract and expect a revert when repeating the attack.
Remember: tests should mimic attacker behavior, not just benign user interactions.
