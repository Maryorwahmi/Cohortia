// Start Ganache (GUI or CLI) and note RPC URL
// Then run node connectLocal.js which contains below examples
Web3.js example
const Web3 = require('web3');
const web3 = new Web3('http://127.0.0.1:7545');

(async () => {
  try {
    const accounts = await web3.eth.getAccounts();
    console.log('Connected accounts (Web3.js):', accounts);
  } catch (err) {
    console.error('Error connecting with Web3.js:', err.message);
  }
})();
Ethers.js example
const { ethers } = require('ethers');
const provider = new ethers.providers.JsonRpcProvider('http://127.0.0.1:7545');

(async () => {
  try {
    const accounts = await provider.listAccounts();
    const block = await provider.getBlockNumber();
    console.log('Connected accounts (Ethers.js):', accounts);
    console.log('Local block number:', block);
  } catch (err) {
    console.error('Error connecting with Ethers.js:', err.message);
  }
})();
Warning
Do not hardcode private keys in production. Use Ganache test accounts only for local testing and never commit secrets to version control.
