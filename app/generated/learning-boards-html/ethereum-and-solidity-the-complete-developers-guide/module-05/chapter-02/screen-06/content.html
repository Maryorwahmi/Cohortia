Web3.js — Sepolia via Infura
const Web3 = require('web3');
const web3 = new Web3('https://sepolia.infura.io/v3/YOUR_INFURA_PROJECT_ID');

(async () => {
  try {
    const block = await web3.eth.getBlockNumber();
    console.log('Current Sepolia block number (Web3.js):', block);
  } catch (err) {
    console.error('Error connecting to Sepolia with Web3.js:', err.message);
  }
})();
Tips
  • Replace the placeholder with your actual project ID before running.
  • Broken or missing keys typically produce authentication or 401-style errors.
  • Expect rate limits on free tiers; handle errors with retries or backoff.
Ethers.js — Sepolia via Alchemy
const { ethers } = require('ethers');
const provider = new ethers.providers.JsonRpcProvider('https://eth-sepolia.g.alchemy.com/v2/YOUR_ALCHEMY_API_KEY');

(async () => {
  try {
    const block = await provider.getBlockNumber();
    console.log('Current Sepolia block number (Ethers.js):', block);

    const balanceBigNumber = await provider.getBalance('TARGET_ADDRESS');
    const balance = ethers.utils.formatEther(balanceBigNumber);
    console.log('Balance:', balance, 'ETH');
  } catch (err) {
    console.error('Error connecting to Sepolia with Ethers.js:', err.message);
  }
})();
Security note
API keys in client code are readable. Use limited-scope keys and avoid storing sensitive server secrets in the frontend.
