hardhat.config.js (example)

require("@nomicfoundation/hardhat-ethers");
require("@nomicfoundation/hardhat-chai-matchers");
require("@nomicfoundation/hardhat-network-helpers");

/** @type import('hardhat/config').HardhatUserConfig */
module.exports = {
  solidity: "0.8.20",
  networks: {
    // network configs go here
  }
};
Why: require statements load plugin features available to tasks and tests.
Safety: avoid hardcoding private keys; read them from process.env in production-like configs.
Key Takeaway

Require plugins and set compiler versions in hardhat.config.js. Use environment variables to protect secrets.
