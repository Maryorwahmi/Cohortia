using UnityEngine;
using UnityEngine.UI;
using TMPro;

public class GameUIController : MonoBehaviour
{
    public TextMeshProUGUI scoreText;
    public Image healthBarFill;
    public Button attackButton;

    private int currentScore = 0;
    private float currentHealth = 1.0f;

    void Start()
    {
        UpdateScoreUI();
        UpdateHealthUI();

        if (attackButton != null)
            attackButton.onClick.AddListener(OnAttackButtonClick);
    }

    void OnAttackButtonClick()
    {
        currentScore += 10;
        currentHealth = Mathf.Max(0, currentHealth - 0.1f);
        UpdateScoreUI();
        UpdateHealthUI();
    }

    void UpdateScoreUI() { if (scoreText != null) scoreText.text = "Score: " + currentScore.ToString(); }
    void UpdateHealthUI() { if (healthBarFill != null) healthBarFill.fillAmount = currentHealth; }
}
Visual snippet: assign scoreText, healthBarFill, and attackButton in the Inspector after attaching this script to a UIManager GameObject. Avoid forgetting the namespace imports for UI and TMPro.
