Key code excerpt
using UnityEngine;
using UnityEngine.EventSystems;
using UnityEngine.UI;

public class DragDropItem : MonoBehaviour, IBeginDragHandler, IDragHandler, IEndDragHandler
{
    private RectTransform rectTransform;
    private Canvas canvas;
    private CanvasGroup canvasGroup;

    void Awake()
    {
        rectTransform = GetComponent<RectTransform>();
        canvas = GetComponentInParent<Canvas>();
        canvasGroup = GetComponent<CanvasGroup>() ?? gameObject.AddComponent<CanvasGroup>();
    }

    public void OnBeginDrag(PointerEventData eventData) { /* set alpha, blocksRaycasts false, reparent, SetAsLastSibling */ }
    public void OnDrag(PointerEventData eventData) { rectTransform.anchoredPosition += eventData.delta / canvas.scaleFactor; }
    public void OnEndDrag(PointerEventData eventData) { /* restore alpha, blocksRaycasts true, finalize parent or leave where dropped */ }
}
Note the dependency on canvas.scaleFactor in OnDrag. This keeps drag motion proportional when Canvas uses different render modes or scale settings.
Debug tips
  • Log eventData.delta and canvas.scaleFactor to validate movement math.
  • Log eventData.pointerCurrentRaycast.gameObject in OnEndDrag to inspect drop target hits.
  • If item disappears behind UI, reparent or SetAsLastSibling during begin.
Common error

Using transform.position instead of anchoredPosition breaks layout and can misplace items. Use RectTransform APIs for UI movement.
