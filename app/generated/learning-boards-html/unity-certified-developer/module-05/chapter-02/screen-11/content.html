Options Panel Architecture

Group related UI elements and handlers under a single manager for small panels so initialization and cleanup happen together.

  • Declare public fields for each UI component and assign in the Inspector.
  • Initialize control values from AudioListener and PlayerPrefs in Start.
  • Subscribe to events with AddListener and unsubscribe in OnDestroy.
Canonical snippet
using UnityEngine;
using UnityEngine.UI;
using TMPro;

public class UIManager : MonoBehaviour
{
    public Toggle musicToggle;
    public Slider volumeSlider;
    public TMP_InputField playerNameInputField;

    void Start()
    {
        if (musicToggle != null)
        {
            musicToggle.onValueChanged.AddListener(SetMusicEnabled);
            musicToggle.isOn = !AudioListener.pause;
        }
        if (volumeSlider != null)
        {
            volumeSlider.onValueChanged.AddListener(SetVolume);
            volumeSlider.value = AudioListener.volume;
        }
        if (playerNameInputField != null)
        {
            playerNameInputField.onEndEdit.AddListener(OnPlayerNameEntered);
            playerNameInputField.text = PlayerPrefs.GetString("PlayerName", "New Player");
        }
    }

    // ... handlers and cleanup ...
}
Key Takeaway
A single manager simplifies wiring and debugging for compact UI panels; split responsibilities as a panel grows.
