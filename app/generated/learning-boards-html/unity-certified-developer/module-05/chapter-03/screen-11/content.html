Script excerpt
public class DynamicMenuManager : MonoBehaviour {
  public GameObject buttonPrefab;
  public VerticalLayoutGroup menuLayoutGroup;

  void Start() {
    AddMenuItem("Start Game", StartGame);
    AddMenuItem("Options", OpenOptions);
    AddMenuItem("Quit", QuitGame);
  }

  void AddMenuItem(string text, UnityAction action) {
    var go = Instantiate(buttonPrefab, menuLayoutGroup.transform);
    // set label and listener
  }
}
Design notes
  • Prefab carries Layout Element / Content Size Fitter so sizing is self-contained.
  • Parent to the layout group's transform to let Unity handle ordering and placement.
  • Do not set child positions after parenting; rely on layout rules.
Key Takeaway

Keep visual and sizing hints in prefabs, and instantiate them under a layout group to get consistent runtime placement.
