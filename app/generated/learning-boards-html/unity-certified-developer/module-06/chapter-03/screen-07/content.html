Module 6 › Chapter 6.3 › HandIKController
Starter Script (visual)
public class HandIKController : MonoBehaviour
{
    public Animator animator;
    public Transform ikTargetSphere;
    public Transform elbowHint;
    [Range(0,1)] public float ikWeight = 0f;

    void Start()
    {
        if (animator == null) animator = GetComponent();
    }

    void OnAnimatorIK(int layerIndex)
    {
        if (animator == null) return;
        animator.SetIKPositionWeight(AvatarIKGoal.RightHand, ikWeight);
        animator.SetIKRotationWeight(AvatarIKGoal.RightHand, ikWeight);
        if (ikTargetSphere != null) {
            animator.SetIKPosition(AvatarIKGoal.RightHand, ikTargetSphere.position);
            animator.SetIKRotation(AvatarIKGoal.RightHand, ikTargetSphere.rotation);
            if (elbowHint != null) {
                animator.SetIKHintPositionWeight(AvatarIKHint.RightElbow, ikWeight);
                animator.SetIKHintPosition(AvatarIKHint.RightElbow, elbowHint.position);
            }
        } else {
            animator.SetIKPositionWeight(AvatarIKGoal.RightHand, 0);
            animator.SetIKRotationWeight(AvatarIKGoal.RightHand, 0);
        }
    }
}
Runtime checklist
  • • Animator reference is not null and Avatar is Humanoid.
  • • Target and hint transforms are assigned in the Inspector.
  • • ikWeight is animated or adjusted in Play to observe blending.
  • • Zero weights when target is removed to prevent residual overrides.
Try This

Animate ikWeight from 0 to 1 over half a second to observe a smooth transition between base animation and IK.
