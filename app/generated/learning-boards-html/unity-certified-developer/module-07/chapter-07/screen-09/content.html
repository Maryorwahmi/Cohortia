Module 8 › Chapter 7.7 › Lab: AnalyticsTest
Lab steps
  1. Create AnalyticsTest.cs and attach to an empty GameObject.
  2. Enable Analytics and Cloud Diagnostics in Services.
  3. Build a development build with networking enabled.
  4. Run the built player, press the event key several times, then press the crash key.
  5. Wait and verify events and the exception on the Dashboard.
Checklist

Confirm service toggles, build settings, and that the built player can reach the network for reporting.

Example script (visual)
using UnityEngine;
using UnityEngine.Analytics;
using System.Collections.Generic;

public class AnalyticsTest : MonoBehaviour
{
    void Start()
    {
        Analytics.CustomEvent("GameStarted", new Dictionary
        {
            { "session_id", System.Guid.NewGuid().ToString() },
            { "device_model", SystemInfo.deviceModel }
        });
    }

    void Update()
    {
        if (Input.GetKeyDown(KeyCode.E))
        {
            Analytics.CustomEvent("PlayerUsedAbility", new Dictionary
            {
                { "ability_name", "Dash" },
                { "cooldown_remaining", 5.0f }
            });
        }

        if (Input.GetKeyDown(KeyCode.C))
        {
            Debug.LogError("Simulating a critical error!");
            string nullRef = null;
            Debug.Log(nullRef.Length); // This will cause a NullReferenceException
        }
    }
}
Run the steps and allow a few minutes for the Dashboard to process and display events and exceptions.
