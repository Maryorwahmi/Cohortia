Example script
using UnityEngine;

public class LegacyPlayerInput : MonoBehaviour
{
    public float moveSpeed = 5f;
    public float jumpForce = 10f;

    private Rigidbody rb;

    void Start()
    {
        rb = GetComponent();
        if (rb == null)
        {
            Debug.LogError("Rigidbody missing!");
            enabled = false;
        }
    }

    void Update()
    {
        float horizontal = Input.GetAxis("Horizontal");
        float vertical = Input.GetAxis("Vertical");

        Vector3 moveDirection = new Vector3(horizontal, 0, vertical);
        transform.Translate(moveDirection * moveSpeed * Time.deltaTime);

        if (Input.GetButtonDown("Jump"))
        {
            rb.AddForce(Vector3.up * jumpForce, ForceMode.Impulse);
        }

        if (Input.GetMouseButtonDown(0))
        {
            Debug.Log("Fire!");
        }
    }
}
Annotations
  • Start(): defensive null check for Rigidbody to avoid runtime errors.
  • Update(): polls GetAxis and GetButtonDown; suitable for frame-accurate input.
  • Movement: uses transform.Translate — easy but not always physics-friendly.
  • Jump: AddForce on Rigidbody — physics-based action applied immediately when detected.
Practice change
Replace transform.Translate with stored input applied in FixedUpdate via rb.MovePosition to observe consistent physics behavior.
