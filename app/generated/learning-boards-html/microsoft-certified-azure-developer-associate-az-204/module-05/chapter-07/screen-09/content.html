Module 6 › Chapter 6.7 › Retrieving Secrets with `DefaultAzureCredential`
C# Code Snippet

using System;
using Azure.Identity;
using Azure.Security.KeyVault.Secrets;

public static class SecretRetriever
{
    private static readonly string KeyVaultUri = Environment.GetEnvironmentVariable("KEY_VAULT_URI");

    public static async Task GetSecretAsync(string secretName)
    {
        if (string.IsNullOrEmpty(KeyVaultUri))
        {
            Console.WriteLine("KEY_VAULT_URI environment variable is not set.");
            return null;
        }

        try
        {
            // DefaultAzureCredential automatically uses Managed Identity on Azure
            var client = new SecretClient(new Uri(KeyVaultUri), new DefaultAzureCredential());
            KeyVaultSecret secret = await client.GetSecretAsync(secretName);
            return secret.Value;
        }
        catch (Exception ex)
        {
            Console.WriteLine($"Error retrieving secret: {ex.Message}");
            return null;
        }
    }
}
Key Libraries

`Azure.Identity` for `DefaultAzureCredential` and `Azure.Security.KeyVault.Secrets` for interacting with Key Vault.

Key Takeaway

Leverage `DefaultAzureCredential` to seamlessly authenticate using Managed Identity when deployed to Azure.
