// Program.cs - Configuring a basic health check endpoint
using Microsoft.AspNetCore.Builder;using Microsoft.Extensions.DependencyInjection;using Microsoft.AspNetCore.Diagnostics.HealthChecks; // Add this using directivevar builder = WebApplication.CreateBuilder(args);// ... existing services ...builder.Services.AddHealthChecks(); // Add basic health checks// ... rest of Program.cs ...var app = builder.Build();app.UseHttpsRedirection(); // Ensure this is before health checks if using HTTPSapp.MapHealthChecks("/health", new HealthCheckOptions{    Predicate = _ => true,    ResponseWriter = HealthChecks.UI.Client.UIResponseWriter.WriteHealthCheckUIResponse});// ... rest of app.MapControllers() etc. ...app.Run();
AddHealthChecks()

Registers the necessary services for health checks in the dependency injection container.

MapHealthChecks()

Maps the /health endpoint to execute the registered health checks and return their status.

Key Takeaway

A simple code configuration enables Azure to monitor your application's health effectively, allowing for automated responses to issues.

Common Mistake

Forgetting to add the HealthChecks.UI.Client NuGet package or the corresponding using directives.
