Module 1 › Chapter 1.1 › __setitem__ & __delitem__
Mutating the container
  def __setitem__(self, key, value):
    if isinstance(key, int):
      if not (-len(self._data) <= key < len(self._data)):
        raise IndexError("Index out of range")
      self._data[key] = value
    elif isinstance(key, slice):
      self._data[key] = list(value)  # accept any iterable
    else:
      raise TypeError("Key must be an integer or slice")

  def __delitem__(self, key):
    if isinstance(key, int):
      del self._data[key]
    elif isinstance(key, slice):
      del self._data[key]
    else:
      raise TypeError("Key must be an integer or slice")
Test assignments with single indices, different-length slice replacements, and deletions to ensure the internal structure remains consistent.

my_list = MyCustomList([1,2,3,4]) my_list[1] = 20 # [1,20,3,4] my_list[1:3] = [7,8] # [1,7,8,4] del my_list[0] # [7,8,4]
Common Mistake

Not handling slice assignment correctly or assuming the right-hand side is a list; convert iterables explicitly.
