# Part 1: Notification Sender Strategy
class NotificationStrategy:
    def send(self, recipient: str, message: str):
        raise NotImplementedError

class EmailSender(NotificationStrategy):
    def send(self, recipient: str, message: str):
        print(f"Sending Email to {recipient}: {message}")

class SMSSender(NotificationStrategy):
    def send(self, recipient: str, message: str):
        print(f"Sending SMS to {recipient}: {message}")

# Part 3: Notification Service (Context for Strategy)
class NotificationService:
    def __init__(self, sender_strategy: NotificationStrategy):
        self._sender = sender_strategy

    def set_sender(self, sender_strategy: NotificationStrategy):
        self._sender = sender_strategy

    def notify(self, recipient: str, message: str):
        print(f"NotificationService: Preparing to send message to {recipient}...")
        self._sender.send(recipient, message)
Expected Factory Implementation (concept)
class NotificationSenderFactory:
    def create_sender(self, sender_type: str) -> NotificationStrategy:
        if sender_type == "email":
            return EmailSender()
        elif sender_type == "sms":
            return SMSSender()
        else:
            raise ValueError(f"Unknown sender type: {sender_type}")
Test the service by creating an EmailSender through the factory, sending a message, then swapping to SMSSender and sending again. Also test that an invalid type raises ValueError.
Key Takeaway

This exercise practices decoupling creation and behavior: factory for creation, strategy for runtime behavior swap.
