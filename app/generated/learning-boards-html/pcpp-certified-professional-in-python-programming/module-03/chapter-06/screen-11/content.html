class AbstractSensor(ABC):
  def __init__(self):
    self._observers = []
    self._value = 0.0

  def attach(self, observer: Observer):
    if observer not in self._observers:
      self._observers.append(observer)

  def _notify(self):
    for observer in self._observers:
      observer.update(self, self._value)

  @property
  def value(self) -> float:
    return self._value

  @value.setter
  def value(self, new_value: float):
    if self._value != new_value:
      self._value = new_value
      self._notify()
Concrete observers
ConsoleDisplay prints the value; GraphDisplay simulates plotting with timestamps; Logger appends readings to a file. Attach them to a RandomSensor and generate values to observe behavior.
