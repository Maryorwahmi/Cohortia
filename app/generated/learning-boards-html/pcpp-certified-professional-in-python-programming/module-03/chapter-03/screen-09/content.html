Abstract Builder (excerpt)
class ReportBuilder(ABC):
  def __init__(self):
    self.report = Report()

  @abstractmethod
  def set_title(self, title: str):
    pass

  @abstractmethod
  def add_introduction(self, content: str):
    pass

  def get_report(self) -> Report:
    return self.report
Concrete Builder (excerpt)
class FinancialReportBuilder(ReportBuilder):
  def __init__(self):
    super().__init__()
    self.report.title = "Financial Report"

  def set_title(self, title: str):
    self.report.title = title
    return self

  def add_introduction(self, content: str):
    self.report.add_section("Introduction", content)
    return self

  def enable_page_numbers(self):
    self.report.page_numbers = True
    return self
Note how each method returns self for fluent calls. The concrete builder keeps control over how the product is modified.
