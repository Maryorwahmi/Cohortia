# Subsystem components
class Amplifier:
  def turn_on(self):
    print("Amplifier: Turning on.")
  def set_volume(self, volume):
    print(f"Amplifier: Setting volume to {volume}.")
  def set_source(self, source):
    print(f"Amplifier: Setting source to {source}.")
  def turn_off(self):
    print("Amplifier: Turning off.")

class DVDPlayer:
  def turn_on(self):
    print("DVD Player: Turning on.")
  def play(self, movie):
    print(f"DVD Player: Playing '{movie}'.")
  def stop(self):
    print("DVD Player: Stopping.")
  def eject(self):
    print("DVD Player: Ejecting disc.")
  def turn_off(self):
    print("DVD Player: Turning off.")

class Projector:
  def turn_on(self):
    print("Projector: Turning on.")
  def wide_screen_mode(self):
    print("Projector: Setting wide screen mode.")
  def turn_off(self):
    print("Projector: Turning off.")

class HomeTheaterFacade:
  def __init__(self, amplifier: Amplifier, dvd_player: DVDPlayer, projector: Projector):
    self._amp = amplifier
    self._dvd = dvd_player
    self._proj = projector

  def watch_movie(self, movie_title: str):
    print("
--- Get ready to watch a movie! ---")
    self._proj.turn_on()
    self._proj.wide_screen_mode()
    self._amp.turn_on()
    self._amp.set_volume(10)
    self._amp.set_source("DVD")
    self._dvd.turn_on()
    self._dvd.play(movie_title)

  def end_movie(self):
    print("
--- Shutting down home theater ---")
    self._dvd.stop()
    self._dvd.eject()
    self._dvd.turn_off()
    self._amp.turn_off()
    self._proj.turn_off()
Explanation
  • The facade receives ready-to-use subsystem objects in its constructor.
  • High-level methods group multiple lower-level operations into a single workflow.
  • Clients can still access subsystem APIs directly when necessary.
Pro tip

Keep sequence and orchestration here; avoid embedding domain rules or heavy logic in the facade.
