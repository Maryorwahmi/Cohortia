from collections import deque

def bfs_with_logging(graph, start):
    visited = set([start])
    queue = deque([start])
    parents = {start: None}
    order = []

    while queue:
        print("Queue:", list(queue), "Visited:", visited)
        node = queue.popleft()
        order.append(node)
        for n in graph[node]:
            if n not in visited:
                visited.add(n)
                parents[n] = node
                queue.append(n)
    return order, parents

Key Takeaway

Understanding queue evolution clarifies why BFS is breadth-first and why it finds shortest edge-count paths.
