# socketserver_threaded_echo_server.py (excerpt)
import socketserver
import threading

HOST = '127.0.0.1'
PORT = 65432

class ThreadedTCPRequestHandler(socketserver.BaseRequestHandler):
    def handle(self):
        print(f"Connected by {self.client_address} in thread {threading.current_thread().name}")
        while True:
            try:
                data = self.request.recv(1024)
                if not data:
                    break
                print(f"[{self.client_address}] Received: {data.decode()}")
                self.request.sendall(data)
            except ConnectionResetError:
                break
            except Exception as e:
                break
        print(f"[{self.client_address}] Connection closed.")
Notes & takeaway
Observe printed thread names to confirm the thread-per-connection model. Remember to protect shared state when handlers access global resources.
Pro tip
Use allow_reuse_address = True during development to avoid waiting for socket timeouts between restarts.
