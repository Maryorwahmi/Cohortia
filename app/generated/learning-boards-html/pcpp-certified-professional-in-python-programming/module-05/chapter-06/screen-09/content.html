# robust client (starter style)
import socket

def robust_client(host, port, message):
  s = None
  try:
    s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    s.setsockopt(socket.SOL_SOCKET, socket.SO_KEEPALIVE, 1)
    s.settimeout(5)
    s.connect((host, port))
    print('Local:', s.getsockname())
    print('Peer :', s.getpeername())
    s.sendall(message.encode())
    data = s.recv(1024)
    if not data:
      print('Server closed connection gracefully')
    else:
      print('Received:', data.decode())
  except ConnectionRefusedError:
    print('Connection refused')
  except TimeoutError:
    print('Operation timed out')
  except socket.error as e:
    print('Socket error', e)
  finally:
    if s:
      s.close()
Explanation & Tips
  • Set timeouts to avoid blocking calls during network issues.
  • Use targeted except blocks so each error yields a clear instruction or retry decision.
  • Log getsockname and getpeername for reproducible debugging information.
  • Always close sockets in finally to free file descriptors.
