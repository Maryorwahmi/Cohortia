Module 5 › Chapter 5.4 › One-way UDP Chat
Activity overview
Receiver binds and prints messages; sender prompts user and sends datagrams until exit.
Receiver starter (udp_receiver.py)
import socket

HOST = '127.0.0.1'
PORT = 12345

def run_receiver():
  with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as s:
    s.bind((HOST, PORT))
    print(f"UDP Receiver listening on {HOST}:{PORT}")
    while True:
      data, addr = s.recvfrom(1024)
      print(f"[{addr[0]}:{addr[1]}] Received: {data.decode()}")

if __name__ == "__main__":
  run_receiver()
Sender starter (udp_sender.py)
import socket

RECEIVER_HOST = '127.0.0.1'
RECEIVER_PORT = 12345

with socket.socket(socket.AF_INET, socket.SOCK_DGRAM) as s:
  print(f"UDP Sender ready to send messages to {RECEIVER_HOST}:{RECEIVER_PORT}")
  while True:
    message = input("Your message (type 'exit' to quit): ")
    if message.lower() == 'exit':
      break
    s.sendto(message.encode(), (RECEIVER_HOST, RECEIVER_PORT))
    print(f"Sent: '{message}'")
  print("Sender finished.")
