Module 5 › Chapter 5.3 › TCP server example explained
Reference code
# tcp_server.py
import socket

HOST = '127.0.0.1'
PORT = 65432

def run_server():
  with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
    s.bind((HOST, PORT))
    s.listen()
    print(f"Server listening on {HOST}:{PORT}")
    conn, addr = s.accept()
    with conn:
      print(f"Connected by {addr}")
      while True:
        data = conn.recv(1024)
        if not data:
          break
        print(f"Received from client: {data.decode()}")
        conn.sendall(data)
      print(f"Connection with {addr} closed.")
  print("Server shutting down.")

if __name__ == "__main__":
  run_server()
Explanation & checks
  • with socket.socket: ensures s.close() runs automatically even if an exception occurs.
  • conn, addr: accept returns a fresh connection socket; closing conn does not stop the listening socket.
  • recv(1024): blocking read up to the buffer size; returns b'' when the peer closes the connection.
  • sendall(data): guarantees the full message is transmitted or raises an exception.
Key Takeaway

This example maps each server call to its runtime effect so you can reason about connections and resource cleanup.
