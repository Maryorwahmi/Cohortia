class InterfaceEnforcer(type):
    def __new__(mcs, name, bases, attrs):
        required = 'process_data'
        if required not in attrs:
            print(f"Warning: Class '{name}' is missing '{required}'. Injecting placeholder.")
            def _placeholder_process_data(self):
                raise NotImplementedError(
                    f"Class '{self.__class__.__name__}' must implement the '{required}' method."
                )
            attrs[required] = _placeholder_process_data
        return super().__new__(mcs, name, bases, attrs)

class DataProcessor(metaclass=InterfaceEnforcer):
    def __init__(self, data):
        self.data = data
    def process_data(self):
        return f"Processing: {self.data.upper()}"

class MissingProcessor(metaclass=InterfaceEnforcer):
    def __init__(self, data):
        self.data = data
Test scenario
Instantiate both processors and call process_data. Expect DataProcessor to succeed and MissingProcessor to raise NotImplementedError.
Key Takeaway

Attributes injected in __new__ are part of the final class. Use this to enforce presence of API methods at definition time.
