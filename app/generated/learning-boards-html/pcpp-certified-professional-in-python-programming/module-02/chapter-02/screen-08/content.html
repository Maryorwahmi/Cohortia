Module 2 › Chapter 2.2 › MyMetaclass example
Full example
class MyMetaclass(type):
    def __new__(mcs, name, bases, attrs):
        print(f"--- Metaclass __new__ called for class {name} ---")
        print(f"  Bases: {bases}")
        print(f"  Attributes before modification: {list(attrs.keys())}")

        attrs['METACLASS_INJECTED_VAR'] = f"Hello from {name}"

        if 'required_method' not in attrs:
            print(f"  WARNING: Class {name} is missing 'required_method'. Adding a placeholder.")
            def _placeholder_method(self):
                raise NotImplementedError(f"'{name}' must implement 'required_method'")
            attrs['required_method'] = _placeholder_method

        cls = super().__new__(mcs, name, bases, attrs)
        print(f"  Class object '{name}' created by __new__.")
        return cls

    def __init__(cls, name, bases, attrs):
        print(f"--- Metaclass __init__ called for class {name} ---")
        print(f"  Class attributes after creation: {list(cls.__dict__.keys())}")
        if not name.startswith('Base'):
            print(f"  INFO: Non-Base class '{name}' initialized.")
        super().__init__(name, bases, attrs)
Notes & behavior
  • Prints at import time: You will see the metaclass prints when Python defines the class, not when you instantiate it.
  • Injected attributes: METACLASS_INJECTED_VAR appears on the class and on instances as a normal attribute lookup.
  • Placeholder methods: The placeholder raises NotImplementedError to force implementers to provide a real method.
Key Takeaway

The example demonstrates how __new__ and __init__ can be combined to both alter and validate classes during creation.
