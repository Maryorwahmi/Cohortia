Starter code

class Vehicle:
  wheels = 4

  def __init__(self, make, model):
    self.make = make
    self.model = model

  def display_info(self):
    return f"{self.make} {self.model} with {self.wheels} wheels."

class Car(Vehicle):
  def drive(self):
    return f"The {self.make} {self.model} is driving."
