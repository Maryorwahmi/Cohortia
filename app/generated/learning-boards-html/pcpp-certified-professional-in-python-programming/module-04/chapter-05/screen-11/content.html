Rate-limited processing
Implement the TODOs: create a semaphore, wrap processing with async with, create tasks, and gather results. Observe printed active counts to confirm limiting behavior.
import asyncio
import time
import random

async def process_item(item_id, semaphore):
  async with semaphore:
    delay = random.uniform(0.5, 2.0)
    print(f"  [Item {item_id}] Started processing (delay: {delay:.2f}s).")
    await asyncio.sleep(delay)
    print(f"  [Item {item_id}] Finished processing.")
    return f"Processed {item_id}"

async def main_rate_limited_processing():
  total_items = 15
  concurrency_limit = 5
  semaphore = asyncio.Semaphore(concurrency_limit)

  start_time = time.perf_counter()
  tasks = [process_item(i, semaphore) for i in range(1, total_items+1)]
  results = await asyncio.gather(*tasks)
  end_time = time.perf_counter()
  print(f"All items processed in {end_time - start_time:.4f} seconds")
