Concurrent API fetch simulation

import asyncio
import time
import random

async def simulate_api_call(url, min_delay=0.5, max_delay=3):
  delay = random.uniform(min_delay, max_delay)
  print(f"Fetching {url} (simulated delay: {delay:.2f}s)...")
  await asyncio.sleep(delay)
  data = f"Data from {url} after {delay:.2f}s"
  print(f"Finished {url}")
  return url, data

async def main_api_fetch():
  urls = ["https://api.example.com/data/1", "https://api.example.com/data/2", "https://api.example.com/data/3"]
  start_time = time.perf_counter()
  tasks = [simulate_api_call(url) for url in urls]
  results = await asyncio.gather(*tasks)
  print(f"All completed in {time.perf_counter() - start_time:.4f} seconds")

asyncio.run(main_api_fetch())
Expected observations
  • Total runtime ≈ the maximum simulated delay, not the sum.
  • Printed start messages appear quickly, finishes interleave based on random delays.
  • Results returned by gather match the order of requested URLs, not completion order.
