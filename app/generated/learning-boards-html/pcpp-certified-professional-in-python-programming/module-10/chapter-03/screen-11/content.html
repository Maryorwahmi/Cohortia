import pytest
from string_reverser import StringReverser

@pytest.mark.parametrize("input_string, expected_output", [
    ("hello", "olleh"),
    ("Python is fun", "nuf si nohtyP"),
    ("", ""),
    ("!@#", "#@!"),
    ("12345", "54321"),
])
def test_reverse_string_various_inputs(input_string, expected_output):
    assert StringReverser.reverse_string(input_string) == expected_output


def test_reverse_string_non_string_input_raises_type_error():
    with pytest.raises(TypeError):
        StringReverser.reverse_string(123)
