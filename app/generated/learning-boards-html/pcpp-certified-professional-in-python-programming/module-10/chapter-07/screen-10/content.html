WeatherService test

Patch weather_service.requests and configure a mock response to test parsing and call parameters without a network call.

# test_weather_service.py
import unittest
from unittest.mock import patch, Mock
from weather_service import WeatherService

class TestWeatherService(unittest.TestCase):

    @patch('weather_service.requests')
    def test_get_current_temperature_success(self, mock_requests):
        mock_response = Mock()
        mock_response.json.return_value = {
            "current": {"temp_c": 25.5, "condition": {"text": "Sunny"}}
        }
        mock_response.raise_for_status.return_value = None
        mock_requests.get.return_value = mock_response

        service = WeatherService(api_key="dummy_key")
        temp = service.get_current_temperature("London")

        self.assertEqual(temp, 25.5)
        mock_requests.get.assert_called_once_with(
            "http://api.weatherapi.com/v1/current.json",
            params={"key": "dummy_key", "q": "London"}
        )
Checklist
  • Patch the correct module name
  • Configure json.return_value and raise_for_status
  • Assert both return value and request call
