# fib_buggy.py
def fibonacci(n):
    """
    Generates the nth Fibonacci number.
    Expected: fibonacci(0) -> 0, fibonacci(1) -> 1
    """
    if n <= 0:
        return 0
    elif n == 1:
        return 1
    else:
        a, b = 0, 1
        for _ in range(n - 1): # Suspect this range or initial values
            a, b = b, a + b
        return b

if __name__ == "__main__":
    print(f"Fibonacci(6): {fibonacci(6)}")
