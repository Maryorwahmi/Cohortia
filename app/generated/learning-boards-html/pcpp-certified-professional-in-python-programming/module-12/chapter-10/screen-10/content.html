Mandelbrot Generation: Python vs. Numba

Observe the significant performance difference when optimizing a CPU-bound fractal calculation.

import numpy as np
import time
from numba import njit

# Original Python function
def generate_mandelbrot_python(width, height, max_iter):
    image = np.zeros((height, width), dtype=np.int32)
    x_min, x_max = -2.0, 1.0
    y_min, y_max = -1.5, 1.5
    for row in range(height):
        for col in range(width):
            c_real = x_min + (col / width) * (x_max - x_min)
            c_imag = y_min + (row / height) * (y_max - y_min)
            c = complex(c_real, c_imag)
            z = 0.0j
            for i in range(max_iter):
                z = z*z + c
                if abs(z) > 2.0:
                    image[row, col] = i
                    break
            else:
                image[row, col] = max_iter
    return image

# Numba-optimized version
@njit
def generate_mandelbrot_numba(width, height, max_iter):
    image = np.zeros((height, width), dtype=np.int32)
    x_min, x_max = -2.0, 1.0
    y_min, y_max = -1.5, 1.5
    for row in range(height):
        for col in range(width):
            c_real = x_min + (col / width) * (x_max - x_min)
            c_imag = y_min + (row / height) * (y_max - y_min)
            c = complex(c_real, c_imag)
            z = 0.0j
            for i in range(max_iter):
                z = z*z + c
                if abs(z) > 2.0:
                    image[row, col] = i
                    break
            else:
                image[row, col] = max_iter
    return image

if __name__ == "__main__":
    WIDTH, HEIGHT, MAX_ITER = 800, 600, 100

    start_py = time.perf_counter()
    generate_mandelbrot_python(WIDTH, HEIGHT, MAX_ITER)
    end_py = time.perf_counter()
    print(f"Python version took: {end_py - start_py:.4f} seconds")

    start_numba_first = time.perf_counter()
    generate_mandelbrot_numba(WIDTH, HEIGHT, MAX_ITER)
    end_numba_first = time.perf_counter()
    print(f"Numba @njit first call took: {end_numba_first - start_numba_first:.4f} seconds (includes compilation)")

    start_numba_second = time.perf_counter()
    generate_mandelbrot_numba(WIDTH, HEIGHT, MAX_ITER)
    end_numba_second = time.perf_counter()
    print(f"Numba @njit second call took: {end_numba_second - start_numba_second:.4f} seconds (compiled)")

Performance Insights

  • Pure Python: Interpreted execution leads to significant overhead for each loop iteration and complex number operation.
  • Numba First Call: Includes the one-time compilation cost, making it potentially slower than subsequent calls but still often faster than pure Python.
  • Numba Subsequent Calls: Executes highly optimized machine code directly, resulting in dramatic speedups (e.g., 100x-1000x faster).
  • CPU-Bound Tasks: Numba is exceptionally effective for tasks dominated by numerical computations and tight loops, where CPython's GIL and dynamic typing are bottlenecks.

Why it Matters

Numba allows Python developers to write high-performance scientific and data processing code without switching to lower-level languages, preserving Python's readability and development speed.

Key Takeaway

For computationally intensive numerical algorithms, Numba's `@njit` can transform Python code into highly efficient machine code, unlocking significant performance gains for scientific computing and data analysis.
