# log_processor.py
import os

def check_file_exists(filepath):
    """Return True if filepath exists, otherwise False."""
    return os.path.exists(filepath)

def read_log_content(filepath):
    """Return simulated content if file exists, otherwise an error string."""
    if check_file_exists(filepath):
        return f"Simulated content of {filepath}: Line 1, Line 2"
    else:
        return "Error: File not found."
Lab Steps
  1. Create log_processor.py and add the imports.
  2. Implement check_file_exists with a docstring.
  3. Implement read_log_content to call the checker and return simulated content.
