def calculate_disk_usage(filesize_bytes):
    total_size = 0
    for size in filesize_bytes:
        print(f"DEBUG: Processing size: {size}")
        total_size += size
    print(f"DEBUG: Total size before conversion: {total_size} bytes")
    return total_size / (1024 * 1024)
