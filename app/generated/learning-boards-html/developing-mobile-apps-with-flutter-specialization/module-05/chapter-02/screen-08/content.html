Google sign-in example
// Example for Google Sign-In
import 'package:google_sign_in/google_sign_in.dart';
import 'package:firebase_auth/firebase_auth.dart';

final GoogleSignIn _googleSignIn = GoogleSignIn();
final FirebaseAuth _auth = FirebaseAuth.instance;

Future signInWithGoogle() async {
  try {
    final GoogleSignInAccount? googleUser = await _googleSignIn.signIn();

    final GoogleSignInAuthentication? googleAuth = await googleUser?.authentication;

    if (googleAuth == null) {
      return null; // User cancelled
    }

    final credential = GoogleAuthProvider.credential(
      accessToken: googleAuth.accessToken,
      idToken: googleAuth.idToken,
    );

    final UserCredential userCredential = await _auth.signInWithCredential(credential);
    print('Signed in with Google: ${userCredential.user?.displayName}');
    return userCredential.user;
  } on FirebaseAuthException catch (e) {
    print('Firebase Auth Error during Google Sign-In: ${e.message}');
    return null;
  } catch (e) {
    print('Error during Google Sign-In: $e');
    return null;
  }
}
Notes and best practices
  • Add google_sign_in to pubspec.yaml and run flutter pub get.
  • Verify OAuth client IDs in the Firebase Console for each platform.
  • Handle the null authentication case when the user cancels the account chooser.
  • Clear the Google session on sign-out to force account choice next time.
Key takeaway
Google Sign-In improves UX but requires both package setup and Console configuration; treat it as a complement to firebase_auth, not a replacement.
