Callable function (Flutter)
import 'package:cloud_functions/cloud_functions.dart';

Future _callGreetUserFunction() async {
  final HttpsCallable callable = FirebaseFunctions.instance.httpsCallable('greetUser');
  final result = await callable.call>({'name': 'user@example.com'});
  print('Function result: ${result.data?['message']}');
}

final response = await http.get(Uri.parse('YOUR_CLOUD_FUNCTION_HTTP_URL'));
