Thumbnail generator (conceptual example)
exports.generateThumbnail = functions.storage.object().onFinalize(async (object) => {
  const fileBucket = object.bucket;
  const filePath = object.name;
  const contentType = object.contentType;

  if (!filePath || object.resourceState === 'not_exists' || !contentType.startsWith('image/')) {
    return null;
  }
  if (filePath.startsWith('thumbnails/')) return null;

  const bucket = admin.storage().bucket(fileBucket);
  const file = bucket.file(filePath);
  const thumbnailFilePath = `thumbnails/${filePath.split('/').pop()}`;
  const thumbnailFile = bucket.file(thumbnailFilePath);

  const [imageBuffer] = await file.download();
  const resizedBuffer = await sharp(imageBuffer).resize(200,200).toBuffer();
  await thumbnailFile.save(resizedBuffer, { contentType });
  return null;
});
Test with the emulator or a staging project, ensure the function skips thumbnails and handles non-image uploads safely. Log actions for observability.
