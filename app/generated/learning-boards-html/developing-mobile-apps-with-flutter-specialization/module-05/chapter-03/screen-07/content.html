Code examples
import 'package:cloud_firestore/cloud_firestore.dart';

final FirebaseFirestore _firestore = FirebaseFirestore.instance;

Future<void> addUser(String name, int age) async {
  try {
    await _firestore.collection('users').add({
      'name': name,
      'age': age,
      'timestamp': FieldValue.serverTimestamp(),
    });
    print('User added successfully!');
  } catch (e) {
    print('Error adding user: $e');
  }
}

Future<void> setUserData(String userId, String email) async {
  try {
    await _firestore.collection('users').doc(userId).set({
      'email': email,
      'createdAt': FieldValue.serverTimestamp(),
      'roles': ['user'],
    });
    print('User data set successfully for $userId!');
  } catch (e) {
    print('Error setting user data: $e');
  }
}
Explanation cards
add()
Creates a document with an auto ID. Good for lists where ID control is unnecessary.
set()
Writes to a document reference. Overwrites existing document unless you use merge.
Common mistake
Using set() unintentionally replaces fields instead of merging. Use SetOptions(merge:true) to add fields safely.
