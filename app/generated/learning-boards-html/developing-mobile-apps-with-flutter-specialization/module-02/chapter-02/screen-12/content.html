Starter Login Screen

// Declare controllers
final TextEditingController _emailController = TextEditingController();
final TextEditingController _passwordController = TextEditingController();

void _performLogin() {
  // Print current values
  print('Email: ${_emailController.text}');
  print('Password: ${_passwordController.text}');
}

@override
void dispose() {
  _emailController.dispose();
  _passwordController.dispose();
  super.dispose();
}
Starter layout notes
  • Use keyboardType: TextInputType.emailAddress for email fields.
  • Use obscureText: true for password fields to hide user input.
  • Add labelText and hintText via InputDecoration for clarity.
