Bottom navigation code
class MyBottomNavScreen extends StatefulWidget {
  const MyBottomNavScreen({super.key});

  @override
  State<MyBottomNavScreen> createState() => _MyBottomNavScreenState();
}

class _MyBottomNavScreenState extends State<MyBottomNavScreen> {
  int _selectedIndex = 0;
  static const List<Widget> _widgetOptions = <Widget>[
    Text('Home Page', style: TextStyle(fontSize: 30, fontWeight: FontWeight.bold)),
    Text('Search Page', style: TextStyle(fontSize: 30, fontWeight: FontWeight.bold)),
    Text('Profile Page', style: TextStyle(fontSize: 30, fontWeight: FontWeight.bold)),
  ];

  void _onItemTapped(int index) {
    setState(() { _selectedIndex = index; });
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: Center(child: _widgetOptions.elementAt(_selectedIndex)),
      bottomNavigationBar: BottomNavigationBar(
        items: const <BottomNavigationBarItem>[...],
        currentIndex: _selectedIndex,
        onTap: _onItemTapped,
      ),
    );
  }
}
To preserve each tab's UI state, replace body with an IndexedStack whose index is _selectedIndex and whose children are the tab pages.
Example test
Place a scrollable list inside the Home tab and a form inside Profile. Switch tabs and verify the scroll position and form contents persist. If they reset, switch to an IndexedStack to preserve widgets.
Common pitfall
Rebuilding tab widgets on every index change causes lost state. Use IndexedStack or automatic keep-alive strategies.
