Example code
import 'package:flutter/material.dart';
import 'package:shared_preferences/shared_preferences.dart';

class SettingsScreen extends StatefulWidget {
  const SettingsScreen({super.key});

  @override
  State<SettingsScreen> createState() => _SettingsScreenState();
}

class _SettingsScreenState extends State<SettingsScreen> {
  bool _isDarkMode = false;
  final String _darkModeKey = 'isDarkMode';

  @override
  void initState() {
    super.initState();
    _loadSettings();
  }

  Future<void> _loadSettings() async {
    final prefs = await SharedPreferences.getInstance();
    setState(() {
      _isDarkMode = prefs.getBool(_darkModeKey) ?? false;
    });
  }

  Future<void> _toggleDarkMode(bool value) async {
    setState(() { _isDarkMode = value; });
    final prefs = await SharedPreferences.getInstance();
    await prefs.setBool(_darkModeKey, _isDarkMode);
  }
}
Notes
  • Load values in initState to initialize UI before user interaction.
  • Update state first for responsiveness, then persist with await.
  • Use descriptive keys and defaults to prevent null issues.
Misconception

Writes always finish instantly. In reality you should await writes if subsequent logic depends on persistence.
