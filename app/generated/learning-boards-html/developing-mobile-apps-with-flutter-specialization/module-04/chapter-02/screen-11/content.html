State and loading
Use a DatabaseHelper instance and fetch todos in initState.
final DatabaseHelper _dbHelper = DatabaseHelper();
List<Todo> _todos = [];

@override
void initState() {
  super.initState();
  _loadTodos();
}

Future<void> _loadTodos() async {
  final todos = await _dbHelper.getTodos();
  setState(() => _todos = todos);
}
UI actions
Add, toggle, and delete operations that call DatabaseHelper methods and then reload or update state.
Future<void> _addTodo() async {
  final newTodo = Todo(title: _textController.text, description: 'No description yet');
  await _dbHelper.insertTodo(newTodo);
  _textController.clear();
  _loadTodos();
}

Future<void> _toggleTodoCompletion(Todo todo) async {
  final updated = Todo(id: todo.id, title: todo.title, description: todo.description, isCompleted: !todo.isCompleted);
  await _dbHelper.updateTodo(updated);
  _loadTodos();
}

Future<void> _removeTodo(int id) async {
  await _dbHelper.deleteTodo(id);
  _loadTodos();
}
