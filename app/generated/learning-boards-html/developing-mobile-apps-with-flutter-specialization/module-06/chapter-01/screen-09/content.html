Widget test pattern

// test/widget/login_button_test.dart
import 'package:flutter_test/flutter_test.dart';
import 'package:flutter/material.dart';
import 'package:your_app_name/widgets/login_button.dart'; // replace package name

void main() {
  testWidgets('LoginButton shows spinner when pressed', (WidgetTester tester) async {
    bool called = false;

    await tester.pumpWidget(MaterialApp(
      home: Scaffold(
        body: LoginButton(onPressed: () async { called = true; }),
      ),
    ));

    // Tap the button
    await tester.tap(find.byType(LoginButton));
    await tester.pump();

    // Expect a loading indicator to appear
    expect(find.byType(CircularProgressIndicator), findsOneWidget);
  });
}
Notes
Wrap widgets with MaterialApp or required parents. Use tester.pump for animations and tester.tap to simulate input.
Common pitfall
Missing parent widgets cause exceptions unrelated to the widget logic—add MaterialApp or MediaQuery as needed.
Key Takeaway
Widget tests give fast UI assurance for components when correctly wrapped and isolated from external services.
