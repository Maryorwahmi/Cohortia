Code: estimate gas and compute cost
This snippet uses the provider to estimate gas and fetch fee data. It’s for estimation only.
// Make sure to install ethers: npm install ethers
const { ethers } = require("ethers");

const provider = new ethers.JsonRpcProvider("https://sepolia.infura.io/v3/YOUR_INFURA_PROJECT_ID");

async function estimateGasForTransfer() {
  const senderAddress = "0xYourSenderAddressHere"; // for estimation only
  const recipientAddress = "0xYourRecipientAddressHere";
  const amount = ethers.parseEther("0.001");

  const tx = {
    to: recipientAddress,
    value: amount,
    from: senderAddress,
  };

  try {
    const gasEstimate = await provider.estimateGas(tx);
    console.log(`Estimated gas for transfer: ${gasEstimate.toString()} units`);

    const feeData = await provider.getFeeData();
    const gasPrice = feeData.gasPrice;
    console.log(`Current estimated gas price: ${ethers.formatUnits(gasPrice, "gwei")} Gwei`);

    const totalCostWei = gasEstimate * gasPrice;
    console.log(`Total estimated cost: ${ethers.formatEther(totalCostWei)} ETH`);
  } catch (error) {
    console.error("Error estimating gas:", error);
  }
}

estimateGasForTransfer();
Notes
Include from for accurate simulation. This does not broadcast a transaction — it only simulates execution to estimate gas.
Example outcome
Estimate: 21,000 units; gasPrice: 30 Gwei → cost ≈ 0.00063 ETH. Compare with real block data to validate.
Misconception check
An estimate can differ from actual gasUsed if the transaction interacts with different contract state during execution.
