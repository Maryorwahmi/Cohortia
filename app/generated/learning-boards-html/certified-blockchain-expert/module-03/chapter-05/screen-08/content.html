Starter Snippet
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract DemoOwner {
  address public owner;
  uint256 public value;

  constructor() { owner = msg.sender; }

  modifier onlyOwner() {
    require(msg.sender == owner, "Not the owner");
    _;
  }

  function updateValue(uint256 _v) public onlyOwner {
    value = _v;
  }
}
Validation Steps
  1. Deploy with account A (owner).
  2. Call updateValue from A — it should succeed and set value.
  3. Switch to account B and call updateValue — it should revert with "Not the owner".
  4. Confirm no state change or logs on the failed call.
Reflection
Modifiers simplify auditing and reduce duplicated checks. Test both paths and use clear revert messages so you know why a transaction failed.
