Starter Contract
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract AccessControlledStorage {
  address public owner;
  uint256 public storedValue;

  event ValueUpdated(address indexed updater, uint256 oldValue, uint256 newValue);

  constructor(uint256 _initialValue) {
    owner = msg.sender;
    storedValue = _initialValue;
  }

  modifier onlyOwner() {
    require(msg.sender == owner, "AccessControlledStorage: Not the owner");
    _;
  }

  function updateValue(uint256 _newValue) public onlyOwner {
    uint256 _oldValue = storedValue;
    storedValue = _newValue;
    emit ValueUpdated(msg.sender, _oldValue, _newValue);
  }

  function getValue() public view returns (uint256) { return storedValue; }

  function transferOwnership(address _newOwner) public onlyOwner {
    require(_newOwner != address(0), "New owner cannot be zero address");
    owner = _newOwner;
  }
}
Remix Steps
  1. Create AccessControlledStorage.sol and paste the code above.
  2. Compile, deploy with an initial value like 100, using JavaScript VM.
  3. Call getValue to confirm initial value, then call updateValue as owner and inspect logs for ValueUpdated.
  4. Switch account, attempt updateValue to see revert, and verify no log entry appears.
Reflection Prompt
When would you store history in state versus emitting events? Consider gas cost, future contract reads, and off-chain analytics needs, then choose the approach that balances on-chain requirements with client efficiency.
