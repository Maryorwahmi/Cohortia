Simplified PriceConsumerV3
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

import "@chainlink/contracts/src/v0.8/interfaces/AggregatorV3Interface.sol";

contract PriceConsumerV3 {
  AggregatorV3Interface internal priceFeed;

  constructor(address feed) {
    priceFeed = AggregatorV3Interface(feed);
  }

  function getLatestPrice() public view returns (int) {
    (, int price, , ,) = priceFeed.latestRoundData();
    return price; // Remember: apply decimals when making UI-ready values
  }
}
Example notes
Provide the correct feed address for your network. The returned price is an integer scaled by the feed's decimals, typically eight for ETH/USD. Convert it in the frontend or contract as needed.
Key Takeaway
Use read-only Aggregator calls to obtain aggregated prices without submitting new transactions when possible; check decimals and timestamps.
