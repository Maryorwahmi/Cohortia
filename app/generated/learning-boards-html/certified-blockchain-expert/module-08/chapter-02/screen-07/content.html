pragma solidity ^0.8.0;

import "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

contract MySecurityToken is ERC20, Ownable {
    mapping(address => bool) public isAccredited; // Simplified: in reality, this would be more robust

    constructor(string memory name, string memory symbol, uint256 initialSupply) ERC20(name, symbol) {
        _mint(msg.sender, initialSupply); // Mints initial supply to deployer
    }

    // Function to mark an address as accredited (only owner can call)
    function setAccredited(address _investor, bool _status) public onlyOwner {
        isAccredited[_investor] = _status;
    }

    // Override transfer function to enforce accreditation
    function _beforeTokenTransfer(address from, address to, uint256 amount) internal virtual override {
        super._beforeTokenTransfer(from, to, amount);

        // Allow minting/burning from/to 0x0 address (for initial supply, etc.)
        if (from == address(0) || to == address(0)) {
            return;
        }

        // Only allow transfers if both sender and receiver are accredited
        require(isAccredited[from], "Sender not accredited");
        require(isAccredited[to], "Receiver not accredited");
    }
}
Note: Angle brackets and imports are shown as in the chapter. This is conceptual code illustrating transfer gating and accreditation checks; production implementations require secure oracles and legal processes.
Explanatory bullets
  • The mapping stores accredited status; only the owner sets it in this example.
  • The _beforeTokenTransfer hook blocks transfers when either party is non-accredited.
  • This enforces regulatory policy at transfer time but limits open-market liquidity.
Key Takeaway

Contract-level controls can operationalize legal requirements, but must be designed with off-chain compliance and governance in mind.

Common mistake
Relying on a simple mapping without secure identity oracles will not satisfy regulatory proof of investor accreditation.
