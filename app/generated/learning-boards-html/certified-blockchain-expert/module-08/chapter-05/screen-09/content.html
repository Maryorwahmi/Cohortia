Design choices

  • Primary model: Hybrid to combine debate with enforceable votes.
  • Proposal threshold: Minimum YIELD deposit to prevent spam.
  • Voting period & quorum: Long enough for discussion; quorum around 20–30%.
  • Delegation: Optional delegation with revocation to enable expertise while preserving agency.

interface IYieldVault {
  function updateInterestRate(uint256 newRate) external;
  function addSupportedAsset(address tokenAddress) external;
}

contract YieldVaultGovernance {
  address public yieldToken; // Address of the YIELD token
  address public yieldVault; // Address of the YieldVault protocol contract
  uint256 public proposalThreshold; // Minimum tokens to submit a proposal
  uint256 public votingPeriod; // Duration of voting
  uint256 public quorumPercentage; // Percentage of total supply needed to vote

  struct Proposal {
    uint256 id;
    address proposer;
    string description;
    address targetContract;
    bytes callData;
    uint256 voteCountFor;
    uint256 voteCountAgainst;
    bool executed;
  }

  mapping(uint256 => Proposal) public proposals;
  uint256 public nextProposalId;

Address a weakness

To mitigate whale dominance consider delegation limits, identity-based reputation layers, or quadratic voting for key decisions.
