Starter code
# calculator.py
def add(a, b):
    # This function might have bugs!
    if not isinstance(a, (int, float)) or not isinstance(b, (int, float)):
        raise TypeError("Both inputs must be numbers.")
    return a + b
Create test_calculator.py, import pytest and the add function, and write four test functions covering the scenarios in the instructions. Run pytest to verify.
Sample tests
# test_calculator.py
import pytest
from calculator import add

def test_add_positive_integers():
    assert add(2, 3) == 5

def test_add_positive_and_negative():
    assert add(5, -2) == 3

def test_add_float_numbers():
    assert add(2.5, 1.25) == 3.75

def test_add_non_numeric_input():
    with pytest.raises(TypeError):
        add("hello", 5)
Key Takeaway

Practice writing clear, single-purpose tests and using pytest.raises for expected errors.
