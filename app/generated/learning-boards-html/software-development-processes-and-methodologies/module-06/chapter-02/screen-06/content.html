# test_app.py
import pytest
from app import calculate_discount

def test_calculate_discount_valid_input():
    assert calculate_discount(100, 10) == 90.00
    assert calculate_discount(50, 0) == 50.00
    assert calculate_discount(200, 100) == 0.00
    assert calculate_discount(100, 7.5) == 92.50

def test_calculate_discount_invalid_price():
    with pytest.raises(ValueError, match="Price must be a non-negative number."):
        calculate_discount(-10, 10)
    with pytest.raises(ValueError, match="Price must be a non-negative number."):
        calculate_discount("abc", 10)

def test_calculate_discount_invalid_percentage():
    with pytest.raises(ValueError, match="Discount percentage must be between 0 and 100."):
        calculate_discount(100, -5)
    with pytest.raises(ValueError, match="Discount percentage must be between 0 and 100."):
        calculate_discount(100, 101)
Run: In the terminal run pytest to discover and run tests.
Assert pattern: One test function per behavior, use pytest.raises for expected errors.
Test names: Use names that describe behavior to act as documentation.
Support
Why use pytest.raises? It checks that code defends against invalid inputs and raises the expected exception with a helpful message. This makes error handling explicit in tests rather than relying on broad exception catching.
Common Mistake

Putting many assertions for different behaviors into one test reduces clarity. Split scenarios across test functions.

Key Takeaway

pytest discovery and pytest.raises make writing clear, self-validating unit tests straightforward and repeatable.
