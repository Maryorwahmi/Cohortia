Unhooking NtAllocateVirtualMemory (Part 1)

Analyzing the setup for restoring a critical API function.

Get Hooked DLL Handle

GetModuleHandleA("ntdll.dll") retrieves the base address of the currently loaded ntdll.dll.

Load Clean DLL Copy

LoadLibraryExA("ntdll.dll", NULL, DONT_RESOLVE_DLL_REFERENCES) loads a fresh, unhooked copy from disk.

Obtain Function Pointers

GetProcAddress is used to get the addresses of NtAllocateVirtualMemory from both the hooked and clean DLLs.

Key Takeaway

The initial setup for unhooking focuses on acquiring references to both the potentially hooked and a guaranteed clean version of the target API function.

#include <windows.h>
#include <stdio.h>

// Function pointer for the original NtAllocateVirtualMemory
typedef NTSTATUS(NTAPI* pNtAllocateVirtualMemory)(
    HANDLE ProcessHandle,
    PVOID* BaseAddress,
    ULONG_PTR ZeroBits,
    PSIZE_T RegionSize,
    ULONG AllocationType,
    ULONG Protect
);

int main() {
    // 1. Get a handle to ntdll.dll in the current process
    HMODULE hNtdll = GetModuleHandleA("ntdll.dll");
    if (hNtdll == NULL) {
        printf("[-] Failed to get ntdll.dll handle.
");
        return 1;
    }
    printf("[+] Got handle to ntdll.dll in current process: %p
", hNtdll);

    // 2. Load a fresh copy of ntdll.dll from disk into a new memory region
    //    This ensures we get the original, unhooked bytes.
    //    DONT_RESOLVE_DLL_REFERENCES prevents DllMain from being called.
    HMODULE hFreshNtdll = LoadLibraryExA("ntdll.dll", NULL, DONT_RESOLVE_DLL_REFERENCES);
    if (hFreshNtdll == NULL) {
        printf("[-] Failed to load fresh ntdll.dll from disk.
");
        return 1;
    }
    printf("[+] Loaded fresh ntdll.dll from disk: %p
", hFreshNtdll);

    // 3. Get the address of NtAllocateVirtualMemory in both copies
    FARPROC pNtAllocHooked = GetProcAddress(hNtdll, "NtAllocateVirtualMemory");
    FARPROC pNtAllocOriginal = GetProcAddress(hFreshNtdll, "NtAllocateVirtualMemory");
    FARPROC pNtFree = GetProcAddress(hNtdll, "NtFreeVirtualMemory"); // Also get free for cleanup

    if (pNtAllocHooked == NULL || pNtAllocOriginal == NULL || pNtFree == NULL) {
        printf("[-] Failed to get function addresses.
");
        FreeLibrary(hFreshNtdll);
        return 1;
    }

    printf("[+] NtAllocateVirtualMemory (Hooked in current process): %p
", pNtAllocHooked);
    printf("[+] NtAllocateVirtualMemory (Original from disk): %p
", pNtAllocOriginal);
    printf("[+] NtFreeVirtualMemory: %p
", pNtFree);
