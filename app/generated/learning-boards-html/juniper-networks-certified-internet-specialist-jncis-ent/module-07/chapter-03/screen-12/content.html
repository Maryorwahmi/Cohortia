Comprehensive policy example
Read and trace the policy to predict how routes are modified and accepted.
policy-options {
    community NO_ADVERTISE members "no-advertise";
    community ISP_A_ROUTES members 65000:100;

    policy-statement INBOUND_ISP_A {
        term PREFER_ISP_A_ROUTES {
            from {
                protocol bgp;
                neighbor 203.0.113.1;
            }
            then {
                local-preference 300;
                community add ISP_A_ROUTES;
                next-term;
            }
        }
        term REJECT_BOGONS {
            from {
                route-filter 10.0.0.0/8 orlonger;
                route-filter 172.16.0.0/12 orlonger;
                route-filter 192.168.0.0/16 orlonger;
            }
            then {
                reject;
            }
        }
        term ACCEPT_ALL_OTHERS {
            then {
                accept;
            }
        }
    }

    policy-statement OUTBOUND_TO_ISP_B {
        term FILTER_INTERNAL_ONLY {
            from {
                protocol direct;
                route-filter 192.168.0.0/16 orlonger;
            }
            then {
                as-path-prepend "65500 65500";
                metric 500;
                community add NO_ADVERTISE;
                accept;
            }
        }
        term REJECT_ALL_OTHERS {
            then { reject; }
        }
    }
}
Trace each term: attributes applied in earlier terms persist unless overwritten, and accept/reject decisions determine final installation and advertisement.
