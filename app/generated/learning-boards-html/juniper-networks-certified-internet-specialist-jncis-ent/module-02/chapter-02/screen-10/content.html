# Assume VLAN_SALES (ID 10) and VLAN_HR (ID 20) are already defined

# 1. Configure the RVI for VLAN_SALES
set interfaces irb unit 10 family inet address 192.168.10.1/24

# 2. Configure the RVI for VLAN_HR
set interfaces irb unit 20 family inet address 192.168.20.1/24

# 3. Associate the RVIs with their respective VLANs
set vlans VLAN_SALES l3-interface irb.10
set vlans VLAN_HR l3-interface irb.20

# Commit the changes
commit
Verification steps
  • show interfaces irb — confirm addresses like 192.168.10.1/24
  • show vlans — verify l3-interface is linked to the VLAN
  • show route — see direct routes to 192.168.10.0/24 and 192.168.20.0/24
