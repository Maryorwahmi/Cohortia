Starter script (visual)
const Web3 = require('web3');
const web3 = new Web3('http://127.0.0.1:7545');

const simpleStorageABI = [ /* ABI for SimpleStorage */ ];
const simpleStorageAddress = '0x...';

async function readContractData() {
  if (simpleStorageAddress === '0x...') {
    console.error('ERROR: Please deploy and update the address.');
    return;
  }

  const simpleStorage = new web3.eth.Contract(simpleStorageABI, simpleStorageAddress);
  const currentData = await simpleStorage.methods.getData().call();
  const contractOwner = await simpleStorage.methods.owner().call();
  console.log('Current stored data:', currentData);
  console.log('Contract Owner:', contractOwner);
}

readContractData();

Common mistake
Leaving the placeholder address causes the script to exit or throw. Always confirm the deployed address.
