Terminal: subscribe to headers
const Web3 = require('web3'); const wsUrl = 'ws://127.0.0.1:7545'; const web3 = new Web3(new Web3.providers.WebsocketProvider(wsUrl)); web3.eth.subscribe('newBlockHeaders') .on('connected', id => console.log('Connected', id)) .on('data', async header => { console.log(' New block', header.number, header.hash); const block = await web3.eth.getBlock(header.number, false); console.log('Tx count:', block.transactions.length); }) .on('error', err => console.error('Sub error', err));
Notes
Ensure ws provider is available; clear subscriptions on process exit to avoid resource leaks. Use headers for triggers and fetch details later.
