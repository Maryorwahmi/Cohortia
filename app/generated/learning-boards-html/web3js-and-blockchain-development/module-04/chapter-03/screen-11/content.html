Starter: monitor.js (excerpt)
const Web3 = require('web3'); const ws = 'ws://127.0.0.1:7545'; const web3 = new Web3(ws); // Subscribe to headers web3.eth.subscribe('newBlockHeaders') .on('connected', id => console.log('Headers sub', id)) .on('data', async header => { console.log('NEW BLOCK', header.number); const block = await web3.eth.getBlock(header.number, false); console.log('Txs in block:', block.transactions.length); }); // Subscribe to pending txs web3.eth.subscribe('pendingTransactions') .on('data', async txHash => { console.log('PENDING TX', txHash); const tx = await web3.eth.getTransaction(txHash); if (tx) console.log(' From:', tx.from, 'To:', tx.to); });
Guidance
Run the script and then create transactions in Ganache. Observe both the header events and pending transaction hashes appearing in the terminal.
Debug tip

If subscriptions fail, verify you used a WebSocket provider and that the node supports subscriptions.
