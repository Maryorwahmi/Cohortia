Starter Code (connect.js)
const Web3 = require('web3');

async function connectToBlockchain() {
    // TODO: Define the Ganache provider URL
    const providerUrl = 'http://127.0.0.1:7545'; 

    // TODO: Create a new Web3 instance with the provider
    const web3 = new Web3(providerUrl);

    try {
        // TODO: Check if the node is listening and log the result
        const isListening = await web3.eth.net.isListening();
        console.log('Is connected to Ethereum node:', isListening);

        if (isListening) {
            // TODO: Get the network ID and log it
            const networkId = await web3.eth.net.getId();
            console.log('Connected to Network ID:', networkId);

            // Get the current block number as an extra check
            const blockNumber = await web3.eth.getBlockNumber();
            console.log('Current Block Number:', blockNumber);
        } else {
            console.error('Failed to connect to the Ethereum node.');
        }
    } catch (error) {
        console.error('An error occurred during connection:', error.message);
        console.error('Please ensure Ganache is running on', providerUrl);
    }
}

connectToBlockchain();
Activity checklist
  1. Create a new directory and run npm init -y.
  2. Install Web3.js with npm install web3.
  3. Start Ganache and run node connect.js to verify output.
  4. Stop Ganache and run the script again to observe the error handling path.
Expected outcome

When Ganache runs, the script prints connected true, the network ID, and the current block number. When Ganache is down, the script logs a connection error with the provider URL for debugging.
