fetchData using BatchRequest

Use web3.BatchRequest inside a useCallback, include web3 and contract in dependencies, and update state once.

const fetchData = useCallback(async () => {
  if (contract && web3) {
    try {
      const batch = new web3.BatchRequest();
      let fetchedCount = 0;
      let fetchedOwner = '';

      batch.add(contract.methods.getCount().call.request({}, (err, data) => {
        if (err) console.error('Batch error fetching count:', err);
        else fetchedCount = parseInt(data);
      }));

      batch.add(contract.methods.getOwner().call.request({}, (err, data) => {
        if (err) console.error('Batch error fetching owner:', err);
        else fetchedOwner = data;
      }));

      await batch.execute();

      setCount(fetchedCount);
      setOwner(fetchedOwner);
      setError('');
    } catch (err) {
      setError(`Error fetching data: ${err.message}`);
      console.error('Error fetching data:', err);
    }
  }
}, [contract, web3]);
Pro tip
Avoid setState in callbacks; update once after execute to minimize rerenders.
Common mistake
Forgetting to include web3 in dependencies can cause stale batches or runtime errors when provider changes.
