import Web3 from 'web3';

const provider = new Web3.providers.HttpProvider("http://127.0.0.1:8545");
const web3 = new Web3(provider);
const counterContract = new web3.eth.Contract(COUNTER_ABI, CONTRACT_ADDRESS);

// 1. Retrieve past events
const past = await counterContract.getPastEvents('CounterChanged', { fromBlock: 0, toBlock: 'latest' });
// parse past.forEach(ev => ev.returnValues.newCount, ev.returnValues.sender)

// 2. Subscribe to real-time events
counterContract.events.CounterChanged()
  .on('data', event => { /* parse event.returnValues.newCount and sender */ })
  .on('changed', ch => { /* handle reorg */ })
  .on('error', e => { /* log and reconnect */ });
Parsing returnValues
Web3.js provides event.returnValues where numbers may be strings. Convert the value to an integer or BigNumber before using it in the UI.
Verification step
After sending an increment transaction, confirm the listener logs the event and that getPastEvents includes the new entry with its block number.
Terminal-demo note
