Starter snippet
async function incrementCounter() {
  const accounts = await web3.eth.getAccounts();
  const sender = accounts[0];
  console.log(`Incrementing from ${sender}`);
  const gasEstimate = await counterContract.methods.increment().estimateGas({ from: sender });
  const receipt = await counterContract.methods.increment().send({ from: sender, gas: gasEstimate + 10000 });
  console.log('Transaction Hash:', receipt.transactionHash);
  console.log('Gas Used:', receipt.gasUsed);
  const newCount = await counterContract.methods.getCount().call();
  console.log('New count:', parseInt(newCount));
}
Checks
  • Ensure Ganache is running and contract deployed.
  • Verify accounts[0] has enough Ether.
  • If out-of-gas, increase the buffer and retry.
