Starter Code
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract MyFirstContract {
    string public greeting = "Hello, Cohortia!"; // A state variable

    function setGreeting(string memory _newGreeting) public {
        greeting = _newGreeting;
    }

    function getGreeting() public view returns (string memory) {
        return greeting;
    }

    event GreetingUpdated(string indexed oldGreeting, string newGreeting, address updater);

    function changeGreetingAndLog(string memory _newGreeting) public {
        string memory _oldGreeting = greeting;
        greeting = _newGreeting;
        emit GreetingUpdated(_oldGreeting, _newGreeting, msg.sender);
    }
}
What to inspect in Remix
  • Confirm the compiler version matches the pragma line before compiling.
  • Open Compilation Details and copy the ABI JSON for use with Web3.js.
  • Copy the bytecode if you plan to deploy programmatically or via scripts.
  • Use the deployed instance in Remix to call read functions for free and send transactions to update state.
Key Takeaway

The ABI connects your human-readable contract to front-end code; the bytecode is what you publish to create the immutable contract account.
