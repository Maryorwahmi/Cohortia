Code template
const Web3 = require('web3');

// Replace with your Infura project ID
const INFURA_PROJECT_ID = 'YOUR_INFURA_PROJECT_ID';
const providerUrl = `https://mainnet.infura.io/v3/${INFURA_PROJECT_ID}`;
const web3 = new Web3(providerUrl);

async function exploreBlock() {
  try {
    const latestBlockNumber = await web3.eth.getBlockNumber();
    console.log(`Latest Block Number: ${latestBlockNumber}`);

    const latestBlock = await web3.eth.getBlock(latestBlockNumber, true);
    console.log('Block Hash:', latestBlock.hash);
    console.log('Parent Hash:', latestBlock.parentHash);
    console.log('Timestamp:', new Date(latestBlock.timestamp * 1000).toUTCString());
    console.log('Gas Used:', latestBlock.gasUsed);
    console.log('Transactions Root:', latestBlock.transactionsRoot);
    console.log('State Root:', latestBlock.stateRoot);

    if (latestBlock.transactions.length > 0) {
      const firstTx = latestBlock.transactions[0];
      console.log('First Tx Hash:', firstTx.hash);
      console.log('From:', firstTx.from);
      console.log('To:', firstTx.to);
      console.log('Value (Wei):', firstTx.value);
    }
  } catch (error) {
    console.error('Error exploring block:', error);
  }
}

exploreBlock();
This exact template prints header fields and a sample transaction so you can map printed roots back to theory discussed earlier in the chapter.
What to observe
  • stateRoot: the authoritative post-block world state hash.
  • transactionsRoot: proves inclusion of any transaction in the block.
  • first transaction fields: from, to, value, input; useful for debugging.
Misconception
Running the script reads state; it doesn't alter the chain. To change on-chain state you must sign and broadcast transactions from an EOA.
Key Takeaway

Use web3.js to connect theory to live data — inspecting headers and transactions solidifies understanding of state commitments.
