// Deploy
const MyContract = new web3.eth.Contract(myContractABI);
const deployedContract = await MyContract.deploy({ data: myContractBytecode }).send({ from: deployerAccount, gas: 1500000 });

// Instantiate with address
const myDeployedContract = new web3.eth.Contract(myContractABI, deployedContract.options.address);

// Call
const currentGreeting = await myDeployedContract.methods.getGreeting().call();

// Send
const receipt = await myDeployedContract.methods.setGreeting("Hello from Web3.js!").send({ from: deployerAccount, gas: 100000 });

// Events
myDeployedContract.events.GreetingUpdated({ fromBlock: receipt.blockNumber }).on('data', event => { /* ... */ });
Run checklist
Start Ganache, paste your ABI and bytecode, replace addresses, then run node interactContract.js.
Debug tip
If send fails, check the from address, unlock the account or configure the provider to sign transactions, and increase gas limit.
