Starter code
// connectWalletConnect (excerpt)
const provider = new WalletConnectProvider({
  infuraId: INFURA_ID,
  rpc: {
    [SEPOLIA_CHAIN_ID]: NETWORKS[SEPOLIA_CHAIN_ID].rpcUrl,
    [MUMBAI_CHAIN_ID]: NETWORKS[MUMBAI_CHAIN_ID].rpcUrl,
  },
  qrcodeModalOptions: { mobileLinks: ['metamask', 'trust', 'rainbow'] }
});

await provider.enable();
const web3Instance = new Web3(provider);
setWeb3(web3Instance);
setAccount((await web3Instance.eth.getAccounts())[0]);
setChainId(await web3Instance.eth.getChainId());
Important: attach provider.on('accountsChanged', ...), provider.on('chainChanged', ...), and provider.on('disconnect', ...) and remove them on cleanup.
