// scripts/deploy-upgradable.js
const { upgrades } = require("hardhat");

async function main() {
  const MyContract = await ethers.getContractFactory("MyContract");
  const myContract = await upgrades.deployProxy(MyContract, [initialArg], { kind: "uups" });
  await myContract.deployed();
  console.log("MyContract deployed to:", myContract.address);
  console.log("Implementation address:", await upgrades.erc1967.getImplementationAddress(myContract.address));
}

main();
Checklist after deploy
  • Confirm proxy address returned by deployProxy.
  • Log and verify the implementation address with erc1967 helper.
  • Interact with retrieve/store functions and verify state is persisted.
Common mistake

Do not call initialize directly on the implementation contract; always initialize through the proxy.
