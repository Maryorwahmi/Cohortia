// scripts/upgrade.js
const { upgrades } = require("hardhat");

async function main() {
  const MyContractV2 = await ethers.getContractFactory("MyContractV2");
  const existingProxyAddress = "0x...";
  const myContractV2 = await upgrades.upgradeProxy(existingProxyAddress, MyContractV2);
  console.log("New implementation:", await upgrades.erc1967.getImplementationAddress(myContractV2.address));
}

main();
Post-upgrade checks
  • Call retrieve() to confirm prior state persists.
  • Call new V2 functions (e.g., increment) to confirm new logic works.
  • Log the implementation address to ensure it changed.
If upgrade fails

An incompatibility error likely means storage layout changed in an unsafe way. Revisit layout changes and use migrations that append variables only.
