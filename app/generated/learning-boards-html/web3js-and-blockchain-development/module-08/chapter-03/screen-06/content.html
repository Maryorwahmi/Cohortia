VulnerableBank vs SecureBank

Compare the exact withdraw flow and see how OpenZeppelin utilities fit in.

VulnerableBank.sol (excerpt)
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract VulnerableBank {
  mapping(address => uint256) public balances;

  function deposit() public payable {
    balances[msg.sender] += msg.value;
  }

  // VULNERABLE WITHDRAW FUNCTION
  function withdraw(uint256 _amount) public {
    require(balances[msg.sender] >= _amount, "Insufficient balance");

    // External call before state update - REENTRANCY VULNERABILITY
    (bool success, ) = msg.sender.call{value: _amount}("");
    require(success, "Transfer failed");

    balances[msg.sender] -= _amount; // State updated AFTER external call
  }
}
Vulnerable because state is decremented after the external call, enabling reentry.
SecureBank.sol (excerpt)
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "@openzeppelin/contracts/access/Ownable.sol";
import "@openzeppelin/contracts/utils/ReentrancyGuard.sol";

contract SecureBank is Ownable, ReentrancyGuard {
  mapping(address => uint256) public balances;

  constructor() Ownable(msg.sender) {}

  function deposit() public payable {
    balances[msg.sender] += msg.value;
  }

  // SECURE WITHDRAW FUNCTION
  function withdraw(uint256 _amount) public nonReentrant {
    require(balances[msg.sender] >= _amount, "Insufficient balance");

    balances[msg.sender] -= _amount; // State updated BEFORE external call

    (bool success, ) = msg.sender.call{value: _amount}("");
    require(success, "Transfer failed");
  }
}
Secure because state is updated first and the nonReentrant modifier blocks recursive entry.
Example scenario
Deploy the vulnerable contract, deposit some ETH, then run a malicious contract that calls withdraw inside its fallback. The attacker can loop until drained. Test the secure contract the same way to see the guard and ordering block the drain.
