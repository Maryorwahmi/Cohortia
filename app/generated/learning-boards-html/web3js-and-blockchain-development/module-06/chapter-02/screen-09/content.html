main.jsx example
import React from 'react'
import ReactDOM from 'react-dom/client'
import App from './App.jsx'
import { Web3Provider } from './contexts/Web3Provider.jsx'

ReactDOM.createRoot(document.getElementById('root')).render(
  <React.StrictMode>
    <Web3Provider>
      <App />
    </Web3Provider>
  </React.StrictMode>
)
App.jsx usage
function App() {
  const { isLoading, isMetaMaskInstalled, currentAccount, chainId, connectWallet } = useWeb3();

  if (isLoading) return <div>Loading Web3...</div>

  return (
    <div>
      {!isMetaMaskInstalled && <div>MetaMask not installed</div>}
      {isMetaMaskInstalled && !currentAccount && <div className="connect-pill">Connect Wallet (call connectWallet)</div>}
      {currentAccount && <div>Connected: <code>{currentAccount}</code> Chain: <code>{chainId}</code></div>}
    </div>
  );
}
Key Takeaway

Place Web3Provider at the app root and read connection state via useWeb3 so the app reacts to connect, disconnect, and network changes consistently.
