Add DataChanged to SimpleStorage
Recompile, redeploy, and update your frontend ABI and address after making the change.
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract SimpleStorage {
  string public storedData;
  event DataChanged(address indexed changer, string oldData, string newData);

  constructor(string memory initialData) {
    storedData = initialData;
  }

  function setData(string memory newData) public {
    string memory _oldData = storedData;
    storedData = newData;
    emit DataChanged(msg.sender, _oldData, newData);
  }
}
Practical checklist
  • Recompile and redeploy the contract after changing events.
  • Update src/contracts/SimpleStorage.json ABI and VITE_SIMPLE_STORAGE_CONTRACT_ADDRESS in your .env.
  • Verify the event is present using getPastEvents or by sending a test transaction.
Key Takeaway

Emit an event whenever a meaningful write occurs so clients can react without polling the chain.
