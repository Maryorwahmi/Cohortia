sendSetDataTransaction
This pattern sets flags, listens for events, and refreshes contract data on success.
Checklist
  • Guard for simpleStorageContract and currentAccount.
  • Set isSendingTransaction and transactionStatus before sending.
  • Chain on('transactionHash'), on('receipt'), on('error') and call fetchContractData on success.
Example hook snippet
const sendSetDataTransaction = useCallback(async (newData) => {
  if (!simpleStorageContract || !currentAccount) {
    setError('Contract or account not initialized for transaction.');
    return;
  }
  setIsSendingTransaction(true);
  setTransactionStatus('pending');
  setTransactionHash(null);
  setError(null);

  try {
    await simpleStorageContract.methods.setData(newData).send({ from: currentAccount })
      .on('transactionHash', hash => {
        setTransactionHash(hash);
      })
      .on('receipt', receipt => {
        setTransactionStatus('success');
        fetchContractData();
      })
      .on('error', (err) => {
        setTransactionStatus('error');
        setError(err.message || 'Transaction failed.');
      });
  } catch (err) {
    setTransactionStatus('error');
    setError(err.message || 'Transaction failed.');
  } finally {
    setIsSendingTransaction(false);
  }
}, [simpleStorageContract, currentAccount, fetchContractData]);
