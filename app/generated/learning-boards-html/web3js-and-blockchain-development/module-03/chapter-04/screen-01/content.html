Starter Example
function checkValue(uint256 _input) public returns (string memory) {
  if (_input > 100) { value = _input; return "Value set to high input."; }
  else if (_input > 50) { value = _input / 2; return "Value set to half of medium input."; }
  else { value = 0; return "Value reset to zero."; }
}
