Starter contract
Paste this into SimpleToken.sol in Remix and follow the steps below.
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract SimpleToken {
    string public name = "Simple Token";
    string public symbol = "STK";
    uint256 public totalSupply;
    mapping(address => uint256) public balances;

    event Transfer(address indexed _from, address indexed _to, uint256 _value);

    constructor(uint225 _initialSupply) {
        require(_initialSupply > 0, "Initial supply must be positive.");
        totalSupply = _initialSupply;
        balances[msg.sender] = _initialSupply;
    }

    function transfer(address _to, uint256 _amount) public returns (bool success) {
        require(balances[msg.sender] >= _amount, "Insufficient balance.");
        require(_to != address(0), "Cannot transfer to zero address.");

        balances[msg.sender] -= _amount;
        balances[_to] += _amount;
        emit Transfer(msg.sender, _to, _amount);
        return true;
    }

    function balanceOf(address _owner) public view returns (uint256) {
        return balances[_owner];
    }
}
Lab steps
  1. Compile SimpleToken and switch to JavaScript VM.
  2. Enter an initial supply (e.g., 1000 followed by 18 zeros) and Deploy.
  3. Call name, symbol, and totalSupply (view calls).
  4. Call balanceOf for deployer, transfer to another account, then re-check balances and events.
Safety note

When moving to public testnets, never expose private keys. Use wallets or environment-managed keys for deployments.
