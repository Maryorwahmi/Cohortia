Program examples
Sum grid
for row in range(5):
    for col in range(5):
        print(f"{row + col:3}", end='')
    print()
Hollow rectangle
width = 8
height = 5
for row in range(height):
    for col in range(width):
        if row == 0 or row == height - 1 or col == 0 or col == width - 1:
            print("*", end='')
        else:
            print(" ", end='')
    print()
Chessboard
for row in range(8):
    for col in range(8):
        if (row + col) % 2 == 0:
            print("X", end=' ')
        else:
            print("O", end=' ')
    print()
Pairs summing to 15
for i in range(1, 11):
    for j in range(i + 1, 11):
        if i + j == 15:
            print(f"Pair: ({i}, {j})")
Key Takeaway

Implement these short programs and add counters to confirm the iteration counts match your expectations.
