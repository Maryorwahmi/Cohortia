Why this helps

Starter code
def get_order():
    size = input("Enter pizza size (small/medium/large): ").lower()
    quantity = int(input("Enter quantity: "))
    return size, quantity

def get_price(size):
    if size == "small":
        return 8.0
    elif size == "medium":
        return 10.0
    elif size == "large":
        return 12.0
    else:
        return 0.0
