Tests for pop and peek
TEST(StackTest, PopRemovesElementAndDecreasesSize) {
    Stack s;
    s.push(10);
    s.push(20);
    EXPECT_EQ(20, s.pop());
    EXPECT_EQ(1, s.size());
    EXPECT_EQ(10, s.pop());
    EXPECT_TRUE(s.isEmpty());
}

TEST(StackTest, PeekReturnsTopElementWithoutRemoving) {
    Stack s;
    s.push(10);
    s.push(20);
    EXPECT_EQ(20, s.peek());
    EXPECT_EQ(2, s.size());
}

TEST(StackTest, PopFromEmptyStackThrowsException) {
    Stack s;
    EXPECT_THROW(s.pop(), std::out_of_range);
}
Implement pop and peek to throw std::out_of_range when the stack is empty, and update tests until green.
Implementation snippet
int Stack::pop() {
  if (isEmpty()) throw std::out_of_range("Stack is empty, cannot pop.");
  int value = data.back();
  data.pop_back();
  return value;
}

int Stack::peek() const {
  if (isEmpty()) throw std::out_of_range("Stack is empty, cannot peek.");
  return data.back();
}
Key Takeaway

Edge-case tests make contracts explicit and prevent undefined behavior later in production.
