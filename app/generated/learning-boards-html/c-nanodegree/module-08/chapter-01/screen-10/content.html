# Define a custom command to generate a version header
add_custom_command(
  OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/version.h"
  COMMAND ${CMAKE_COMMAND} -E echo "#pragma once" > "${CMAKE_CURRENT_BINARY_DIR}/version.h"
  COMMAND ${CMAKE_COMMAND} -E echo "#define MY_LIB_VERSION \"1.0.0\"" >> "${CMAKE_CURRENT_BINARY_DIR}/version.h"
  COMMENT "Generating version.h"
)

add_library(mylib STATIC
  src/lib.cpp
)

add_custom_target(generate_version_header ALL
  DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/version.h"
)
add_dependencies(mylib generate_version_header)

target_include_directories(mylib PUBLIC
  $
  $
  $
)

set_target_properties(mylib PROPERTIES
  VERSION 1.0.0
  SOVERSION 1
)
app/CMakeLists.txt (excerpt)
add_executable(myapp main.cpp)

target_link_libraries(myapp PRIVATE mylib)

# mylib exposes include dirs via PUBLIC, so app can find headers
Important checks
  • Ensure ${CMAKE_CURRENT_BINARY_DIR} (binary dir) is in include path so generated headers are found.
  • Use add_dependencies so compilation waits for generation.
  • Link mylib as PRIVATE to avoid propagating its implementation details.
Key Takeaway

Connect generation steps to targets and include the binary dir so generated artifacts are visible at compile time.
