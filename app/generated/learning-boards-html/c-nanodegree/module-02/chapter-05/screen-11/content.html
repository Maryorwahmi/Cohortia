Constructor & destructor (pattern)
Matrix(int r, int c) : rows(r), cols(c) {
  data = new double*[rows];
  for (int i = 0; i < rows; ++i) data[i] = new double[cols]();
}

~Matrix() {
  for (int i = 0; i < rows; ++i) delete[] data[i];
  delete[] data;
}
Operator+ and operator* (patterns)
Matrix operator+(const Matrix& other) const {
  if (rows != other.rows || cols != other.cols) throw std::invalid_argument("dimensions");
  Matrix result(rows, cols);
  for (int i=0;i<rows;++i) for (int j=0;j<cols;++j)
    result.data[i][j] = data[i][j] + other.data[i][j];
  return result;
}

Matrix operator*(const Matrix& other) const {
  if (cols != other.rows) throw std::invalid_argument("dimensions");
  Matrix result(rows, other.cols);
  for (int i=0;i<rows;++i) for (int j=0;j<other.cols;++j)
    for (int k=0;k<cols;++k) result.data[i][j] += data[i][k] * other.data[k][j];
  return result;
}
Concrete example
Add two 2x2 matrices to get element-wise sums; multiply with dimension checks to prevent invalid math.
Misconception
Assuming the compiler will handle resource copying is unsafe when your class uses raw pointers.
Key Takeaway
The Matrix lab is a practical check of correct memory ownership and operator design under real constraints.
