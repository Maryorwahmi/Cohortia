Reference code
#include <iostream>
#include <string>
#include <cmath>

class Shape {
protected:
  std::string color;
public:
  Shape(std::string c) : color(c) {
    std::cout << "Shape constructor called for color: " << color << std::endl;
  }
  void displayColor() const {
    std::cout << "Shape Color: " << color << std::endl;
  }
  ~Shape() {
    std::cout << "Shape destructor called for color: " << color << std::endl;
  }
};

class Circle : public Shape {
private:
  double radius;
public:
  Circle(std::string c, double r) : Shape(c), radius(r) {
    if (radius <= 0) {
      std::cout << "Invalid radius, setting to 1.0" << std::endl;
      radius = 1.0;
    }
    std::cout << "Circle constructor called with radius: " << radius << std::endl;
  }

  double getArea() const {
    return M_PI * radius * radius;
  }

  void displayColor() const {
    Shape::displayColor();
    std::cout << "Radius: " << radius << std::endl;
  }

  ~Circle() {
    std::cout << "Circle destructor called with radius: " << radius << std::endl;
  }
};

int main() {
  Circle myCircle("Blue", 5.0);
  myCircle.displayColor();
  std::cout << "Circle Area: " << myCircle.getArea() << std::endl;

  Circle invalidCircle("Red", -2.0);
  invalidCircle.displayColor();
  return 0;
}
Why this works
Initializer list constructs Shape subobject and radius before constructor body runs, ensuring safe use of members.
Validation
Validation happens in the constructor body after initialization so you can adjust invalid values deterministically.
