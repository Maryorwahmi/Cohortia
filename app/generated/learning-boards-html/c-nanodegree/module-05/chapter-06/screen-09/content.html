Processing with std::visit

#include <variant>
#include <string>
#include <iostream>
#include <type_traits>

std::variant<int,double,std::string> v = "Hello";

std::visit([](auto&& arg){
  using T = std::decay_t<decltype(arg)>;
  if constexpr (std::is_same_v<T,int>) {
    std::cout << "int: " << arg * 2 << '
';
  } else if constexpr (std::is_same_v<T,double>) {
    std::cout << "double: " << arg * 2.0 << '
';
  } else if constexpr (std::is_same_v<T,std::string>) {
    std::cout << "string: " << arg << "!" << '
';
  }
}, v);
Notes
  • Visitor lambdas can use if constexpr to get type-specific code paths.
  • Overloaded function objects provide another clean pattern for visitors.
  • Handle all alternatives to avoid unexpected behavior or exceptions.
