Convert and compute duration units

auto start = std::chrono::high_resolution_clock::now();
// ... work ...
auto end = std::chrono::high_resolution_clock::now();
auto d = end - start; // d is a duration

// Convert to milliseconds
auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(d);
std::cout << ms.count() << " ms
";
Pitfall
Casting from milliseconds to seconds truncates fractional parts; consider using floating-point durations for precision.
Practice
Measure a loop in microseconds, convert to milliseconds, and report both for comparison.
