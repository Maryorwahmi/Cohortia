#include <iostream>
#include <vector>
#include <algorithm>
#include <numeric>
#include <random>
#include <ctime>

int main() {
  // 1. Initialize Data
  std::vector<int> readings;
  readings.reserve(10);
  std::srand(static_cast<unsigned int>(std::time(nullptr)));
  for (int i = 0; i < 10; ++i) {
    readings.push_back(std::rand() % 100 + 1);
  }

  // 2. Display Readings
  std::cout << "Original Readings: ";
  std::for_each(readings.begin(), readings.end(), [](int r){ std::cout << r << ' '; });
  std::cout << std::endl;

}
