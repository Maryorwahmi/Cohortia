#include <iostream>
#include <string>
#include <map>
#include <unordered_map>
#include <set>
#include <sstream>
#include <algorithm>

std::string toLower(std::string s) {
  std::transform(s.begin(), s.end(), s.begin(), [](unsigned char c){ return std::tolower(c); });
  return s;
}

std::string findSynonym(const std::map<std::string, std::string>& thes, const std::string& word) {
  auto it = thes.find(word);
  if (it != thes.end()) return it->second;
  return "Not found";
}

int main() {
  std::map<std::string, std::string> thesaurus;
  thesaurus["happy"] = "joyful";
  thesaurus["sad"] = "unhappy";

  std::unordered_map<std::string, int> wordCounts;
  std::set<std::string> uniqueWords;

  std::string sentence = "The quick brown fox jumps over the lazy dog the quick brown fox";
  std::stringstream ss(sentence);
  std::string word;
  while (ss >> word) {
    word = toLower(word);
    wordCounts[word]++;
    uniqueWords.insert(word);
  }

  // Demonstration outputs omitted for brevity
  return 0;
}
