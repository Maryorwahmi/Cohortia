C++20 Concepts: Defining and Using

Crafting custom constraints and leveraging built-in concepts for templates.

Custom Concepts

Defined with concept and a requires clause to specify type capabilities.

Built-in Concepts

Standard library provides concepts like std::integral for common type categories.

Key Takeaway

Concepts allow you to explicitly define and apply type constraints, leading to more readable template code and significantly clearer compiler error messages.

#include <iostream>
#include <concepts> // For std::integral, std::same_as

// Custom concept: AddableAndPrintable
template <typename T>
concept AddableAndPrintable = requires(T a, T b) {
    { a + b } -> std::same_as<T>; // a + b must compile and return T
    { std::cout << a };           // std::cout << a must compile
};

// Function constrained by custom concept
template <AddableAndPrintable T>
T sumAndPrint(T a, T b) {
    T result = a + b;
    std::cout << "Sum: " << result << std::endl;
    return result;
}

// Function constrained by built-in concept
template <std::integral T>
void processIntegral(T value) {
    std::cout << "Processing integral: " << value << std::endl;
}

int main() {
    sumAndPrint(10, 20);
    sumAndPrint(3.5, 4.2);
    // sumAndPrint("hello", "world"); // Fails with clear concept error

    processIntegral(50);
    // processIntegral(3.14); // Fails with clear concept error
    return 0;
}
