#include <iostream>
#include <thread>
#include <shared_mutex>
#include <vector>
#include <string>

std::string shared_resource = "Initial data";
std::shared_mutex rw_mutex;

void reader(int id) {
  for (int i = 0; i < 3; ++i) {
    std::shared_lock<std::shared_mutex> lock(rw_mutex);
    std::cout << "Reader " << id << " reads: " << shared_resource << std::endl;
    lock.unlock();
    std::this_thread::sleep_for(std::chrono::milliseconds(100));
  }
}

void writer(int id, const std::string& new_data) {
  std::this_thread::sleep_for(std::chrono::milliseconds(50));
  std::unique_lock<std::shared_mutex> lock(rw_mutex);
  shared_resource = new_data + " (by Writer " + std::to_string(id) + ")";
  std::cout << "Writer " << id << " wrote: " << shared_resource << std::endl;
}
Observations to make
  • Readers print concurrently without blocking each other.
  • Writers obtain exclusive access and their update is visible to subsequent reads.
  • Short write critical sections minimize read blocking.
Try this
Increase writer frequency and observe whether readers suffer. If writes dominate, a plain mutex may be simpler and faster due to reduced lock complexity.
