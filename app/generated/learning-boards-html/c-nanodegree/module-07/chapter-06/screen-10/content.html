Map iteration
std::map<std::string,int> scores = {{"Alice",90}};
for (const auto& [name, score] : scores) {
  std::cout << name << " - " << score << std::endl;
}
Tuple unpack
std::tuple<int,std::string,double> t = {1,"C",4.2};
auto [id, name, val] = t;
std::cout << id << name << val;
Mnemonic
Structured bindings = direct names for multiple return values. Choose const auto& to avoid copies when iterating.
