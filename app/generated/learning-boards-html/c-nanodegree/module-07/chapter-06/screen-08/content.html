Before
std::map<std::string,int> grades = {{"Alice",90}};
for (std::map<std::string,int>::const_iterator it = grades.begin(); it != grades.end(); ++it) {
  std::cout << it->first << ": " << it->second << std::endl;
}
After
std::map<std::string,int> grades = {{"Alice",90}};
for (const auto& [name, score] : grades) {
  std::cout << name << ": " << score << std::endl;
}
Key takeaway
The after form communicates intent: iterate entries and use name and score directly. It avoids manual iterator handling and long type names.
