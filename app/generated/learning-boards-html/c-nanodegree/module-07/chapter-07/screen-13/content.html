Module 8 › Chapter 7.7 › Hands-on Activity
Task steps
  1. Define NodeA and NodeB; NodeA has std::shared_ptr<NodeB> next; NodeB has std::weak_ptr<NodeA> prev;
  2. In main create shared_ptr instances with std::make_shared and connect next and prev;
  3. Print use_count values and call a check_prev method that locks the weak_ptr and reports liveness;
  4. Exit the scope and observe destructors running; then repeat with NodeB::prev as shared_ptr to observe the leak.
Starter code
#include <iostream>
#include <memory>

class NodeA; // forward

class NodeB {
public:
  std::weak_ptr<NodeA> prev; // avoid cycle
  NodeB() { std::cout << "NodeB created!" << std::endl; }
  ~NodeB() { std::cout << "NodeB destroyed!" << std::endl; }
  void check_prev();
};

class NodeA {
public:
  std::shared_ptr<NodeB> next;
  NodeA() { std::cout << "NodeA created!" << std::endl; }
  ~NodeA() { std::cout << "NodeA destroyed!" << std::endl; }
};

int main() {
  std::cout << "Starting main..." << std::endl;
  {
    auto node_a = std::make_shared<NodeA>();
    auto node_b = std::make_shared<NodeB>();
    node_a->next = node_b;
    node_b->prev = node_a; // weak_ptr breaks cycle
  }
  std::cout << "Exiting main." << std::endl;
}
