Return values via futures

// Modified task
int calculate_square(int num) {
  std::this_thread::sleep_for(std::chrono::milliseconds(50));
  int result = num * num;
  std::cout << "Task: Calculating square of " << num << " -> " << result << std::endl;
  return result;
}

// In main: collect futures
std::vector<std::future<int>> results;
for (int i = 0; i < 10; ++i) {
  results.emplace_back(pool.enqueue(calculate_square, i+1));
}

int total = 0;
for (auto &f : results) {
  total += f.get();
}
