Simulate Order Processor (send_order.py)

import os
from azure.servicebus import ServiceBusClient, ServiceBusMessage

CONNECTION_STR = "YOUR_SERVICE_BUS_CONNECTION_STRING"
QUEUE_NAME = "orders"

def send_single_message(sender):
    message = ServiceBusMessage("Order ID: 12345, Item: Laptop, Quantity: 1")
    sender.send_messages(message)
    print("Sent a single message.")

def send_batch_messages(sender):
    messages = [ServiceBusMessage(f"Order ID: {i}, Item: Mouse, Quantity: {i%3 + 1}") for i in range(100, 105)]
    with sender:
        for message in messages:
            sender.send_messages(message)
    print("Sent a batch of messages.")

print("Connecting to Service Bus...")
servicebus_client = ServiceBusClient.from_connection_string(conn_str=CONNECTION_STR, logging_enable=True)

with servicebus_client:
    sender = servicebus_client.get_queue_sender(queue_name=QUEUE_NAME)
    with sender:
        send_single_message(sender)
        send_batch_messages(sender)

print("Done sending messages.")
Action

Run pip install azure-servicebus, then execute python send_order.py after replacing the connection string.

Simulate Inventory Updater (receive_order.py)

import os
from azure.servicebus import ServiceBusClient, ServiceBusReceiveMode

CONNECTION_STR = "YOUR_SERVICE_BUS_CONNECTION_STRING"
QUEUE_NAME = "orders"

def receive_messages():
    servicebus_client = ServiceBusClient.from_connection_string(conn_str=CONNECTION_STR, logging_enable=True)
    with servicebus_client:
        receiver = servicebus_client.get_queue_receiver(queue_name=QUEUE_NAME, receive_mode=ServiceBusReceiveMode.PEEK_LOCK)
        with receiver:
            for msg in receiver:
                print(f"Received: {str(msg)}")
                # Simulate processing the order
                print(f"Processing order: {str(msg)}")
                # Complete the message to remove it from the queue
                receiver.complete_message(msg)
                print("Message completed.")
                break # For this demo, we'll just process one message at a time

print("Connecting to Service Bus to receive messages...")
receive_messages()
print("Done receiving messages.")
Key Takeaway

This simulation demonstrates how Azure Service Bus effectively decouples applications, allowing them to communicate reliably and asynchronously, which is crucial for scalable distributed systems.
