import os import shutil project_root = 'my_project' src_dir = os.path.join(project_root,'src') docs_dir = os.path.join(project_root,'docs') os.makedirs(src_dir,exist_ok=True) os.makedirs(docs_dir,exist_ok=True) with open(os.path.join(src_dir,'main.py'),'w') as f: f.write("print('Hello, Project!')") with open(os.path.join(docs_dir,'README.md'),'w') as f: f.write('# My Project') print(os.listdir(project_root)) shutil.copytree(docs_dir,'docs_backup') shutil.move(os.path.join(src_dir,'main.py'),project_root) shutil.rmtree(project_root) shutil.rmtree('docs_backup')
Guided checklist
  1. Create project folders using os.makedirs with exist_ok
  2. Create files with with open to ensure closure
  3. Verify contents using os.listdir
  4. Create docs backup with shutil.copytree and ensure destination does not exist
  5. Move main.py with shutil.move and clean up with shutil.rmtree after checks
