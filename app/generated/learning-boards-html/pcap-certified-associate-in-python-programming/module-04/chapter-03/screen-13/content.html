Step 1 — Raise built-in
def check_age(age):
  if not isinstance(age, int):
    raise ValueError("age must be an integer")
  if age < 0:
    raise ValueError("Age cannot be negative")
Step 2 — Custom exception
class InvalidCredentialsError(Exception):
  def __init__(self, username, reason):
    self.username = username
    self.reason = reason
    super().__init__(f"Invalid credentials for {username}: {reason}")

def login(user, password):
  # Simulated check
  if user != 'admin' or password != 'secret':
    raise InvalidCredentialsError(user, "invalid username or password")
