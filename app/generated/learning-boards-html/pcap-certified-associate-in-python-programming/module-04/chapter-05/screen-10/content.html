Starter Code Template
import csv

input_filename = "sales_data.csv"
output_filename = "product_revenue.csv"

# reading and aggregating
product_revenues = {}
try:
  with open(input_filename, 'r', newline='') as infile:
    reader = csv.DictReader(infile)
    for row in reader:
      try:
        product = row['Product']
        qty = int(row['Quantity'])
        price = float(row['Price'])
        product_revenues[product] = product_revenues.get(product, 0.0) + qty * price
      except (KeyError, ValueError):
        # skip malformed rows
        continue
except FileNotFoundError:
  print(f"Error: {input_filename} not found.")
Write Output
if product_revenues:
  with open(output_filename, 'w', newline='') as outfile:
    fieldnames = ['Product', 'Total Revenue']
    writer = csv.DictWriter(outfile, fieldnames=fieldnames)
    writer.writeheader()
    for p, r in product_revenues.items():
      writer.writerow({'Product': p, 'Total Revenue': f"{r:.2f}"})
Key Takeaway

Validate numeric conversions and skip or log malformed rows to keep the aggregation accurate.
