Module 5 › Chapter 5.3 › Polling example
Walkthrough notes
  • • PollRawInput simulates OS input and updates key states each frame.
  • • InputManager maps KeyCode to GameAction and queries current states.
  • • The game loop polls, asks for active action, then handles it and waits for next frame.
Tip: The example returns the first active action found. To support simultaneous inputs, collect all active actions instead.

#include <iostream>
#include <map>
#include <chrono>
#include <thread>

enum class KeyCode { W, A, S, D, Space, Escape, Unknown };
std::map<KeyCode,bool> g_keyStates;

void PollRawInput() {
  static int frameCounter = 0;
  frameCounter++;
  if (frameCounter % 10 == 1) {
    g_keyStates[KeyCode::W] = true;
    g_keyStates[KeyCode::Space] = false;
  } else if (frameCounter % 10 == 5) {
    g_keyStates[KeyCode::W] = false;
    g_keyStates[KeyCode::Space] = true;
  } else {
    g_keyStates[KeyCode::Space] = false;
  }
}

bool IsKeyPressed(KeyCode key) { return g_keyStates[key]; }

enum class GameAction { MoveForward, MoveBackward, MoveLeft, MoveRight, Jump, Quit, None };

class InputManager {
public:
  InputManager() { /* default bindings */ }
  GameAction GetActiveAction() {
    for (const auto& pair : keyBindings) {
      if (IsKeyPressed(pair.first)) return pair.second;
    }
    return GameAction::None;
  }
  void SetKeyBinding(KeyCode key, GameAction action) { keyBindings[key] = action; }
private:
  std::map<KeyCode,GameAction> keyBindings;
};

int main() {
  bool running = true;
  InputManager inputManager;
  while (running) {
    PollRawInput();
    GameAction currentAction = inputManager.GetActiveAction();
    // Handle action...
    std::this_thread::sleep_for(std::chrono::milliseconds(50));
  }
}
