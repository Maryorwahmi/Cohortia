#include <iostream>
#include <thread>
#include <vector>
#include <numeric>

void processChunk(std::vector<int>& data, int start, int end) {
  for (int i = start; i < end; ++i) {
    data[i] *= 2; // simulate heavy computation
  }
  std::cout << "Thread processing chunk [" << start << ", " << end << ") finished." << std::endl;
}

int main() {
  std::vector<int> gameData(10000);
  std::iota(gameData.begin(), gameData.end(), 1);

  const int numThreads = 4;
  std::vector<std::thread> threads;
  int chunkSize = gameData.size() / numThreads;

  for (int i = 0; i < numThreads; ++i) {
    int start = i * chunkSize;
    int end = (i == numThreads - 1) ? gameData.size() : (i + 1) * chunkSize;
    threads.emplace_back(processChunk, std::ref(gameData), start, end);
  }

  for (auto & t : threads) t.join();
  return 0;
}
Explanation
  • std::thread starts a new execution context when constructed with a callable.
  • Arguments are copied by default; use std::ref to pass references to large containers.
  • Always join threads to ensure they finish and to avoid std::terminate on destruction.
Concrete example
Doubling 10,000 integers in four parallel chunks reduces wall-clock time compared to a single-threaded loop on multi-core hardware.
Key Takeaway

Raw threads give control but require careful argument passing and lifetime management. Profile thread count to match hardware.
