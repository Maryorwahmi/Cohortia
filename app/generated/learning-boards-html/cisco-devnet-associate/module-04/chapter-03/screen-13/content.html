Python Script: Webex API Core Operations

Putting Authentication, Spaces, and Messages into Practice

Script Structure & Key Functions

  • Configuration: Securely manage WEBEX_ACCESS_TOKEN.
  • webex_api_call: Centralized API request handling with error and rate limit retry logic.
  • create_webex_space: Uses POST /v1/rooms to create a new space.
  • send_webex_message: Uses POST /v1/messages to send content to a space.
  • list_webex_spaces: Uses GET /v1/rooms to retrieve existing spaces.
Pro Tip

Always use environment variables for sensitive information like API tokens. This prevents them from being exposed in your code repository.

Python Code Example
import requests
import json
import os
import time

WEBEX_ACCESS_TOKEN = os.getenv("WEBEX_ACCESS_TOKEN", "YOUR_WEBEX_PERSONAL_ACCESS_TOKEN")
BASE_URL = "https://api.webex.com/v1"

def webex_api_call(method, endpoint, data=None, params=None):
    headers = {
        "Authorization": f"Bearer {WEBEX_ACCESS_TOKEN}",
        "Content-Type": "application/json"
    }
    url = f"{BASE_URL}{endpoint}"
    try:
        response = requests.request(method, url, headers=headers, json=data, params=params)
        response.raise_for_status()
        return response.json() if response.content else {}
    except requests.exceptions.RequestException as e:
        print(f"Webex API Error: {e}")
        if hasattr(e, 'response') and e.response is not None:
            if e.response.status_code == 429:
                retry_after = int(e.response.headers.get("Retry-After", 60))
                time.sleep(retry_after)
                return webex_api_call(method, endpoint, data, params)
        return None

def create_webex_space(title):
    endpoint = "/rooms"
    payload = {"title": title}
    return webex_api_call("POST", endpoint, payload)

def send_webex_message(room_id, message_text):
    endpoint = "/messages"
    payload = {
        "roomId": room_id,
        "text": message_text
    }
    return webex_api_call("POST", endpoint, payload)

def list_webex_spaces(max_results=5):
    endpoint = "/rooms"
    params = {"max": max_results}
    return webex_api_call("GET", endpoint, params=params)

if __name__ == "__main__":
    if not WEBEX_ACCESS_TOKEN:
        print("Please set WEBEX_ACCESS_TOKEN environment variable.")
    else:
        # Example usage: list, create, send message
        print("Listing spaces...")
        spaces = list_webex_spaces()
        if spaces and 'items' in spaces:
            for space in spaces['items']:
                print(f"  {space.get('title')}")

        new_space_title = "DevNet Automation Demo"
        new_space = create_webex_space(new_space_title)
        if new_space:
            print(f"Created space: {new_space.get('title')}")
            send_webex_message(new_space['id'], "Hello from DevNet API!")
            print("Message sent.")
