Meraki Webhook Receiver: Code Implementation

Integrating signature verification and event processing logic.

Signature Verification

Uses hmac and hashlib.sha256 to compare the calculated signature with the X-Meraki-API-Signature header.

Event Processing

Extracts alertType from the payload to trigger specific actions, such as logging or initiating further automation.

Key Takeaway

A robust webhook receiver combines strong security measures with intelligent event processing to drive automation.

Python Flask Receiver (Verification & Logic)
import hmac
import hashlib
import os
from flask import Flask, request, abort

WEBHOOK_SECRET = os.getenv("WEBHOOK_SECRET", "my_super_secret_key")
app = Flask(__name__)

@app.route('/meraki-webhook', methods=['POST'])
def meraki_webhook_receiver():
    if not request.is_json: abort(400)
    payload = request.get_json()
    meraki_signature = request.headers.get('X-Meraki-API-Signature')

    if WEBHOOK_SECRET:
        if not meraki_signature: abort(401)
        calculated_signature = hmac.new(
            WEBHOOK_SECRET.encode('utf-8'),
            request.data,
            hashlib.sha256
        ).hexdigest()
        if not hmac.compare_digest(calculated_signature, meraki_signature):
            print(f"Invalid signature. Expected: {calculated_signature}, Received: {meraki_signature}")
            abort(401)
        else: print("Meraki webhook signature verified successfully.")
    else: print("WARNING: WEBHOOK_SECRET not set. Skipping verification.")

    print(f"Received Meraki Webhook Event: {payload.get('alertType')}")
    alert_type = payload.get("alertType")
    device_hostname = payload.get("deviceHostname")

    if alert_type == "Device offline":
        print(f"ALERT: Device '{device_hostname}' just went OFFLINE!")
        # e.g., send_webex_message(f"CRITICAL: {device_hostname} is offline!")
    elif alert_type == "Device online":
        print(f"INFO: Device '{device_hostname}' is now ONLINE.")
    else:
        print(f"Received unhandled alert type: {alert_type}")

    return "Webhook received successfully", 200
