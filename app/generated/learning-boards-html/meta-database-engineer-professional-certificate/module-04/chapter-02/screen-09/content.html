Module 4 › Chapter 4.2 › ERD → Schema
Guideline
  • Create tables for each entity and underline PK.
  • Add FK columns on the many side for 1:N relationships.
  • Create junction tables for M:N and include association attributes.
  • Set NULL/NOT NULL according to optionality.
Visual Example (schema sketch)
CREATE TABLE Users (
  user_id SERIAL PRIMARY KEY,
  username VARCHAR(100) NOT NULL,
  email VARCHAR(255) NOT NULL
);

CREATE TABLE Posts (
  post_id SERIAL PRIMARY KEY,
  user_id INT NOT NULL REFERENCES Users(user_id),
  content TEXT NOT NULL,
  created_at TIMESTAMP
);

CREATE TABLE Enrollment (
  student_id INT NOT NULL REFERENCES Students(student_id),
  course_id INT NOT NULL REFERENCES Courses(course_id),
  enrolled_on DATE,
  PRIMARY KEY (student_id, course_id)
);
