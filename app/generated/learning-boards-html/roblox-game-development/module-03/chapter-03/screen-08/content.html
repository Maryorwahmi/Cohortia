Module 3 › Chapter 3.3 › Teleporter Pad
Activity steps
  1. Create TeleportPadA and TeleportPadB in Workspace, anchored and non-colliding.
  2. Insert a Script into TeleportPadA and get a reference to padB.
  3. Validate otherPart, humanoid and player in Touched handler.
  4. Find HumanoidRootPart and set its CFrame to padB.CFrame plus an upward offset.
  5. Use a teleportCooldowns table and os.time to prevent rapid teleportation.
Starter code (template)
-- Script placed inside "TeleportPadA"
local padA = script.Parent
local padB = game.Workspace:FindFirstChild("TeleportPadB")

if not padB then
    warn("TeleportPadB not found! Teleporter will not work.")
    return
end

local teleportCooldowns = {}

local function onPadATouched(otherPart)
    local character = otherPart.Parent
    local humanoid = character and character:FindFirstChildOfClass("Humanoid")
    local player = game.Players:GetPlayerFromCharacter(character)

    if humanoid and player then
        local lastTeleportTime = teleportCooldowns[player.UserId] or 0
        if os.time() - lastTeleportTime < 2 then return end

        local hrp = character:FindFirstChild("HumanoidRootPart")
        if hrp then
            hrp.CFrame = padB.CFrame + Vector3.new(0, 5, 0)
            teleportCooldowns[player.UserId] = os.time()
            print(player.Name .. " teleported to Pad B!")
        end
    end
end

padA.Touched:Connect(onPadATouched)
