Secret Door script template
-- Script placed inside "SecretDoor"
local door = script.Parent
local prompt = door:FindFirstChildOfClass("ProximityPrompt")

if not prompt then
    warn("SecretDoor is missing a ProximityPrompt!")
    return
end

prompt.ActionText = "Open"
prompt.ObjectText = "Secret Door"
prompt.KeyboardKeyCode = Enum.KeyCode.E
prompt.HoldDuration = 0.5

local isDoorOpen = false

local function onPromptTriggered(player)
    if isDoorOpen then return end

    print(player.Name .. " opened the secret door!")
    isDoorOpen = true
    prompt.Enabled = false

    door.Transparency = 1
    door.CanCollide = false

    task.wait(5)

    door.Transparency = 0
    door.CanCollide = true
    isDoorOpen = false
    prompt.Enabled = true
end

prompt.Triggered:Connect(onPromptTriggered)
Testing checklist
  • Confirm prompt shows ActionText and ObjectText correctly.
  • Press and hold to activate when HoldDuration > 0.
  • Ensure door disappears and reappears after the set time.
Pro tip: Disable the prompt while the door is open to avoid competing activations.
