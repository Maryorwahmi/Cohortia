-- Script placed inside a Part named "DamageZone"
local damageZone = script.Parent
local DAMAGE_AMOUNT = 10
local COOLDOWN_TIME = 1 -- seconds

local playersInZone = {}

local function onDamageZoneTouched(otherPart)
    local character = otherPart.Parent
    if not character then return end

    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end

    local player = game.Players:GetPlayerFromCharacter(character)
    if not player then return end

    local lastDamageTime = playersInZone[player.UserId] or 0
    if os.time() - lastDamageTime < COOLDOWN_TIME then
        return
    end

    humanoid:TakeDamage(DAMAGE_AMOUNT)
    playersInZone[player.UserId] = os.time()

    if humanoid.Health <= 0 then
        playersInZone[player.UserId] = nil
    end
end

damageZone.Touched:Connect(onDamageZoneTouched)
Explanation & checks
  • Guard checks prevent non-player triggers and early-exit quickly.
  • Per-player timestamps avoid blocking and correctly scope cooldowns.
  • Cleaning the table on death prevents stale entries.
Common mistake: using task.wait inside the handler which pauses the whole script.
Pro tip: Use UserId as key to handle name changes safely.
