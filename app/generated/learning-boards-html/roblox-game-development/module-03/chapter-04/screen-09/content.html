Example Script
-- Script placed inside a Part named "Coin"
local coin = script.Parent
local clickDetector = coin:FindFirstChildOfClass("ClickDetector") or Instance.new("ClickDetector")
if not clickDetector.Parent then clickDetector.Parent = coin end -- Ensure ClickDetector exists

local SCORE_AMOUNT = 10
local collected = false -- Debounce for collection

local function onCoinCollected(player)
  if collected then return end
  collected = true

  coin.Transparency = 1
  coin.CanCollide = false
  coin.CanTouch = false -- Prevent further Touched events

  local leaderstats = player:FindFirstChild("leaderstats")
  if not leaderstats then
    leaderstats = Instance.new("Folder")
    leaderstats.Name = "leaderstats"
    leaderstats.Parent = player
  end

  local score = leaderstats:FindFirstChild("Score")
  if not score then
    score = Instance.new("IntValue")
    score.Name = "Score"
    score.Parent = leaderstats
  end

  score.Value = score.Value + SCORE_AMOUNT
  task.wait(1)
  coin:Destroy()
end

clickDetector.MouseClick:Connect(onCoinCollected)
print("Coin script initialized!")
Explanation
  • The script ensures a ClickDetector exists and is parented to the coin.
  • collected prevents duplicate collection when multiple players interact fast.
  • leaderstats and Score are created if missing to avoid nil errors.
  • Setting CanTouch = false prevents additional Touched events after hiding.
Key Takeaway

Protect collection paths and always validate or create player data containers before writing values.
