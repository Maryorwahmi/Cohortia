Step 1 — Generate activity
Log in and run sample queries using different users or roles to populate LOGIN_HISTORY and QUERY_HISTORY.
-- Example commands to run in your worksheet
USE ROLE SYSADMIN;
SELECT CURRENT_USER(), CURRENT_ROLE(), CURRENT_WAREHOUSE();
SELECT COUNT(*) FROM SNOWFLAKE.ACCOUNT_USAGE.TABLES;
USE ROLE PUBLIC;
SELECT 1+1;
Step 2 — Inspect logs
Query LOGIN_HISTORY and QUERY_HISTORY to find recent events, focusing on timestamps and user names.
-- Last 10 successful logins
SELECT event_timestamp, user_name, client_ip, reported_client_type, is_success
FROM SNOWFLAKE.ACCOUNT_USAGE.LOGIN_HISTORY
WHERE is_success = TRUE
ORDER BY event_timestamp DESC
LIMIT 10;
Step 3 — Simulate incident and recover
Create a table, insert rows, DROP it, then use UNDROP to restore. Alternatively, clone it at a timestamp.
CREATE TABLE important_data (id INT, value VARCHAR);
INSERT INTO important_data VALUES (1, 'Alpha'), (2, 'Beta');
SELECT * FROM important_data;

-- Simulate accidental deletion
DROP TABLE important_data;

-- Recover
UNDROP TABLE important_data;
SELECT * FROM important_data;
