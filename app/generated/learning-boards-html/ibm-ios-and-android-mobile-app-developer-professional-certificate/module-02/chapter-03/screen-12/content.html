Starter template
import SwiftUI

struct ContentView: View {
  @State private var greetingMessage = "Hello, Cohortia!"

  var body: some View {
    VStack(spacing: 20) {
      Image(systemName: "hand.wave.fill")
        .resizable()
        .scaledToFit()
        .frame(width:100,height:100)
        .foregroundStyle(.orange)

      Text(greetingMessage)
        .font(.largeTitle)
        .fontWeight(.semibold)
        .multilineTextAlignment(.center)

      Button("Change Greeting") {
        if greetingMessage == "Hello, Cohortia!" {
          greetingMessage = "Welcome, Developer!"
        } else {
          greetingMessage = "Hello, Cohortia!"
        }
      }
      .font(.title2)
      .padding(.horizontal,30)
      .padding(.vertical,15)
      .background(Color.blue)
      .foregroundColor(.white)
      .cornerRadius(15)
    }
    .padding()
  }
}
Run & verify
Open Canvas Live Preview and tap the button. The greeting text should toggle. If not, check for missing @State or missing import SwiftUI.
Extension idea
Add a second button for decrementing a counter, or animate the text change using withAnimation.
Troubleshooting
  • Make sure import SwiftUI is present.
  • Confirm @State is used for mutable greetingMessage.
  • Use Canvas Live Preview to interact quickly before simulator testing.
Key Takeaway

A small @State-driven view plus Canvas interaction is a powerful way to validate UI behavior quickly.
