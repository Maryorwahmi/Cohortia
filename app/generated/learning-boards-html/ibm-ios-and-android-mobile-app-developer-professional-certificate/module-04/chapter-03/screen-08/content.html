Module 4 › Chapter 4.3 › Counter example
Model (lib/models/counter_model.dart)
import 'package:flutter/foundation.dart';

class CounterModel extends ChangeNotifier {
  int _count = 0;

  int get count => _count;

  void increment() {
    _count++;
    notifyListeners();
  }

  void decrement() {
    _count--;
    notifyListeners();
  }
}
Provider (lib/main.dart)
void main() {
  runApp(
    ChangeNotifierProvider(
      create: (context) => CounterModel(),
      child: const MyApp(),
    ),
  );
}
Consume (lib/screens/counter_screen.dart)
final counter = context.watch<CounterModel>();

// UI shows counter.count
// Buttons call:
context.read<CounterModel>().increment();
context.read<CounterModel>().decrement();
This pattern keeps UI code simple: watch for display, read for actions. Avoid using watch inside event handlers.
