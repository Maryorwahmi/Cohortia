param_reader.cpp
#include <ros/ros.h>

int main(int argc, char** argv) {
  ros::init(argc, argv, "param_reader");
  ros::NodeHandle nh;

  std::string robot_name;
  int sensor_freq;
  bool simulation;
  double kp;

  nh.param("/robot_name", robot_name, "default_robot");
  nh.param("/sensor_frequency", sensor_freq, 10);
  nh.param("/simulation_mode", simulation, true);
  nh.param("/pid_gains/kp", kp, 1.0);

  ROS_INFO("Robot: %s, Freq: %d Hz, Sim: %d, Kp: %.2f",
           robot_name.c_str(), sensor_freq, simulation, kp);

  nh.setParam("/status", "running");
  return 0;
}

Takeaway
Use consistent parameter names and types across languages to avoid integration issues between nodes.
