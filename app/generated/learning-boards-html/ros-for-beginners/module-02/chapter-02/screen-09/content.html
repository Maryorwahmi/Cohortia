#!/usr/bin/env python3
import rospy
from std_msgs.msg import Int64

class NumberCounter:
  def __init__(self):
    self.count = 0
    self.sub = rospy.Subscriber('/number', Int64, self.callback)
    self.pub = rospy.Publisher('/number_count', Int64, queue_size=10)

  def callback(self, msg):
    self.count += 1
    count_msg = Int64()
    count_msg.data = self.count
    self.pub.publish(count_msg)
    rospy.loginfo("Received: %d, Count: %d", msg.data, self.count)

if __name__ == '__main__':
  rospy.init_node('number_counter', anonymous=True)
  node = NumberCounter()
  rospy.spin()
This class-based node keeps state and ensures the publisher is available to callbacks.
Explanation & Example
Run a number publisher that sends Int64 values. Each incoming message increases the internal counter and publishes the updated count to /number_count for verification with rostopic echo.
Why this pattern is useful
This pattern lets nodes act as filters, aggregators, or converters in a pipeline of transformations, where each node performs a single, testable task.
Key Takeaway

Create the publisher early, update state inside callbacks, and use rostopic echo to confirm published results.
