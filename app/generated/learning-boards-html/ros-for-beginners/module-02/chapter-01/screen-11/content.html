Step-by-step lab

Python: number_publisher.py
#!/usr/bin/env python3
import rospy
from std_msgs.msg import Int64

if __name__ == '__main__':
  rospy.init_node('number_publisher')
  pub = rospy.Publisher('/number', Int64, queue_size=10)
  rate = rospy.Rate(1)
  i = 1
  while not rospy.is_shutdown():
    msg = Int64()
    msg.data = i
    pub.publish(msg)
    i += 1
    rate.sleep()
C++: twist_publisher.cpp
// Build and run this node after adding to CMakeLists
#include <ros/ros.h>
#include <geometry_msgs/Twist.h>

int main(int argc, char** argv) {
  ros::init(argc, argv, "twist_publisher");
  ros::NodeHandle nh;
  ros::Publisher pub = nh.advertise<geometry_msgs::Twist>("/cmd_vel", 10);
  ros::Rate rate(2);
  while (ros::ok()) {
    geometry_msgs::Twist t;
    t.linear.x = 0.5;
    t.angular.z = 0.1;
    pub.publish(t);
    rate.sleep();
  }
  return 0;
}
Verification steps
  1. Run both publishers.
  2. Use rostopic echo /number and rostopic echo /cmd_vel to view messages.
  3. Use rostopic hz to confirm 1 Hz and 2 Hz rates.
  4. Publish a manual Twist with linear.x = 2.0 and observe behavior.
