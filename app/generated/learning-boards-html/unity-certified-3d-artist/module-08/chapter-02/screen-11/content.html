using UnityEngine; using System.IO; public class IconRenderer : MonoBehaviour { public Camera iconCamera; public RenderTexture renderTexture; public string outputFolderPath = "Assets/GeneratedIcons/"; public string iconName = "MyItemIcon"; [ContextMenu("Render And Save Icon")] public void RenderAndSaveIcon() { if (iconCamera == null || renderTexture == null) { Debug.LogError("Icon Camera or Render Texture not assigned!"); return; } if (!Directory.Exists(outputFolderPath)) { Directory.CreateDirectory(outputFolderPath); } iconCamera.targetTexture = renderTexture; iconCamera.Render(); RenderTexture.active = renderTexture; Texture2D iconTexture = new Texture2D(renderTexture.width, renderTexture.height, TextureFormat.ARGB32, false); iconTexture.ReadPixels(new Rect(0, 0, renderTexture.width, renderTexture.height), 0, 0); iconTexture.Apply(); RenderTexture.active = null; byte[] bytes = iconTexture.EncodeToPNG(); DestroyImmediate(iconTexture); string filePath = Path.Combine(outputFolderPath, iconName + ".png"); File.WriteAllBytes(filePath, bytes); Debug.Log($"Icon saved to: {filePath}"); UnityEditor.AssetDatabase.Refresh(); } }
Explanation
The script is editor-focused: it renders the camera to the RenderTexture, reads and encodes pixels, writes a PNG, and refreshes the asset database so the file appears in the Project window.
Gotcha

Always reset RenderTexture.active and destroy temporary Texture2D to avoid leaks in the editor environment.
