Billboarding Script

A compact C# template rotates the Canvas to face the camera and updates a fill image for health.

using UnityEngine;
using UnityEngine.UI; // Required for Image component

public class BillboardUI : MonoBehaviour
{
  public Transform cameraTransform;
  public Image healthFillImage; // Assign your HealthBarFill Image here
  public float maxHealth = 100f;
  private float currentHealth;

  void Start()
  {
    if (cameraTransform == null)
    {
      Camera mainCam = Camera.main;
      if (mainCam != null)
      {
        cameraTransform = mainCam.transform;
      }
      else
      {
        Debug.LogError("Main Camera not found! Please assign cameraTransform manually.");
        enabled = false;
        return;
      }
    }
    currentHealth = maxHealth;
    UpdateHealthBar();
  }

  void LateUpdate()
  {
    if (cameraTransform != null)
    {
      Vector3 lookAtDir = transform.position - cameraTransform.position;
      lookAtDir.y = 0; // Keep it upright
      transform.rotation = Quaternion.LookRotation(lookAtDir);
    }
  }

  public void TakeDamage(float damageAmount)
  {
    currentHealth -= damageAmount;
    currentHealth = Mathf.Clamp(currentHealth, 0, maxHealth);
    UpdateHealthBar();
    if (currentHealth <= 0)
    {
      Debug.Log(transform.parent.name + " defeated!");
      gameObject.SetActive(false);
    }
  }

  void UpdateHealthBar()
  {
    if (healthFillImage != null)
    {
      healthFillImage.fillAmount = currentHealth / maxHealth;
    }
  }

  void Update()
  {
    if (Input.GetKeyDown(KeyCode.T))
    {
      TakeDamage(10f);
    }
  }
}
Note: Attach this to the World Space Canvas and assign the Fill Image to healthFillImage. Use LateUpdate for rotation after object movement.
Concrete scenario
With the Canvas parented to an enemy capsule, press play and move the camera around. The bar remains oriented to the camera and upright because we zero the y component of the look direction.
Misconception
Rotating all axes makes the UI tilt toward the camera which reduces readability. Lock vertical rotation for stable reading.
Key Takeaway

Billboarding improves legibility but should preserve upright orientation to remain readable.
