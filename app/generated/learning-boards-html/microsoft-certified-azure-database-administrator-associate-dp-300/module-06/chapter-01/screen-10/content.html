Step-by-step Code
-- Create the AdminAuditLog table
CREATE TABLE AdminAuditLog (
  LogID INT PRIMARY KEY IDENTITY(1,1),
  ActionType NVARCHAR(50) NOT NULL,
  ActionDescription NVARCHAR(MAX) NOT NULL,
  PerformedBy NVARCHAR(128) NOT NULL,
  ActionTimestamp DATETIME DEFAULT GETUTCDATE()
);
GO

-- Create the LogAdminAction stored procedure
CREATE PROCEDURE LogAdminAction
  @ActionType NVARCHAR(50),
  @ActionDescription NVARCHAR(MAX)
AS
BEGIN
  SET NOCOUNT ON;
  INSERT INTO AdminAuditLog (ActionType, ActionDescription, PerformedBy)
  VALUES (@ActionType, @ActionDescription, SUSER_SNAME());
END;
GO
Execute and Verify
-- Execute examples
EXEC LogAdminAction @ActionType = 'User Created',
  @ActionDescription = 'Created new database user AppUser with CONNECT permission.';
GO

EXEC LogAdminAction @ActionType = 'Permission Granted',
  @ActionDescription = 'Granted SELECT on Sales.Orders to ReportingRole.';
GO

-- Verify
SELECT * FROM AdminAuditLog;
GO
After running, inspect PerformedBy and ActionTimestamp to confirm the audit captured the executing user and time. Audit logs are essential for accountability and troubleshooting.
