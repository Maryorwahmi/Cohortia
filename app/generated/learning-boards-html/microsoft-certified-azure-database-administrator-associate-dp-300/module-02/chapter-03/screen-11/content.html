# Define variables
RESOURCE_GROUP="cohortia-mysql-flex-rg"
LOCATION="westus2"
MYSQL_SERVER_NAME="cohortiamysqlflex$(openssl rand -hex 4)"
MYSQL_ADMIN_USER="mysqladmin"
MYSQL_ADMIN_PASSWORD="P@ssword12345!"
VNET_NAME="cohortia-mysql-vnet"
SUBNET_NAME="mysql-subnet"
MYSQL_VERSION="8.0"

# Create resource group and VNet
az group create --name $RESOURCE_GROUP --location $LOCATION
az network vnet create --resource-group $RESOURCE_GROUP --name $VNET_NAME --location $LOCATION --address-prefix 10.1.0.0/16
az network vnet subnet create --resource-group $RESOURCE_GROUP --vnet-name $VNET_NAME --name $SUBNET_NAME --address-prefix 10.1.1.0/24 --service-endpoints Microsoft.Sql

# Get subnet ID
SUBNET_ID=$(az network vnet subnet show --resource-group $RESOURCE_GROUP --vnet-name $VNET_NAME --name $SUBNET_NAME --query id -o tsv)

# Deploy MySQL Flexible Server with VNet integration
az mysql flexible-server create --resource-group $RESOURCE_GROUP --name $MYSQL_SERVER_NAME --location $LOCATION --version $MYSQL_VERSION --sku-name Standard_D4ds_v4 --tier GeneralPurpose --storage-size 64 --private-dns-zone "privatelink.mysql.database.azure.com" --vnet $VNET_NAME --subnet $SUBNET_ID --admin-user $MYSQL_ADMIN_USER --admin-password $MYSQL_ADMIN_PASSWORD --high-availability ZoneRedundant
Deployment Notes
  • Use General Purpose sizing for balanced production performance.
  • Provision sufficient storage and IOPS for expected traffic spikes.
  • Plan a jump box or VPN for admin access to private servers.
Practical Tip
Test connectivity from a VM in the same VNet immediately after deployment to verify DNS and networking configuration.
Key Takeaway

Scripted deployments save time and ensure consistency; always validate network resolution and access after create.
