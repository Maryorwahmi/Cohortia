Module 7 › Chapter 7.4 › Timer Trigger Pattern
Polling and scheduled work
Use Timer triggers to run repeatable checks and short jobs against relational databases.
Safe processing checklist
  • Fetch a small batch and use a lease/status column.
  • Wrap row updates in short transactions to prevent race conditions.
  • Make processing idempotent so retries are safe.
Key Takeaway
Combine short leases, idempotency, and retries to make timer-based processing reliable under load.
C# Timer Trigger example — cleanup
// Runs daily at 3 AM UTC
[FunctionName("DailyCleanup")]
public static async Task Run([TimerTrigger("0 0 3 * * *")]TimerInfo myTimer, ILogger log)
{
  log.LogInformation($"C# Timer trigger function executed at: {DateTime.Now}");

  string connectionString = Environment.GetEnvironmentVariable("SqlConnectionString");
  string keyVaultUri = Environment.GetEnvironmentVariable("KeyVaultUri");
  string dbPasswordSecretName = Environment.GetEnvironmentVariable("DbPasswordSecretName");

  // Retrieve password from Key Vault using DefaultAzureCredential
  var client = new SecretClient(new Uri(keyVaultUri), new DefaultAzureCredential());
  KeyVaultSecret secret = await client.GetSecretAsync(dbPasswordSecretName);
  string dbPassword = secret.Value;

  connectionString = connectionString.Replace("{db_password}", dbPassword);

  using (SqlConnection conn = new SqlConnection(connectionString))
  {
    await conn.OpenAsync();
    var text = "UPDATE dbo.UserAccounts SET IsActive = 0 WHERE LastLoginDate < DATEADD(month, -6, GETDATE()) AND IsActive = 1;";
    using (SqlCommand cmd = new SqlCommand(text, conn))
    {
      int rowsAffected = await cmd.ExecuteNonQueryAsync();
      log.LogInformation($"Updated {rowsAffected} inactive user accounts.");
    }
  }
}
