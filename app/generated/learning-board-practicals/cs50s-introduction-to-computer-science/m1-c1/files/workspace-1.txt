TEST_BINARY = '11010'
TEST_DECIMAL = 27
PADDING_BITS = 8


def binary_to_decimal(b):
    """Convert a binary string to its decimal integer."""
    return int(b, 2)


def decimal_to_binary(n):
    """Convert a non-negative integer to an unpadded binary string."""
    return bin(n)[2:]


def decimal_to_padded_binary(n, bits):
    """Return the binary representation of n padded with leading zeros to 'bits' length."""
    return format(n, f'0{bits}b')


def values_for_bits(n):
    """Number of unique values representable with n bits (2^n)."""
    return 2 ** n


def main():
    print("Baseline conversions:")
    print(f"1) Binary {TEST_BINARY} -> {binary_to_decimal(TEST_BINARY)}")
    print(f"2) Decimal {TEST_DECIMAL} -> {decimal_to_binary(TEST_DECIMAL)}")
    print(f"3) 6 bits can represent {values_for_bits(6)} unique values")
    print(f"4) 24-bit color max per component -> {values_for_bits(8)-1}")
    print(f"Bonus: {TEST_DECIMAL} as {PADDING_BITS}-bit binary -> {decimal_to_padded_binary(TEST_DECIMAL, PADDING_BITS)}")


if __name__ == '__main__':
    main()
