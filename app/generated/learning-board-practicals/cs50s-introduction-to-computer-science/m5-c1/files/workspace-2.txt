  id INTEGER PRIMARY KEY AUTOINCREMENT,
  title TEXT NOT NULL,
  department TEXT NOT NULL,
  -- Current constraint: allows credits >= 1 (no upper bound). Modify this to enforce 1..6.
  credits INTEGER NOT NULL CHECK (credits >= 1)
